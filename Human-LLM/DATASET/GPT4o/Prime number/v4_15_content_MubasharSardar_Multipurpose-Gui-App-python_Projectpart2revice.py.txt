from tkinter import *
from tkinter import messagebox

def check_prime_number():
    """Window to check if a number is prime."""
    cp = Tk()
    cp.title("Check Prime Number")
    cp.geometry("500x500")
    cp.configure(background='yellow')

    entry1 = StringVar()
    entry_box = Entry(cp, textvariable=entry1, font=('Arial', 20, 'bold'))
    entry_box.pack()

    def prime_number():
        try:
            no = int(entry_box.get())
            if no <= 1:
                result = f'{no} is not a prime number'
            else:
                for i in range(2, no):
                    if no % i == 0:
                        result = f'{no} is not a prime number'
                        break
                else:
                    result = f'{no} is a prime number'
            Label(cp, text=result, font=('Arial', 15)).pack()
        except ValueError:
            messagebox.showerror("Invalid input", "Please enter a valid number")

    btn = Button(cp, text='Calculate', command=prime_number, font=('Arial', 20, 'bold'))
    btn.pack()

def radian_to_degree():
    """Window to convert radians to degrees."""
    rd = Tk()
    rd.title('Radian to Degree')
    rd.geometry("500x500")
    rd.configure(background='yellow')

    Label(rd, text="Enter Radian value to convert into degrees", bg='yellow', font=('Arial', 15, 'bold')).pack()

    r = StringVar()
    take_input = Entry(rd, textvariable=r, font=('Arial', 15, 'bold'))
    take_input.pack()

    def convert_radian_to_degree():
        try:
            radian = float(take_input.get())
            degree = radian * 180 / math.pi
            Label(rd, text=f"Degrees = {degree}", font=('Arial', 15)).pack()
        except ValueError:
            messagebox.showerror("Invalid input", "Please enter a valid radian value")

    btn = Button(rd, text='Calculate', font=('Arial', 20, 'bold'), command=convert_radian_to_degree)
    btn.pack()

def compute_arc_length():
    """Window to compute the arc length of an angle."""
    cal = Tk()
    cal.title('Compute Arc Length of an Angle')
    cal.geometry("500x500")
    cal.configure(background="yellow")

    Label(cal, text="Please Enter Values To Find Arc Length", bg='yellow', font=('Arial', 15, 'bold')).pack()

    diameter_var = StringVar()
    angle_var = StringVar()

    Label(cal, text="Please Enter Diameter here: ", bg='yellow', font=('Arial', 12, 'bold')).pack()
    diameter_entry = Entry(cal, textvariable=diameter_var, font=('Arial', 20, 'bold'))
    diameter_entry.pack()

    Label(cal, text="Please Enter Angle here: ", bg='yellow', font=('Arial', 12, 'bold')).pack()
    angle_entry = Entry(cal, textvariable=angle_var, font=('Arial', 20, 'bold'))
    angle_entry.pack()

    def calculate_arc_length():
        try:
            diameter = float(diameter_entry.get())
            angle = float(angle_entry.get())
            if angle >= 360:
                messagebox.showerror("Invalid input", "Angle not possible")
            else:
                arc_length = (math.pi * diameter) * (angle / 360)
                Label(cal, text=f"Arc Length is: {arc_length}", font=('Arial', 15, 'bold')).pack()
        except ValueError:
            messagebox.showerror("Invalid input", "Please enter valid numerical values")

    btn = Button(cal, text="Calculate", font=('Arial', 20, 'bold'), command=calculate_arc_length)
    btn.pack()

def area_of_sector():
    """Window to compute the area of a sector."""
    aos = Tk()
    aos.title('Compute Area Of Sector')
    aos.geometry("500x500")
    aos.configure(background="yellow")

    Label(aos, text="Please Enter Radius Here", bg='yellow', font=('Arial', 20, 'bold')).pack()

    radius_var = StringVar()
    angle_var = StringVar()

    radius_entry = Entry(aos, textvariable=radius_var, font=('Arial', 20, 'bold'))
    radius_entry.pack()

    Label(aos, text="Please Enter Angle Here", bg='yellow', font=('Arial', 20, 'bold')).pack()
    angle_entry = Entry(aos, textvariable=angle_var, font=('Arial', 20, 'bold'))
    angle_entry.pack()

    def calculate_area_of_sector():
        try:
            radius = float(radius_entry.get())
            angle = float(angle_entry.get())
            area = 0.5 * radius ** 2 * (angle * math.pi / 180)
            Label(aos, text=f"Area of Sector: {area}", font=('Arial', 15, 'bold')).pack()
        except ValueError:
            messagebox.showerror("Invalid input", "Please enter valid numerical values")

    btn = Button(aos, text="Calculate", font=('Arial', 20, 'bold'), command=calculate_area_of_sector)
    btn.pack()
