def print_prime_numbers(limit):
    """Prints all prime numbers up to a specified limit."""
    for n in range(2, limit):
        for x in range(2, n):
            if n % x == 0:
                print(f"{n} equals {x} * {n // x}")
                break
        else:
            print(f"{n} is a prime number")

# Define the limit up to which we want to check for prime numbers
LIMIT = 100

# Print prime numbers up to the specified limit
print_prime_numbers(LIMIT)
