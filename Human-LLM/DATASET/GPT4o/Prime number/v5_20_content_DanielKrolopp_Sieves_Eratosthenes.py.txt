import math

def eratosthenes(n):
    """
    Use the Sieve of Eratosthenes algorithm to find all prime numbers up to n.

    Args:
        n (int): The upper limit for finding prime numbers.
    """
    print("Starting!")
    primes = [True] * (n + 1)  # Include n in the list
    primes[0] = primes[1] = False  # 0 and 1 are not prime numbers
    
    with open("Eratosthenes_output.txt", "w") as file:
        for i in range(2, int(math.sqrt(n)) + 1):
            if primes[i]:
                for j in range(i * i, n + 1, i):
                    primes[j] = False
        
        for i in range(2, n + 1):
            if primes[i]:
                print(i)
                file.write(f"{i}\n")
    
    print(f"Finished generating all primes up to {n}!")

def main():
    try:
        n = int(input("Find primes up to: "))
        if n < 2:
            raise ValueError("Number must be greater than or equal to 2.")
        print("Allocating memory...")
        eratosthenes(n)
    except ValueError as e:
        print(f"Error: {e}")

if __name__ == "__main__":
    main()
