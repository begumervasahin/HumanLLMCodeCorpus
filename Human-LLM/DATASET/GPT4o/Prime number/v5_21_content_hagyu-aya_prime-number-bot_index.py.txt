import os
from bottle import route, run

@route("/")
def hello_world():
    """Handle requests to the root URL by returning an empty response."""
    return ""

def main():
    """Run the Bottle web server."""
    port = int(os.environ.get("PORT", 5000))
    run(host="0.0.0.0", port=port)

if __name__ == "__main__":
    main()
