def check_prime(number):
    """
    Check if a given number is prime.
    
    :param number: The integer to check for primality.
    :return: True if the number is prime, False otherwise.
    """
    if number <= 1:
        return False

    for i in range(2, int(number ** 0.5) + 1):
        if number % i == 0:
            return False

    return True

def main():
    """
    Main function to interact with the user and check for primality.
    """
    mynum = int(input("Please enter an integer you want to check: "))
    if check_prime(mynum):
        print(f"{mynum} is a prime number")
    else:
        print(f"{mynum} is not a prime number")

if __name__ == "__main__":
    main()
