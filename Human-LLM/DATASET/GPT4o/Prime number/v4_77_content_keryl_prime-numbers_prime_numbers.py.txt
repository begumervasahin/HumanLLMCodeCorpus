def is_prime_number(num):
    """
    Returns True if the given input is a prime number, False otherwise.
    """
    if num < 2:
        return False
    for x in range(2, int(num ** 0.5) + 1):
        if num % x == 0:
            return False
    return True

def get_prime_numbers(n):
    """
    Returns a list of all prime numbers between 2 and the given input (inclusive).
    """
    if n < 2:
        return "Number should be greater than or equal to 2"
    
    prime_numbers = [num for num in range(2, n + 1) if is_prime_number(num)]
    return prime_numbers

    print(get_prime_numbers(10))
