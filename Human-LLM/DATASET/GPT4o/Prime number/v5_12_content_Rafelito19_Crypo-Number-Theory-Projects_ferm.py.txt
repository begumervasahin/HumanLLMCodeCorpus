def simple_modular_exponentiation(x, e, m):
    """
    Calculate (x^e) % m using modular exponentiation.
    """
    base = x
    exponent = e
    result = 1
    while exponent > 0:
        if exponent % 2 == 0:
            base = (base * base) % m
            exponent //= 2
        else:
            result = (base * result) % m
            exponent -= 1
    return result

def is_prime_trial_division(num):
    """
    Check if a number is prime using trial division.
    """
    if num < 2:
        return False
    if num == 2:
        return True
    if num % 2 == 0:
        return False
    for i in range(3, int(num ** 0.5) + 1, 2):
        if num % i == 0:
            return False
    return True

def gcd(a, b):
    """
    Calculate the Greatest Common Divisor (GCD) of a and b.
    """
    while b:
        a, b = b, a % b
    return a

def chinese_test(num):
    """
    Test for primality using an ancient Chinese method.
    """
    a = 2
    if gcd(a, num) != 1:
        return False
    if simple_modular_exponentiation(a, num - 1, num) != 1:
        return False
    return True

def main():
    bound = 100000
    for n in range(2, bound):
        if chinese_test(n) and not is_prime_trial_division(n):
            print(f"{n} passed the Chinese test but it is not prime")

if __name__ == "__main__":
    main()
