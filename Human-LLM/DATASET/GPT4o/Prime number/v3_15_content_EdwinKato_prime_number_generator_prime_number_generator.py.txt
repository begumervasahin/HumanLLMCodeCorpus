def generate_prime_numbers(n):
    """
    Generate a list of prime numbers up to a given number n.
    
    :param n: The upper limit to generate prime numbers.
    :return: A list of prime numbers up to n.
    """
    if n < 2:
        return []

    def is_prime(num):
        """Check if a number is prime."""
        if num < 2:
            return False
        for divisor in range(2, int(num ** 0.5) + 1):
            if num % divisor == 0:
                return False
        return True

    prime_numbers = [number for number in range(2, n + 1) if is_prime(number)]
    
    return prime_numbers

# Example usage
print(generate_prime_numbers(10))
