import numpy as np
import csv

def prime_check(x, prime_list):
    """
    Check if a number x is prime using a list of known primes.
    
    Args:
        x (int): The number to check.
        prime_list (numpy.array): An array of known prime numbers.
        
    Returns:
        bool: True if x is a prime number, False otherwise.
    """
    for prime in prime_list:
        if prime > np.sqrt(x):
            break
        if x % prime == 0:
            return False
    return True

def generate_primes_up_to(target):
    """
    Generate a list of prime numbers up to the specified target.
    
    Args:
        target (int): The upper limit to find prime numbers (inclusive).
        
    Returns:
        numpy.array: An array of prime numbers up to and including the target.
    """
    primes = np.array([2])  # Start with the first prime number
    for i in range(3, target + 1, 2):  # Skip even numbers
        if prime_check(i, primes):
            primes = np.append(primes, i)
    return primes

def save_primes_to_csv(primes, target):
    """
    Save the list of primes to a CSV file.
    
    Args:
        primes (numpy.array): An array of prime numbers.
        target (int): The upper limit used to generate the primes.
    """
    filename = f'prime_{target}_({len(primes)}).csv'
    with open(filename, 'w', newline='') as csvfile:
        prime_writer = csv.writer(csvfile, delimiter=',')
        prime_writer.writerow(primes)

def main():
    """
    Main function to generate prime numbers up to a target and save them to a CSV file.
    """
    target = 100
    primes = generate_primes_up_to(target)
    save_primes_to_csv(primes, target)
    
    print(f"Target: {target}")
    print(f"Number of primes found: {len(primes)}")

if __name__ == "__main__":
    main()
