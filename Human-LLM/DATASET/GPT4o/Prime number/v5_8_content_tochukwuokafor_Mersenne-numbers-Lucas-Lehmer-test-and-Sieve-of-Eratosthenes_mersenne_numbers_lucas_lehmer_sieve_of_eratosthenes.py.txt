import math

def mersenne_number(p):
    """
    Calculate the Mersenne number for a given p.
    
    Args:
        p (int): The exponent.
        
    Returns:
        int: The Mersenne number (2^p - 1).
    """
    return (2 ** p) - 1

def is_prime(number):
    """
    Check if a number is prime.
    
    Args:
        number (int): The number to check.
        
    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if number <= 1:
        return False
    for i in range(2, int(math.sqrt(number)) + 1):
        if number % i == 0:
            return False
    return True

def get_primes(start, end):
    """
    Get a list of prime numbers in the range [start, end).
    
    Args:
        start (int): The lower bound of the range (inclusive).
        end (int): The upper bound of the range (exclusive).
        
    Returns:
        list: A list of prime numbers in the range [start, end).
    """
    return [i for i in range(start, end) if is_prime(i)]

def lucas_lehmer(p):
    """
    Perform the Lucas-Lehmer test for Mersenne numbers.
    
    Args:
        p (int): The exponent.
        
    Returns:
        bool: True if the Mersenne number 2^p - 1 is prime, False otherwise.
    """
    if p == 2:
        return True
    s = 4
    m = (2 ** p) - 1
    for _ in range(p - 2):
        s = (s * s - 2) % m
    return s == 0

def is_prime_fast(number):
    """
    Fast check if a number is prime using an optimized algorithm.
    
    Args:
        number (int): The number to check.
        
    Returns:
        bool: True if the number is prime, False otherwise.
    """
    if number <= 1:
        return False
    if number % 2 == 0:
        return number == 2
    for factor in range(3, int(math.sqrt(number)) + 1, 2):
        if number % factor == 0:
            return False
    return True

def get_primes_fast(limit):
    """
    Get a list of prime numbers up to a given limit using a fast algorithm.
    
    Args:
        limit (int): The upper limit.
        
    Returns:
        list: A list of prime numbers up to the given limit.
    """
    return [number for number in range(limit) if is_prime_fast(number)]

def list_true(n):
    """
    Create a list of boolean values where index i is True if i >= 2.
    
    Args:
        n (int): The upper limit of the list.
        
    Returns:
        list: A list of boolean values.
    """
    return [i >= 2 for i in range(n + 1)]

def mark_false(bool_list, p):
    """
    Mark multiples of p as False in the boolean list.
    
    Args:
        bool_list (list): The list of boolean values.
        p (int): The number whose multiples are to be marked False.
        
    Returns:
        list: The updated list of boolean values.
    """
    for i in range(2 * p, len(bool_list), p):
        bool_list[i] = False
    return bool_list

def find_next(bool_list, p):
    """
    Find the next True value in the boolean list after index p.
    
    Args:
        bool_list (list): The list of boolean values.
        p (int): The current index.
        
    Returns:
        int: The index of the next True value, or None if not found.
    """
    for i in range(p + 1, len(bool_list)):
        if bool_list[i]:
            return i
    return None

def prime_from_list(bool_list):
    """
    Get the indices of True values in the boolean list.
    
    Args:
        bool_list (list): The list of boolean values.
        
    Returns:
        list: A list of indices where the value is True.
    """
    return [i for i, is_prime in enumerate(bool_list) if is_prime]

def sieve(limit):
    """
    Sieve of Eratosthenes to find all prime numbers up to a given limit.
    
    Args:
        limit (int): The upper limit.
        
    Returns:
        list: A list of prime numbers up to the given limit.
    """
    bool_list = list_true(limit)
    p = 2
    while p is not None:
        bool_list = mark_false(bool_list, p)
        p = find_next(bool_list, p)
    return prime_from_list(bool_list)
