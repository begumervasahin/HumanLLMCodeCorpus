def sieve_of_eratosthenes(limit):
    """
    Sieve of Eratosthenes to find all prime numbers up to a given limit.
    """
    if limit < 2:
        return []

    sieve = [True] * limit
    sieve[0] = sieve[1] = False  # 0 and 1 are not primes

    for num in range(3, int(limit ** 0.5) + 1, 2):
        if sieve[num]:
            sieve[num * num::2 * num] = [False] * ((limit - num * num - 1) // (2 * num) + 1)
    
    return [2] + [num for num in range(3, limit, 2) if sieve[num]]

def optimized_sieve_of_eratosthenes(limit):
    """
    Optimized Sieve of Eratosthenes to find all prime numbers up to a given limit.
    """
    if limit < 2:
        return []

    sieve = [True] * (limit // 2)
    
    for num in range(3, int(limit ** 0.5) + 1, 2):
        if sieve[num // 2]:
            sieve[num * num // 2::num] = [False] * ((limit - num * num - 1) // (2 * num) + 1)
    
    return [2] + [2 * num + 1 for num in range(1, limit // 2) if sieve[num]]

if __name__ == "__main__":
    limit = 2000000000
    list_of_primes = optimized_sieve_of_eratosthenes(limit)
    print(f"There are {len(list_of_primes)} primes from 0 to {limit}")
