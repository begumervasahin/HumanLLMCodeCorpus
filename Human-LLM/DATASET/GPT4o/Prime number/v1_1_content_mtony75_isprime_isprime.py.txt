def is_prime(number):
    """
    Function to check if a number is prime.
    """
    if number <= 1:
        return False
    elif number == 2:
        return True
    else:
        for element in range(2, number // 2 + 1):
            if number % element == 0:
                return False
        return True

    Function to find all prime numbers within a given range.
    """
