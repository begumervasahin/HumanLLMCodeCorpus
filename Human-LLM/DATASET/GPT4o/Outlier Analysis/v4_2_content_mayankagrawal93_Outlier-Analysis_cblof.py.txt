import pandas as pd
import time
import numpy as np

def to_float(x):
    """Converts the input to float."""
    return float(x)

def mutate_dict(f, d):
    """Applies function f to all values in dictionary d."""
    for k, v in d.items():
        d[k] = f(v)

def new_cluster(key, current_tuple):
    """Creates a new cluster with the given key and tuple."""
    summary = []
    cluster_structure = []
    for k in current_tuple:
        value_summary = {current_tuple[k]: 1}
        summary.append([value_summary])
    cluster_structure.append([key])
    cluster_structure.append(summary)
    return cluster_structure

def add_to_cluster(cluster, current_tuple):
    """Adds the current tuple to an existing cluster."""
    for k in current_tuple:
        if current_tuple[k] in cluster[k][0]:
            cluster[k][0][current_tuple[k]] += 1
        else:
            cluster[k][0][current_tuple[k]] = 1

def calculate_similarity(cluster_summary, current_tuple):
    """Calculates similarity between a cluster summary and the current tuple."""
    similarity = 0
    for k in current_tuple:
        if current_tuple[k] in cluster_summary[k][0]:
            support = cluster_summary[k][0][current_tuple[k]]
        else:
            support = 0
        similarity += support / float(sum(cluster_summary[k][0].values()))
    return similarity

def main():
    # Load data
    df = pd.read_csv("fsd.csv")
    data_array = np.array(df)
    
    # Count value frequencies
    value_counts = {}
    for col in df:
        col_array = np.array(df[col])
        for key in col_array:
            value_counts[key] = value_counts.get(key, 0) + 1
        break

    print("1")

    # Process data into a dictionary
    data_dict = {}
    for i, row in enumerate(data_array):
        data_dict[i] = {j: to_float(val) for j, val in enumerate(row)}

    print("2")

    # Clustering
    clusters = []
    for key, current_tuple in data_dict.items():
        if key == 0:
            clusters.append(new_cluster(key, current_tuple))
        else:
            similarities = [calculate_similarity(c[1], current_tuple) for c in clusters]
            max_similarity = max(similarities)
            max_index = similarities.index(max_similarity)
            similarity_threshold = 5
            
            if max_similarity >= similarity_threshold:
                clusters[max_index][0].append(key)
                add_to_cluster(clusters[max_index][1], current_tuple)
            else:
                clusters.append(new_cluster(key, current_tuple))
                print(key)

    # Categorize clusters
    cluster_dict = {len(c[0]): c for c in clusters}
    sorted_cluster_sizes = sorted(cluster_dict.keys())
    
    a = 0.1
    cumulative_size = 0
    large_clusters = []
    small_clusters = []
    for size in sorted_cluster_sizes:
        cumulative_size += size
        if cumulative_size <= a * len(data_dict):
            small_clusters.append(size)
        else:
            large_clusters.append(size)

    # Compute CBLOF
    cblof = {}
    for key, current_tuple in data_dict.items():
        cluster_size = None
        for size, cluster in cluster_dict.items():
            if key in cluster[0]:
                cluster_size = size
                break
        
        if cluster_size in large_clusters:
            similarity = calculate_similarity(cluster_dict[cluster_size][1], current_tuple)
            lof = cluster_size * similarity
            cblof[key] = round(lof, 6)
        else:
            similarities = [calculate_similarity(cluster_dict[size][1], current_tuple) for size in large_clusters]
            max_similarity = min(similarities)
            lof = cluster_size * max_similarity
            cblof[key] = round(lof, 6)

    # Find outliers
    sorted_cblof = sorted(cblof, key=cblof.get, reverse=True)
    top_n = 5
    num_outliers = int((top_n / 100.0) * len(sorted_cblof))
    outliers = sorted_cblof[:num_outliers]

    print(outliers)

if __name__ == "__main__":
    main()
