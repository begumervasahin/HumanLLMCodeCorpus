import math

# Constants for direction
DIRECTION_UNKNOWN = 0
DIRECTION_UP = 1
DIRECTION_DOWN = 2

# Threshold value for peak detection
R = 0.1

def choose_peak(data, max_d):
    """
    Identify peaks in the given data.

    Args:
        data (list): A list of numerical values.
        max_d (float): Maximum difference value for peak detection.    

    Returns:
        tuple: A tuple containing a list of peak values and their corresponding indices.
    """
    peak_values = [data[0]]
    peak_indices = [0]
    current_dir = DIRECTION_UNKNOWN

    for index in range(1, len(data)):
        if current_dir == DIRECTION_UNKNOWN:
            if data[index] > data[index - 1]:
                current_dir = DIRECTION_UP
            elif data[index] < data[index - 1]:
                current_dir = DIRECTION_DOWN
            continue

        if current_dir == DIRECTION_UP and data[index] < data[index - 1]:
            current_dir = DIRECTION_DOWN
            if (data[index - 1] - peak_values[-1]) / max_d >= R:
                peak_values.append(data[index - 1])
                peak_indices.append(index - 1)
            continue

        if current_dir == DIRECTION_DOWN and data[index] > data[index - 1]:
            current_dir = DIRECTION_UP
            if (peak_values[-1] - data[index - 1]) / max_d >= R:
                peak_values.append(data[index - 1])
                peak_indices.append(index - 1)
            continue

    return peak_values, peak_indices

def calc_slope(peaks):
    """
    Calculate the slope between consecutive peaks.

    Args:
        peaks (tuple): A tuple containing a list of peak values and their corresponding indices.

    Returns:
        list: A list of slopes between consecutive peaks.
    """
    peak_values, peak_indices = peaks
    slopes = [
        (peak_values[i] - peak_values[i - 1]) / (peak_indices[i] - peak_indices[i - 1])
        for i in range(1, len(peak_values))
    ]
    return slopes

def calc_mean(data, peaks):
    """
    Calculate the mean values between consecutive peaks.

    Args:
        data (list): A list of numerical values.
        peaks (tuple): A tuple containing a list of peak values and their corresponding indices.

    Returns:
        list: A list of mean values between consecutive peaks.
    """
    peak_values, peak_indices = peaks
    means = [
        sum(data[peak_indices[i - 1]:peak_indices[i]]) / (peak_indices[i] - peak_indices[i - 1])
        for i in range(1, len(peak_values))
    ]
    return means

def calc_dist(slopes, peaks, means):
    """
    Calculate the distance between consecutive slopes, peaks, and means.

    Args:
        slopes (list): A list of slopes between consecutive peaks.     
        peaks (tuple): A tuple containing a list of peak values and their corresponding indices.
        means (list): A list of mean values between consecutive peaks. 

    Returns:
        list: A list of distances between consecutive slopes, peaks, and means.
    """
    _, peak_indices = peaks
    distances = [
        math.sqrt(
            (slopes[i] - slopes[i - 1]) ** 2 +
            (peak_indices[i] - peak_indices[i - 1]) ** 2 +
            (means[i] - means[i - 1]) ** 2
        )
        for i in range(1, len(slopes))
    ]
    return distances

# Sample data for testing
data0 = [
    10.0, 11.0, 14.0, 13.0, 40.0, 39.0, 38.0, 60.0, 59.0, 62.0, 80.0, 81.0,
    79.0, 78.0, 50.0, 52.0, 51.0, 53.0, 10.0, 9.0, 8.0, 50.0
]    

# Execute functions with sample data
peaks = choose_peak(data0, max_d=100.0)
slopes = calc_slope(peaks)
means = calc_mean(data0, peaks)
distances = calc_dist(slopes, peaks, means)

# Print the distances
print(distances)
