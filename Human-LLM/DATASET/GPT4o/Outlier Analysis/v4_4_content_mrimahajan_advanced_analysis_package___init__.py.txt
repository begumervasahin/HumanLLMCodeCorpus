def analyze_stock_data(stock_data, moving_average_window=30):
    """
    Analyzes stock data and calculates moving averages and signal indicators.

    Args:
        stock_data (pd.DataFrame): DataFrame containing stock prices with a 'Close' column.
        moving_average_window (int): Window size for calculating moving average. Default is 30.

    Returns:
        pd.DataFrame: DataFrame with original stock data, moving averages, and buy/sell signals.
    """
    # Ensure the data is sorted by date
    stock_data = stock_data.sort_index()

    # Calculate moving average
    stock_data['Moving_Average'] = stock_data['Close'].rolling(window=moving_average_window).mean()

    # Generate buy/sell signals
    stock_data['Signal'] = 0  # Default to no signal
    stock_data['Signal'][moving_average_window:] = \
        np.where(stock_data['Close'][moving_average_window:] > stock_data['Moving_Average'][moving_average_window:], 1, -1)

    # Calculate daily returns
    stock_data['Daily_Return'] = stock_data['Close'].pct_change()

    # Calculate strategy returns based on signals
    stock_data['Strategy_Return'] = stock_data['Daily_Return'] * stock_data['Signal'].shift(1)

    return stock_data
