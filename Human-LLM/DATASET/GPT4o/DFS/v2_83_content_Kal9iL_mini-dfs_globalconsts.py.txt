from enum import Enum

# Define the available operations
class Operations(Enum):
    LS = "ls"
    FETCH = "fetch"
    UPLOAD = "upload"
    READ = "read"
    QUIT = "quit"

# File paths and configuration settings
NAME_NODE_PATH = "./dfs/NameNode"
DATA_NODE_PATH = "./dfs/DataNode"
INFO_FILE_PATH = f"{NAME_NODE_PATH}/info.pkl"  # Path to the info file

# Constants for configuration
NUM_OF_DATA_SERVERS = 4
NUM_OF_REPLICAS = 3
CHUNK_SIZE = 2 * 1024 * 1024  # 2 MB in bytes

def main():
    """
    Main function to display the configuration and available operations.
    """
    print("Available Operations:")
    for operation in Operations:
        print(f"- {operation.value}")

    print("\nConfiguration Settings:")
    print(f"NameNode Path: {NAME_NODE_PATH}")
    print(f"DataNode Path: {DATA_NODE_PATH}")
    print(f"Info File Path: {INFO_FILE_PATH}")
    print(f"Number of Data Servers: {NUM_OF_DATA_SERVERS}")
    print(f"Number of Replicas: {NUM_OF_REPLICAS}")
    print(f"Chunk Size: {CHUNK_SIZE} bytes")

if __name__ == "__main__":
    main()
