class ChordMessage:
    """
    Defines various types of messages used in a Chord-based distributed system.
    """
    FIND_SUCCESSOR = "FIND_SUCCESSOR"
    RETURN_SUCCESSOR = "RETURN_SUCCESSOR"
    GET_PREDECESSOR = "GET_PREDECESSOR"
    RETURN_PREDECESSOR = "RETURN_PREDECESSOR"
    NOTIFY_PREDECESSOR = "NOTIFY_PREDECESSOR"
    CHECK_ALIVE = "CHECK_ALIVE"
    AM_ALIVE = "AM_ALIVE"
    SEND_FILE = "SEND_FILE"
    REQUEST_FILE = "REQUEST_FILE"
    SOMEONE_DIED = "SOMEONE_DIED"
    LEAVING = "LEAVING"
    INSERT_FILE = "INSERT"
    GET_FILE = "GET"
    GET_FILE_LIST = "LIST"
    ERR = "ERR"
    SUCCESS = "SUCCESS"
    ENTRIES = "ENTRIES"
    OP_SEND_FILE = "SEND"
    OP_REQ_FILE = "REQUEST"
    OP_INSERT_FILE = "INSERT"
    EXIT = "EXIT"
    HELP = "HELP"

def create_message_dict():
    """
    Initializes and returns a dictionary with default values for Chord messages.

    Returns:
        dict: A dictionary initialized with default values for Chord messages.
    """
    return {
        'msg_type': None,
        'filename': None,
        'finger': None,
        'client_ip': None,
        'suc_ip': None,
        'key': None,
        'target': None,
        'pred_ip': None,
        'content': None,
        'hash': None,
        'dead_node': None,
        'file_list': None,
        'hops': 0
    }

def print_message(message):
    """
    Prints the content of the message dictionary in a readable format.

    Args:
        message (dict): The message dictionary to print.
    """
    print("Message Dictionary:")
    for key, value in message.items():
        print(f"{key}: {value}")

# Example usage
if __name__ == "__main__":
    # Create a new message dictionary with default values
    message = create_message_dict()
    
    # Set some example values in the message dictionary
    message['msg_type'] = ChordMessage.SEND_FILE
    message['filename'] = 'example.txt'
    message['client_ip'] = '192.168.1.1'
    
    # Print the message dictionary
    print_message(message)
