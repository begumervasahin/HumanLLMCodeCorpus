# Initialize a global counter
move_counter = 1

def tower_of_hanoi_sol(height, start, end, auxiliary):
    """
    Solve the Tower of Hanoi puzzle and print each move.
    
    Args:
        height (int): The number of disks.
        start (int): The starting rod.
        end (int): The destination rod.
        auxiliary (int): The auxiliary rod.
    """
    global move_counter
    if height >= 1:
        # Move a tower of height-1 to the auxiliary rod
        tower_of_hanoi_sol(height - 1, start, auxiliary, end)
        
        # Move the remaining disk to the destination rod
        print(f"{move_counter}: Move from {start} to {end}")
        move_counter += 1
        
        # Move the tower of height-1 from the auxiliary rod to the destination rod
        tower_of_hanoi_sol(height - 1, auxiliary, end, start)

# Set the number of disks
num_disks = 3

# Call the function to solve the puzzle
tower_of_hanoi_sol(num_disks, 1, 2, 3)
