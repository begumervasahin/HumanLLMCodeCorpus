import os
from invoke import task, run

# Define directories for documentation
docs_dir = 'docs'
build_dir = os.path.join(docs_dir, '_build')

@task
def test(c):
    """
    Run the tests using setup.py.
    """
    run('python setup.py test', pty=True)

@task
def clean(c):
    """
    Clean build artifacts and documentation builds.
    """
    # Remove build artifacts
    run("rm -rf build dist dfs_parser.egg-info")
    # Clean documentation builds
    clean_docs(c)
    print("Cleaned up.")

@task
def clean_docs(c):
    """
    Clean documentation builds.
    """
    run(f"rm -rf {build_dir}")

@task
def browse_docs(c):
    """
    Open the documentation index.html in the browser.
    """
    index_path = os.path.join(build_dir, 'index.html')
    run(f"open {index_path}")

@task
def build_docs(c, clean=False, browse=False):
    """
    Build the documentation.
    """
    if clean:
        clean_docs(c)
    run(f"sphinx-build {docs_dir} {build_dir}", pty=True)
    if browse:
        browse_docs(c)

@task
def readme(c, browse=False):
    """
    Convert README.rst to HTML.
    """
    run('rst2html.py README.rst > README.html')
    if browse:
        run("open README.html")

@task
def publish(c, test=False):
    """
    Publish to the Python Package Index (PyPI).
    """
    if test:
        run('python setup.py register -r test sdist upload -r test')
    else:
        run("python setup.py register sdist upload")
