import MySQLdb

def read_sql_file(filepath):
    """
    Read an SQL file and return the query as a single string with newlines replaced by spaces.
    
    Args:
        filepath (str): The path to the SQL file.
    
    Returns:
        str: The SQL query as a single string.
    """
    with open(filepath, 'r') as file:
        return file.read().replace('\n', ' ')

def execute_query(connection, query):
    """
    Execute a given SQL query on the MySQL database and print the results.
    
    Args:
        connection (MySQLdb.Connection): The MySQL database connection.
        query (str): The SQL query to execute.
    """
    with connection.cursor() as cursor:
        cursor.execute(query)
        rows = cursor.fetchall()
        
        # Print the results
        for row in rows:
            print(row)

def main():
    """Main function to read an SQL file, execute the query, and print the results."""
    sqlfile = 'sql/consistency.sql'
    query = read_sql_file(sqlfile)
    
    # Print the query
    print("Executing query:")
    print(query)
    
    # Connect to the MySQL database
    connection = MySQLdb.connect(host='localhost', user='root', passwd='', db='dfs-nba')
    
    try:
        # Execute the query and print the results
        execute_query(connection, query)
    finally:
        # Ensure the connection is closed
        connection.close()

if __name__ == "__main__":
    main()
