
from PIL import Image

def create_image():
    """
    Create and display an image with a specific pattern.
    """
    # Define the color pattern
    colors = [(0, 0, 0), (0, 0, 0), (128, 128, 128), (256, 256, 256), (256, 256, 256)]
    
    # Repeat the pattern to fill the image
    data = colors * 2000
    
    # Create a new image with RGB mode and size 100x100
    img = Image.new("RGB", (100, 100), "white")
    
    # Put the data into the image
    img.putdata(data)
    
    # Display the image
    img.show()

if __name__ == "__main__":
    create_image()
