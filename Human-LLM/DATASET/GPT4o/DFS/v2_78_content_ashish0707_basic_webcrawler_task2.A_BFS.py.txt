from webcrawler import Crawler  # Import the Crawler class from the webcrawler module
from bfs import BFS  # Import the BFS class from the bfs module

def main():
    """
    Main function to initialize and run the web crawler with BFS search.
    """
    # Set the target URL for the web crawler
    target_url = "https://en.wikipedia.org/wiki/Sustainable_energy"
    
    # Initialize the BFS searcher
    bfs_searcher = BFS()
    
    # Create a Crawler instance with the target URL and BFS searcher
    crawler = Crawler(target_url, bfs_searcher)
    
    # Configure the crawler with specific rules and parameters
    crawler.add_to_mandatory_list('/wiki/')
    crawler.add_to_disallowed_list(':')
    crawler.set_focused_text('solar')
    
    # Begin the web crawling process
    crawler.start()

if __name__ == "__main__":
    main()
