class Solution:
    def addDigits(self, num: int) -> int:
        """
        Given a non-negative integer num, repeatedly add all its digits until the result has only one digit.
        :param num: int
        :return: int
        """
        while num >= 10:  # Continue the process while num has more than one digit
            num = self.sum_of_digits(num)
        return num

    def sum_of_digits(self, num: int) -> int:
        """
        Helper function to calculate the sum of digits of a given number.
        :param num: int
        :return: int
        """
        total = 0
        while num > 0:
            total += num % 10  # Add the last digit of num to total
            num //= 10  # Remove the last digit from num
        return total

# Example usage:
solution = Solution()

# Test cases
print(solution.addDigits(38))   # Output: 2
print(solution.addDigits(123))  # Output: 6
print(solution.addDigits(0))    # Output: 0
