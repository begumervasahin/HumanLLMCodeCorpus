from webcrawler import Crawler  # Assuming the class is in a file named webcrawler.py
from bfs import BFS

def main():
    """
    Main function to set up and start the web crawler with BFS search.
    """
    # Define the URL to crawl
    url = "https://en.wikipedia.org/wiki/Sustainable_energy"
    
    # Create an instance of the BFS searcher
    searcher = BFS()
    
    # Create an instance of the Crawler with the URL and searcher
    my_crawler = Crawler(url, searcher)
    
    # Configure the crawler with mandatory and disallowed paths, and focused text
    my_crawler.add_to_mandatory_list('/wiki/')
    my_crawler.add_to_disallowed_list(':')
    my_crawler.set_focused_text('solar')
    
    # Start the crawling process
    my_crawler.start()

if __name__ == "__main__":
    main()
