from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import Select
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from time import sleep

def setup_driver():
    """
    Sets up the Selenium WebDriver with Firefox.
    
    Returns:
        webdriver.Firefox: Configured WebDriver instance.
    """
    driver_path = 'D:/geckodriver/geckodriver.exe'
    driver = webdriver.Firefox(executable_path=driver_path)
    return driver

def select_date(driver, date):
    """
    Selects a date in the date input field.

    Args:
        driver (webdriver.Firefox): The WebDriver instance.
        date (str): The date to input in 'YYYY-MM-DD' format.
    """
    date_input = driver.find_element(By.ID, 'fecha_salida')
    date_input.click()
    date_input.send_keys(date)
    date_input.click()

def print_dropdown_options(driver, dropdown_id):
    """
    Prints the values of options in a dropdown menu.

    Args:
        driver (webdriver.Firefox): The WebDriver instance.
        dropdown_id (str): The ID of the dropdown element.
    """
    dropdown_element = WebDriverWait(driver, 10).until(
        EC.presence_of_element_located((By.XPATH, f"//select[@id='{dropdown_id}']"))
    )
    options = dropdown_element.find_elements(By.TAG_NAME, 'option')
    for option in options:
        option.click()
        print(option.get_attribute('value'))
        sleep(1)

def main():
    """
    Main function to execute the script logic.
    """
    driver = setup_driver()
    try:
        driver.get("http://checkyourflight.info")
        
        select_date(driver, '2019-08-14')
        sleep(1)  # Wait for any dynamic content to load
        
        print_dropdown_options(driver, 'flight')
        print_dropdown_options(driver, 'hoteles')

        # Sleep for 100 seconds to observe results before closing
        sleep(100)
    finally:
        driver.quit()  # Ensure the browser closes after script completion

if __name__ == "__main__":
    main()
