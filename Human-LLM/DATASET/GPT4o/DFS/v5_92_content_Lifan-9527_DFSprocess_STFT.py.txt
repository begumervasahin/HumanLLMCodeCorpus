This script provides a simplified implementation of the Short-Time Fourier Transform (STFT).
The original STFT method in NumPy may be too complex for some use cases, so this version is recomposed.

import math
import numpy as np

def stft(signal, window_length, slide_step):
    """
    Computes the Short-Time Fourier Transform (STFT) of a given signal.

    Args:
        signal (list or np.ndarray): The input signal data.
        window_length (int): The length of each window for the STFT.
        slide_step (int): The number of samples to slide the window.

    Returns:
        np.ndarray: The STFT result with shape (window_length, num_windows).
    """
    signal = np.asarray(signal)  # Ensure signal is a NumPy array

    # Determine the nearest power of 2 greater than or equal to the length of the signal
    padded_length = 2 ** nearest_power_of_2(len(signal))
    
    # Pad the signal with zeros to match the padded length
    if padded_length > len(signal):
        signal = np.pad(signal, (0, padded_length - len(signal)), mode='constant')

    num_windows = (len(signal) - window_length) // slide_step + 1
    
    # Initialize matrix to hold the windowed segments of the signal
    window_matrix = np.zeros((window_length, num_windows))

    # Populate the matrix with windowed segments
    for i in range(num_windows):
        start_index = slide_step * i
        end_index = start_index + window_length
        window_matrix[:, i] = signal[start_index:end_index]
    
    # Compute the FFT for each window
    stft_result = np.fft.fft(window_matrix, axis=0)
    
    return stft_result

def nearest_power_of_2(n):
    """
    Finds the smallest power of 2 that is greater than or equal to n.

    Args:
        n (int): The input number.

    Returns:
        int: The smallest power of 2 such that 2^power >= n.
    """
    if n <= 0:
        raise ValueError("Input must be greater than 0")
    
    power = 0
    while (1 << power) < n:
        power += 1

    return power

    slide_step = 32
    
    # Compute the STFT
    stft_result = stft(example_signal, window_length, slide_step)
