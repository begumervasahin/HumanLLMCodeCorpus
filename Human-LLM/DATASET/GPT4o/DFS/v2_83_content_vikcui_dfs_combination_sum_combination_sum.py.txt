def find_combinations(candidates, target):
    """
    Find all unique combinations of candidate numbers that sum up to the target.
    
    Args:
        candidates (list of int): A list of candidate numbers (no duplicates).
        target (int): The target sum.
    
    Returns:
        list of list of int: A list of unique combinations that sum up to the target.
    """
    # Sort the candidates to streamline the combination process
    candidates.sort()
    
    # Initialize the list to store the results and a temporary list for the current combination
    result_list = []
    current_combination = []
    
    # Begin the recursive search for combinations
    find_combinations_recursive(candidates, target, 0, current_combination, result_list)
    
    return result_list

def find_combinations_recursive(candidates, target, start_index, current_combination, result_list):
    """
    Recursive helper function to explore combinations that sum up to the target.
    
    Args:
        candidates (list of int): A list of candidate numbers.
        target (int): The remaining sum needed to reach the target.
        start_index (int): The index in candidates to start from for this recursive call.
        current_combination (list of int): The current sequence of numbers being considered.
        result_list (list of list of int): The list where valid combinations are added.
    """
    for i in range(start_index, len(candidates)):
        number = candidates[i]
        
        # If the current number is greater than the remaining target, skip to the next
        if number > target:
            break
        
        # Add the current number to the combination
        current_combination.append(number)
        
        # If the number is less than the target, continue with the reduced target
        if number < target:
            find_combinations_recursive(candidates, target - number, i, current_combination, result_list)
        else:
            # If the number matches the target, add the current combination to results
            result_list.append(list(current_combination))
        
        # Backtrack by removing the last number added
        current_combination.pop()

# Example usage
if __name__ == "__main__":
    candidates = [2, 3, 6, 7]
    target = 7
    combinations = find_combinations(candidates, target)
    print("Unique combinations:", combinations)
