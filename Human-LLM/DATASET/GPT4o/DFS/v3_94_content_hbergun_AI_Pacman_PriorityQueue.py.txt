import heapq

class PriorityQueue:
    """
    A priority queue implemented using a binary heap.
    
    This class manages items with associated priorities, where lower values indicate higher priority.

    Attributes:
        elements (list): List of tuples where each tuple contains a priority and an item.
    """

    def __init__(self):
        """
        Initializes an empty priority queue.
        """
        self.elements = []

    def is_empty(self):
        """
        Checks if the priority queue is empty.

        Returns:
            bool: True if the queue is empty, False otherwise.
        """
        return not self.elements

    def add(self, item, priority):
        """
        Adds an item to the priority queue with a given priority.

        Args:
            item (any): The item to add to the queue.
            priority (int or float): The priority of the item; lower values mean higher priority.
        """
        heapq.heappush(self.elements, (priority, item))

    def pop(self):
        """
        Removes and returns the item with the highest priority (lowest priority value).

        Returns:
            any: The item with the highest priority.

        Raises:
            IndexError: If the queue is empty.
        """
        if self.is_empty():
            raise IndexError("Cannot pop from an empty priority queue")
        return heapq.heappop(self.elements)[1]

# Example usage
if __name__ == "__main__":
    # Initialize the priority queue
    pq = PriorityQueue()
    
    # Add items with different priorities
    pq.add("task1", priority=5)
    pq.add("task2", priority=1)
    pq.add("task3", priority=3)
    
    # Retrieve and print items in order of their priority
    while not pq.is_empty():
        item = pq.pop()
        print(item)
