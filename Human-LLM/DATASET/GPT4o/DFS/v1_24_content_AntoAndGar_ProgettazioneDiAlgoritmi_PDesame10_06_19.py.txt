# Initialize the value of n
n = 5

# Initialize the array T with zero values
T = [0 for _ in range(n + 1)]

# Define the PD function to compute the desired values
def PD(n):
    # Base cases
    T[0] = 1
    T[1] = 4
    T[2] = 10
    
    # Compute values for T[i] using the given recurrence relation
    for i in range(3, n + 1):
        T[i] = 3 * T[i - 1] - T[i - 2]
    
    # Return the nth value
    return T[n]

# Call the PD function and print the result
print(PD(n))
