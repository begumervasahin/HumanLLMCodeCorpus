import os
import sys

def get_main_path():
    """
    Get the absolute path of the main script.
    
    Returns:
        str: The absolute path of the main script.
    """
    return os.path.abspath(__file__)

def get_main_dir(main_path):
    """
    Get the directory of the main script.
    
    Args:
        main_path (str): The absolute path of the main script.
        
    Returns:
        str: The directory of the main script.
    """
    return os.path.dirname(main_path)

def append_sys_path(main_dir):
    """
    Append the main directory to the system path.
    
    Args:
        main_dir (str): The directory to append to the system path.
    """
    sys.path.append(main_dir)

def initialize_app(main_dir):
    """
    Initialize and run the application.
    
    Args:
        main_dir (str): The directory of the main script.
    """
    from gui.app import Application
    app = Application(main_dir)
    app.mainloop()

def main():
    """
    Main function to initialize and run the application.
    """
    main_path = get_main_path()
    main_dir = get_main_dir(main_path)
    
    append_sys_path(main_dir)
    
    print("[ START PROGRAM ]")
    print(f"Start main file: {main_path}")
    print(f"System append path: {main_dir}")
    
    initialize_app(main_dir)

if __name__ == "__main__":
    main()
