def generate_combinations(length, index=0, current_combination=None):
    if current_combination is None:
        current_combination = []
        
    # Base case: If the current index equals the desired length, print the combination
    if index == length:
        print(''.join(current_combination))
        return

    # Iterate over the set of allowed characters
    for char in {'0', '1', '2', '3'}:
        # Check if the character can be appended without violating the consecutive difference rule
        if index == 0 or (char != str(int(current_combination[index - 1]) + 1) and char != str(int(current_combination[index - 1]) - 1)):
            current_combination.append(char)
            generate_combinations(length, index + 1, current_combination)
            current_combination.pop()

# Generate and print all valid combinations of length 6
generate_combinations(6)
