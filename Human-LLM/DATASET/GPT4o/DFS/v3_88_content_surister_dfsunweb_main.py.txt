from selenium import webdriver
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
from time import sleep

def initialize_webdriver():
    """
    Initializes and returns a Selenium WebDriver for Firefox.
    
    Returns:
        webdriver.Firefox: Configured WebDriver instance.
    """
    driver_path = 'D:/geckodriver/geckodriver.exe'
    return webdriver.Firefox(executable_path=driver_path)

def input_date(driver, date):
    """
    Inputs a date into the date field on the webpage.

    Args:
        driver (webdriver.Firefox): The WebDriver instance.
        date (str): The date to input, in 'YYYY-MM-DD' format.
    """
    date_field = driver.find_element(By.ID, 'fecha_salida')
    date_field.click()
    date_field.clear()  # Clear any existing value
    date_field.send_keys(date)

def print_options_from_dropdown(driver, dropdown_id):
    """
    Prints each option's value from a dropdown menu.

    Args:
        driver (webdriver.Firefox): The WebDriver instance.
        dropdown_id (str): The ID of the dropdown menu.
    """
    dropdown = WebDriverWait(driver, 10).until(
        EC.presence_of_element_located((By.ID, dropdown_id))
    )
    options = dropdown.find_elements(By.TAG_NAME, 'option')
    for option in options:
        print(option.get_attribute('value'))
        sleep(1)  # Pause for readability

def main():
    """
    Main function to execute web interactions.
    """
    driver = initialize_webdriver()
    try:
        driver.get("http://checkyourflight.info")
        
        input_date(driver, '2019-08-14')
        sleep(1)  # Wait for the page to update
        
        print("Available flight options:")
        print_options_from_dropdown(driver, 'flight')
        
        print("Available hotel options:")
        print_options_from_dropdown(driver, 'hoteles')

        # Pause to allow observation of results
        sleep(100)
    finally:
        driver.quit()  # Ensure the browser is closed properly

if __name__ == "__main__":
    main()
