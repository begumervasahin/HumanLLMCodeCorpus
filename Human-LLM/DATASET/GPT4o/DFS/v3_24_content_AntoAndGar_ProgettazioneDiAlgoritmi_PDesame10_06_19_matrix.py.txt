def pd_sequence(n):
    """
    Compute the nth value in the sequence based on the given recurrence relations
    and initial conditions.
    """
    set_size = 4  # Size of the set {'1', '2', '3', '4'}
    
    # Initialize the 2D array T with zeros
    T = [[0] * set_size for _ in range(n + 1)]
    
    # Populate the array based on the given conditions
    for x in range(n + 1):
        for y in range(set_size):
            if x == 0:
                T[x][y] = 0
            elif x == 1:
                T[x][y] = 1
            elif x == 2:
                if y == 0:
                    T[x][y] = 3
                elif y in {1, 2}:
                    T[x][y] = 2
                elif y == 3:
                    T[x][y] = 3
            else:
                if y == 0:
                    T[x][y] = T[x-1][y] + T[x-1][y+2] + T[x-1][y+3]
                elif y == 1:
                    T[x][y] = T[x-1][y] + T[x-1][y+2]
                elif y == 2:
                    T[x][y] = T[x-1][y] + T[x-1][y-2]
                elif y == 3:
                    T[x][y] = T[x-1][y] + T[x-1][y-2] + T[x-1][y-3]
    
    return T[n]

# Example usage
if __name__ == "__main__":
    n = 6
    result = pd_sequence(n)
    print(f"The values in the sequence at n={n} are: {result}")
