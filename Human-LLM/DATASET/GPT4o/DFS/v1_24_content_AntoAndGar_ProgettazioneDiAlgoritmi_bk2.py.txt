def generate_binary_strings(n, k):
    """
    Generate all binary strings of length n with at most k ones.
    :param n: int - The length of the binary strings.
    :param k: int - The maximum number of ones in the binary strings.
    """
    def backtrack(i, tot1, sol):
        """
        Helper function to perform the backtracking.
        :param i: int - The current index in the binary string.
        :param tot1: int - The current count of ones in the binary string.
        :param sol: List[str] - The current binary string being constructed.
        """
        if i == n:
            print(''.join(sol))
        else:
            sol.append('0')
            backtrack(i + 1, tot1, sol)
            sol.pop()
            if tot1 < k:
                sol.append('1')
                backtrack(i + 1, tot1 + 1, sol)
                sol.pop()

    backtrack(0, 0, [])

# Example usage:
generate_binary_strings(4, 1)
