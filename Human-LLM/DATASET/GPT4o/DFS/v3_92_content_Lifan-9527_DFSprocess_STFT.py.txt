STFT Method:
This script provides a simplified implementation of the Short-Time Fourier Transform (STFT).
The original numpy STFT method might be too complex for some use cases.

import math
import numpy as np

def stft(signal, window_length, slide_step):
    """
    Computes the Short-Time Fourier Transform (STFT) of the given signal.

    Args:
        signal (list or np.ndarray): The input signal data.
        window_length (int): The length of each window for the STFT.
        slide_step (int): The number of samples to slide the window.

    Returns:
        np.ndarray: STFT result with shape (window_length, num_windows).
    """
    # Ensure the signal is a NumPy array
    signal = np.asarray(signal)

    # Compute the nearest power of 2 greater than or equal to the length of the signal
    padded_length = 2 ** find_nearest_power_of_2(len(signal))
    
    # Pad the signal with zeros if needed
    if padded_length > len(signal):
        signal = np.pad(signal, (0, padded_length - len(signal)), mode='constant')

    # Compute the number of windows
    num_windows = (len(signal) - window_length) // slide_step + 1
    
    # Initialize a matrix to hold windowed data
    window_matrix = np.zeros((window_length, num_windows))

    # Fill the matrix with windowed segments of the signal
    for i in range(num_windows):
        start_idx = slide_step * i
        end_idx = start_idx + window_length
        window_matrix[:, i] = signal[start_idx:end_idx]
    
    # Perform FFT on each window
    stft_result = np.fft.fft(window_matrix, axis=0)
    
    return stft_result

def find_nearest_power_of_2(n):
    """
    Finds the smallest power of 2 that is greater than or equal to n.

    Args:
        n (int): The input number.

    Returns:
        int: The smallest power of 2 such that 2^power >= n.
    """
    if n <= 0:
        raise ValueError("Input must be greater than 0")
    
    power = 0
    while (1 << power) < n:
        power += 1

    return power

    # Generate a sinusoidal signal

    # Define parameters for STFT
    window_length = 64
    slide_step = 32
    
    # Compute STFT
    stft_result = stft(signal, window_length, slide_step)

    print("STFT result shape:", stft_result.shape)
