class TreeNode(object):
    def __init__(self, x):
        self.val = x
        self.left = None  # Initialize the left child as None
        self.right = None  # Initialize the right child as None

class Solution(object):
    def postorderTraversal(self, root):
        """
        Perform a post-order traversal of a binary tree.
        :type root: TreeNode
        :rtype: List[int]
        """
        result = []  # List to store the post-order traversal
        stack = []  # Stack to manage tree nodes
        node = root  # Start with the root node
        visited = set()  # Set to keep track of visited nodes
        
        while node or stack:
            if node:
                stack.append(node)  # Push the current node onto the stack
                node = node.left  # Move to the left child
            else:
                node = stack.pop()  # Pop a node from the stack
                if node.right and node.right not in visited:
                    stack.append(node)  # Push the current node back onto the stack
                    node = node.right  # Move to the right child
                else:
                    visited.add(node)  # Mark the node as visited
                    result.append(node.val)  # Add the node's value to the result list
                    node = None  # Reset the node to None to continue with the stack

        return result

# Example usage:
# Constructing the binary tree
#        1
#         \
#          2
#         /
#        3

root = TreeNode(1)
root.right = TreeNode(2)
root.right.left = TreeNode(3)

solution = Solution()
result = solution.postorderTraversal(root)
print(result)  # Output: [3, 2, 1]
