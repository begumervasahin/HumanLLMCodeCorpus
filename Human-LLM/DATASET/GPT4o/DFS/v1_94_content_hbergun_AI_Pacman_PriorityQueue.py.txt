import heapq

class PriorityQueue:
    """
    A simple priority queue implementation using a heap.
    
    Attributes:
        elements (list): The internal list used by heapq to maintain the priority queue.
    """

    def __init__(self):
        """
        Initializes an empty priority queue.
        """
        self.elements = []

    def empty(self):
        """
        Checks if the priority queue is empty.

        Returns:
            bool: True if the queue is empty, False otherwise.
        """
        return len(self.elements) == 0

    def put(self, item, priority):
        """
        Adds an item to the priority queue with a given priority.

        Args:
            item (any): The item to be added to the queue.
            priority (int or float): The priority of the item. Lower values indicate higher priority.
        """
        heapq.heappush(self.elements, (priority, item))

    def get(self):
        """
        Removes and returns the item with the highest priority (lowest priority value).

        Returns:
            any: The item with the highest priority.
        """
        return heapq.heappop(self.elements)[1]

# Example usage
if __name__ == "__main__":
    # Create a priority queue
    pq = PriorityQueue()
    
    # Add items with different priorities
    pq.put("task1", priority=5)
    pq.put("task2", priority=1)
    pq.put("task3", priority=3)
    
    # Retrieve and print items in priority order
    while not pq.empty():
        item = pq.get()
        print(item)
