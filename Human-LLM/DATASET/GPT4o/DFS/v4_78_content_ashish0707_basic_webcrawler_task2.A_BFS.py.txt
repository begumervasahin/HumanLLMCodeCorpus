from Webcrawler import Crawler  # Import the Crawler class from the Webcrawler module
from bfs import BFS  # Import the BFS class from the bfs module

def main():
    """
    Main function to set up and run the web crawler with BFS search.
    """
    # Define the URL to be crawled
    target_url = "https://en.wikipedia.org/wiki/Sustainable_energy"
    
    # Create an instance of the BFS searcher
    bfs_searcher = BFS()
    
    # Initialize the Crawler with the target URL and the BFS searcher
    crawler = Crawler(target_url, bfs_searcher)
    
    # Configure the crawler with specific settings
    crawler.add_to_mandatory_list('/wiki/')
    crawler.add_to_disallowed_list(':')
    crawler.set_focused_text('solar')
    
    # Start the web crawling process
    crawler.start()

if __name__ == "__main__":
    main()
