def pd_sequence(n):
    """
    Compute the nth value in the sequence defined by:
    T[0] = 1
    T[1] = 4
    T[2] = 10
    T[i] = 3 * T[i - 1] - T[i - 2] for i >= 3
    """
    if n < 0:
        raise ValueError("n must be a non-negative integer")

    # Initialize the array T with base cases
    T = [0] * (n + 1)
    base_cases = {0: 1, 1: 4, 2: 10}
    
    for i in range(min(3, n + 1)):
        T[i] = base_cases[i]

    # Compute values for T[i] using the given recurrence relation
    for i in range(3, n + 1):
        T[i] = 3 * T[i - 1] - T[i - 2]
    
    return T[n]

# Example usage
if __name__ == "__main__":
    n = 5
    result = pd_sequence(n)
    print(f"The {n}th value in the sequence is: {result}")
