import pymysql.cursors

def get_cursor():
    """
    Creates and returns a database cursor object.

    Returns:
        pymysql.cursors.Cursor: Cursor object for database operations.
    """
    connection = pymysql.connect(
        host='localhost',
        user='root',
        password='password',
        db='test_db',
        charset='utf8',
        cursorclass=pymysql.cursors.DictCursor
    )
    return connection.cursor()
