class Solution:
    def addDigits(self, num: int) -> int:
        """
        Given a non-negative integer num, repeatedly add all its digits until the result has only one digit.
        :type num: int
        :rtype: int
        """
        while num >= 10:  # Continue the process while num has more than one digit
            num = sum(int(digit) for digit in str(num))  # Convert num to a string, sum its digits, and update num
        return num

# Example usage:
solution = Solution()

# Test cases
print(solution.addDigits(38))   # Output: 2
print(solution.addDigits(123))  # Output: 6
print(solution.addDigits(0))    # Output: 0
