import cmd

class CLI(cmd.Cmd):
    """A simple command line interface."""

    def __init__(self):
        """Initialize the CLI with a custom prompt and current path."""
        super().__init__()
        self.current_path = ''
        self.prompt = "FYP >>> "

    def do_hi(self, arg):
        """Print a greeting message."""
        print("Hello World.")

    def do_exit(self, arg):
        """Exit the CLI with a goodbye message."""
        print("Goodbye!")
        return True

def main():
    """Main function to start the command line interface."""
    cli = CLI()
    cli.cmdloop()

if __name__ == '__main__':
    main()
