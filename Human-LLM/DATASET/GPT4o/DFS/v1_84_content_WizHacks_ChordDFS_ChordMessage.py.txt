class ChordMessage:
    """
    Class to define various types of messages used in a Chord-based distributed system.
    """
    FIND_SUCCESSOR = "FIND_SUCCESSOR"
    RETURN_SUCCESSOR = "RETURN_SUCCESSOR"
    GET_PREDECESSOR = "GET_PREDECESSOR"
    RETURN_PREDECESSOR = "RETURN_PREDECESSOR"
    NOTIFY_PREDECESSOR = "NOTIFY_PREDECESSOR"
    CHECK_ALIVE = "CHECK_ALIVE"
    AM_ALIVE = "AM_ALIVE"
    SEND_FILE = "SEND_FILE"
    REQUEST_FILE = "REQUEST_FILE"
    SOMEONE_DIED = "SOMEONE_DIED"
    LEAVING = "LEAVING"
    INSERT_FILE = "INSERT"
    GET_FILE = "GET"
    GET_FILE_LIST = "LIST"
    ERR = "ERR"
    SUCCESS = "SUCCESS"
    ENTRIES = "ENTRIES"
    OP_SEND_FILE = "SEND"
    OP_REQ_FILE = "REQUEST"
    OP_INSERT_FILE = "INSERT"
    EXIT = "EXIT"
    HELP = "HELP"

def new_msg_dict():
    """
    Creates and initializes a new message dictionary with default values.

    Returns:
        dict: A dictionary with keys related to Chord messages and default values.
    """
    msg = {
        'msg_type': None,
        'filename': None,
        'finger': None,
        'client_ip': None,
        'suc_ip': None,
        'key': None,
        'target': None,
        'pred_ip': None,
        'content': None,
        'hash': None,
        'dead_node': None,
        'file_list': None,
        'hops': 0
    }
    return msg

# Example usage
if __name__ == "__main__":
    # Create a new message dictionary
    message = new_msg_dict()
    
    # Example of setting some values
    message['msg_type'] = ChordMessage.SEND_FILE
    message['filename'] = 'example.txt'
    message['client_ip'] = '192.168.1.1'
    
    # Print the message dictionary
    print("Message Dictionary:")
    for key, value in message.items():
        print(f"{key}: {value}")
