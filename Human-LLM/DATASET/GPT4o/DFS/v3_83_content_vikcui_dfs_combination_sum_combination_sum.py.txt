def find_combinations(candidates, target):
    """
    Find all unique combinations of candidate numbers that sum up to the target.
    
    Args:
        candidates (list of int): A list of candidate numbers (no duplicates).
        target (int): The target sum.
    
    Returns:
        list of list of int: A list of unique combinations that sum up to the target.
    """
    # Sort candidates to facilitate combination generation
    candidates.sort()
    
    # Prepare to collect results and initiate recursive search
    result_list = []
    current_combination = []
    _find_combinations_recursive(candidates, target, 0, current_combination, result_list)
    
    return result_list

def _find_combinations_recursive(candidates, target, start_index, current_combination, result_list):
    """
    Recursive function to find combinations that sum up to the target.
    
    Args:
        candidates (list of int): A list of candidate numbers.
        target (int): The remaining sum needed to reach the target.
        start_index (int): The index to start considering candidates from.
        current_combination (list of int): The current sequence of numbers being considered.
        result_list (list of list of int): The list where valid combinations are added.
    """
    for i in range(start_index, len(candidates)):
        number = candidates[i]
        
        # If the number exceeds the remaining target, no need to proceed further
        if number > target:
            break
        
        # Include the number in the current combination
        current_combination.append(number)
        
        # Recur with the updated target and the same index (allowing repeated numbers)
        if number < target:
            _find_combinations_recursive(candidates, target - number, i, current_combination, result_list)
        else:
            # If the number equals the target, record the combination
            result_list.append(list(current_combination))
        
        # Remove the last number to backtrack
        current_combination.pop()

# Example usage
if __name__ == "__main__":
    candidates = [2, 3, 6, 7]
    target = 7
    combinations = find_combinations(candidates, target)
    print("Unique combinations:", combinations)
