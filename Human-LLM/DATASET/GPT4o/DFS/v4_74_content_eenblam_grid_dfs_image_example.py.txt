from PIL import Image

def create_and_display_image():
    """
    Create a 100x100 image with a repeating color pattern and display it.
    """
    # Define the color pattern
    color_pattern = [
        (0, 0, 0),       # Black
        (0, 0, 0),       # Black
        (128, 128, 128), # Gray
        (255, 255, 255), # White (corrected from (256,256,256))
        (255, 255, 255)  # White (corrected from (256,256,256))
    ]
    
    # Repeat the pattern to cover the entire image (100x100 pixels)
    num_pixels = 100 * 100
    repeated_pattern = color_pattern * (num_pixels // len(color_pattern))
    
    # Create a new image with RGB mode and size 100x100
    image = Image.new("RGB", (100, 100), "white")
    
    # Apply the color pattern to the image
    image.putdata(repeated_pattern)
    
    # Display the image
    image.show()

if __name__ == "__main__":
    create_and_display_image()
