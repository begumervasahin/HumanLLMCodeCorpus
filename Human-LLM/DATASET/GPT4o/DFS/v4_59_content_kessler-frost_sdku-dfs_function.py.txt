from utils import grid_values, search, display

def main():
    # Define the Sudoku puzzle as a string
    puzzle = '4.....8.5.3..........7......2.....6.....8.4......1.......6.3.7.5..2.....1.4......'
    
    # Convert the puzzle string into a dictionary of possible values for each square
    values = grid_values(puzzle)
    
    # Solve the puzzle using the search function
    solution = search(values)
    
    # Display the solved puzzle
    display(solution)

if __name__ == "__main__":
    main()
