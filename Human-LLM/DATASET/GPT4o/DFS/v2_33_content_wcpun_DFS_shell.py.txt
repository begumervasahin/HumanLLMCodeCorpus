import cmd

class CLI(cmd.Cmd):
    """Simple command line interface."""

    def __init__(self):
        super().__init__()
        self.current_path = ''
        self.prompt = "FYP >>> "  # Custom command prompt

    def do_hi(self, arg):
        """Respond with a greeting."""
        print("Hello World.")

    def do_exit(self, arg):
        """Exit the command line interface."""
        print("Goodbye!")
        return True  # Returning True exits the command loop

def main():
    """Main function to start the CLI."""
    term = CLI()
    term.cmdloop()  # Start the command loop

if __name__ == '__main__':
    main()  # Run the main function if this script is executed
