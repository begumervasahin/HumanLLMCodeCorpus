class Meta:
    """A class to hold metadata information."""

    def __init__(self, user_id, file_name, file_size=None, iv=None):
        """
        Initialize the Meta class with user ID, file name, file size, and IV.

        Args:
            user_id (int): The ID of the user.
            file_name (str): The name of the file.
            file_size (int, optional): The size of the file. Defaults to None.
            iv (str, optional): The initialization vector. Defaults to None.
        """
        self.user_id = user_id
        self.file_name = file_name
        self.file_size = file_size
        self.iv = iv

    def __str__(self):
        """Return a formatted string representation of the metadata."""
        return (f"User ID: {self.user_id}\n"
                f"File Name: {self.file_name}\n"
                f"File Size: {self.file_size}\n"
                f"IV: {self.iv}")

class Code:
    """A class to hold a key."""

    def __init__(self, key):
        """
        Initialize the Code class with a key.

        Args:
            key (str): The key to be stored.
        """
        self.key = key

    def __str__(self):
        """Return a string representation of the key."""
        return f"Key: {self.key}"

class Cipher:
    """A class to hold a cipher."""

    def __init__(self, cipher):
        """
        Initialize the Cipher class with a cipher.

        Args:
            cipher (str): The cipher text to be stored.
        """
        self.cipher = cipher

    def __str__(self):
        """Return a string representation of the cipher."""
        return f"Cipher: {self.cipher}"

def main():
    """Main function to demonstrate the usage of Meta, Code, and Cipher classes."""
    meta = Meta(user_id=123, file_name="example.txt", file_size=2048, iv="abcdef123456")
    code = Code(key="mysecretkey")
    cipher = Cipher(cipher="encryptedtext")

    print(meta)
    print(code)
    print(cipher)

if __name__ == "__main__":
    main()
