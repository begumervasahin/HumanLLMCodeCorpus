from typing import List
import collections

class Solution:
    def intersect(self, nums1: List[int], nums2: List[int]) -> List[int]:
        """
        Given two integer arrays nums1 and nums2, return an array of their intersection.
        Each element in the result must appear as many times as it shows in both arrays.
        :param nums1: List[int]
        :param nums2: List[int]
        :return: List[int]
        """
        result = []
        counter_nums2 = collections.Counter(nums2)

        for num in nums1:
            if counter_nums2.get(num, 0) > 0:
                result.append(num)
                counter_nums2[num] -= 1

        return result

# Example usage:
solution = Solution()

nums1 = [1, 2, 2, 1]
nums2 = [2, 2]
print(f"Intersection of {nums1} and {nums2}: {solution.intersect(nums1, nums2)}")  # Output should be [2, 2]

nums1 = [4, 9, 5]
nums2 = [9, 4, 9, 8, 4]
print(f"Intersection of {nums1} and {nums2}: {solution.intersect(nums1, nums2)}")  # Output should be [4, 9]
