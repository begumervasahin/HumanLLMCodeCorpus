from typing import List

class Node:
    def __init__(self, name: str, x: float = 0, y: float = 0, dest_nodes: List['Node'] = None):
        """
        Initialize a Node instance.

        Args:
            name (str): The name of the node.
            x (float): The x-coordinate of the node. Default is 0.
            y (float): The y-coordinate of the node. Default is 0.
            dest_nodes (List['Node']): A list of destination nodes. Default is None.
        """
        if dest_nodes is None:
            dest_nodes = []
        self.name = name
        self.x = x
        self.y = y
        self.dest_nodes = dest_nodes

    def __eq__(self, other):
        """
        Check if this node is equal to another node based on their names.
        
        Args:
            other (Node): The other node to compare with.

        Returns:
            bool: True if the names are the same, False otherwise.
        """
        return isinstance(other, Node) and self.name == other.name

    def dest_nodes_to_string(self):
        """
        Convert the list of destination nodes to a string representation.

        Returns:
            str: A string representation of the destination nodes.
        """
        next_nodes_str = '['
        if self.dest_nodes:
            next_nodes_str += ','.join(node.name for node in self.dest_nodes)
        next_nodes_str += ']'
        return next_nodes_str

# Example usage
if __name__ == "__main__":
    # Create some nodes
    node_a = Node('A')
    node_b = Node('B')
    node_c = Node('C')
    node_d = Node('D', dest_nodes=[node_a, node_b])
    node_e = Node('E', dest_nodes=[node_b, node_c])
    
    # Print destination nodes as a string
    print(f"Node D destination nodes: {node_d.dest_nodes_to_string()}")
    print(f"Node E destination nodes: {node_e.dest_nodes_to_string()}")
    
    # Check equality of nodes
    print(f"Node A equals Node B: {node_a == node_b}")
    print(f"Node A equals Node A: {node_a == node_a}")
