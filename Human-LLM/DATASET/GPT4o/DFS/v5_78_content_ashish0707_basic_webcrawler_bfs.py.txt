from searcher import Searcher

class BFS(Searcher):
    """
    A class for Breadth-First Search (BFS) algorithm, inheriting from Searcher.
    """
    name = '2.A_bfs.txt'

    def merge_lists(self, list1, list2):
        """
        Merge two lists into a single list.

        Args:
            list1 (list): The first list to merge.
            list2 (list): The second list to merge.

        Returns:
            list: A list containing all elements from list1 followed by elements from list2.
        """
        return list1 + list2
