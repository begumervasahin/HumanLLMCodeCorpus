from PIL import Image

def create_image():
    """
    Create an image with a repeating color pattern and display it.
    """
    # Define a color pattern
    color_pattern = [
        (0, 0, 0),       # Black
        (0, 0, 0),       # Black
        (128, 128, 128), # Gray
        (255, 255, 255), # White
        (255, 255, 255)  # White
    ]
    
    # Repeat the color pattern to fill the image
    pattern_length = len(color_pattern)
    data = color_pattern * (100 * 100 // pattern_length)
    
    # Create a new image with RGB mode and size 100x100
    image_size = (100, 100)
    image = Image.new("RGB", image_size, "white")
    
    # Apply the color data to the image
    image.putdata(data)
    
    # Display the image
    image.show()

if __name__ == "__main__":
    create_image()
