import argparse
import matplotlib.pyplot as plt
import pandas as pd

def parse_arguments():
    """
    Parse command-line arguments.
    Returns:
        argparse.Namespace: Parsed command-line arguments.
    """
    parser = argparse.ArgumentParser(description='Plot scores from a given file.')
    parser.add_argument('scores', type=str, help='Specify the path of scores.txt')
    parser.add_argument('--title', type=str, default=None, help='Title of the plot')
    return parser.parse_args()

def read_scores(file_path):
    """
    Read scores from the specified file.
    Args:
        file_path (str): Path to the scores file.
    Returns:
        pd.DataFrame: DataFrame containing the scores data.
    """
    return pd.read_csv(file_path, delimiter='\t')

def plot_scores(scores, title=None):
    """
    Plot mean and median scores.
    Args:
        scores (pd.DataFrame): DataFrame containing the scores data.
        title (str, optional): Title of the plot. Defaults to None.
    """
    for col in ['mean', 'median']:
        plt.plot(scores['steps'], scores[col], label=col)
    
    if title:
        plt.title(title)
    
    plt.xlabel('Steps')
    plt.ylabel('Score')
    plt.legend(loc='best')

def save_plot(file_path):
    """
    Save the plot to a file.
    Args:
        file_path (str): Path to save the plot image.
    """
    fig_fname = f"{file_path}.png"
    plt.savefig(fig_fname)
    print(f'Saved a figure as {fig_fname}')

def main():
    """
    Main function to parse arguments, read scores, plot data, and save the plot.
    """
    args = parse_arguments()
    scores = read_scores(args.scores)
    plot_scores(scores, args.title)
    save_plot(args.scores)

if __name__ == '__main__':
    main()
