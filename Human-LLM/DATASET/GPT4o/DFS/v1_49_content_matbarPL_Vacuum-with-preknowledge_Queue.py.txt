class Queue:
    """A class representing a queue using a list with enqueue operation at the front."""

    def __init__(self):
        """Initialize an empty queue."""
        self.items = []

    def is_empty(self):
        """Check if the queue is empty."""
        return self.items == []

    def enqueue(self, item):
        """Add an item to the front of the queue."""
        self.items.insert(0, item)

    def dequeue(self):
        """Remove and return the item from the end of the queue."""
        if not self.is_empty():
            return self.items.pop()
        else:
            raise IndexError("dequeue from empty queue")

    def first(self):
        """Return the first item in the queue without removing it."""
        if not self.is_empty():
            return self.items[0]
        else:
            raise IndexError("first from empty queue")

    def size(self):
        """Return the number of items in the queue."""
        return len(self.items)

    def enqueue_items(self, items):
        """Enqueue multiple items into the queue."""
        for el in items:
            self.enqueue(el)

class BQueue:
    """A class representing a queue using a list with enqueue operation at the end."""

    def __init__(self):
        """Initialize an empty queue."""
        self.items = []

    def is_empty(self):
        """Check if the queue is empty."""
        return self.items == []

    def enqueue(self, item):
        """Add an item to the end of the queue."""
        self.items.append(item)

    def dequeue(self):
        """Remove and return the item from the end of the queue."""
        if not self.is_empty():
            return self.items.pop()
        else:
            raise IndexError("dequeue from empty queue")

    def size(self):
        """Return the number of items in the queue."""
        return len(self.items)

def main():
    """Main function to demonstrate the usage of Queue and BQueue classes."""
    # Example usage of Queue
    q = Queue()
    q.enqueue(1)
    q.enqueue(2)
    q.enqueue(3)
    print("Queue items:", q.items)
    print("Dequeue from Queue:", q.dequeue())
    print("First item in Queue:", q.first())
    print("Queue size:", q.size())

    # Example usage of BQueue
    bq = BQueue()
    bq.enqueue(1)
    bq.enqueue(2)
    bq.enqueue(3)
    print("BQueue items:", bq.items)
    print("Dequeue from BQueue:", bq.dequeue())
    print("BQueue size:", bq.size())

if __name__ == "__main__":
    main()
