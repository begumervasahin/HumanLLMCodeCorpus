def remove_element(nums, val):
    """
    Removes all instances of val from the list nums.
    :param nums: List[int]
    :param val: int
    :return: List[int]
    """
    while val in nums:
        nums.remove(val)
        print(f"Current list after removing {val}: {nums}")  # Debugging statement to show the list after each removal
    return nums

# Example usage:
nums = [3, 2, 2, 3]
result = remove_element(nums, 2)
print(f"Final list after removing 2: {result}")  # Output should be [3, 3]

# Additional test cases
nums = [1, 2, 3, 4, 2, 2, 5]
result = remove_element(nums, 2)
print(f"Final list after removing 2: {result}")  # Output should be [1, 3, 4, 5]

nums = [4, 4, 4, 4]
result = remove_element(nums, 4)
print(f"Final list after removing 4: {result}")  # Output should be []

nums = [1, 2, 3, 4, 5]
result = remove_element(nums, 6)
print(f"Final list after removing 6: {result}")  # Output should be [1, 2, 3, 4, 5] (no change)
