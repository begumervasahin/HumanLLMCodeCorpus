import os
import sys

def get_script_path():
    """
    Get the absolute path of the current script.
    
    Returns:
        str: The absolute path of the current script.
    """
    return os.path.abspath(__file__)

def get_script_directory(script_path):
    """
    Get the directory of the current script.
    
    Args:
        script_path (str): The absolute path of the current script.
        
    Returns:
        str: The directory of the current script.
    """
    return os.path.dirname(script_path)

def add_to_sys_path(directory):
    """
    Add the specified directory to the system path.
    
    Args:
        directory (str): The directory to add to the system path.
    """
    sys.path.append(directory)

def run_application(directory):
    """
    Initialize and run the application.
    
    Args:
        directory (str): The directory of the current script.
    """
    from gui.app import Application
    app = Application(directory)
    app.mainloop()

def main():
    """
    Main function to initialize and run the application.
    """
    script_path = get_script_path()
    script_directory = get_script_directory(script_path)
    
    add_to_sys_path(script_directory)
    
    print("[ START PROGRAM ]")
    print(f"Start main file: {script_path}")
    print(f"System append path: {script_directory}")
    
    run_application(script_directory)

if __name__ == "__main__":
    main()
