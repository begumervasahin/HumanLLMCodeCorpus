def generate_binary_strings(n, k):
    """
    Generate all binary strings of length n with at most k ones.
    :param n: int - The length of the binary strings.
    :param k: int - The maximum number of ones in the binary strings.
    """
    def backtrack(index, total_ones, solution):
        """
        Helper function to perform the backtracking.
        :param index: int - The current index in the binary string.
        :param total_ones: int - The current count of ones in the binary string.
        :param solution: List[str] - The current binary string being constructed.
        """
        if index == n:
            print(''.join(solution))
            return

        # Add '0' and proceed to the next position
        solution.append('0')
        backtrack(index + 1, total_ones, solution)
        solution.pop()
        
        # Add '1' and proceed to the next position if total_ones is less than k
        if total_ones < k:
            solution.append('1')
            backtrack(index + 1, total_ones + 1, solution)
            solution.pop()

    backtrack(0, 0, [])

# Example usage:
generate_binary_strings(4, 1)
