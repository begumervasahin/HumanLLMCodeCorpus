class Meta:
    """Class to hold metadata information."""

    def __init__(self, user_id, fileName, fileSize=None, iv=None):
        """Initialize the Meta class with user ID, file name, file size, and IV."""
        self.user_id = user_id
        self.fileName = fileName
        self.fileSize = fileSize
        self.iv = iv

    def __str__(self):
        """Return a formatted string representation of the metadata."""
        return (f"User ID: {self.user_id}\n"
                f"File Name: {self.fileName}\n"
                f"File Size: {self.fileSize}\n"
                f"IV: {self.iv}")

class Code:
    """Class to hold a key."""

    def __init__(self, key):
        """Initialize the Code class with a key."""
        self.key = key

    def __str__(self):
        """Return a string representation of the key."""
        return f"Key: {self.key}"

class Cipher:
    """Class to hold a cipher."""

    def __init__(self, cipher):
        """Initialize the Cipher class with a cipher."""
        self.cipher = cipher

    def __str__(self):
        """Return a string representation of the cipher."""
        return f"Cipher: {self.cipher}"

# Example usage
if __name__ == "__main__":
    meta = Meta(user_id=123, fileName="example.txt", fileSize=2048, iv="abcdef123456")
    code = Code(key="mysecretkey")
    cipher = Cipher(cipher="encryptedtext")

    print(meta)
    print(code)
    print(cipher)
