from fs.memoryfs import MemoryFS

def create_directories(memory_fs):
    """
    Create directories in the provided filesystem.

    Args:
        memory_fs (MemoryFS): The memory filesystem instance.
    """
    memory_fs.makedir('test')
    memory_fs.makedir('foo')

def list_directory_contents(memory_fs):
    """
    List the contents of the directories in the provided filesystem.

    Args:
        memory_fs (MemoryFS): The memory filesystem instance.
    """
    for path, info in memory_fs.walk.info(max_depth=1):
        if info.is_dir:
            print(f"[dir] {path}")
        else:
            print(f"[file] {path}")

def main():
    """
    Main function to create directories and list their contents.
    """
    memory_fs = MemoryFS()
    create_directories(memory_fs)
    list_directory_contents(memory_fs)

if __name__ == "__main__":
    main()
