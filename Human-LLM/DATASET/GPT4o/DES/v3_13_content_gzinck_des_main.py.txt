from cli.menus.main_menu import main_menu
import global_settings
from tempfile import TemporaryDirectory

"""
This main file runs the DES application on the command line.
"""

def run_des_application():
    """
    Run the DES application, initializing settings and handling screen transitions.
    """
    # Initialize global settings
    global_settings.initialize()

    # Initialize the stack of screens to display and the list of automata
    screens = [main_menu]
    automata = []

    # Use a temporary directory for temporary file storage
    with TemporaryDirectory() as temp_dir:
        # Continue running while there are screens to display
        while screens:
            # Get the current screen function to execute
            current_screen = screens.pop()
            # Execute the screen function, passing the screen stack, automata stack, and temp directory
            current_screen(screens, automata, temp_dir)

if __name__ == "__main__":
    run_des_application()
