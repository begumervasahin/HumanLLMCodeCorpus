import crypt

def generate_hash(input_filename, output_filename, salt):
    """
    Generate hashes for each line in the input file using the provided salt
    and write the results to the output file.
    """
    with open(input_filename, 'r') as infile, open(output_filename, 'w') as outfile:
        for line in infile:
            line = line.strip()
            if line:  # Ensure the line is not empty
                hashed_line = crypt.crypt(line, salt)
                result = f"{hashed_line} {line}"
                print(result)
                outfile.write(result + "\n")

def filter_lines_of_length(input_filename, output_filename, length=8):
    """
    Process the input file and write lines of the specified length to the output file.
    """
    with open(input_filename, 'r') as infile, open(output_filename, 'w') as outfile:
        for line in infile:
            line = line.strip()
            if len(line) == length:
                print(line)
                outfile.write(line + "\n")

def des_dictionary(input_filename, output_filename):
    """
    Wrapper function to filter lines of a specific length.
    """
    filter_lines_of_length(input_filename, output_filename)

def main():
    """
    Main function to filter lines of exactly 8 characters from the input file
    and generate hashed output using two different salts.
    """
    input_file = 'rockyou.txt'
    intermediate_file = 'pec1.txt'
    hashed_file1 = 'hashed1.txt'
    hashed_file2 = 'hashed2.txt'
    salt1 = 'tl'
    salt2 = 'as'

    # Filter lines of exactly 8 characters
    des_dictionary(input_file, intermediate_file)

    # Generate hashes using two different salts
    generate_hash(intermediate_file, hashed_file1, salt1)
    generate_hash(intermediate_file, hashed_file2, salt2)

if __name__ == "__main__":
    main()
