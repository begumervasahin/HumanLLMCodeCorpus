"""
Pattern name - Singleton
Pattern type - Creational Design Pattern
"""

class SingletonDecorator:
    """
    A decorator class to implement the Singleton pattern.
    Ensures that only one instance of the decorated class is created.
    """
    def __init__(self, klass):
        self.klass = klass
        self.instance = None

    def __call__(self, *args, **kwargs):
        if self.instance is None:
            self.instance = self.klass(*args, **kwargs)
        return self.instance

@SingletonDecorator
class Logger:
    """
    A Logger class that follows the Singleton pattern.
    Only one instance of this class can be created.
    """
    def __init__(self):
        self.start = None

    def write(self, message):
        """
        Prints a message, optionally prefixed by the 'start' attribute.
        """
        if self.start:
            print(f"{self.start} {message}")
        else:
            print(message)

def main():
    """
    The main function to demonstrate the Singleton pattern using the Logger class.
    """
    # Creating the first logger instance
    logger1 = Logger()
    logger1.start = "> "
    print("Logger 1:", logger1)
    logger1.write("Logger1 object is created.")

    # Creating the second logger instance (will return the same instance as logger1)
    logger2 = Logger()
    logger2.start = "$ >"
    print("Logger 2:", logger2)
    logger1.write("Logger2 object is created.")

if __name__ == "__main__":
    main()
