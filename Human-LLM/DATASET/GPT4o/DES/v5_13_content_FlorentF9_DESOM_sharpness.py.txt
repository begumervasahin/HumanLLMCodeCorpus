import numpy as np

def prototype_sharpness_ratio(X, som):
    """
    Calculate the prototype sharpness ratio for image datasets.

    Parameters
    ----------
    X : array, shape = [n_samples, input_dim]
        Input samples.
    som : array, shape = [n_prototypes, input_dim]
        SOM prototype vectors.

    Returns
    -------
    float
        The ratio of SOM sharpness to data sharpness.
    """
    
    # Determine the size of the images (assuming square images)
    image_size = int(np.sqrt(X.shape[-1]))
    
    # Reshape the input samples and SOM prototypes to 2D images
    X_reshaped = X.reshape(-1, image_size, image_size, 1)
    som_reshaped = som.reshape(-1, image_size, image_size, 1)
    
    # Calculate the sharpness for the data samples and SOM prototypes
    data_sharpness = calculate_average_sharpness(X_reshaped)
    som_sharpness = calculate_average_sharpness(som_reshaped)
    
    # Return the ratio of SOM sharpness to data sharpness
    return som_sharpness / data_sharpness

def calculate_average_sharpness(images):
    """
    Calculate the average sharpness for a set of images.

    Parameters
    ----------
    images : array, shape = [n_images, size, size, 1]
        Input images.

    Returns
    -------
    float
        The average sharpness of the images.
    """
    total_sharpness = 0.0
    
    for img in images:
        gx, gy = np.gradient(img[:, :, 0])  # Compute gradients
        gradient_norm = np.sqrt(gx**2 + gy**2)  # Gradient norm
        total_sharpness += np.mean(gradient_norm)  # Mean gradient norm
    
    # Average sharpness over all images
    return total_sharpness / images.shape[0]
