from openerp import models, fields

class ResVille(models.Model):
    """
    Model representing a ville (city).
    """
    _name = 'res.ville'
    _description = 'Ville'

    name = fields.Char(string='Ville', required=True)

class ResPartner(models.Model):
    """
    Extends the res.partner model to include a reference to res.ville.
    """
    _inherit = 'res.partner'

    ville_id = fields.Many2one('res.ville', string='Ville Client')
