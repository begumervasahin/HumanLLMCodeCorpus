class TOperation:
    """
    Class representing a single operation performed in the program.

    Attributes:
        o_type (str): Operation type.
        shape (object): Shape on which the operation was performed.
        shape_num (int): Shape list index.
    """

    def __init__(self, o_type=None, shape=None, shape_num=None):
        """
        Initialize the TOperation object.

        Args:
            o_type (str): Operation type.
            shape (object): Shape on which the operation was performed.
            shape_num (int): Shape list index.
        """
        self.o_type = o_type
        self.shape = shape
        self.shape_num = shape_num

    def set_operation(self, o_type=None, shape=None, shape_num=None):
        """
        Set the parameters of an operation.

        Args:
            o_type (str): Operation type.
            shape (object): Shape on which the operation was performed.
            shape_num (int): Shape list index.
        """
        self.o_type = o_type
        self.shape = shape
        self.shape_num = shape_num

def main():
    """
    Main function to demonstrate the usage of the TOperation class.
    """
    # Define some mock shape classes for demonstration
    class TShape:
        def __str__(self):
            return self.__class__.__name__

    class TRect(TShape):
        pass

    class TCylin(TShape):
        pass

    class TCylinSector(TShape):
        pass

    class TPolygon(TShape):
        pass

    # Create a shape instance
    shape_instance = TRect()

    # Create an instance of TOperation with sample data
    operation = TOperation(o_type="create", shape=shape_instance, shape_num=1)

    # Print the operation details
    print(f"Operation Type: {operation.o_type}")
    print(f"Shape: {operation.shape}")
    print(f"Shape Number: {operation.shape_num}")

if __name__ == "__main__":
    main()
