def expected_value_for_line(m, b, x_value):
    """
    Calculate the expected y-value for a given x-value on a line defined by slope (m) and y-intercept (b).

    Parameters
    ----------
    m : float
        The slope of the line.
    b : float
        The y-intercept of the line.
    x_value : float
        The x-value for which to calculate the y-value.

    Returns
    -------
    float
        The expected y-value.
    """
    return m * x_value + b

def sorted_points(x_values, y_values):
    """
    Sort points by their x-values.

    Parameters
    ----------
    x_values : list of float
        The list of x-values.
    y_values : list of float
        The list of y-values.

    Returns
    -------
    list of tuple
        The sorted list of (x, y) points.
    """
    values = list(zip(x_values, y_values))
    sorted_values = sorted(values, key=lambda value: value[0])
    return sorted_values

def y_intercept(x_values, y_values, m=None):
    """
    Calculate the y-intercept of a line given points and an optional slope.

    Parameters
    ----------
    x_values : list of float
        The list of x-values.
    y_values : list of float
        The list of y-values.
    m : float, optional
        The slope of the line. If not provided, it will be calculated.

    Returns
    -------
    float
        The y-intercept of the line.
    """
    sorted_values = sorted_points(x_values, y_values)
    highest = sorted_values[-1]
    if m is None:
        m = slope(x_values, y_values)
    b = highest[1] - m * highest[0]
    return b

def slope(x_values, y_values):
    """
    Calculate the slope of a line given points.

    Parameters
    ----------
    x_values : list of float
        The list of x-values.
    y_values : list of float
        The list of y-values.

    Returns
    -------
    float
        The slope of the line.
    """
    sorted_values = sorted_points(x_values, y_values)
    x1, y1 = sorted_values[0]
    x2, y2 = sorted_values[-1]
    m = (y2 - y1) / (x2 - x1)
    return m

def build_regression_line(x_values, y_values):
    """
    Build the regression line for the given points.

    Parameters
    ----------
    x_values : list of float
        The list of x-values.
    y_values : list of float
        The list of y-values.

    Returns
    -------
    dict
        A dictionary containing the slope (m) and y-intercept (b) of the regression line.
    """
    m = slope(x_values, y_values)
    b = y_intercept(x_values, y_values, m)
    return {'m': m, 'b': b}

# Example usage
if __name__ == "__main__":
    x_vals = [1, 2, 3, 4, 5]
    y_vals = [2, 4, 5, 4, 5]

    regression_line = build_regression_line(x_vals, y_vals)
    print(f"Slope (m): {regression_line['m']}")
    print(f"Y-intercept (b): {regression_line['b']}")

    x_value = 6
    expected_y = expected_value_for_line(regression_line['m'], regression_line['b'], x_value)
    print(f"Expected y-value for x={x_value}: {expected_y}")
