import slang

# Define constants for metadata and spreadsheets
METADATA_FILE = "poc.slang"
SPREADSHEETS = ["office-supplies-order.ods", "office-supplies-order.ods"]

def render_value(cell):
    """
    Render the value of a cell based on its type.

    Parameters
    ----------
    cell : slang.Cell
        The cell to render.

    Returns
    -------
    str or any
        The rendered value of the cell.
    """
    if isinstance(cell.type, slang.slang_String):
        return f"\"{cell.value()}\""
    return cell.value()

def render_json(result, row):
    """
    Render a row as a JSON object.

    Parameters
    ----------
    result : any
        The result of the extraction process.
    row : list of slang.Cell
        The row to render.

    Returns
    -------
    bool
        Always returns True.
    """
    print("  {")
    for cell in row:
        print(f"    \"{cell.name}\": {render_value(cell)},")
    print("  },")
    return True

def process_spreadsheets(metadata, spreadsheets):
    """
    Process the given spreadsheets against the provided metadata.

    Parameters
    ----------
    metadata : slang.Metadata
        The metadata to validate against.
    spreadsheets : list of str
        The list of spreadsheets to process.
    """
    for sheet in spreadsheets:
        print(f"Validating spreadsheet {sheet} against metadata...")
        with open(sheet) as spreadsheet:
            instance = metadata.validate(spreadsheet)
            print("Extracting typed data from spreadsheet...")
            print("[")
            instance.extract(render_json)
            print("]")

def main():
    """
    Main function to read metadata and process the spreadsheets.
    """
    print(f"Reading metadata from {METADATA_FILE}...")
    with open(METADATA_FILE) as metadata_file:
        metadata = slang.slang(metadata_file)
        metadata.parse()
        process_spreadsheets(metadata, SPREADSHEETS)

if __name__ == "__main__":
    main()
