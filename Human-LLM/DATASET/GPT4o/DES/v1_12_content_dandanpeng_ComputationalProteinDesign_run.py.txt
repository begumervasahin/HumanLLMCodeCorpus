# Import necessary classes and functions from the modules
from function import Protein
from seqpred import Matches, genetic_algorithm

# List of protein IDs
ids = ['1aba', '1bxv', '1by2', '1hyp', '1opc',
       '1tmy', '2acy', '2mcm', '3e0e', '3k63']

# Initialize score
score = 0

# Loop through each protein ID, create Protein and Matches objects, and run the genetic algorithm
for i in ids:
    protein = Protein(i)
    match = Matches(protein.match)
    genetic_algorithm(protein, match, 100, 50, 2, 0.03, 2)
