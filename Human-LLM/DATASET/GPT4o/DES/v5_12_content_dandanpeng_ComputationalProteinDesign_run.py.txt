# Import necessary classes and functions from the modules
from function import Protein
from seqpred import Matches, genetic_algorithm

# List of protein IDs to process
PROTEIN_IDS = [
    '1aba', '1bxv', '1by2', '1hyp', '1opc',
    '1tmy', '2acy', '2mcm',
]

def process_proteins(protein_ids):
    """
    Process a list of protein IDs using the genetic algorithm.
    
    Parameters:
        protein_ids (list): List of protein IDs to process.
    """
    for protein_id in protein_ids:
        # Create a Protein object for the given ID
        protein = Protein(protein_id)
        
        # Create a Matches object using the protein's match attribute
        match = Matches(protein.match)
        
        # Run the genetic algorithm with the specified parameters
        genetic_algorithm(
            protein,          # Protein object
            match,            # Matches object
            POPULATION_SIZE,  # Population size
            NUM_GENERATIONS,  # Number of generations
            TOURNAMENT_SIZE,  # Tournament size
            MUTATION_RATE,    # Mutation rate
            ELITISM_COUNT     # Elitism count
        )

    score = 0
