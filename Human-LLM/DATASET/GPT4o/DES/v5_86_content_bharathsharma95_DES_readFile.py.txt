def read_file_as_bytes(file_path):
    """
    Reads a file in binary mode and returns its content as a list of bytes.

    :param file_path: The path to the file to be read.
    :return: A list of bytes representing the file content.
    """
    with open(file_path, 'rb') as file:
        return list(file.read())

def main():
    """
    Main function to read a file and print its content as bytes.
    """
    file_path = 'path/to/your/file'  # Replace with your actual file path
    try:
        file_bytes = read_file_as_bytes(file_path)
        print(file_bytes)
    except FileNotFoundError:
        print(f"Error: The file at '{file_path}' was not found.")
    except Exception as e:
        print(f"An error occurred: {e}")

if __name__ == "__main__":
    main()
