class TMaterial:
    """
    Class represents a material with its properties.

    :param epsilon_r: relative permittivity.
    :type epsilon_r: float
    :param sigma: conductivity.
    :type sigma: float
    :param mu_r: relative permeability.
    :type mu_r: float
    :param sigma_mag: magnetic loss.
    :type sigma_mag: float
    :param name: material identifier.
    :type name: str
    """
    
    def __init__(self, epsilon_r=0.0, sigma=0.0, mu_r=0.0, sigma_mag=0.0, name=""):
        """
        Initialize the properties of the material.
        
        Properties of a material are explained in the gprMax user manual.
        """
        self.epsilon_r = epsilon_r
        self.sigma = sigma
        self.mu_r = mu_r
        self.sigma_mag = sigma_mag
        self.name = name

    def velocity(self):
        """
        Calculate the electromagnetic wave velocity in the material.

        :return: The velocity of the electromagnetic wave in the material.
        :rtype: float
        """
        c = 299792458  # Speed of light in vacuum (in meters per second)
        return c / (self.epsilon_r**0.5)


# Example usage
if __name__ == "__main__":
    # Create an instance of TMaterial
    material = TMaterial(epsilon_r=4.0, sigma=0.01, mu_r=1.0, sigma_mag=0.0, name="SampleMaterial")

    # Calculate and print the velocity of the electromagnetic wave in the material
    velocity = material.velocity()
    print(f"The velocity of the electromagnetic wave in {material.name} is {velocity:.2f} m/s")
