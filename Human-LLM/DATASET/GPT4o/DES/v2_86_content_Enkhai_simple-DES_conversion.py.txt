import misc
import permutations

def primary_key_to_subkeys(key):
    """
    Converts a primary key into a list of subkeys using specified permutations and shifts.

    :param key: The primary key as a hexadecimal string.
    :return: A list of subkeys as binary strings.
    """
    subkeys = []

    # Convert the hexadecimal key to a binary string
    binary_prim_key = misc.hex_to_bits(key)
    
    # Remove every eighth bit (parity bits)
    binary_prim_key = ''.join([binary_prim_key[i:i + 7] for i in range(0, len(binary_prim_key), 8)])
    
    # Split the binary key into left and right parts
    half_length = len(binary_prim_key) // 2
    left_part = binary_prim_key[:half_length]
    right_part = binary_prim_key[half_length:]
    
    # Generate 16 subkeys
    for i in range(16):
        # Determine the number of shifts
        shift_amount = 1 if i in {0, 1, 8, 15} else 2
        
        # Perform the left shifts
        left_part = misc.left_shift(left_part, shift_amount)
        right_part = misc.left_shift(right_part, shift_amount)
        
        # Combine the left and right parts and apply the compression permutation
        combined_key = left_part + right_part
        subkeys.append(permutations.permutation(combined_key, "compression"))
    
    return subkeys

def main():
    """
    Main function to demonstrate the generation of subkeys from a primary key.
    """
    primary_key = "133457799BBCDFF1"  # Example primary key (hexadecimal string)
    subkeys = primary_key_to_subkeys(primary_key)
    
    print("Subkeys:")
    for idx, subkey in enumerate(subkeys, 1):
        print(f"Subkey {idx}: {subkey}")

if __name__ == "__main__":
    main()
