import re

def concat_files(readme_path):
    """
    Concatenates the contents of files included in the README file.

    Parameters
    ----------
    readme_path : str
        The path to the README file.
    """
    result_content = []
    include_mode = False

    with open(readme_path, "r", encoding="utf-8") as readme_file:
        lines = readme_file.readlines()

    for line in lines:
        if not include_mode:
            result_content.append(line)
        
        include_match = re.search(r"<!--\s*include\s+(.+?)\s*-->", line)
        if include_match and not include_mode:
            include_mode = True
            include_filename = include_match.group(1).strip()

            with open(include_filename, "r", encoding="utf-8") as include_file:
                result_content.extend(include_file.readlines())

        end_match = re.search(r"<!--\s*end\s*-->", line)
        if end_match:
            include_mode = False
            result_content.append("\n" + line)

    with open(readme_path, "w", encoding="utf-8") as readme_file:
        readme_file.writelines(result_content)

if __name__ == "__main__":
    concat_files("README.md")
