def bfs(graph, start, goal):
    """Perform Breadth-First Search (BFS) to find the shortest path in a graph."""
    queue = [[start]]
    visited = set()

    while queue:
        # Dequeue the first path from the queue
        path = queue.pop(0)
        # Get the last node from the path
        state = path[-1]

        # Check if the last node is the goal
        if state == goal:
            return path

        # If the state has not been visited, explore adjacent nodes
        elif state not in visited:
            for branch in graph.get(state, []):
                new_path = list(path)  # Copy current path
                new_path.append(branch)  # Append the adjacent node to the path
                queue.append(new_path)  # Enqueue the new path

            # Mark the state as visited
            visited.add(state)

    # Return a message if the goal is not found
    return "Path not found"
