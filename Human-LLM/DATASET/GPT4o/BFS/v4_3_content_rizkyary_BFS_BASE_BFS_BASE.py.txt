# Define the graph using an adjacency list
graph = {
    'A': {'B', 'C'},
    'B': {'A', 'H', 'J'},
    'C': {'A', 'D', 'G'},
    'D': {'C', 'E'},
    'E': {'D', 'F'},
    'F': {'E', 'G', 'K', 'L'},
    'G': {'C', 'F'},
    'H': {'B', 'I'},
    'I': {'H', 'J', 'K'},
    'J': {'B', 'I'},
    'K': {'F', 'I', 'L'},
    'L': {'F', 'K'}
}

def bfs(graph, start, goal):
    """
    Perform Breadth-First Search (BFS) on the graph to find the shortest path from start to goal.

    Args:
        graph (dict): The graph represented as an adjacency list.
        start (str): The starting node.
        goal (str): The goal node.

    Returns:
        list: The shortest path from start to goal if found, otherwise None.
    """
    queue = [[start]]  # Initialize the queue with the starting node
    visited = set()  # Set to keep track of visited nodes

    while queue:
        # Get the first path from the queue
        path = queue.pop(0)
        # Get the last node from the path
        state = path[-1]

        # Check if the last node is the goal
        if state == goal:
            return path  # Return the path if goal is reached

        # If the state is not visited
        elif state not in visited:
            # Explore the neighbors of the state
            for neighbor in graph.get(state, []):
                new_path = list(path)  # Create a new path
                new_path.append(neighbor)  # Add the neighbor to the new path
                queue.append(new_path)  # Add the new path to the queue

            # Mark the state as visited
            visited.add(state)

    # If the queue is empty and no path is found
    print("Path not found")
    return None

# Example usage
start_node = 'A'
goal_node = 'L'
path = bfs(graph, start_node, goal_node)
if path:
    print("Path found:", path)
else:
    print("No path found")
