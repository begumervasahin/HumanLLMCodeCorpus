class Queue:
    """
    A simple Queue implementation using a list.
    """
    def __init__(self):
        self.items = []
        self.state = set()
        
    def enqueue(self, item):
        """
        Add an item to the queue and track its state.

        Args:
            item: The item to be added to the queue.
        """
        self.items.append(item)  # Add item to the queue
        self.state.add(tuple(item.state_list))  # Track the state of the item
        
    def dequeue(self):
        """
        Remove the first item from the queue (FIFO) and return it.

        Returns:
            The first item in the queue, or None if the queue is empty.
        """
        if not self.is_empty():
            item = self.items.pop(0)  # Remove and return the first item
            self.state.remove(tuple(item.state_list))  # Remove the state of the dequeued item
            return item
        return None
        
    def is_empty(self):
        """
        Check if the queue is empty.

        Returns:
            bool: True if the queue is empty, False otherwise.
        """
        return len(self.items) == 0

class PuzzleState:
    """
    A simple class to represent a puzzle state.
    """
    def __init__(self, state_list):
        self.state_list = state_list

    def __repr__(self):
        return str(self.state_list)

# Example usage of the Queue class
if __name__ == "__main__":
    # Create a Queue instance
    queue = Queue()

    # Enqueue some puzzle states
    queue.enqueue(PuzzleState([1, 2, 3, 4, 5, 6, 7, 8, 0]))
    queue.enqueue(PuzzleState([1, 2, 3, 4, 5, 6, 0, 7, 8]))
    queue.enqueue(PuzzleState([1, 2, 3, 4, 5, 0, 6, 7, 8]))

    # Dequeue and print each item
    while not queue.is_empty():
        state = queue.dequeue()
        print("Dequeued:", state)
