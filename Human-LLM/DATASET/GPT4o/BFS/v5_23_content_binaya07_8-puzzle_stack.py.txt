class Stack:
    """A simple implementation of a stack with state tracking."""
    
    def __init__(self):
        self.items = []
        self.state = set()
    
    def push(self, item):
        """Add an item to the stack."""
        self.items.append(item)
        self.state.add(tuple(item.state))
    
    def pop(self):
        """Remove and return the top item from the stack (LIFO)."""
        if not self.is_empty():
            top_item = self.items.pop()
            self.state.remove(tuple(top_item.state))
            return top_item
        return None
        
    def is_empty(self):
        """Check if the stack is empty."""
        return len(self.items) == 0
        
    def clear(self):
        """Clear all items from the stack."""
        self.items.clear()
        self.state.clear()

class PuzzleState:
    """Example PuzzleState class to be used with Stack."""
    
    def __init__(self, state_list):
        self.state = state_list
        
    def __repr__(self):
        return f"PuzzleState({self.state})"
    
if __name__ == "__main__":
    # Create a Stack instance
    stack = Stack()
    
    # Push some PuzzleState instances onto the stack
    stack.push(PuzzleState([1, 2, 3, 4, 5, 6, 7, 8, 0]))
    stack.push(PuzzleState([1, 2, 3, 4, 5, 6, 7, 0, 8]))
    
    # Check if the stack is empty
    print(f"Stack is empty: {stack.is_empty()}")
    
    # Pop an item and print it
    popped_item = stack.pop()
    print(f"Popped: {popped_item}")
    
    # Check if the stack is empty again
    print(f"Stack is empty: {stack.is_empty()}")
    
    # Pop another item and print it
    popped_item = stack.pop()
    print(f"Popped: {popped_item}")
    
    # Check if the stack is empty one last time
    print(f"Stack is empty: {stack.is_empty()}")
    
    # Clear the stack and check if it's empty
    stack.clear()
    print(f"Stack cleared. Stack is empty: {stack.is_empty()}")
