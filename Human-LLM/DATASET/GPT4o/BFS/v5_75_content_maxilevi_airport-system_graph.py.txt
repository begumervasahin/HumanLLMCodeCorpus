import random

class Graph:
    def __init__(self, is_undirected=False):
        """
        Initialize the graph.

        Args:
            is_undirected (bool): Flag to indicate if the graph is undirected.
        """
        self._adjacency_list = {}
        self._is_weighted = False
        self._is_undirected = is_undirected

    def vertices(self):
        """
        Return a list of vertices in the graph.

        Returns:
            list: List of vertices.
        """
        return list(self._adjacency_list.keys())

    def edges(self):
        """
        Return a list of edges in the graph.

        Returns:
            list: List of edges with their weights.
        """
        visited = set()
        edge_list = []
        for vertex in self.vertices():
            new_edges = [(vertex, adjacent, self.weight((vertex, adjacent))) 
                         for adjacent in self.adjacent(vertex)]
            for edge in new_edges:
                u, v, weight = edge
                if (u, v) not in visited and (v, u) not in visited:
                    edge_list.append(edge)
                    visited.add((u, v))
        return edge_list

    def adjacent(self, vertex):
        """
        Return a list of adjacent vertices for a given vertex.

        Args:
            vertex (str): The vertex to get adjacent vertices for.

        Returns:
            list: List of adjacent vertices.
        """
        return list(self._adjacency_list.get(vertex, {}).keys())

    def is_adjacent(self, origin, vertex):
        """
        Check if two vertices are adjacent.

        Args:
            origin (str): The origin vertex.
            vertex (str): The vertex to check adjacency with.

        Returns:
            bool: True if vertices are adjacent, False otherwise.
        """
        return vertex in self._adjacency_list.get(origin, {})

    def add_vertex(self, vertex):
        """
        Add a vertex to the graph.

        Args:
            vertex (str): The vertex to add.
        """
        if vertex not in self._adjacency_list:
            self._adjacency_list[vertex] = {}

    def add_edge(self, edge, weight=1):
        """
        Add an edge to the graph with an optional weight.

        Args:
            edge (tuple): The edge to add, as a tuple of two vertices.
            weight (int, optional): The weight of the edge. Defaults to 1.
        """
        x, y = edge
        if x not in self._adjacency_list:
            self.add_vertex(x)
        if y not in self._adjacency_list:
            self.add_vertex(y)

        self._adjacency_list[x][y] = weight
        if self._is_undirected:
            self._adjacency_list[y][x] = weight

        self._is_weighted = self._is_weighted or weight != 1

    def weight(self, edge):
        """
        Return the weight of an edge.

        Args:
            edge (tuple): The edge to get the weight of, as a tuple of two vertices.

        Returns:
            int: The weight of the edge.
        """
        x, y = edge
        return self._adjacency_list[x][y]

    def random_vertex(self):
        """
        Return a random vertex from the graph.

        Returns:
            str: A random vertex.
        """
        vertex_list = self.vertices()
        return vertex_list[random.randint(0, len(vertex_list) - 1)]

    def is_weighted(self):
        """
        Check if the graph is weighted.

        Returns:
            bool: True if the graph is weighted, False otherwise.
        """
        return self._is_weighted

    def is_undirected(self):
        """
        Check if the graph is undirected.

        Returns:
            bool: True if the graph is undirected, False otherwise.
        """
        return self._is_undirected
