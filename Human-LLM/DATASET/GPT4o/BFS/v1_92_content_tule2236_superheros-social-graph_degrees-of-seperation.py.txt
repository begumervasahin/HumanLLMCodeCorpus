from pyspark import SparkConf, SparkContext

def convertToBFS(line):
    fields = line.split()
    connections = []
    heroID = int(fields[0])
    for connection in fields[1:]:
        connections.append(int(connection))
    distance = 9999  # infinity
    color = "WHITE"
    if heroID == startCharacterID:
        color = "GRAY"
        distance = 0
    return (heroID, (connections, distance, color))

def bfsMap(node):
    heroID = node[0]
    connections = node[1][0]
    distance = node[1][1]
    color = node[1][2]
    results = []

    if color == "GRAY":
        for connection in connections:
            newDistance = distance + 1
            if connection == targetCharacterID:
                hitCounter.add(1)
            results.append((connection, ([], newDistance, "GRAY")))
        color = "BLACK"
    results.append((heroID, (connections, distance, color)))
    return results

def bfsReduce(data1, data2):
    edges1, edges2 = data1[0], data2[0]
    dist1, dist2 = data1[1], data2[1]
    color1, color2 = data1[2], data2[2]

    edges = edges1 if len(edges1) > 0 else edges2
    distance = min(dist1, dist2)
    color = color1

    if (color1 == 'WHITE' and color2 in ('GRAY', 'BLACK')):
        color = color2
    elif (color1 == 'GRAY' and color2 == 'BLACK'):
        color = color2
    elif (color2 == 'WHITE' and color1 in ('GRAY', 'BLACK')):
        color = color1
    elif (color2 == 'GRAY' and color1 == 'BLACK'):
        color = color1

    return (edges, distance, color)

# Initialize Spark
conf = SparkConf().setMaster("local").setAppName("DegreesOfSeparation")
sc = SparkContext(conf=conf)

# The characters we wish to find the degree of separation between:
startCharacterID = 5306  # SpiderMan
targetCharacterID = 14  # ADAM 3,031 (who?)
hitCounter = sc.accumulator(0)

# Load the graph
graph = sc.textFile("Data/Marvel-Graph.txt")
rdd = graph.map(convertToBFS)

for iteration in range(10):
    print(f"Running BFS iteration# {iteration + 1}")
    mapper = rdd.flatMap(bfsMap)
    print(f"Processing {mapper.count()} values")
    if hitCounter.value > 0:
        print(f"Hit the target character! From {hitCounter.value} different directions.")
        break
    rdd = mapper.reduceByKey(bfsReduce)
