"""
Breadth First Search (BFS): Algorithm to find the degree of separation from one node to other nodes in the graph.

Steps:
1. Initiate the starting node as GRAY, all other nodes as WHITE.
2. From the GRAY node, mark all connected nodes to the GRAY node as BLACK.
3. Iteratively process the RDD:
   - Look for GRAY nodes to expand.
   - Mark nodes we're done with as BLACK.
   - Update the distance as we go.
4. A BFS iteration as a Map and Reduce job.
5. The reducer combines all nodes for the same hero ID, preserves the shortest distance, the darkest color found, and the list of connections from the original node.
6. An Accumulator.
"""

from pyspark import SparkConf, SparkContext

# Configuration for the Spark Context
conf = SparkConf().setMaster("local").setAppName("DegreesOfSeparation")
sc = SparkContext(conf=conf)

# The characters we wish to find the degree of separation between:
startCharacterID = 5306  # SpiderMan
targetCharacterID = 14   # ADAM 3,031 (who?)
hitCounter = sc.accumulator(0)

# Step 1: Represent each line as a node with connections, color, distance (ID, listOfAssociatedHero, distance, color)
def convert_to_bfs(line):
    fields = line.split()
    hero_id = int(fields[0])
    connections = [int(connection) for connection in fields[1:]]
    distance = 0 if hero_id == startCharacterID else 9999  # 0 if startCharacterID, else infinity
    color = "GRAY" if hero_id == startCharacterID else "WHITE"
    return (hero_id, (connections, distance, color))

# The mapper function for BFS
def bfs_map(node):
    hero_id, (connections, distance, color) = node
    results = []

    if color == "GRAY":
        for connection in connections:
            new_distance = distance + 1
            if connection == targetCharacterID:
                hitCounter.add(1)
            results.append((connection, ([], new_distance, "GRAY")))
        color = "BLACK"  # Mark the processed node as BLACK

    # Emit the input node so we don't lose it
    results.append((hero_id, (connections, distance, color)))
    return results

# The reducer function for BFS
def bfs_reduce(data1, data2):
    connections1, distance1, color1 = data1
    connections2, distance2, color2 = data2

    edges = connections1 + connections2  # Combine all connections of 2 nodes
    distance = min(distance1, distance2)  # Preserve the shortest distance
    color = max(color1, color2, key=lambda c: ["WHITE", "GRAY", "BLACK"].index(c))  # Preserve the darkest color

    return (edges, distance, color)

# Load the graph data
graph = sc.textFile("Data/Marvel-Graph.txt")
rdd = graph.map(convert_to_bfs)

# Perform BFS iterations
for iteration in range(10):
    print(f"Running BFS iteration #{iteration + 1}")

    mapper = rdd.flatMap(bfs_map)
    print(f"Processing {mapper.count()} values")

    if hitCounter.value > 0:
        print(f"Hit the target character! From {hitCounter.value} different directions.")
        break

    rdd = mapper.reduceByKey(bfs_reduce)

sc.stop()
