class Queue:
    """
    A simple Queue implementation using a list.
    """
    def __init__(self):
        self.items = []

    def enqueue(self, item):
        """
        Add an item to the queue.
        """
        self.items.insert(0, item)

    def dequeue(self):
        """
        Remove and return the first item from the queue.
        """
        return self.items.pop()

    def is_empty(self):
        """
        Check if the queue is empty.
        """
        return self.items == []

    def size(self):
        """
        Get the size of the queue.
        """
        return len(self.items)

class Stack:
    """
    A simple Stack implementation using a list.
    """
    def __init__(self):
        self.items = []

    def push(self, item):
        """
        Add an item to the stack.
        """
        self.items.append(item)

    def pop(self):
        """
        Remove and return the top item from the stack.
        """
        return self.items.pop()

    def is_empty(self):
        """
        Check if the stack is empty.
        """
        return self.items == []

    def size(self):
        """
        Get the size of the stack.
        """
        return len(self.items)

def bfs(graph, start, goal):
    """
    Perform Breadth-First Search (BFS) on a graph.

    Args:
        graph (dict): The graph to search, represented as an adjacency list.
        start: The starting node.
        goal: The goal node.

    Returns:
        list: The path from start to goal, if found. None otherwise.
    """
    frontier = Queue()
    frontier.enqueue(start)
    came_from = {start: None}

    while not frontier.is_empty():
        current = frontier.dequeue()

        if current == goal:
            return reconstruct_path(came_from, start, goal)

        for next_node in graph[current]:
            if next_node not in came_from:
                frontier.enqueue(next_node)
                came_from[next_node] = current

    return None

def dfs(graph, start, goal):
    """
    Perform Depth-First Search (DFS) on a graph.

    Args:
        graph (dict): The graph to search, represented as an adjacency list.
        start: The starting node.
        goal: The goal node.

    Returns:
        list: The path from start to goal, if found. None otherwise.
    """
    frontier = Stack()
    frontier.push(start)
    came_from = {start: None}

    while not frontier.is_empty():
        current = frontier.pop()

        if current == goal:
            return reconstruct_path(came_from, start, goal)

        for next_node in graph[current]:
            if next_node not in came_from:
                frontier.push(next_node)
                came_from[next_node] = current

    return None

def reconstruct_path(came_from, start, goal):
    """
    Reconstruct the path from start to goal.

    Args:
        came_from (dict): A dictionary mapping nodes to their predecessors.
        start: The starting node.
        goal: The goal node.

    Returns:
        list: The reconstructed path.
    """
    current = goal
    path = []
    while current != start:
        path.append(current)
        current = came_from[current]
    path.append(start)
    path.reverse()
    return path

def main():
    """
    Main function to demonstrate BFS and DFS on a sample graph.
    """
    graph = {
        'A': ['B', 'C', 'E'],
        'B': ['A', 'D', 'F'],
        'C': ['A', 'G'],
        'D': ['B'],
        'E': ['A', 'F'],
        'F': ['B', 'E', 'H'],
        'G': ['C', 'H'],
        'H': ['F', 'G']
    }
    start = 'A'
    goal = 'H'

    print("BFS Path from {} to {}:".format(start, goal))
    bfs_path = bfs(graph, start, goal)
    print(bfs_path)

    print("\nDFS Path from {} to {}:".format(start, goal))
    dfs_path = dfs(graph, start, goal)
    print(dfs_path)

if __name__ == '__main__':
    main()
