import random

class Graph:
    def __init__(self, is_undirected=False):
        """
        Initialize a graph object.
        
        :param is_undirected: Boolean indicating if the graph is undirected.
        """
        self._adjacency_list = {}
        self._is_weighted = False
        self._is_undirected = is_undirected

    def vertices(self):
        """
        Get a list of all vertices in the graph.

        :return: List of vertices.
        """
        return list(self._adjacency_list.keys())

    def edges(self):
        """
        Get a list of all edges in the graph.

        :return: List of tuples representing edges (start, end, weight).
        """
        visited = set()
        edge_list = []
        for v in self.vertices():
            new_edges = [(v, u, self.weight((v, u))) for u in self.adjacent(v)]
            for edge in new_edges:
                u, v, weight = edge
                if (u, v) not in visited and (v, u) not in visited:
                    edge_list.append(edge)
                    visited.add((u, v))
        return edge_list

    def adjacent(self, vertex):
        """
        Get a list of all adjacent vertices to the given vertex.

        :param vertex: The vertex to query.
        :return: List of adjacent vertices.
        """
        return list(self._adjacency_list[vertex].keys())

    def is_adjacent(self, origin, vertex):
        """
        Check if two vertices are adjacent.

        :param origin: The starting vertex.
        :param vertex: The vertex to check adjacency.
        :return: Boolean indicating if the vertices are adjacent.
        """
        return vertex in self._adjacency_list[origin]

    def add_vertex(self, vertex):
        """
        Add a vertex to the graph.

        :param vertex: The vertex to add.
        """
        if vertex not in self._adjacency_list:
            self._adjacency_list[vertex] = {}

    def add_edge(self, edge, weight=1):
        """
        Add an edge to the graph with an optional weight.

        :param edge: Tuple representing the edge (start, end).
        :param weight: The weight of the edge. Default is 1.
        """
        x, y = edge
        if x not in self._adjacency_list:
            self.add_vertex(x)
        if y not in self._adjacency_list:
            self.add_vertex(y)
        self._adjacency_list[x][y] = weight
        if self.is_undirected():
            self._adjacency_list[y][x] = weight
        self._is_weighted = self._is_weighted or weight != 1

    def weight(self, edge):
        """
        Get the weight of an edge.

        :param edge: Tuple representing the edge (start, end).
        :return: The weight of the edge.
        """
        x, y = edge
        return self._adjacency_list[x][y]

    def random_vertex(self):
        """
        Get a random vertex from the graph.

        :return: A randomly selected vertex.
        """
        vertex_list = self.vertices()
        return vertex_list[random.randint(0, len(vertex_list) - 1)]

    def is_weighted(self):
        """
        Check if the graph is weighted.

        :return: Boolean indicating if the graph is weighted.
        """
        return self._is_weighted

    def is_undirected(self):
        """
        Check if the graph is undirected.

        :return: Boolean indicating if the graph is undirected.
        """
        return self._is_undirected


if __name__ == "__main__":
    # Example usage of the Graph class
    graph = Graph(is_undirected=True)
    
    # Add vertices
    graph.add_vertex("A")
    graph.add_vertex("B")
    graph.add_vertex("C")
    
    # Add edges
    graph.add_edge(("A", "B"), 5)
    graph.add_edge(("B", "C"), 3)
    graph.add_edge(("C", "A"), 2)
    
    # Print vertices
    print("Vertices:", graph.vertices())
    
    # Print edges
    print("Edges:", graph.edges())
    
    # Check adjacency
    print("Is A adjacent to B?", graph.is_adjacent("A", "B"))
    print("Is A adjacent to C?", graph.is_adjacent("A", "C"))
    
    # Print random vertex
    print("Random vertex:", graph.random_vertex())
    
    # Check if graph is weighted
    print("Is the graph weighted?", graph.is_weighted())
    
    # Check if graph is undirected
    print("Is the graph undirected?", graph.is_undirected())
