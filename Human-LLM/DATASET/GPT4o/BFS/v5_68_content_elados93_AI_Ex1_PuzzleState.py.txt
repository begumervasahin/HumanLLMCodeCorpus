class State:
    """
    Represents the state of the puzzle.

    Attributes:
        _board (list): The current board configuration.
        _move (str): The move that led to this state.
        came_from (State): The previous state.
    """
    def __init__(self, board=None, move=None, came_from=None, state=None):
        if state is None:
            # Normal constructor
            self._board = board
            self._move = move
            self.came_from = came_from
        else:
            # Copy constructor
            self._board = state.board
            self._move = state.move
            self.came_from = state.came_from

    def __copy__(self):
        """
        Create a shallow copy of the current state.

        :return: A new State instance with a copied board.
        """
        import copy
        copy_board = copy.copy(self._board)
        return State(board=copy_board, move=self._move, came_from=self.came_from)

    def __hash__(self):
        """
        Generate a hash value for the state based on the board configuration.

        :return: Hash value of the board.
        """
        return hash(self._board)

    @property
    def board(self):
        """
        Get the board configuration.

        :return: The current board configuration.
        """
        return self._board

    @property
    def move(self):
        """
        Get the move that led to this state.

        :return: The move that led to this state.
        """
        return self._move

    def __eq__(self, other):
        """
        Compare two states for equality based on their board configurations.

        :param other: Another State instance.
        :return: True if the boards are equal, False otherwise.
        """
        return self._board == other.board

class HeuristicState(State):
    """
    Represents the state of the puzzle with heuristic values for A* algorithm.

    Attributes:
        h_cost (float): The heuristic cost.
        g_cost (float): The cost from the start state.
    """
    def __init__(self, board=None, move=None, came_from=None, state=None):
        super().__init__(board, move, came_from, state)
        self.h_cost = 0
        self.g_cost = 0

    def __lt__(self, other):
        """
        Compare two heuristic states to determine which has a lower f_cost.

        :param other: Another HeuristicState instance.
        :return: True if this instance has a lower f_cost than the other, False otherwise.
        """
        return self.f_cost < other.f_cost

    @property
    def f_cost(self):
        """
        Calculate the total cost (f_cost) for the A* algorithm.

        :return: The sum of h_cost and g_cost.
        """
        return self.h_cost + self.g_cost
