from collections import defaultdict, deque

class EdmondsKarp:
    def __init__(self, graph):
        self.graph = graph
        self.num_vertices = len(graph)

    def bfs(self, source, sink, parent):
        visited = [False] * self.num_vertices
        queue = deque([source])
        visited[source] = True

        while queue:
            u = queue.popleft()

            for v in range(self.num_vertices):
                if visited[v] is False and self.graph[u][v] > 0:
                    queue.append(v)
                    visited[v] = True
                    parent[v] = u
                    if v == sink:
                        return True

        return False

    def edmonds_karp(self, source, sink):
        parent = [-1] * self.num_vertices
        max_flow = 0

        while self.bfs(source, sink, parent):
            path_flow = float('Inf')
            s = sink

            while s != source:
                path_flow = min(path_flow, self.graph[parent[s]][s])
                s = parent[s]

            max_flow += path_flow
            v = sink

            while v != source:
                u = parent[v]
                self.graph[u][v] -= path_flow
                self.graph[v][u] += path_flow
                v = parent[v]

        return max_flow

    def min_cut(self, source):
        visited = [False] * self.num_vertices
        queue = deque([source])
        visited[source] = True
        min_cut_vertices = []

        while queue:
            u = queue.popleft()
            min_cut_vertices.append(u)

            for v in range(self.num_vertices):
                if visited[v] is False and self.graph[u][v] > 0:
                    queue.append(v)
                    visited[v] = True

        return min_cut_vertices

def read_graph(filename):
    graph = defaultdict(lambda: defaultdict(int))
    with open(filename, 'r') as file:
        first_line = file.readline().strip().split()
        num_vertices = int(first_line[0])
        num_edges = int(first_line[1])

        for _ in range(num_edges):
            line = file.readline().strip().split()
            u = int(line[0])
            v = int(line[1])
            weight = int(line[2])
            graph[u][v] = weight
            graph[v][u] = weight  # Assuming undirected graph

    return graph, num_vertices

def main():
    input_file = 'mincut_input/XXXX.in'  # Replace with your input file path
    output_file = 'output_file.txt'  # Replace with your output file path

    graph, num_vertices = read_graph(input_file)
    source = 0  # Assuming the source vertex is 0, adjust if necessary
    sink = num_vertices - 1  # Assuming the sink vertex is the last vertex, adjust if necessary

    ek = EdmondsKarp(graph)
    max_flow = ek.edmonds_karp(source, sink)
    min_cut_vertices = ek.min_cut(source)

    with open(output_file, 'w') as file:
        file.write(f"{len(min_cut_vertices)}\n")
        file.write(" ".join(map(str, min_cut_vertices)) + "\n")
        file.write(f"{max_flow}\n")

    print("Successfully wrote output to", output_file)

if __name__ == "__main__":
    main()
