from Queue import Queue
from copy import deepcopy

class Node:
    """
    Node class for the 8-puzzle problem. Represents a state in the puzzle.
    """
    
    def __init__(self, puzzle, parent=None, move=""):
        self.state = puzzle
        self.parent = parent
        self.move = move
        self.depth = 0 if parent is None else parent.depth + 1
        self.moves = move if parent is None else parent.moves + move

    def goal_state(self):
        """
        Checks if the Node's state is a goal state.
        """
        return self.state.check_puzzle()

    def succ(self):
        """
        Generates the node's children states.
        """
        successors = Queue()
        for move in self.state.moves:
            new_puzzle = deepcopy(self.state)
            new_puzzle.do_move(move)
            if new_puzzle.zero != self.state.zero:
                successors.put(Node(new_puzzle, self, move))
        return successors

    def cost_heuristic(self, heuristic):
        """
        Chooses between the two available heuristics.
        """
        return self.n_wrong_tiles() if heuristic == 0 else self.manhattan_distance()

    def n_wrong_tiles(self):
        """
        First heuristic - number of wrong tiles.
        """
        result = 0
        count = 1
        for i in range(self.state.size):
            for j in range(self.state.size):
                if self.state.puzzle[i][j] != (count % (self.state.size * self.state.size)):
                    result += 1
                count += 1
        return result

    def manhattan_distance(self):
        """
        Second heuristic - distance of wrong tiles to their right position.
        """
        result = 0
        for i in range(self.state.size):
            for j in range(self.state.size):
                index = self.state.puzzle[i][j] - 1
                if index == -1:  # The blank tile (0) should be in the bottom right corner
                    distance = (2 - i) + (2 - j)
                else:
                    distance = abs(i - (index // self.state.size)) + abs(j - (index % self.state.size))
                result += distance
        return result

    def __str__(self):
        """
        When printing the node, return the moves from the starting state to this specific one.
        """
        return str(self.moves)
