import collections

# Create an empty adjacency list
adjList = {}

# File name to change as needed
fileName = "wrestler2.txt"

# Read in the file and build the adjacency list
with open(fileName, "r") as f:
    lines = f.readlines()
    num = int(lines[0])  # Number of players
    numRivals = int(lines[num + 1])  # Number of rivalries

    for count, line in enumerate(lines):
        if count > num + 2 and count <= num + 2 + numRivals:
            wrestler1, wrestler2 = line.strip().split()
            adjList.setdefault(wrestler1, []).append(wrestler2)
            adjList.setdefault(wrestler2, []).append(wrestler1)

# Define teams
babyFaces = []
heels = []

def BFS(graph, source):
    """
    Perform a Breadth-First Search (BFS) on the graph starting from the source.

    Args:
        graph (dict): The adjacency list of the graph.
        source (str): The starting vertex.

    Returns:
        list: List of vertices that have been checked.
    """
    doneChecking = []
    queue = collections.deque([source])
    depth = {source: 0}
    visited = {source}

    while queue:
        vertex = queue.popleft()
        doneChecking.append(vertex)

        # Assign team based on depth
        if depth[vertex] % 2 == 0:
            babyFaces.append(vertex)
        else:
            heels.append(vertex)

        for neighbor in graph[vertex]:
            if neighbor not in visited:
                queue.append(neighbor)
                visited.add(neighbor)
                depth[neighbor] = depth[vertex] + 1

    return doneChecking

# Start BFS from the first key in the adjacency list
first_key = list(adjList.keys())[0]
checked = BFS(adjList, first_key)

# Check for any unvisited vertices and run BFS on them
for key in adjList:
    if key not in checked:
        checked += BFS(adjList, key)

# Determine the number of players in each group
numBabyFaces = len(babyFaces)
numHeels = len(heels)

# Print the results
if numBabyFaces == numHeels:
    print("Yes, possible")
    print("Baby Faces:", babyFaces)
    print("Heels:", heels)
else:
    print("Not possible")
