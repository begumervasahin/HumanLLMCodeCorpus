import collections

def bfs(grid, start):
    """
    Perform a breadth-first search to find the shortest path from the start to the goal in a grid.
    """
    queue = collections.deque([[start]])  # Initialize the queue with the start position
    seen = set([start])  # Set to keep track of visited nodes
    
    while queue:
        path = queue.popleft()  # Get the current path from the queue
        x, y = path[-1]  # Get the current position
        
        # Check if the goal has been reached
        if grid[y][x] == goal:
            return path
        
        # Explore the neighbors (right, left, down, up)
        for x2, y2 in ((x + 1, y), (x - 1, y), (x, y + 1), (x, y - 1)):
            if 0 <= x2 < columns and 0 <= y2 < rows and grid[y2][x2] != wall and (x2, y2) not in seen:
                queue.append(path + [(x2, y2)])  # Append the new path with the neighbor
                seen.add((x2, y2))  # Mark the neighbor as seen
    
    return None  # Return None if no path is found

# Constants representing different types of grid cells
wall = "#"
clear = "."
goal = "*"

# Dimensions of the grid
columns, rows = 10, 5

# The grid representation
grid = [
    "..........",
    "...#*..##.",
    "..##...#..",
    ".....###..",
    ".........."
]

# Starting position
start = (0, 0)

# Find the path using BFS
path = bfs(grid, start)

# Print the result
if path:
    print("Path to goal:", path)
else:
    print("No path found")
