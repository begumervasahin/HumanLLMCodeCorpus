class Node:
    def __init__(self, parent=None, world_state=None, priority=None):
        """
        Initialize a new node in the search tree.

        Args:
            parent (Node, optional): The parent node of the current node. Defaults to None.
            world_state (object, optional): The state of the world represented by this node. Defaults to None.
            priority (int or float, optional): The priority value for the node. Used in priority queues. Defaults to None.
        """
        self.parent = parent
        self.world_state = world_state
        self.priority = priority

    def __lt__(self, other):
        """
        Less-than comparison method for node priority.
        This is used by priority queues to compare nodes.

        Args:
            other (Node): The other node to compare against.

        Returns:
            bool: True if this node's priority is less than the other node's priority, False otherwise.
        """
        if self.priority is None or other.priority is None:
            raise ValueError("Priority values must be set for comparison.")
        return self.priority < other.priority
