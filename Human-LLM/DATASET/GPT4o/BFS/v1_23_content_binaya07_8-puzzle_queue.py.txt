class Queue:
    """
    A simple Queue implementation using a list.
    """
    def __init__(self):
        self.list = []
        self.state = set()
        
    def enqueue(self, item):
        """
        Add item to the queue and track its state.
        """
        self.list.append(item)  # Add item to the queue
        self.state.add(tuple(item.list))  # Track the state of the item
        
    def dequeue(self):
        """
        Remove the first item from the queue (FIFO) and return it.
        """
        if not self.isEmpty():
            item = self.list[0]  # Get the first item
            for i in range(len(self.list) - 1):  # Shift the remaining items forward
                self.list[i] = self.list[i + 1]
            del self.list[-1]  # Delete the last element to prevent duplication
            self.state.remove(tuple(item.list))  # Remove the state of the dequeued item
            return item
        
    def isEmpty(self):
        """
        Check if the queue is empty.
        """
        return len(self.list) == 0
