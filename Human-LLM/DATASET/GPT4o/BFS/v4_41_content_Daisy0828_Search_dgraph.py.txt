from searchproblem import SearchProblem

class DGraph(SearchProblem):
    def __init__(self, matrix, goal_indices, start_state=0):
        """
        Initialize the directed graph.

        :param matrix: The matrix representation of the directed graph.
        :param goal_indices: A set of indices representing the goal states.
        :param start_state: The index of the start state. Defaults to 0.
        """
        self.matrix = matrix
        self.goal_indices = goal_indices
        self.start_state = start_state

    def get_start_state(self):
        """
        Return the start state index.

        :return: The start state index.
        """
        return self.start_state

    def is_goal_state(self, state):
        """
        Check if the given state is a goal state.

        :param state: The state to check.
        :return: True if the state is a goal state, False otherwise.
        """
        return state in self.goal_indices

    def get_successors(self, state):
        """
        Get the successors of the given state.

        :param state: The state for which to find successors.
        :return: A dictionary of successors where keys are state indices and values are costs.
        """
        row = self.matrix[state]
        successors = {}
        for index, cost in enumerate(row):
            if cost is not None:
                successors[index] = cost
        return successors

# Example usage
if __name__ == "__main__":
    matrix = [
        [None, 1, None, None],
        [None, None, 2, None],
        [None, None, None, 3],
        [None, None, None, None]
    ]
    goal_indices = {3}
    graph = DGraph(matrix, goal_indices)

    print("Start state:", graph.get_start_state())
    print("Is goal state (2):", graph.is_goal_state(2))
    print("Is goal state (3):", graph.is_goal_state(3))
    print("Successors of state 1:", graph.get_successors(1))
