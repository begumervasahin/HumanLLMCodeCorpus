"""
Find captured black items
"""

from collections import deque
from enum import Enum

class Node(Enum):
    BLACK = 5
    WHITE = 2
    EMPTY = 0

class Grid:
    def __init__(self, size, node_pairs):
        """
        Initialize the grid with the given size and node pairs.

        Args:
            size (int): The size of the grid (n x n).
            node_pairs (list): List of tuples with coordinates and Node type.
        """
        self.size = size
        self.board = [[Node.EMPTY for _ in range(size)] for _ in range(size)]
        for location, value in node_pairs:
            self.set_value(location, value)

    def get_value(self, location):
        """
        Get the value of a cell in the grid.

        Args:
            location (tuple): The coordinates of the cell.

        Returns:
            Node: The value of the cell.
        """
        return self.board[location[0]][location[1]]

    def set_value(self, location, value):
        """
        Set the value of a cell in the grid.

        Args:
            location (tuple): The coordinates of the cell.
            value (Node): The value to set in the cell.
        """
        self.board[location[0]][location[1]] = value

    def is_captured(self, location):
        """
        Check if a black node is captured.

        Args:
            location (tuple): The coordinates of the cell.

        Returns:
            bool: True if the black node is captured, False otherwise.
        """
        if self.get_value(location) != Node.BLACK or not self.is_within_bounds(location):
            return False

        q = deque([location])
        visited = [[False for _ in range(self.size)] for _ in range(self.size)]
        visited[location[0]][location[1]] = True

        while q:
            curr = q.popleft()
            if not self.is_surrounded(q, visited, curr):
                return False

        return True

    def is_within_bounds(self, location):
        """
        Check if a location is within the bounds of the grid.

        Args:
            location (tuple): The coordinates of the cell.

        Returns:
            bool: True if the location is within bounds, False otherwise.
        """
        return 0 <= location[0] < self.size and 0 <= location[1] < self.size

    def is_surrounded(self, q, visited, curr):
        """
        Check if a node is surrounded.

        Args:
            q (deque): The queue of nodes to visit.
            visited (list): The grid of visited nodes.
            curr (tuple): The current node being checked.

        Returns:
            bool: True if the node is surrounded, False otherwise.
        """
        neighbors = [
            (curr[0], curr[1] + 1), 
            (curr[0], curr[1] - 1),
            (curr[0] + 1, curr[1]), 
            (curr[0] - 1, curr[1])
        ]

        for loc in neighbors:
            if self.is_within_bounds(loc):
                if not visited[loc[0]][loc[1]]:
                    cell_value = self.get_value(loc)
                    if cell_value == Node.WHITE:
                        visited[loc[0]][loc[1]] = True
                    elif cell_value == Node.EMPTY:
                        return False
                    elif cell_value == Node.BLACK:
                        visited[loc[0]][loc[1]] = True
                        q.append(loc)
            else:
                return False

        return True

    ]

    new_grid = Grid(5, node_pairs)
