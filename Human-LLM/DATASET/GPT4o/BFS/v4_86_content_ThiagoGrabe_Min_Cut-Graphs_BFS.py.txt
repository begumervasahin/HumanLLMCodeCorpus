class BFS:
    def bfs(self, graph, source, terminal, parent):
        """Perform BFS to find if there is a path from source to terminal in the given graph.

        Args:
            graph (list of list of int): The adjacency matrix of the graph.
            source (int): The source vertex.
            terminal (int): The terminal vertex.
            parent (list of int): The parent list to store the path.

        Returns:
            bool: True if there is a path from source to terminal, False otherwise.
        """
        visited = [False] * len(graph)
        queue = []

        # Mark the source node as visited and enqueue it
        queue.append(source)
        visited[source] = True

        while queue:
            # Dequeue a vertex from the queue
            current_vertex = queue.pop(0)

            # Get all adjacent vertices of the dequeued vertex
            for idx, capacity in enumerate(graph[current_vertex]):
                # If an adjacent vertex is not visited and there is a capacity in the edge
                if capacity > 0 and not visited[idx]:
                    # Mark the vertex as visited and enqueue it
                    queue.append(idx)
                    visited[idx] = True
                    parent[idx] = current_vertex

        # If we reached the terminal vertex in BFS starting from source, then return True
        return visited[terminal]

# Example usage
if __name__ == "__main__":
    # Example graph represented as an adjacency matrix
    graph = [
        [0, 16, 13, 0, 0, 0],
        [0, 0, 10, 12, 0, 0],
        [0, 4, 0, 0, 14, 0],
        [0, 0, 9, 0, 0, 20],
        [0, 0, 0, 7, 0, 4],
        [0, 0, 0, 0, 0, 0]
    ]
    source = 0
    terminal = 5
    parent = [-1] * len(graph)

    bfs_instance = BFS()
    if bfs_instance.bfs(graph, source, terminal, parent):
        print("There is a path from source to terminal")
    else:
        print("No path from source to terminal")
