import sys
import time
import heapq
from collections import deque

class State:
    def __init__(self, state, parent, move, depth, cost, key):
        self.state = state
        self.parent = parent
        self.move = move
        self.depth = depth
        self.cost = cost
        self.key = key

        if self.state:
            self.map = ''.join(str(e) for e in self.state)

    def __eq__(self, other):
        return self.map == other.map

    def __lt__(self, other):
        return self.map < other.map

def bfs(start_state):
    frontier = deque([State(start_state, None, None, 0, 0, 0)])
    explored = set()
    nodes_expanded = 0
    max_fringe_size = 0

    while frontier:
        current_state = frontier.popleft()
        explored.add(current_state.map)
        nodes_expanded += 1

        if is_goal(current_state.state):
            return construct_path(current_state), nodes_expanded, len(frontier), max(len(frontier), max_fringe_size), current_state.depth

        neighbors = get_neighbors(current_state)

        for neighbor in neighbors:
            if neighbor.map not in explored and neighbor not in frontier:
                frontier.append(neighbor)
                max_fringe_size = max(max_fringe_size, len(frontier))

    return None, nodes_expanded, len(frontier), max_fringe_size, current_state.depth

def dfs(start_state):
    frontier = [State(start_state, None, None, 0, 0, 0)]
    explored = set()
    nodes_expanded = 0
    max_fringe_size = 0

    while frontier:
        current_state = frontier.pop()
        explored.add(current_state.map)
        nodes_expanded += 1

        if is_goal(current_state.state):
            return construct_path(current_state), nodes_expanded, len(frontier), max(len(frontier), max_fringe_size), current_state.depth

        neighbors = get_neighbors(current_state)

        for neighbor in reversed(neighbors):
            if neighbor.map not in explored and neighbor not in frontier:
                frontier.append(neighbor)
                max_fringe_size = max(max_fringe_size, len(frontier))

    return None, nodes_expanded, len(frontier), max_fringe_size, current_state.depth

def ast(start_state):
    frontier = []
    explored = set()
    start_node = State(start_state, None, None, 0, 0, 0)
    heapq.heappush(frontier, (0, start_node))
    nodes_expanded = 0
    max_fringe_size = 0

    while frontier:
        current_state = heapq.heappop(frontier)[1]
        explored.add(current_state.map)
        nodes_expanded += 1

        if is_goal(current_state.state):
            return construct_path(current_state), nodes_expanded, len(frontier), max(len(frontier), max_fringe_size), current_state.depth

        neighbors = get_neighbors(current_state)

        for neighbor in neighbors:
            if neighbor.map not in explored:
                heapq.heappush(frontier, (neighbor.cost + heuristic(neighbor.state), neighbor))
                max_fringe_size = max(max_fringe_size, len(frontier))

    return None, nodes_expanded, len(frontier), max_fringe_size, current_state.depth

def ida(start_state):
    def ida_search(path, g, bound):
        current_node = path[-1]
        f = g + heuristic(current_node.state)

        if f > bound:
            return f
        if is_goal(current_node.state):
            return path

        min_bound = float('inf')

        for neighbor in get_neighbors(current_node):
            if neighbor not in path:
                path.append(neighbor)
                result = ida_search(path, g + 1, bound)
                if isinstance(result, list):
                    return result
                if result < min_bound:
                    min_bound = result
                path.pop()

        return min_bound

    start_node = State(start_state, None, None, 0, 0, 0)
    bound = heuristic(start_state)
    path = [start_node]
    nodes_expanded = 0
    max_fringe_size = 0

    while True:
        result = ida_search(path, 0, bound)
        if isinstance(result, list):
            return construct_path(result[-1]), nodes_expanded, len(path), max_fringe_size, result[-1].depth
        if result == float('inf'):
            return None, nodes_expanded, len(path), max_fringe_size, 0
        bound = result

def is_goal(state):
    return state == [1, 2, 3, 4, 5, 6, 7, 8, 0]

def get_neighbors(current_state):
    neighbors = []
    moves = {
        'Up': -3,
        'Down': 3,
        'Left': -1,
        'Right': 1
    }
    index = current_state.state.index(0)

    for move, position in moves.items():
        new_index = index + position

        if 0 <= new_index < len(current_state.state):
            new_state = current_state.state[:]
            new_state[index], new_state[new_index] = new_state[new_index], new_state[index]
            neighbors.append(State(new_state, current_state, move, current_state.depth + 1, current_state.cost + 1, 0))

    return neighbors

def construct_path(state):
    path = []
    while state.parent:
        path.append(state.move)
        state = state.parent
    return path[::-1]

def heuristic(state):
    return sum(abs((val - 1) % 3 - i % 3) + abs((val - 1) // 3 - i // 3)
               for i, val in enumerate(state) if val != 0)

def write_output(result, nodes_expanded, fringe_size, max_fringe_size, search_depth, running_time):
    with open('output.txt', 'w') as f:
        if result:
            f.write(f"path_to_goal: {result}\n")
        f.write(f"cost_of_path: {len(result) if result else 0}\n")
        f.write(f"nodes_expanded: {nodes_expanded}\n")
        f.write(f"fringe_size: {fringe_size}\n")
        f.write(f"max_fringe_size: {max_fringe_size}\n")
        f.write(f"search_depth: {search_depth}\n")
        f.write(f"max_search_depth: {search_depth}\n")
        f.write(f"running_time: {running_time}\n")

if __name__ == "__main__":
    if len(sys.argv) != 3:
        print("Usage: python driver.py <algorithm> <initial_state>")
        sys.exit(1)

    algorithm = sys.argv[1]
    initial_state = list(map(int, sys.argv[2].split(',')))

    start_time = time.time()

    if algorithm == 'bfs':
        result, nodes_expanded, fringe_size, max_fringe_size, search_depth = bfs(initial_state)
    elif algorithm == 'dfs':
        result, nodes_expanded, fringe_size, max_fringe_size, search_depth = dfs(initial_state)
    elif algorithm == 'ast':
        result, nodes_expanded, fringe_size, max_fringe_size, search_depth = ast(initial_state)
    elif algorithm == 'ida':
        result, nodes_expanded, fringe_size, max_fringe_size, search_depth = ida(initial_state)
    else:
        print("Unknown algorithm")
        sys.exit(1)

    end_time = time.time()
    running_time = end_time - start_time

    write_output(result, nodes_expanded, fringe_size, max_fringe_size, search_depth, running_time)
