def binary_search(input_array, value):
    """
    Perform binary search on a sorted array to find the index of a given value.

    :param input_array: List of sorted elements.
    :param value: The value to search for.
    :return: Index of the value in the array if found, otherwise -1.
    """
    low = 0
    high = len(input_array) - 1

    while low <= high:
        mid = (low + high) // 2
        if input_array[mid] == value:
            return mid
        elif input_array[mid] < value:
            low = mid + 1
        else:
            high = mid - 1

    return -1

# Example usage
if __name__ == "__main__":
    test_list = [1, 3, 9, 11, 15, 19, 29]
    test_val1 = 25
    test_val2 = 15

    print(f"Index of {test_val1}: {binary_search(test_list, test_val1)}")
    print(f"Index of {test_val2}: {binary_search(test_list, test_val2)}")
