class Node:
    def __init__(self, val):
        """
        Initialize a new Node.

        Args:
            val: The value of the node.
        """
        self.left = None
        self.right = None
        self.parent = None
        self.value = val

    def add_left(self, child_node):
        """
        Add a left child to the node.

        Args:
            child_node (Node): The node to be added as the left child.
        """
        self.left = child_node
        child_node.parent = self

    def add_right(self, child_node):
        """
        Add a right child to the node.

        Args:
            child_node (Node): The node to be added as the right child.
        """
        self.right = child_node
        child_node.parent = self

    def __str__(self):
        """Return a string representation of the node's value."""
        return str(self.value)

    def print_left(self):
        """Print the left child of the node."""
        print("Left child:", self.left)

    def print_right(self):
        """Print the right child of the node."""
        print("Right child:", self.right)

    def print_value(self):
        """Print the value of the node."""
        print("Node value:", self.value)

    def print_parent(self):
        """Print the parent of the node."""
        print("Parent:", self.parent)

# Example usage
if __name__ == "__main__":
    root = Node(10)
    left_child = Node(5)
    right_child = Node(15)

    root.add_left(left_child)
    root.add_right(right_child)

    root.print_value()       # Should print: Node value: 10
    root.print_left()        # Should print: Left child: 5
    root.print_right()       # Should print: Right child: 15
    left_child.print_parent()  # Should print: Parent: 10
    right_child.print_parent() # Should print: Parent: 10
