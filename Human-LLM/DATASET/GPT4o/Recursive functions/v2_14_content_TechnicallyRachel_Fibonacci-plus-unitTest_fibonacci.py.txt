def fibonacci_sequence(limit):
    sequence = []
    a, b = 0, 1
    while b < limit:
        sequence.append(b)
        a, b = b, a + b
    return sequence

def main():
    limit = 10
    fib_sequence = fibonacci_sequence(limit)
    print(f"Fibonacci sequence up to {limit}: {fib_sequence}")

if __name__ == '__main__':
    main()
