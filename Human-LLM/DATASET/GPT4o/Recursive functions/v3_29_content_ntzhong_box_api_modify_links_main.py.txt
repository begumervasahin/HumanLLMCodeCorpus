import time
import box_module.interface_box as box
import setup
import utils

def update_folder_link_access(console_url, access_token, folder_id_list):
    """
    Update all folder link access to company level.

    Parameters:
    - console_url (str): The URL for the Box console.
    - access_token (str): The access token for authentication.
    - folder_id_list (list): List of folder IDs to update.

    Returns:
    - failed_updates (list): List of folder IDs that failed to update.
    """
    failed_updates = []
    
    for folder_id in folder_id_list:
        if not box.box_update_folder_shared_link_company(console_url, access_token, folder_id):
            time.sleep(1)
            if not box.box_update_folder_shared_link_company(console_url, access_token, folder_id):
                failed_updates.append(folder_id)
    
    print(f"Failed folder updates: {failed_updates}")
    return failed_updates

def lock_items_with_pattern(console_url, access_token, pattern, access_level):
    """
    Lock all items containing a specific pattern to a specified access level.

    Parameters:
    - console_url (str): The URL for the Box console.
    - access_token (str): The access token for authentication.
    - pattern (str): The pattern to match in folder names.
    - access_level (str): The access level to set for the matching items.
    """
    all_users = box.box_get_all_users(console_url, access_token)
    
    for user in all_users:
        folders = box.box_get_user_folders(console_url, access_token, user['id'])
        for folder in folders:
            if pattern.lower() in folder['name'].lower():
                box.box_update_all_shared_links_access_in_folder(console_url, access_token, folder['id'], access_level)
    
    print(f"Locked all items containing pattern '{pattern}' to access level '{access_level}'.")

def main():
    """
    Main function to run the Box update and lock operations.
    """
    # Read API credentials and refresh token
    client_id, client_secret = box.box_read_api_credentials(setup.box_cred_file)
    refresh_token = utils.read_txt_file(setup.box_refresh_token_file)

    # Refresh access token
    access_token, refresh_token = box.box_refresh_access_token(client_id, client_secret, refresh_token)

    # Save the new refresh token
    utils.write_to_txt_file(refresh_token, setup.box_refresh_token_file)

    # Define the pattern and access level for the lock operation
    example_pattern = 'finance'
    access_level = 'company'

    # Lock all items containing the pattern to the specified access level
    lock_items_with_pattern(setup.console_url, access_token, example_pattern, access_level)

if __name__ == '__main__':
    main()
