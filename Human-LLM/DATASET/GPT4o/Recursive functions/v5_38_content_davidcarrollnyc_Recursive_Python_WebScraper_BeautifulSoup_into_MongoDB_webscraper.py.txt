from bs4 import BeautifulSoup
from bs4.element import Comment
import urllib.request
from urllib.error import HTTPError

def tag_visible(element):
    """
    Check if an HTML element is visible.
    """
    if element.parent.name in ['style', 'script', 'head', 'title', 'meta', '[document]']:
        return False
    if isinstance(element, Comment):
        return False
    return True

def text_from_html(body):
    """
    Extract visible text from HTML content.
    """
    soup = BeautifulSoup(body, 'html.parser')
    texts = soup.findAll(text=True)
    visible_texts = filter(tag_visible, texts)
    return " ".join(t.strip() for t in visible_texts)

def fetch_html(url):
    """
    Fetch HTML content from a given URL.
    """
    try:
        response = urllib.request.urlopen(url)
        return response.read()
    except HTTPError as e:
        print(f"HTTP Error: {e.code} for URL: {url}")
        return None

def extract_links(html):
    """
    Extract all http links from HTML content.
    """
    soup = BeautifulSoup(html, "html.parser")
    return [link.attrs['href'] for link in soup.select('a[href^="http"]') if link.has_attr('href')]

def process_url(url):
    """
    Fetch HTML content from the given URL, extract and print the text, and find all links.
    """
    html = fetch_html(url)
    if html:
        print(text_from_html(html))
        links = extract_links(html)
        for link in links:
            print(link)
            process_url(link)

def main():
    """
    Main function to extract and print text and links from a website.
    """
    url = 'http://www.awebsite.com'
    process_url(url)

if __name__ == "__main__":
    main()
