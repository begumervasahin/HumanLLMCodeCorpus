import sys

def recursive_func(the_list, indent=False, level=0, fh=sys.stdout):
    """
    A recursive function to print nested lists.

    Parameters:
    the_list (list): The list to be printed.
    indent (bool): Whether to indent the nested lists.
    level (int): The current level of nesting.
    fh (file handle): The file handle to write the output to (default is sys.stdout).
    """
    for each_item in the_list:
        if isinstance(each_item, list):
            recursive_func(each_item, indent, level + 1, fh)
        else:
            if indent:
                print('\t' * level, end='', file=fh)
            print(each_item, file=fh)

if __name__ == '__main__':
    # Example usage
    example_list = [
        'a',
        ['b', 'c', ['d', 'e', 'f']],
        ['g', 'h'],
        'i'
    ]
    recursive_func(example_list, indent=True)
