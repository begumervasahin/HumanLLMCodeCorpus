import re
from queue import Queue
from concurrent.futures import ThreadPoolExecutor
import requests
from bs4 import BeautifulSoup
import logging

class Ingestion:
    def __init__(self, sitemap, regex, concurrency=20, proxy=None):
        """
        Initialize the Ingestion class.

        :param sitemap: Initial sitemap URL to start parsing from
        :param regex: Regular expression pattern to filter URLs
        :param concurrency: Number of threads for concurrent processing
        :param proxy: Proxy settings for requests
        """
        self.regex = re.compile(regex)
        self.links = set()
        self.queue = Queue()
        self.queue.put(sitemap)
        self.thread_pool = ThreadPoolExecutor(max_workers=concurrency)
        self.user_agent = ('Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 '
                           '(KHTML, like Gecko) Chrome/64.0.3282.186 Safari/537.36')
        self.proxy = {'https': proxy, 'http': proxy} if proxy else None

    def parser(self, response):
        """
        Method to be implemented by subclasses for processing the response content.

        :param response: Response object from the HTTP request
        """
        raise NotImplementedError

    def _parser_callback(self, future):
        """
        Callback function to handle the result of the thread pool future.

        :param future: Future object representing the asynchronous execution
        """
        response = future.result()
        if response:
            self.parser(response)

    def _recursive_sitemap_parse(self):
        """
        Recursively parse sitemaps to gather URLs matching the regex pattern.
        """
        while not self.queue.empty():
            sitemap = self.queue.get_nowait()
            self.__parse_sitemap_xml(sitemap)

    def __parse_sitemap_xml(self, sitemap):
        """
        Parse a sitemap XML to extract URLs.

        :param sitemap: URL of the sitemap to be parsed
        """
        try:
            response = requests.get(sitemap)
            response.raise_for_status()
            soup = BeautifulSoup(response.text, 'lxml-xml')
            raw_urls = soup.find_all('loc')

            for raw in raw_urls:
                url = raw.text
                if url.endswith('.xml'):
                    self.queue.put(url)
                elif re.match(self.regex, url):
                    self.links.add(url)
        except requests.RequestException as e:
            logging.warning(f"Exception retrieving sitemap: {sitemap}, Exception: {e}")

    def _get_response_object(self, url):
        """
        Fetch the response object for a given URL.

        :param url: URL to be fetched
        :return: Response object or None in case of an exception
        """
        try:
            response = requests.get(url, headers={'User-Agent': self.user_agent}, proxies=self.proxy, timeout=(30, 60))
            response.raise_for_status()
            return response
        except requests.RequestException as e:
            logging.warning(f"Request failed for {url}: {e}")
            return None

    def digest(self):
        """
        Start the ingestion process by recursively parsing the sitemaps and processing the URLs.
        """
        self._recursive_sitemap_parse()
        for link in self.links:
            future = self.thread_pool.submit(self._get_response_object, link)
            future.add_done_callback(self._parser_callback)
