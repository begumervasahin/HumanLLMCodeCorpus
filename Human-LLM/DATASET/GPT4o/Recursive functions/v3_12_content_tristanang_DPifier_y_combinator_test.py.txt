import nose
from y_combinator import Y

class DPifier:
    """
    A class that memoizes recursive functions to optimize their performance.

    Attributes:
        dp (dict): A dictionary to store computed values.
        lambda_mode (bool): A flag to indicate if the function is in lambda mode.
        f (function): The function to be memoized.
        Y (function): The Y combinator applied to the function.
    """
    def __init__(self, f, lambda_mode=False):
        self.dp = {}
        self.lambda_mode = lambda_mode
        self.f = f if lambda_mode else self._parse_function(f)
        self.Y = Y(self._memoize(self.f))

    def __call__(self, *args):
        return self.Y(*args)

    def _parse_function(self, f):
        """
        Parses the function to be used with the Y combinator.

        Args:
            f (function): The function to be parsed.

        Returns:
            function: The parsed function.
        """
        def g(func):
            return lambda *args: func(*args)
        return g(f)
    
    def _memoize(self, f):
        """
        Memoizes the function to store computed values.

        Args:
            f (function): The function to be memoized.

        Returns:
            function: The memoized function.
        """
        def memoized_func(func):
            def wrapper(*args):
                if args in self.dp:
                    return self.dp[args]
                result = f(func)(*args)
                self.dp[args] = result
                return result
            return wrapper
        return memoized_func

def fac(n):
    """Calculate the factorial of n."""
    return 1 if n == 0 else n * fac(n - 1)

def fac_lambda(fac):
    """Return a lambda function for calculating factorial."""
    def h(n):
        return 1 if n == 0 else n * fac(n - 1)
    return h

def match_lambda(match):
    """Return a lambda function for regex-like matching."""
    def h(text, pattern):
        if not pattern:
            return not text
        if len(pattern) > 1 and pattern[1] == '*':
            if text == "" or ((text[0] != pattern[0]) and pattern[0] != '.'):
                return match(text, pattern[2:])
            return match(text, pattern[2:]) or match(text[1:], pattern) or match(text[1:], pattern[2:])
        if not text or ((text[0] != pattern[0]) and pattern[0] != '.'):
            return False
        return match(text[1:], pattern[1:])
    return h

def match(text, pattern):
    """Simple regex-like pattern matching."""
    if not pattern:
        return not text
    if len(pattern) > 1 and pattern[1] == '*':
        if text == "" or ((text[0] != pattern[0]) and pattern[0] != '.'):
            return match(text, pattern[2:])
        return match(text, pattern[2:]) or match(text[1:], pattern) or match(text[1:], pattern[2:])
    if not text or ((text[0] != pattern[0]) and pattern[0] != '.'):
        return False
    return match(text[1:], pattern[1:])

def test_basic():
    """Test the basic functionality of DPifier with the factorial function."""
    f = DPifier(fac)
    assert f(7) == 5040

def test_match():
    """Test the DPifier with the match function."""
    g = DPifier(match)
    assert not g("aaaaaaaaaaaaab", "a*a*a*a*a*a*a*a*a*a*c")

def test_lambda_mode():
    """Test the DPifier in lambda mode."""
    f = DPifier(fac_lambda, True)
    assert f(7) == 5040
    g = DPifier(match_lambda, True)
    assert not g("aaaaaaaaaaaaab", "a*a*a*a*a*a*a*a*a*a*c")
