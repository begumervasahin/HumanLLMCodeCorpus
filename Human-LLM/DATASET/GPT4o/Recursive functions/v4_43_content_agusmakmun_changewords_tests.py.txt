import os
import unittest
from changewords.changewords import ChangeWords

class ChangeWordsTestCase(unittest.TestCase):
    """Unit tests for `changewords/changewords.py`."""

    def setUp(self):
        """
        Set up the test case environment.
        """
        change = ChangeWords()
        self.parser = change.create_parser()
        self.change_words = change.change_words

    def test_is_words_changed(self):
        """
        Test if words are successfully changed in the specified file.
        """
        args = [
            ('--path', 'changewords_test'),
            ('--file_type', '.py'),
            ('--from_string', 'helloworld'),
            ('--to_string', 'mantabjiwa')
        ]
        parsed_args = {k: self.parser.parse_args([k, v]) for k, v in args}
        
        result = self.change_words(
            parsed_args['--path'].path,
            parsed_args['--file_type'].file_type,
            parsed_args['--from_string'].from_string,
            parsed_args['--to_string'].to_string
        )

        self.assertTrue(result)

if __name__ == '__main__':
    unittest.main()
