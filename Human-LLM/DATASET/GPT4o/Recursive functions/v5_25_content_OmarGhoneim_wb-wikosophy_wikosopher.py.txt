import requests
from bs4 import BeautifulSoup
import regex as re
import time

def get_to_philosophy(url, cache=None):
    """
    Recursively follow the first non-parenthetical, non-italicized link in 
    the main text of a Wikipedia article until reaching the Philosophy page,
    an article with no outgoing links, or a loop.

    Args:
        url (str): The starting URL of the Wikipedia article.
        cache (list): A list to keep track of visited URLs to detect loops.

    Returns:
        str: The final URL reached in the search process.
    """
    if cache is None:
        cache = []

    cache.append(url)
    excluded_classes = ["new", "extiw", 'nowrap', 'IPA', 'reference', 'mw-selflink selflink']
    excluded_ids = ["coordinates"]

    time.sleep(0.5)  # Respectful delay to avoid overwhelming the server
    response = requests.get(url)
    soup = BeautifulSoup(response.text, 'lxml')

    no_wikilinks = True

    # Find the first valid link in the main text
    for paragraph in soup.find(class_='mw-parser-output').find_all('p', recursive=False):
        for cls in excluded_classes:
            for element in paragraph.find_all(class_=cls):
                element.extract()
        for id_ in excluded_ids:
            for element in paragraph.find_all(id=id_):
                element.extract()
        
        # Remove text inside parentheses
        cleaned_paragraph = re.sub(r'(?<!_)(?<rec>\((?:[^()]++|(?&rec))*\))', '', str(paragraph))
        cleaned_soup = BeautifulSoup(cleaned_paragraph, 'lxml')
        
        for italic in cleaned_soup.find_all('i'):
            italic.extract()
        
        first_link = cleaned_soup.find('a', href=True)
        if first_link:
            link_suffix = first_link['href']
            no_wikilinks = False
            break

    if no_wikilinks:
        print(f"No outgoing wikilinks in the article: {url[30:]}")
        return url

    link = "https://en.wikipedia.org" + link_suffix
    print(link)

    if link == "https://en.wikipedia.org/wiki/Philosophy":
        print("Philosophy page found!")
        return link
    elif link in cache:
        print("Stuck in a loop!")
        return link
    else:
        return get_to_philosophy(link, cache)

if __name__ == "__main__":
    random_url = "https://en.wikipedia.org/wiki/Special:Random"
    get_to_philosophy(random_url)
