def replace_sep(my_string, sep1, sep2):
    """
    Replaces occurrences of sep1 with sep2 in the given string.

    Args:
        my_string (str): The input string.
        sep1 (str): The separator to be replaced.
        sep2 (str): The separator to replace with.

    Returns:
        str: The modified string with sep1 replaced by sep2.
    """
    if not my_string:
        return ""
    if my_string.startswith(sep1):
        return sep2 + replace_sep(my_string[len(sep1):], sep1, sep2)
    return my_string[0] + replace_sep(my_string[1:], sep1, sep2)

def count_siblings(number_elves):
    """
    Returns the total number of siblings that the elves have.

    Args:
        number_elves (int): The number of elves.

    Returns:
        int: The total number of siblings.
    """
    if number_elves <= 0:
        return 0
    if number_elves % 2 == 1:
        return 3 + count_siblings(number_elves - 1)
    return 1 + count_siblings(number_elves - 1)

def extractor(string):
    """
    Extracts and returns the substring enclosed within parentheses.

    Args:
        string (str): The input string.

    Returns:
        str: The substring enclosed within parentheses.
    """
    if not string:
        return ""
    if string[0] != "(":
        return extractor(string[1:])
    if string[-1] != ")":
        return extractor(string[:-1])
    return string

def is_palindrome(sequence):
    """
    Checks if the given sequence is a palindrome.

    Args:
        sequence (str): The input sequence.

    Returns:
        bool: True if the sequence is a palindrome, False otherwise.
    """
    sequence = sequence.replace(" ", "")
    if len(sequence) < 2:
        return True
    if sequence[0] != sequence[-1]:
        return False
    return is_palindrome(sequence[1:-1])
