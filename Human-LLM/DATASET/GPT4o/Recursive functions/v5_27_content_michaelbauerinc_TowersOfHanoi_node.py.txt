class Node:
    def __init__(self, value, next_node=None):
        """
        Initialize a new node.

        Args:
        value (any): The value to store in the node.
        next_node (Node, optional): The next node in the list. Defaults to None.
        """
        self.value = value
        self.next_node = next_node
        self.previous_node = None
        self.image = None

    def set_next_node(self, next_node):
        """
        Set the next node.

        Args:
        next_node (Node): The node to set as the next node.
        """
        self.next_node = next_node

    def get_next_node(self):
        """
        Get the next node.

        Returns:
        Node: The next node.
        """
        return self.next_node

    def set_previous_node(self, previous_node):
        """
        Set the previous node.

        Args:
        previous_node (Node): The node to set as the previous node.
        """
        self.previous_node = previous_node

    def get_previous_node(self):
        """
        Get the previous node.

        Returns:
        Node: The previous node.
        """
        return self.previous_node

    def get_value(self):
        """
        Get the value of the node.

        Returns:
        any: The value stored in the node.
        """
        return self.value

    def assign_image(self, image):
        """
        Assign an image to the node.

        Args:
        image (any): The image to assign to the node.
        """
        self.image = image

    def get_image(self):
        """
        Get the assigned image of the node.

        Returns:
        any: The image assigned to the node.
        """
        return self.image
