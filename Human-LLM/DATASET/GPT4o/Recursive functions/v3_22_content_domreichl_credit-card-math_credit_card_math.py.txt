def calc_balance(balance, min_payment_rate, monthly_interest_rate, months):
    """
    Calculate the credit card balance after a given number of months for a person
    who only pays the minimum monthly payment. Also builds a list of monthly payments.
    
    :param balance: Initial credit card balance
    :param min_payment_rate: Minimum monthly payment rate
    :param monthly_interest_rate: Monthly interest rate
    :param months: Number of months to calculate for
    :return: Final balance after given months, list of monthly payments
    """
    payments = []
    for _ in range(months):
        min_payment = min_payment_rate * balance
        payments.append(min_payment)
        unpaid_balance = balance - min_payment
        balance = unpaid_balance + monthly_interest_rate * unpaid_balance
    return round(balance, 2), payments

def pay_off_debt(balance, monthly_interest_rate, months, lower, upper):
    """
    Calculate the minimum fixed monthly payment needed to pay off a credit card
    balance within a given number of months using bisection search.
    
    :param balance: Initial credit card balance
    :param monthly_interest_rate: Monthly interest rate
    :param months: Number of months to pay off the debt
    :param lower: Lower bound of the guess for the monthly payment
    :param upper: Upper bound of the guess for the monthly payment
    :return: The minimum fixed monthly payment required
    """
    while lower <= upper:
        guess = (lower + upper) / 2
        current_balance = balance
        
        for _ in range(months):
            unpaid_balance = current_balance - guess
            current_balance = unpaid_balance + monthly_interest_rate * unpaid_balance
        
        if round(current_balance, 2) == 0:
            return round(guess, 2)
        elif current_balance > 0:
            lower = guess
        else:
            upper = guess
    return round(guess, 2)

def main():
    # Initialize parameters
    balance = 100000
    months = 12
    annual_interest_rate = 0.22
    min_payment_rate = 0.05
    monthly_interest_rate = annual_interest_rate / 12.0

    # Calculate balance after making only minimum payments
    new_balance, payments = calc_balance(balance, min_payment_rate, monthly_interest_rate, months)
    average_payment = round(sum(payments) / len(payments), 2)
    loss1 = round(abs(balance - new_balance - sum(payments)), 2)

    # Calculate the fixed monthly payment needed to pay off debt
    min_guess = balance / 12
    max_guess = (balance * (1 + monthly_interest_rate) ** months) / months
    payment = pay_off_debt(balance, monthly_interest_rate, months, min_guess, max_guess)
    loss2 = round(abs(balance - 0 - payment * months), 2)

    # Output the results
    print('Current Balance:', balance, '€')
    print('Duration:', months, 'months')
    print(f'With a monthly payment rate of {min_payment_rate:.2f}:')
    print(f'   - You will pay {average_payment} € per month on average.')
    print(f'   - Your loss will be {loss1} € for that time period.')
    print(f'   - Your remaining balance will be {new_balance} €.')
    print(f'If you want to pay off your balance within {months} months:')
    print(f'   - You will have to pay {payment} € per month.')
    print(f'   - Your loss will be {loss2} € in total.')
    print('   - Your remaining balance will be 0 €.')

if __name__ == "__main__":
    main()
