# Create a dictionary where each key-value pair is (p, p*p) for p in the range 0 to 9
x = {p: p * p for p in range(10)}
print(x)

# Create a dictionary where each key-value pair is (q, q**2) for q in the range 10 to 19, only if q is even
y = {q: q ** 2 for q in range(10, 20) if q % 2 == 0}
print(y)

# Create a dictionary where each key-value pair is (r, r*2) for r in the range 20 to 29, only if r is odd
z = {r: r * 2 for r in range(20, 30) if r % 2 != 0}
print(z)
