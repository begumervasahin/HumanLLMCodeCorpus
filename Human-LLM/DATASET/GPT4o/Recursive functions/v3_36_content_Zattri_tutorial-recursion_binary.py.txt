class Node:
    def __init__(self, val):
        """
        Initialize a new node with the given value.

        Parameters:
        val (int): The value of the node.
        """
        self.left = None    # Left child
        self.right = None   # Right child
        self.parent = None  # Parent
        self.value = val    # Value

    def add_left(self, child_node):
        """
        Add a left child to the current node.

        Parameters:
        child_node (Node): The node to be added as the left child.
        """
        self.left = child_node
        child_node.parent = self

    def add_right(self, child_node):
        """
        Add a right child to the current node.

        Parameters:
        child_node (Node): The node to be added as the right child.
        """
        self.right = child_node
        child_node.parent = self

    def print_left(self):
        """Print the value of the left child of the current node."""
        print("Left child:", self.left.value if self.left else None)

    def print_right(self):
        """Print the value of the right child of the current node."""
        print("Right child:", self.right.value if self.right else None)

    def print_value(self):
        """Print the value of the current node."""
        print("Value:", self.value)

    def print_parent(self):
        """Print the value of the parent of the current node."""
        print("Parent:", self.parent.value if self.parent else None)

# Example usage
if __name__ == "__main__":
    # Create nodes
    root = Node(10)
    left_child = Node(5)
    right_child = Node(15)
    left_grandchild = Node(3)
    
    # Add children to the tree
    root.add_left(left_child)
    root.add_right(right_child)
    left_child.add_left(left_grandchild)
    
    # Print node values and relationships
    root.print_value()             # Should print: Value: 10
    root.print_left()              # Should print: Left child: 5
    root.print_right()             # Should print: Right child: 15
    
    left_child.print_value()       # Should print: Value: 5
    left_child.print_parent()      # Should print: Parent: 10
    left_child.print_left()        # Should print: Left child: 3
    
    left_grandchild.print_value()  # Should print: Value: 3
    left_grandchild.print_parent() # Should print: Parent: 5
