class LN:
    def __init__(self, value, next=None):
        self.value = value
        self.next = next

def len_ll(ll):
    """Iteratively calculate the length of a linked list."""
    count = 0
    while ll is not None:
        count += 1
        ll = ll.next
    return count

def len_ll_r(ll):
    """Recursively calculate the length of a linked list."""
    if ll is None:
        return 0
    else:
        return 1 + len_ll_r(ll.next)

def sum_ll(ll):
    """Iteratively calculate the sum of values in a linked list."""
    total = 0
    while ll is not None:
        total += ll.value
        ll = ll.next
    return total

def sum_ll_r(ll):
    """Recursively calculate the sum of values in a linked list."""
    if ll is None:
        return 0
    else:
        return ll.value + sum_ll_r(ll.next)

def str_ll(ll):
    """Iteratively create a string representation of a linked list."""
    answer = ''
    while ll is not None:
        answer += str(ll.value) + '->'
        ll = ll.next
    return answer + 'None'

def str_ll_r(ll):
    """Recursively create a string representation of a linked list."""
    if ll is None:
        return 'None'
    else:
        return str(ll.value) + '->' + str_ll_r(ll.next)

def list_to_ll(lst):
    """Convert a list to a linked list iteratively."""
    if not lst:
        return None
    front = rear = LN(lst[0])
    for v in lst[1:]:
        rear.next = LN(v)
        rear = rear.next
    return front

def list_to_ll_r(lst):
    """Convert a list to a linked list recursively."""
    if not lst:
        return None
    else:
        return LN(lst[0], list_to_ll_r(lst[1:]))

def find_ll(ll, avalue):
    """Iteratively find a value in a linked list."""
    while ll is not None:
        if ll.value == avalue:
            return ll
        ll = ll.next
    return None

def find_ll_r(ll, avalue):
    """Recursively find a value in a linked list."""
    if ll is None:
        return None
    elif ll.value == avalue:
        return ll
    else:
        return find_ll_r(ll.next, avalue)

def find_ll_r2(ll, avalue):
    """Recursively find a value in a linked list (alternative implementation)."""
    if ll is None or ll.value == avalue:
        return ll
    else:
        return find_ll_r2(ll.next, avalue)

def copy_ll(ll):
    """Iteratively copy a linked list."""
    if ll is None:
        return None
    front = rear = LN(ll.value)
    while ll.next is not None:
        ll = ll.next
        rear.next = LN(ll.value)
        rear = rear.next
    return front

def copy_ll_r(ll):
    """Recursively copy a linked list."""
    if ll is None:
        return None
    else:
        return LN(ll.value, copy_ll_r(ll.next))

def iterator(ll):
    """Create an iterator for a linked list."""
    while ll is not None:
        yield ll.value
        ll = ll.next

def append_ll(ll, value):
    """Append a value to the end of a linked list iteratively."""
    if ll is None:
        return LN(value)
    front = ll
    while ll.next is not None:
        ll = ll.next
    ll.next = LN(value)
    return front

def append_ll_r(ll, value):
    """Append a value to the end of a linked list recursively."""
    if ll is None:
        return LN(value)
    else:
        ll.next = append_ll_r(ll.next, value)
        return ll

def add_after_ll(ll, value):
    """Add a value after the current node in a linked list."""
    ll.next = LN(value, ll.next)

def remove_after_ll(ll):
    """Remove the node after the current node in a linked list."""
    if ll.next is not None:
        ll.next = ll.next.next
