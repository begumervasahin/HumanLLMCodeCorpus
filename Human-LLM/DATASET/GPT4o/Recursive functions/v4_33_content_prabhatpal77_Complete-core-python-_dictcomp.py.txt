# Create a dictionary with numbers from 0 to 9 as keys and their squares as values
x = {p: p * p for p in range(10)}
print(x)

# Create a dictionary with even numbers from 10 to 19 as keys and their squares as values
y = {q: q ** 2 for q in range(10, 20) if q % 2 == 0}
print(y)

# Create a dictionary with odd numbers from 20 to 29 as keys and their doubles as values
z = {r: r * 2 for r in range(20, 30) if r % 2 != 0}
print(z)
