class Node:
    def __init__(self, val):
        """
        Initialize a new node with the given value.

        Parameters:
        val (int): The value of the node.
        """
        self.l = None  # Left child
        self.r = None  # Right child
        self.p = None  # Parent
        self.v = val   # Value

    def addLeft(self, childNode):
        """
        Add a left child to the current node.

        Parameters:
        childNode (Node): The node to be added as the left child.
        """
        self.l = childNode
        childNode.p = self

    def addRight(self, childNode):
        """
        Add a right child to the current node.

        Parameters:
        childNode (Node): The node to be added as the right child.
        """
        self.r = childNode
        childNode.p = self

    def printLeft(self):
        """Print the value of the left child of the current node."""
        print("Left child:", self.l.v if self.l else None)

    def printRight(self):
        """Print the value of the right child of the current node."""
        print("Right child:", self.r.v if self.r else None)

    def printVal(self):
        """Print the value of the current node."""
        print("Value:", self.v)

    def printParent(self):
        """Print the value of the parent of the current node."""
        print("Parent:", self.p.v if self.p else None)

# Example usage
if __name__ == "__main__":
    # Create nodes
    root = Node(10)
    left_child = Node(5)
    right_child = Node(15)
    left_grandchild = Node(3)
    
    # Add children to the tree
    root.addLeft(left_child)
    root.addRight(right_child)
    left_child.addLeft(left_grandchild)
    
    # Print node values and relationships
    root.printVal()             # Should print: Value: 10
    root.printLeft()            # Should print: Left child: 5
    root.printRight()           # Should print: Right child: 15
    
    left_child.printVal()       # Should print: Value: 5
    left_child.printParent()    # Should print: Parent: 10
    left_child.printLeft()      # Should print: Left child: 3
    
    left_grandchild.printVal()  # Should print: Value: 3
    left_grandchild.printParent()  # Should print: Parent: 5
