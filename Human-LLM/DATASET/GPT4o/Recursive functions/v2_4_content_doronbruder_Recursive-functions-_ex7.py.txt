def to_string(lst):
    """Convert a list to a string."""
    return ''.join(lst)

def print_to_n(n):
    """Print all numbers from 1 to n in increasing order."""
    if n >= 1:
        print_to_n(n - 1)
        print(n)

def print_reversed(n):
    """Print all numbers from n to 1 in decreasing order."""
    if n >= 1:
        print(n)
        print_reversed(n - 1)

def has_divisor_smaller_than(n, i):
    """Check if a number has a divisor smaller than a given number."""
    if i <= 1:
        return False
    elif n % i == 0:
        return True
    return has_divisor_smaller_than(n, i - 1)

def is_prime(n):
    """Check if a number is prime."""
    if n < 2:
        return False
    return not has_divisor_smaller_than(n, int(n ** 0.5))

def factorial(n):
    """Calculate the factorial of a number."""
    if n == 1:
        return 1
    return n * factorial(n - 1)

def exp_n_x(n, x):
    """Calculate the value of e^x as n goes to infinity."""
    if n == 0:
        return 1
    return (x ** n) / factorial(n) + exp_n_x(n - 1, x)

def play_hanoi(hanoi, n, src, dest, temp):
    """Solve the Towers of Hanoi puzzle using a graphic class."""
    if n < 1:
        return
    if n == 1:
        hanoi.move(src, dest)
        return
    play_hanoi(hanoi, n - 1, src, temp, dest)
    hanoi.move(src, dest)
    play_hanoi(hanoi, n - 1, temp, dest, src)

def print_sequences(char_list, n):
    """Print all possible sequences of length n from a given list of characters."""
    if n == 0:
        return
    print_sequences_rec(char_list, "", len(set(char_list)), n)

def print_sequences_rec(char_list, prefix, n, k):
    """Helper function to print all possible sequences of length k from a given list of characters."""
    if k == 0:
        print(prefix)
        return
    for i in range(n):
        new_prefix = prefix + char_list[i]
        print_sequences_rec(char_list, new_prefix, n, k - 1)

def print_no_repetition_sequences(char_list, n):
    """Print all possible sequences of length n from a given list of characters without repetitions."""
    if n == 0:
        return
    print_no_repetition_sequences_rec(char_list, "", n)

def print_no_repetition_sequences_rec(char_list, prefix, n):
    """Helper function to print all possible sequences of length n from a given list of characters without repetitions."""
    if n == 0:
        print(prefix)
        return
    for i, char in enumerate(char_list):
        new_prefix = prefix + char
        print_no_repetition_sequences_rec(char_list[:i] + char_list[i + 1:], new_prefix, n - 1)

def parentheses(n):
    """Generate all combinations of n pairs of properly opened and closed parentheses."""
    result = []
    if n > 0:
        parentheses_rec([""] * 2 * n, 0, n, 0, 0, result)
    return result

def parentheses_rec(curr, pos, n, open, close, result):
    """Helper function to generate all combinations of n pairs of properly opened and closed parentheses."""
    if close == n:
        result.append(to_string(curr))
        return
    if open > close:
        curr[pos] = ')'
        parentheses_rec(curr, pos + 1, n, open, close + 1, result)
    if open < n:
        curr[pos] = '('
        parentheses_rec(curr, pos + 1, n, open + 1, close, result)

def up_and_right(n, k):
    """Print all possible paths from (0,0) to (n,k) using only ups and rights."""
    if n < 0 or k < 0:
        return
    up_and_right_rec(n, k, "")

def up_and_right_rec(n, k, prefix):
    """Helper function to print all possible paths from (0,0) to (n,k) using only ups and rights."""
    if n == 0 and k == 0:
        print(prefix)
        return
    if n > 0:
        up_and_right_rec(n - 1, k, prefix + 'r')
    if k > 0:
        up_and_right_rec(n, k - 1, prefix + 'u')

def flood_fill(image, start):
    """Flood fill algorithm starting at a given point."""
    x, y = start
    flood_fill_rec(image, x, y, '.', '*')

def flood_fill_rec(image, x, y, empty_char, fill_char):
    """Helper function for the flood fill algorithm."""
    width = len(image)
    height = len(image[0])
    if x < 0 or x >= width or y < 0 or y >= height or image[x][y] != empty_char:
        return
    image[x][y] = fill_char
    flood_fill_rec(image, x - 1, y, empty_char, fill_char)
    flood_fill_rec(image, x + 1, y, empty_char, fill_char)
    flood_fill_rec(image, x, y - 1, empty_char, fill_char)
    flood_fill_rec(image, x, y + 1, empty_char, fill_char)
