# Declare a dictionary to cache Fibonacci calculations
fibonacci_cache = {}

def fibonacci(num):
    """
    Compute the Fibonacci number using recursion with caching.

    Args:
    num (int): The position in the Fibonacci sequence.

    Returns:
    int: The Fibonacci number at the given position.
    """
    # Ensure the input is an integer
    if not isinstance(num, int):
        return False
    
    # Check if the value is in the cache
    if num in fibonacci_cache:
        return fibonacci_cache[num]
    
    # Base cases: F(0) = 1, F(1) = 1
    if num == 0:
        return 1
    elif num == 1:
        return 1
    else:
        # Recursive computation with caching
        value = fibonacci(num - 1) + fibonacci(num - 2)
        fibonacci_cache[num] = value
        return value

def main():
    """Print the first 1000 Fibonacci numbers."""
    for num in range(1000):
        print(f"{num}: {fibonacci(num)}")

if __name__ == "__main__":
    main()
