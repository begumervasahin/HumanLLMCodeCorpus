from direct.showbase.ShowBase import ShowBase
from panda3d.core import loadPrcFileData, WindowProperties
from editorLogHandler import setupLog
from DirectGuiDesigner.DirectGuiDesigner import DirectGuiDesigner

# Set Panda3D configuration parameters
PANDA3D_CONFIG = """
sync-video
textures-power-2 none
window-title DirectGUI Designer
maximized
win-size 1280 720
ime-aware
"""
loadPrcFileData("", PANDA3D_CONFIG)

# Set up logging
log_filepath, config_file = setupLog("DirectGuiDesigner")

# Create ShowBase instance
base = ShowBase()

# Function to set window title to indicate unsaved changes
def set_dirty_name():
    set_window_title("*DirectGUI Designer")

# Function to set window title to indicate saved changes
def set_clean_name():
    set_window_title("DirectGUI Designer")

# Set window title
def set_window_title(title):
    window_properties = WindowProperties()
    window_properties.setTitle(title)
    base.win.requestProperties(window_properties)

# Accept events to change window title
base.accept("request_dirty_name", set_dirty_name)
base.accept("request_clean_name", set_clean_name)

# Initialize DirectGuiDesigner
direct_gui_designer = DirectGuiDesigner(base.pixel2d)
direct_gui_designer.log_file = log_filepath
direct_gui_designer.config_file = config_file
base.direct_gui_designer = direct_gui_designer

# Run the application
base.run()
