import requests
import pandas as pd
import os
import sys

def fetch_comments(fb_token, fb_page_id, fb_post_id):
    comments = []
    dates = []
    url = f"https://graph.facebook.com/v2.9/{fb_page_id}_{fb_post_id}/comments?limit=100&access_token={fb_token}"

    while True:
        response = requests.get(url)
        data = response.json()

        for comment_data in data['data']:
            comments.append(comment_data.get('message').encode('utf-8').strip())
            dates.append(comment_data.get('created_time'))

        next_page = data.get('paging', {}).get('next')
        if not next_page:
            break

        url = next_page

    return comments, dates

def main():
    # Retrieve Facebook token from environment variables       
    fb_token = os.environ.get('FB_TOKEN')
    if not fb_token:
        print("Please provide the Facebook token as an environment variable (FB_TOKEN)")
        sys.exit(-1)

    # Facebook page and post IDs
    fb_page_id = "624258451196"
    fb_post_id = "12488754962145785"

    # Fetch comments
    comments, dates = fetch_comments(fb_token, fb_page_id, fb_post_id)

    # Generate DataFrame
    df = pd.DataFrame({'comment': comments, 'date': dates})    
    df['date'] = pd.to_datetime(df['date'])
    df['day_of_week'] = df['date'].dt.day_name()
    df['year'] = df['date'].dt.year
    df['month'] = df['date'].dt.month
    df['count'] = 1

    # Save as JSON file
    print("\nGenerating JSON file...")
    df.to_json('comment_data.json')
    print("JSON file generated successfully.")

if __name__ == "__main__":
    main()
