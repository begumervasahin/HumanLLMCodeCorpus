from collections import deque

def column_break(input_string):
    """
    Breaks the input string into a 4x4 array of columns.

    :param input_string: Input string
    :return: 4x4 array of columns
    """
    columns = []
    letters = []
    index = 0
    for _ in range(4):
        for _ in range(4):
            letters.append(input_string[index:index + 2])
            index += 2
        columns.append(letters)
        letters = []
    return columns

def bytes_to_array(bytes_string):
    """
    Converts bytes to a 4x4 array.

    :param bytes_string: Bytes string
    :return: 4x4 array
    """
    array = []
    for i, byte in enumerate(bytes_string):
        if i % 4 == 0:
            array.append([byte])
        else:
            array[i // 4].append(byte)
    return array

def generate_key_schedule(key):
    """
    Generates the key schedule.

    :param key: Input key
    :return: Key schedule
    """
    key_schedule = []
    keys = bytes_to_array(bytes.fromhex(key))
    for _ in range(4, 44):
        x0, x1, x2, x3 = keys[-4]
        round_key = [x3, x0, x1, x2]
        keys.append(round_key)
    return keys
