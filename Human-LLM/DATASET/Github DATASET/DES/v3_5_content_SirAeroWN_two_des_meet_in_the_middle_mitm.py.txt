from multiprocessing import Pool

PLAIN_TEXT = '89504E470D0A1A0A'
CIPHER_TEXT = '89A3F4E3A99337A4'

def encrypt(key, plain_text):
    cipher = des(key)
    return cipher.encrypt(plain_text)

def decrypt(key, cipher_text):
    cipher = des(key)
    return cipher.decrypt(cipher_text)

def generate_keys(num_keys):
    for n in range(num_keys):
        key_bytes = bytearray([0] * 8)
        powers = [7, 6, 5, 4, 3, 2, 1, 0]
        for i, power in enumerate(powers):
            key_bytes[i] = n // (128 ** power)
            n -= key_bytes[i] * (128 ** power) if n >= (128 ** power) else n
        yield bytes(key_bytes)

def convert_hex_string_to_bytes(hex_string):
    if len(hex_string) % 2 != 0:
        raise ValueError("Hex string length must be even")
    else:
        byte_list = [hex_string[i:i + 2] for i in range(0, len(hex_string), 2)]
        return bytes([int(c, 16) for c in byte_list])

def meet_in_the_middle_attack(num_keys, plain_text, cipher_text, pool=None):
    def composed_encrypt(plain_text, key):
        return (encrypt(key, plain_text), key)

    if pool is None:
        table = {}
        for key in generate_keys(num_keys):
            encrypted_text = encrypt(key, plain_text)
            table[encrypted_text] = key
    else:
        table = dict(pool.map(partial(composed_encrypt, plain_text), generate_keys(num_keys)))

    for key in generate_keys(num_keys):
        decrypted_text = decrypt(key, cipher_text)
        if decrypted_text in table:
            k1 = table[decrypted_text].hex()
            k2 = key.hex()
            print(f'Found keys: (k1: {k1}, k2: {k2})')
            print('Apply k2 then k1 to decrypt.')
            print('Use a tool such as this one (http://des.online-domain-tools.com/) to decrypt the whole file with the keys.')
            return (k1, k2)

    print('Did not find keys.')
