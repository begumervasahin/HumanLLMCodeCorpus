import cv2
import os

# Input and output directories
input_directory = "/images"
output_directory = "/images/cropped/"

def facecrop(image_path):
    """Detects faces in an image and saves the cropped faces in the output directory."""
    # Load the face detection classifier
    facedata = "haarcascade_frontalface_alt.xml"
    cascade = cv2.CascadeClassifier(facedata)
    
    # Read the image
    img = cv2.imread(image_path)
    
    try:
        # Resize the image for faster processing
        minisize = (img.shape[1], img.shape[0])
        miniframe = cv2.resize(img, minisize)
        
        # Detect faces in the image
        faces = cascade.detectMultiScale(miniframe)
        
        for f in faces:
            x, y, w, h = f
            # Draw rectangle around the face
            cv2.rectangle(img, (x, y), (x + w, y + h), (0, 255, 0), 2)
            # Crop the face region
            sub_face = img[y:y + h, x:x + w]
            
            # Extract image filename
            filename = os.path.basename(image_path)
            
            # Save the cropped face in the output directory
            cv2.imwrite(output_directory + filename, sub_face)
            print("Writing: " + filename)
    except Exception as e:
        print("Error:", e)

if __name__ == '__main__':
    # Get a list of image files in the input directory
    images = os.listdir(input_directory)
    
    for i, img in enumerate(images, 1):
        # Construct the full path to each image
        image_path = os.path.join(input_directory, img)
        print(i)
        facecrop(image_path)
