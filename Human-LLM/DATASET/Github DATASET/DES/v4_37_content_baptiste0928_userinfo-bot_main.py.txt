import json
import discord
from discord.ext import commands
import glob
import os

# Load bot configuration from config.json
with open('config.json', 'r') as f:
    config = json.load(f)

# Create bot instance with a customizable command prefix
bot = commands.AutoShardedBot(command_prefix=commands.when_mentioned_or(config.get("prefix", ".")))

# Remove the default help command to customize it later
bot.remove_command("help")

# Store the loaded configuration in the bot object
bot.config = config

# Find and load cog modules
modules = []
for file in glob.glob("cogs/*.py"):
    modules.append(file.replace(os.path.sep, '.').replace('.py', ''))

# Load each cog extension
if __name__ == '__main__':
    for extension in modules:
        try:
            bot.load_extension(extension)
        except Exception as e:
            print(f"Failed to load extension {extension}: {e}")

# Event handler for when the bot is ready
@bot.event
async def on_ready():
    # Set bot's presence
    await bot.change_presence(activity=discord.Activity(status=discord.Status.idle,
                                                        name=config["activity"],
                                                        type=discord.ActivityType.watching))
    # Print bot's connection details
    print(f"Connected as {bot.user}")
    print("Ready!")

# Run the bot with the token from the configuration
bot.run(config.get("token"))
