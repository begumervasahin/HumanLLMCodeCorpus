import random
from math import sqrt

# Define the size of the list
n = 10000

# Create a list of numbers and shuffle it
val = list(range(n))
random.shuffle(val)

# Create pointers for linked list structure
ptr = [-1] * n
for i in range(n - 1):
    ptr[val[i]] = val[i + 1]
head = val[0]

# Function to search for an element in the list
def search(x, i):
    count = 0
    while x > val[i]:
        i = ptr[i]
        count += 1
    return i, count

# Search function type A
def search_type_a(x):
    return search(x, head)

# Search function type B
def search_type_b(x):
    i = head
    max_val = val[i]
    for j in range(int(sqrt(n))):
        y = val[j]
        if max_val < y <= x:
            i = j
            max_val = y
    return search(x, i)

# Search function type C
def search_type_c(x):
    i = head
    max_val = val[i]
    for _ in range(int(sqrt(n))):
        jj = random.randint(0, n - 1)
        y = val[jj]
        if max_val < y <= x:
            i = jj
            max_val = y
    return search(x, i)

# Search function type D
def search_type_d(x):
    i = random.randint(0, n - 1)
    y = val[i]
    if x < y:
        return search(x, head)
    elif x > y:
        return search(x, ptr[i])
    else:
        return i, 0

# Perform 100 iterations of each search type and collect the counts
search_counts_a = [search_type_a(random.randint(0, n - 1))[1] for _ in range(100)]
search_counts_b = [search_type_b(random.randint(0, n - 1))[1] for _ in range(100)]
search_counts_c = [search_type_c(random.randint(0, n - 1))[1] for _ in range(100)]
search_counts_d = [search_type_d(random.randint(0, n - 1))[1] for _ in range(100)]

# Calculate and print the average search counts for each type
print("Average Search Count Type A:", sum(search_counts_a) / len(search_counts_a))
print("Average Search Count Type B:", sum(search_counts_b) / len(search_counts_b))
print("Average Search Count Type C:", sum(search_counts_c) / len(search_counts_c))
print("Average Search Count Type D:", sum(search_counts_d) / len(search_counts_d))
