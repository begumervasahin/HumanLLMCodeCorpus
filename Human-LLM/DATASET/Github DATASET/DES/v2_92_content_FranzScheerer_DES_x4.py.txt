# Function to write a number to a file
def write_number(number, filename):
    with open(filename, 'w') as file:
        file.write(str(number))

# Function to read a number from a file
def read_number(filename):
    with open(filename, 'r') as file:
        number_str = file.read()
        number = 0
        for char in number_str:
            number = (number * 10) + ord(char) - 48
    return number

    index_j = permutation[(index_j + permutation[index_i]) % 256]

# Helper function to convert number to hexadecimal text
def num_to_hex_text(number):
    hex_chars = '0123456789abcdef'
    hex_text = ''
    while number > 0:
        hex_text = hex_chars[number % 16] + hex_text
        number >>= 4
    return hex_text

# Extended Euclidean Algorithm to compute GCD
def gcd(a, b):
    while b > 0:
        a, b = b, a % b
    return a

# Function to find the next prime number
def next_prime(current_prime):

    return find_next_prime(current_prime)

# Elliptic curve addition operation
def elliptic_curve_addition(point_p, point_q):
    global prime_modulus

    x1, y1 = point_p
    x2, y2 = point_q

    if x1 == x2:
        slope = ((3 * x1 * x1 - 1) * pow(2 * y1, prime_modulus - 2, prime_modulus)) % prime_modulus
    else:
        if x1 < x2:
            x1 += prime_modulus
        slope = ((y1 - y2) * pow(x1 - x2, prime_modulus - 2, prime_modulus)) % prime_modulus

    xr = (slope * slope) - x1 - x2
    yr = slope * (x1 - xr) - y1

    return [xr % prime_modulus, yr % prime_modulus]

# Elliptic curve multiplication operation
def elliptic_curve_multiplication(point_p, scalar_n):
    global prime_modulus

    result_point = 'ZERO'
    current_point = point_p

    while scalar_n != 0:
        if scalar_n % 2 != 0:
            if result_point == 'ZERO':
                result_point = current_point
            else:
                result_point = elliptic_curve_addition(result_point, current_point)

        current_point = elliptic_curve_addition(current_point, current_point)
        scalar_n >>= 1

    return result_point

# Schnorr signature generation function
def schnorr_signature_generation(generator_point, message, private_key):
    global prime_modulus

    k = custom_hash(message + 'key value')
    R = elliptic_curve_multiplication(generator_point, k)
    e = custom_hash(str(R[0]) + message) % ((prime_modulus + 1) >> 2)

    return [(k - private_key * e) % ((prime_modulus + 1) >> 2), e]
