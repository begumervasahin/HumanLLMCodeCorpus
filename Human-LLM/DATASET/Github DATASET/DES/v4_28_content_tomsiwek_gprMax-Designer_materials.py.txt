class Material:
    """
    Represents a material with its properties.

    Args:
        epsilon_r (float): Relative permittivity.
        sigma (float): Conductivity.
        mu_r (float): Relative permeability.
        sigma_mag (float): Magnetic loss.
        name (str): Material identifier.
    """
    def __init__(self, epsilon_r=0.0, sigma=0.0, mu_r=0.0, sigma_mag=0.0, name=""):
        """
        Initializes a material with specified properties.

        Properties of a material are explained in the gprMax user manual.
        """
        self.epsilon_r = epsilon_r
        self.sigma = sigma
        self.mu_r = mu_r
        self.sigma_mag = sigma_mag
        self.name = name

    def calculate_wave_velocity(self):
        """
        Calculates the electromagnetic wave velocity in the material.

        Returns:
            float: Electromagnetic wave velocity.
        """
        speed_of_light = 299792458
        return speed_of_light / (self.epsilon_r ** 0.5)
