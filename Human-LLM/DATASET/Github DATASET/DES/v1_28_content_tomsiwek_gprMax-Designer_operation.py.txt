class TOperation(object):
    """
    Class represents a single operation performed in the program.
    
    Attributes:
        o_type (str): The type of operation.
        shape (object): The shape on which the operation was performed.
        num (int): The index of the shape in the shape list.
    """
    
    def __init__(self, o_type=None, **kwargs):
        """
        Initialize the TOperation object.
        
        Args:
            o_type (str): The type of operation.
            **kwargs: Additional keyword arguments.
        """
        self.set_operation(o_type, **kwargs)
    
    def set_operation(self, o_type=None, **kwargs):
        """
        Set the parameters of an operation.
        
        Args:
            o_type (str): The type of operation.
            **kwargs: Additional keyword arguments.
                shape (object): The shape on which the operation was performed.
                num (int): The index of the shape in the shape list.
        """
        self.o_type = o_type
        self.shape = kwargs.pop('shape', None)
        self.num = kwargs.pop('num', None)

# Example usage:
# Create a TOperation object
operation = TOperation(o_type="resize", shape="rectangle", num=1)

# Print the attributes of the TOperation object
print("Operation Type:", operation.o_type)
print("Shape:", operation.shape)
print("Number:", operation.num)
