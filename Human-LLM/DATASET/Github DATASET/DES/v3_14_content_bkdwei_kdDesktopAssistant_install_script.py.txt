import sys
import os
from setuptools.command.install import install

project_name = "kdDesktopAssistant"

class InstallCmd(install):
    def run(self):
        # Run the default install process
        install.run(self)

        # Try creating the startup script based on the platform
        try:
            if sys.platform == "win32":
                self._create_windows_startup_script()
            elif sys.platform == "linux":
                self._create_linux_startup_script()
        except Exception as e:
            # Print an error message if script creation fails
            print("Cannot create start script: " + str(e)) 

    def _create_windows_startup_script(self):
        # For Windows, create a .bat script        
        script_file = os.path.join(self._get_desktop_folder(), project_name + '.bat')
        with open(script_file, "w") as f:
            f.write("@echo off\r\nstart " + project_name + ".exe")

    def _create_linux_startup_script(self):
        # For Linux, create a .sh script
        script_file = os.path.join(self._get_desktop_folder(), project_name + '.sh')
        with open(script_file, "w") as f:
            f.write("#!/bin/bash\n")
            f.write(project_name + ".exe &")       
            st = os.stat(script_file)
            os.chmod(script_file, st.st_mode | 0o111)  # Add execute permission

    def _get_desktop_folder(self):
        # Get the path to the user's desktop folder        
        try:
            return os.path.join(os.path.expanduser('~'), 'Desktop')
        except Exception:
            # Return an empty string if the desktop folder cannot be accessed
            return ""

if __name__ == "__main__":
    # Create an instance of InstallCmd and run its run() method
    install_cmd = InstallCmd()
    install_cmd.run()
