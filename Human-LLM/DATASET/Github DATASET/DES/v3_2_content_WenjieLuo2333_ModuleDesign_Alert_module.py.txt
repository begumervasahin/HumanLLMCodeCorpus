import numpy as np

class AlertSystem:
    def __init__(self):
        self.blood_oxygen_data = []
        self.blood_pressure_data = []
        self.pulse_rate_data = []
        self.average_values = [[] for _ in range(3)]
        self.alert_flag = -1

    def check_threshold_exceedance(self, value, category):
        thresholds = {
            'bo': (0.1, 0.3),
            'bp': (80, 120),
            'pul': (60, 90)
        }
        min_threshold, max_threshold = thresholds[category]
        if not min_threshold <= value <= max_threshold:
            return list(thresholds.keys()).index(category)
        return -1

    def get_alert_status(self):
        return self.alert_flag

    def process_input_data(self, data):
        value, category = data
        data_list = {
            0: self.blood_oxygen_data,
            1: self.blood_pressure_data,
            2: self.pulse_rate_data
        }

        category_data = data_list[category]
        category_data.append(float(value))

        if len(category_data) > 20:
            category_data.pop(0)

        average_value = np.mean(category_data)
        if len(category_data) > 2:
            alert_type = self.check_threshold_exceedance(average_value, list(data_list.keys())[category])
            if alert_type != -1:
                self.alert_flag = alert_type

        return self.alert_flag

# Example Usage
alert_system = AlertSystem()

# Simulate data input
data_input = [
    (0.15, 0),  # Example blood oxygen data
    (110, 1),   # Example blood pressure data
    (70, 2)     # Example pulse data
]

# Process data and check for alerts
for data in data_input:
    alert_flag = alert_system.process_input_data(data)
    if alert_flag != -1:
        print(f"Alert Flag: {alert_flag}")
    else:
        print("No alert")
