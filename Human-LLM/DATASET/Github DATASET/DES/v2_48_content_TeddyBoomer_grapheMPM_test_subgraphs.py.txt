import pytest
import re
from grapheMPM import GrapheMPM

# Define a fixture for MPM data with show_level enabled
@pytest.fixture
def MPM_showlevel_data():
    # Define precedence and weight dictionaries
    precedence = {"A": "", "B": "", "C": "A", "D": "AB", "E": "B",
                  "F": "DE", "G": "E", "H": "CF", "I": "FG", "J": "HI"}        
    weights = {"A": 7, "B": 3, "C": 4, "D": 2, "E": 8,
               "F": 6, "G": 5, "H": 7, "I": 5, "J": 3}

    # Create GrapheMPM object with show_level set to True
    G = GrapheMPM(pred=precedence, pond=weights, show_level=True)

    # Calculate earliest date and generate Graphviz representation
    G.earliestdate()
    G.makeGraphviz()

    return G

# Define a fixture to extract subgraph attributes
@pytest.fixture
def subgraph_attrs(MPM_showlevel_data):
    # Use regular expression to extract attributes of subgraph from Graphviz source
    subgraph_pattern = re.compile("subgraph cluster_[0-9]+ {\n(.*)$", flags=re.M)
    return subgraph_pattern.search(MPM_showlevel_data.gv.source)

# Test subgraph rank
def test_subgraph_rank(subgraph_attrs):
    # Check if subgraph attributes are found and if 'rank=same' is present     
    assert subgraph_attrs
    assert "rank=same" in subgraph_attrs.group(1)

# Test Graphviz output with show_level enabled
def test_showlevel_on(MPM_showlevel_data):
    # Test rendering Graphviz output with show_level set to True
    assert isinstance(MPM_showlevel_data.gv.render("test_sl_on"), str)

# Test Graphviz output with show_level disabled
def test_showlevel_off(MPM_showlevel_data):
    # Disable show_level, recalculate earliest date, and generate Graphviz representation
    MPM_showlevel_data.show_level = False
    MPM_showlevel_data.earliestdate()
    MPM_showlevel_data.makeGraphviz()

    # Test rendering Graphviz output with show_level set to False
    assert isinstance(MPM_showlevel_data.gv.render("test_sl_off"), str)
