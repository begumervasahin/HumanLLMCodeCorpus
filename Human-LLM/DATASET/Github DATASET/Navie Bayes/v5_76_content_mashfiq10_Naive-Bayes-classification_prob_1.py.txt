import numpy as np
import matplotlib.pyplot as plt

def load_images(file_path):
    with open(file_path, 'rb') as file:
        data = bytearray(file.read())[16:]
    images = np.array(data, dtype=np.float64) > 100.
    return images

def load_labels(file_path):
    with open(file_path, 'rb') as file:
        data = bytearray(file.read())[8:]
    labels = np.array(data)
    labels = labels.reshape(labels.shape[0], 1)
    return labels

def display_images(images):
    fig = plt.figure()
    for k in range(20 * 20):
        a = fig.add_subplot(20, 20, k + 1)
        plt.imshow(images[k].reshape(28, 28), cmap=plt.cm.bone)
        a.set_axis_off()
    plt.show()

def preprocess_images(images):
    d = 28 * 28
    n = images.shape[0] // d
    return images.reshape(int(n), d)

def initialize_parameters(labels, num_classes):
    counts = np.bincount(labels)
    prior = counts / labels.shape[0]
    log_prior = np.log10(prior)
    theta = np.zeros((num_classes, d), dtype=np.float64)
    return counts, prior, log_prior, theta

def compute_class_probabilities(images, labels, counts, num_classes):
    theta = np.zeros((num_classes, d), dtype=np.float64)
    for i in range(num_classes):
        mask = (labels == i)
        theta[i] += (np.sum(images[mask], axis=0, dtype=np.float64) + 1.) / (counts[i] + num_classes)
    log_theta = np.log10(theta)
    log_complement = np.log10(1. - theta)
    return log_theta, log_complement

def load_and_preprocess_data(images_file, labels_file):
    images = load_images(images_file)
    labels = load_labels(labels_file)
    images = preprocess_images(images)
    return images, labels

def classify_images(images, log_theta, log_complement, log_prior):
    test_labels_hat = np.zeros((images.shape[0], 1))
    for i in range(images.shape[0]):
        log_likelihood = np.sum(log_theta[:, images[i].reshape(-1)], axis=1) + np.sum(log_complement[:, np.logical_not(images[i].reshape(-1))], axis=1)
        log_posterior = log_prior + log_likelihood
        test_labels_hat[i] = np.argmax(log_posterior)
    return test_labels_hat

def calculate_accuracy(predicted_labels, actual_labels):
    correct_predictions = np.sum(predicted_labels == actual_labels)
    accuracy = (correct_predictions / len(actual_labels)) * 100.
    return accuracy

# Load and preprocess training data
train_images, train_labels = load_and_preprocess_data('train-images.idx3-ubyte', 'train-labels.idx1-ubyte')

# Display some training images
display_images(train_images)

# Initialize parameters
num_classes = 10
d = 28 * 28
counts, prior, log_prior, theta = initialize_parameters(train_labels, num_classes)

# Compute class-wise probabilities
log_theta, log_complement = compute_class_probabilities(train_images, train_labels, counts, num_classes)

# Load and preprocess test data
test_images, test_labels = load_and_preprocess_data('t10k-images.idx3-ubyte', 't10k-labels.idx1-ubyte')

# Perform classification on test images
test_labels_hat = classify_images(test_images, log_theta, log_complement, log_prior)

# Calculate classification accuracy
accuracy = calculate_accuracy(test_labels_hat, test_labels)
print("Naive Bayes classifier classification accuracy:", accuracy, '%')
