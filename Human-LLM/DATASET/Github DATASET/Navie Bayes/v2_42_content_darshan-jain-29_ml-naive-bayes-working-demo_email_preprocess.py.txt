import pickle
import numpy as np
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.feature_selection import SelectPercentile, f_classif

def preprocess(words_file="../tools/word_data.pkl", authors_file="../tools/email_authors.pkl"):
    """
    Preprocess email data for classification.
    
    Parameters:
    - words_file: Path to the file containing email text data.
    - authors_file: Path to the file containing corresponding authors' labels.
    
    Returns:
    - features_train_transformed: Transformed training features.
    - features_test_transformed: Transformed test features.
    - labels_train: Training labels.
    - labels_test: Test labels.
    """
    # Load authors data
    with open(authors_file, "rb") as f:
        authors = pickle.load(f)

    # Load word data
    with open(words_file, "rb") as f:
        word_data = pickle.load(f)

    # Split data into training and testing sets
    features_train, features_test, labels_train, labels_test = train_test_split(word_data, authors, test_size=0.1, random_state=42)

    # Vectorize into tfidf matrix
    vectorizer = TfidfVectorizer(sublinear_tf=True, max_df=0.5, stop_words='english')
    features_train_transformed = vectorizer.fit_transform(features_train)
    features_test_transformed = vectorizer.transform(features_test)

    # Select/keep most helpful features
    selector = SelectPercentile(f_classif, percentile=10)
    selector.fit(features_train_transformed, labels_train)
    features_train_transformed = selector.transform(features_train_transformed).toarray()
    features_test_transformed = selector.transform(features_test_transformed).toarray()

    # Print number of Chris and Sara training emails
    print("Number of Chris training emails:", sum(labels_train))
    print("Number of Sara training emails:", len(labels_train) - sum(labels_train))

    return features_train_transformed, features_test_transformed, labels_train, labels_test
