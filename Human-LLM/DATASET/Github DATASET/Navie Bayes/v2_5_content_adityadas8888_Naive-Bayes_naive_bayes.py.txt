import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import math
from scipy.stats import norm

def generate_data(tsize0, tsize1):
    mean1 = [1, 0]
    mean2 = [0, 1]
    Sigma1 = [[1, 0.75], [0.75, 1]]
    Sigma2 = [[1, 0.75], [0.75, 1]]
    
    x1, y1 = np.random.multivariate_normal(mean1, Sigma1, tsize0).T
    x2, y2 = np.random.multivariate_normal(mean2, Sigma2, tsize1).T
    x3, y3 = np.random.multivariate_normal(mean1, Sigma1, 100).T
    x4, y4 = np.random.multivariate_normal(mean2, Sigma2, 100).T
    
    data_train = pd.DataFrame({'x': np.concatenate([x1, x2]), 
                               'y': np.concatenate([y1, y2]), 
                               'label': np.concatenate([np.zeros(tsize0), np.ones(tsize1)])})
    
    data_test = pd.DataFrame({'x': np.concatenate([x3, x4]), 
                              'y': np.concatenate([y3, y4]), 
                              'actual': np.concatenate([np.zeros(100), np.ones(100)])})
    
    return data_train, data_test

def calculate_statistics(data_train, data_test):
    meanx0, meany0 = data_train[data_train['label'] == 0][['x', 'y']].mean()
    meanx1, meany1 = data_train[data_train['label'] == 1][['x', 'y']].mean()
    stdx0, stdy0 = data_train[data_train['label'] == 0][['x', 'y']].std()
    stdx1, stdy1 = data_train[data_train['label'] == 1][['x', 'y']].std()
    
    return meanx0, meany0, stdx0, stdy0, meanx1, meany1, stdx1, stdy1

def classify_data(data_test, meanx0, meany0, stdx0, stdy0, meanx1, meany1, stdx1, stdy1):
    prob0 = len(data_train[data_train['label'] == 0]) / len(data_train)
    prob1 = len(data_train[data_train['label'] == 1]) / len(data_train)
    
    data_test['post0'] = norm.pdf(data_test['x'], loc=meanx0, scale=stdx0) * norm.pdf(data_test['y'], loc=meany0, scale=stdy0) * prob0
    data_test['post1'] = norm.pdf(data_test['x'], loc=meanx1, scale=stdx1) * norm.pdf(data_test['y'], loc=meany1, scale=stdy1) * prob1
    
    data_test['predicted'] = np.where(data_test['post0'] > data_test['post1'], 0, 1)
    data_test['color'] = np.where(data_test['predicted'] == 0, 'r', 'b')
    
    return data_test

def evaluate_classification(data_test):
    true_positive = ((data_test['predicted'] == 1) & (data_test['actual'] == 1)).sum()
    true_negative = ((data_test['predicted'] == 0) & (data_test['actual'] == 0)).sum()
    false_positive = ((data_test['predicted'] == 1) & (data_test['actual'] == 0)).sum()
    false_negative = ((data_test['predicted'] == 0) & (data_test['actual'] == 1)).sum()
    
    accuracy = (true_positive + true_negative) / len(data_test)
    error_rate = 1 - accuracy
    recall = true_positive / (true_positive + false_negative)
    precision = true_positive / (true_positive + false_positive)
    
    print("Accuracy: {:.4f}".format(accuracy))
    print("Error Rate: {:.4f}".format(error_rate))
    print("Recall: {:.4f}".format(recall))
    print("Precision: {:.4f}".format(precision))
    
    return accuracy

def plot_data(data_test):
    plt.figure(figsize=(10, 10))
    plt.scatter(data_test['x'], data_test['y'], color=data_test['color'], alpha=0.5, edgecolor='k', marker="x")
    plt.show()

def main():
    data_train, data_test = generate_data(500, 500)
    meanx0, meany0, stdx0, stdy0, meanx1, meany1, stdx1, stdy1 = calculate_statistics(data_train, data_test)
    data_test = classify_data(data_test, meanx0, meany0, stdx0, stdy0, meanx1, meany1, stdx1, stdy1)
    accuracy = evaluate_classification(data_test)
    plot_data(data_test)

if __name__ == "__main__":
    main()
