import numpy as np
from sklearn.svm import LinearSVC
from sklearn.datasets import fetch_20newsgroups
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn import metrics
from naive_feature_selection import NaiveFeatureSelection

# Print testing message
print("Testing NFS ...")

# Define categories for classification
categories = ['sci.med', 'sci.space']

# Specify which parts of the text to remove
remove = ('headers', 'footers', 'quotes')

# Load 20 newsgroups dataset for specified categories
print("Loading 20 newsgroups dataset for categories:")
print(categories if categories else "all")
print()

# Load training and testing data
data_train = fetch_20newsgroups(subset='train', categories=categories,
                                shuffle=True, random_state=42, remove=remove)
data_test = fetch_20newsgroups(subset='test', categories=categories,
                               shuffle=True, random_state=42, remove=remove)

# Extract labels for training and testing data
y_train, y_test = data_train.target, data_test.target

# Extract features from the training data using TF-IDF vectorizer
print("Extracting features from the training data using a sparse vectorizer")
vectorizer = TfidfVectorizer(sublinear_tf=True, max_df=0.5,
                             stop_words='english')
X_train = vectorizer.fit_transform(data_train.data)
print("n_samples: %d, n_features: %d" % X_train.shape)
print()

# Extract features from the test data using the same vectorizer
print("Extracting features from the test data using the same vectorizer")
X_test = vectorizer.transform(data_test.data)
print("n_samples: %d, n_features: %d" % X_test.shape)
print()

# Get feature names from the vectorizer
feature_names = vectorizer.get_feature_names()

# Perform Naive Feature Selection (NFS)
kv = 100  # Number of selected features
nfs = NaiveFeatureSelection(k=kv, alpha=1e-4)
X_new = nfs.fit_transform(X_train, y_train)

# Train Linear SVM classifier on the selected features
clfsv = LinearSVC(random_state=0, tol=1e-5)
clfsv.fit(X_new, y_train == 1)

# Transform test data using NFS
X_testnew = nfs.transform(X_test)

# Predict using the trained model
y_pred_NFS = clfsv.predict(X_testnew)

# Compute accuracy of NFS
score_nfs = metrics.accuracy_score(y_test == 1, y_pred_NFS)
print("NFS accuracy:\t%0.3f" % score_nfs)
print("")

# Print selected features for 'space' category
print('Space features:')
print([feature_names[np.nonzero(nfs.mask_)[0][i]]
       for i in range(kv) if clfsv.coef_[0][i] >= 0])
print()

# Print selected features for 'med' category
print('Med features:')
print([feature_names[np.nonzero(nfs.mask_)[0][i]] for i in range(kv) if clfsv.coef_[0][i] < 0])
print("")
