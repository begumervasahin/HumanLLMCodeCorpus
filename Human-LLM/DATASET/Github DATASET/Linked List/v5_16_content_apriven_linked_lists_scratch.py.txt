class Hedgehog:
    """Class representing a Hedgehog node."""
    def __init__(self, number):
        self.number = number
        self.next = None

class HedgehogList:
    """Class representing a linked list of Hedgehogs."""
    def __init__(self):
        self.head = None

    def add_first(self, number):
        """Add a Hedgehog with the given number to the beginning of the list."""
        new_hedgehog = Hedgehog(number)
        new_hedgehog.next = self.head
        self.head = new_hedgehog

    def add_last(self, number):
        """Add a Hedgehog with the given number to the end of the list."""
        current = self.head
        if current is None:
            self.head = Hedgehog(number)
            return
        while current.next:
            current = current.next
        current.next = Hedgehog(number)

    def get_length(self):
        """Get the length of the list."""
        length = 0
        current = self.head
        while current:
            length += 1
            current = current.next
        return length

    def is_empty(self):
        """Check if the list is empty."""
        return self.head is None

    def find_max(self):
        """Find the maximum value in the list."""
        if self.is_empty():
            return None
        current = self.head
        max_value = current.number
        while current:
            if current.number > max_value:
                max_value = current.number
            current = current.next
        return max_value

    def remove(self, number):
        """Remove the Hedgehog with the given number from the list."""
        if self.is_empty():
            return None
        if self.head.number == number:
            removed = self.head
            self.head = self.head.next
            return removed
        current = self.head
        while current.next:
            if current.next.number == number:
                removed = current.next
                current.next = current.next.next
                return removed
            current = current.next
        return None


def test_list(actual, expected):
    """Test function to compare a linked list with an expected list."""
    item_index = 1
    actual_item = actual
    for item in expected:
        actual_value = actual_item.number if actual_item else None
        err_msg = f"Element number {item_index} should have the value of {item}, got {actual_value}"
        assert actual_item and item == actual_item.number, err_msg
        actual_item = actual_item.next
        item_index += 1


def test_length(linked_list, expected_length):
    """Test function to check the length of a linked list."""
    assert linked_list.get_length() == expected_length, f"Length should be {expected_length}, got {linked_list.get_length()}"


def test_value(actual, expected, msg=""):
    """Test function to check the equality of two values."""
    assert actual == expected, f"{msg} should be {expected}, got {actual}"


def test_max(linked_list, expected):
    """Test function to check the maximum value in a linked list."""
    max_value = linked_list.find_max()
    test_value(max_value, expected, "Wrong maximum")


def create_list(numbers):
    """Helper function to create a linked list from a list of numbers."""
    hedgehog_list = HedgehogList()
    for number in numbers:
        hedgehog_list.add_last(number)
    return hedgehog_list
