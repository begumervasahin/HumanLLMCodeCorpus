import urllib.request
from bs4 import BeautifulSoup
import ssl

def retrieve_url(url, count, position):
    """Retrieve URLs from the given webpage recursively based on count and position.

    Args:
    url (str): The URL of the webpage to start with.
    count (int): The number of times to retrieve URLs.
    position (int): The position of the URL to follow on the webpage.

    Returns:
    None
    """
    # Create SSL context to ignore SSL certificate errors
    ctx = ssl.create_default_context()
    ctx.check_hostname = False
    ctx.verify_mode = ssl.CERT_NONE

    for i in range(count):
        print('Retrieving:', url)
        try:
            # Open and read the webpage
            html = urllib.request.urlopen(url, context=ctx).read()
            soup = BeautifulSoup(html, 'html.parser')

            # Extract all anchor tags from the webpage
            tags = soup('a')

            # Get the URL at the specified position
            url = tags[position - 1].get('href', None)
        except Exception as e:
            print('Error:', e)
            break

if __name__ == "__main__":
    # Input URL, count, and position from the user
    url = input('Enter URL: ')
    count = int(input('Enter count: '))
    position = int(input('Enter position: '))

    # Retrieve URLs recursively
    retrieve_url(url, count, position)
