def inOrderTraverse(root):
    """
    Performs in-order traversal of the binary tree and prints the data.

    Parameters:
        root: The root node of the binary tree.
    """
    temp = root
    stack = []
    while 1:
        while temp:
            stack.append(temp)
            temp = temp.left
        while temp == None and len(stack):
            temp = stack.pop()
            print(temp.data, end=' ')
            temp = temp.right
        if len(stack) == 0 and temp == None:
            break

def convertBT2DLL(root):
    """
    Converts the binary tree to a doubly linked list.

    Parameters:
        root: The root node of the binary tree.

    Returns:
        Node: The head of the doubly linked list.
    """
    temp = root
    stack = []
    prevNode = None
    head = None
    tail = None
    while 1:
        while temp:
            stack.append(temp)
            temp = temp.left
        while temp == None and len(stack):
            temp = stack.pop()
            temp.left = prevNode
            if prevNode:
                prevNode.right = temp
            else:
                head = temp
            prevNode = temp
            print(temp.data, end=' ')
            if temp.right == None:
                tail = temp
            temp = temp.right
        if len(stack) == 0 and temp == None:
            tail.right = head
            head.left = tail
            break
    return head

def llTraversal(root):
    """
    Traverses the doubly linked list and prints the data.

    Parameters:
        root: The head of the doubly linked list.
    """
    temp = root
    start = True
    while temp and (start or temp != root):
