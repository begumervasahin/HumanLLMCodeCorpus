class ListNode:
    """Definition for a singly-linked list node."""
    def __init__(self, value):
        self.val = value  # Value of the node
        self.next = None  # Reference to the next node

class Solution:
    """Class containing the solution for adding two numbers represented as linked lists."""
    def addTwoNumbers(self, l1, l2):
        """
        Add two numbers represented by linked lists.

        Args:
            l1 (ListNode): Head node of the first linked list.
            l2 (ListNode): Head node of the second linked list.

        Returns:
            ListNode: Head node of the resulting linked list representing the sum.
        """
        carry = 0  # Initialize carry to 0
        dummy_head = ListNode(0)  # Create a dummy head node for the result
        current = dummy_head  # Initialize a current pointer to build the result linked list

        # Loop until both input linked lists are traversed and carry is 0
        while l1 or l2 or carry:
            # Add the values of the current nodes and the carry
            if l1:
                carry += l1.val
                l1 = l1.next
            if l2:
                carry += l2.val
                l2 = l2.next

            # Create a new node with the digit of the sum (carry % 10)
            current.next = ListNode(carry % 10)
            current = current.next  # Move the current pointer to the next node
            carry //= 10  # Update the carry for the next iteration

        return dummy_head.next  # Return the head of the resulting linked list
