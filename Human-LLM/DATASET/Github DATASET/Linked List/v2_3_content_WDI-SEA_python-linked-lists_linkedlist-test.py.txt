# Definition of a single node in the linked list
class ListNode:
    def __init__(self, data):
        self.data = data  # Data stored in the node
        self.next = None  # Reference to the next node in the linked list


# Definition of the Linked List data structure
class LinkedList:
    def __init__(self):
        self.head = None  # Initialize an empty linked list with no nodes

    # Method to insert a new node at the beginning of the linked list
    def insert_in_front(self, data):
        new_node = ListNode(data)  # Create a new node with the provided data
        new_node.next = self.head  # Set the next of the new node to the current head
        self.head = new_node  # Update the head to point to the new node

    # Method to insert a new node at the end of the linked list
    def insert_at_end(self, data):
        new_node = ListNode(data)  # Create a new node with the provided data
        if not self.head:  # If the linked list is empty
            self.head = new_node  # Set the new node as the head
            return
        last_node = self.head
        while last_node.next:  # Traverse the linked list to find the last node
            last_node = last_node.next
        last_node.next = new_node  # Set the next of the last node to the new node

    # Method to remove the last node from the linked list
    def remove_last(self):
        if not self.head:  # If the linked list is empty
            return False
        if not self.head.next:  # If there's only one node in the linked list
            self.head = None  # Set the head to None
            return True
        second_last = self.head
        while second_last.next.next:  # Traverse the linked list to find the second last node
            second_last = second_last.next
        second_last.next = None  # Remove the reference to the last node
        return True

    # Method to insert a new node at a specified index in the linked list
    def insert_at_index(self, data, index=0):
        if index == 0:  # If index is 0, insert at the beginning
            self.insert_in_front(data)
            return
        new_node = ListNode(data)  # Create a new node with the provided data
        current = self.head
        for _ in range(index - 1):  # Traverse to the node before the specified index
            if current is None:
                raise IndexError("Index out of range")
            current = current.next
        new_node.next = current.next  # Adjust pointers to insert the new node
        current.next = new_node

    # Method to remove the node at a specified index from the linked list
    def remove_at_index(self, index=None):
        if not self.head:  # If the linked list is empty
            return False
        if index is None:  # If index is not specified
            return False
        if index == 0:  # If removing the first node
            self.head = self.head.next
            return True
        current = self.head
        for _ in range(index - 1):  # Traverse to the node before the specified index
            if current.next is None:
                raise IndexError("Index out of range")
            current = current.next
        if current is None or current.next is None:
            raise IndexError("Index out of range")
        current.next = current.next.next  # Adjust pointers to remove the node
        return True

    # Method to check if the linked list is empty
    def is_empty(self):
        return self.head is None

    # Method to get the length of the linked list
    def __len__(self):
        count = 0
        current = self.head
        while current:
            count += 1
            current = current.next
        return count

    # Method to represent the linked list as a string
    def __str__(self):
        result = "["
        current = self.head
        while current:
            result += str(current.data)  # Append the data of each node to the result
            current = current.next
            if current:
                result += " -> "  # Add arrow between node values
        result += "]"
        return result


if __name__ == "__main__":
    # Testing the LinkedList class
    linked_list = LinkedList()
    print(linked_list.is_empty())  # True

    linked_list.insert_in_front(1)
    linked_list.insert_at_end(2)
    linked_list.insert_at_end(3)
    print(linked_list)  # [1 -> 2 -> 3]
    print(len(linked_list))  # 3

    linked_list.remove_last()
    print(linked_list)  # [1 -> 2]
    print(len(linked_list))  # 2

    linked_list.insert_at_index(4, 1)
    print(linked_list)  # [1 -> 4 -> 2]
    print(len(linked_list))  # 3

    linked_list.remove_at_index(1)
    print(linked_list)  # [1 -> 2]
    print(len(linked_list))  # 2

    print(linked_list.is_empty())  # False
