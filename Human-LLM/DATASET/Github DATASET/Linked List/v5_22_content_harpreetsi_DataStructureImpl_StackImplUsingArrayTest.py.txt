from StackImplUsingArray import Stack
import unittest

class TestStackImplUsingArray(unittest.TestCase):

    def test_push(self):
        """Test push operation."""
        stack = Stack(10)
        self.assertEqual(stack.push(5), "success")

    def test_stack_overflow(self):
        """Test stack overflow."""
        stack = Stack(3)
        stack.push(5)
        stack.push(5)
        stack.push(5)
        self.assertEqual(stack.push(5), "Stack full")

    def test_top_element(self):
        """Test top element retrieval."""
        stack = Stack(10)
        stack.push(10)
        self.assertEqual(stack.top_element(), 10)

    def test_top_element_stack_empty(self):
        """Test top element retrieval from an empty stack."""
        stack = Stack(10)
        self.assertEqual(stack.top_element(), "Stack empty")

    def test_pop(self):
        """Test pop operation."""
        stack = Stack(10)
        stack.push(10)
        self.assertEqual(stack.pop(), 10)

    def test_stack_underflow(self):
        """Test stack underflow."""
        stack = Stack(10)
        self.assertEqual(stack.pop(), "Stack empty")

    def test_is_empty(self):
        """Test stack emptiness."""
        stack = Stack(10)
        self.assertTrue(stack.is_empty())

    def test_is_not_empty(self):
        """Test stack non-emptiness."""
        stack = Stack(10)
        stack.push(10)
        self.assertFalse(stack.is_empty())

    def test_print_empty_stack(self):
        """Test printing an empty stack."""
        stack = Stack(10)
        stack.print()

    def test_print(self):
        """Test printing a non-empty stack."""
        stack = Stack(10)
        stack.push(10)
        stack.push(20)
        stack.print()

if __name__ == '__main__':
    unittest.main()
