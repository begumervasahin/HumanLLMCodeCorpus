class Node:
    def __init__(self, datavalue=None):
        self.datavalue = datavalue
        self.nextvalue = None

class LinkedList:
    def __init__(self):
        self.headvalue = None

    def printing(self):
        printvalue = self.headvalue
        while printvalue is not None:
            print(printvalue.datavalue)
            printvalue = printvalue.nextvalue

    def begin(self, data_a):
        new_node = Node(data_a)
        new_node.nextvalue = self.headvalue
        self.headvalue = new_node

    def between(self, nodes, data_a):
        new_node = Node(data_a)
        new_node.nextvalue = nodes.nextvalue
        nodes.nextvalue = new_node

    def end(self, data_a):
        newest_node = Node(data_a)
        if self.headvalue is None:
            self.headvalue = newest_node
            return
        last_node = self.headvalue
        while last_node.nextvalue is not None:
            last_node = last_node.nextvalue
        last_node.nextvalue = newest_node

    def delete(self, data_a):
        current = self.headvalue
        if current is not None and current.datavalue == data_a:
            self.headvalue = current.nextvalue
            current = None
            return
        prev = None
        while current is not None and current.datavalue != data_a:
            prev = current
            current = current.nextvalue
        if current is None:
            return
        prev.nextvalue = current.nextvalue
        current = None
