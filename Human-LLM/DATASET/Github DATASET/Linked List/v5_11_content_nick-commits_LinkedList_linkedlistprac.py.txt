class Node:
    def __init__(self, data=None):
        """
        Initializes a node with the given data.

        Parameters:
            data: The data to be stored in the node.
        """
        self.data = data
        self.next = None

class LinkedList:
    def __init__(self):
        """
        Initializes a linked list.
        """
        self.head = Node()  # Dummy head node

    def append(self, data):
        """
        Appends a new node with the given data to the end of the linked list.

        Parameters:
            data: The data to be appended to the linked list.
        """
        new_node = Node(data)
        cur = self.head
        while cur.next:
            cur = cur.next
        cur.next = new_node

    def length(self):
        """
        Computes the length of the linked list.

        Returns:
            int: The length of the linked list.
        """
        cur = self.head
        total = 0
        while cur.next:
            total += 1
            cur = cur.next
        return total

    def display(self):
        """
        Displays the elements of the linked list.
        """
        cur_node = self.head.next  # Skip the dummy head node
        elems = []
        while cur_node:
            elems.append(cur_node.data)
            cur_node = cur_node.next
        print(elems)

    def get(self, index):
        """
        Gets the data at the given index in the linked list.

        Parameters:
            index: The index of the node to retrieve.

        Returns:
            Any: The data stored in the node at the given index.
        """
        if index >= self.length():
            print('Index out of range')
            return None
        cur_idx = 0
        cur_node = self.head.next  # Skip the dummy head node
        while cur_node:
            if cur_idx == index:
                return cur_node.data
            cur_node = cur_node.next
            cur_idx += 1

    def erase(self, index):
        """
        Removes the node at the given index from the linked list.

        Parameters:
            index: The index of the node to remove.
        """
        if index >= self.length():
            print('Index out of range')
            return
        cur_idx = 0
        cur_node = self.head
        while cur_node.next:
            last_node = cur_node
            cur_node = cur_node.next
            if cur_idx == index:
                last_node.next = cur_node.next
                return
            cur_idx += 1

# Example usage
my_list = LinkedList()
my_list.append(2)
my_list.append(1)
my_list.erase(1)
my_list.display()
