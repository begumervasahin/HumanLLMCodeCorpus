import requests
from bs4 import BeautifulSoup

# Output file name
output_file = "rentpad_list.csv"

def scrape_rentpad(city):
    # Payload for POST request
    payload = {
        "a": "31",
        "cityName": city,
        "propertyTypeIDs": [2, 3, 5],
        "furnishTypeIDs": [1, 2, 3],
        "placeIDs": [],
        "statusTypeIDs": [],
        "amenityIDs": [],
        "longMonthRateLow": "0",
        "longMonthRateHigh": "30,000",
        "numBedroomsLow": "0",
        "numBedroomsHigh": "0",
        "itemsPerPage": "1000",
        "pageNumber": "1",
        "lengthOfStay": "",
        "ham": "ham"
    }

    try:
        # Send POST request to Rentpad
        response = requests.post("https://rentpad.com.ph/long-term-rentals/mandaluyong/apartment", data=payload)
        response.raise_for_status()
    except requests.RequestException as e:
        print("Error fetching data:", e)
        return

    # Parse HTML response
    html = BeautifulSoup(response.text, 'html.parser')
    prices = html.find_all(itemprop="price")
    apartments = html.find_all(itemprop="name")
    links = html.find_all(itemprop="offers")

    # Write data to CSV file
    with open(output_file, "w+") as output:
        i = 0
        for apartment in apartments:
            if apartment.string is None:
                continue
            price = prices[i].string[1:].replace(",", "")  # Remove currency symbol and commas
            link = links[i].find('a').get('href')
            output.write(f"{apartment.string},{price},{link}\n")
            print(f"{apartment.string},{price},{link}")  # Print details to console
            i += 1

# Scrape Rentpad for Mandaluyong city
scrape_rentpad("Mandaluyong")
