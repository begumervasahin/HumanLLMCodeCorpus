from StackImplUsingArray import Stack
import unittest

class TestStackImplUsingArray(unittest.TestCase):
    
    def test_push(self):
        """Test push operation."""
        myStack = Stack(10)
        self.assertEqual(myStack.push(5), "success")
        
    def test_stackOverflow(self):
        """Test stack overflow."""
        myStack = Stack(3)
        myStack.push(5)
        myStack.push(5)
        myStack.push(5)
        self.assertEqual(myStack.push(5), "Stack full")
        
    def test_topElement(self):
        """Test top element retrieval."""
        myStack = Stack(10)
        myStack.push(10)
        self.assertEqual(myStack.topElement(), 10)
        
    def test_topElement_stackEmpty(self):
        """Test top element retrieval from an empty stack."""
        myStack = Stack(10)
        self.assertEqual(myStack.topElement(), "Stack empty")
        
    def test_pop(self):
        """Test pop operation."""
        myStack = Stack(10)
        myStack.push(10)
        self.assertEqual(myStack.pop(), 10)
        
    def test_stackUnderflow(self):
        """Test stack underflow."""
        myStack = Stack(10)
        self.assertEqual(myStack.pop(), "Stack empty")
        
    def test_isEmpty(self):
        """Test stack emptiness."""
        myStack = Stack(10)
        self.assertTrue(myStack.isEmpty())
        
    def test_isNotEmpty(self):
        """Test stack non-emptiness."""
        myStack = Stack(10)
        myStack.push(10)
        self.assertFalse(myStack.isEmpty())
        
    def test_print_empty_stack(self):
        """Test printing an empty stack."""
        myStack = Stack(10)
        myStack.print()
        
    def test_print(self):
        """Test printing a non-empty stack."""
        myStack = Stack(10)
        myStack.push(10)
        myStack.push(20)
        myStack.print()

if __name__ == '__main__':
    unittest.main()
