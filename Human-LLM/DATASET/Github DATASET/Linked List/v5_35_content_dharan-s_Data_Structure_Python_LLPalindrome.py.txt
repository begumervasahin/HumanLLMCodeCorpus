class Node:
    """
    Class to represent a node in a linked list.
    """

    def __init__(self, value):
        """
        Initialize a node with the given value.

        Args:
            value: The value to be stored in the node.
        """
        self.info = value
        self.next = None


class LinkedList:
    """
    Class to manage a linked list and perform various operations on it.
    """

    def __init__(self):
        """
        Initialize the linked list with head and tail pointers.
        """
        self.head = None

    def insert_at_end(self, data):
        """
        Insert a new node with the given data at the end of the linked list.

        Args:
            data: The data to be stored in the new node.
        """
        temp = Node(data)
        if not self.head:
            self.head = temp
        else:
            p = self.head
            while p.next:
                p = p.next
            p.next = temp

    def display(self):
        """
        Display the elements of the linked list.
        """
        if not self.head:
            print("List is empty")
        else:
            p = self.head
            while p:
                print(p.info, end=' ')
                p = p.next
            print()

    def create(self):
        """
        Create a linked list by taking input from the user.
        """
        num_nodes = int(input("Enter the number of nodes: "))
        for _ in range(num_nodes):
            data = int(input("Enter data to be inserted: "))
            self.insert_at_end(data)

    def palindrome(self):
        """
        Check if the linked list is a palindrome.
        """
        if not self.head:
            print("List is Empty")
            return

        # Reverse the linked list
        prev = None
        p = self.head
        while p:
            next_node = p.next
            p.next = prev
            prev = p
            p = next_node
        self.head = prev

        # Check if the reversed list matches the original list
        p1 = self.head
        p2 = self.head
        while p2.next:
            p1 = p1.next
            p2 = p2.next
            if p2.next:
                p2 = p2.next

        while p1:
            if p1.info != prev.info:
                print("Not a palindrome")
                return
            p1 = p1.next
            prev = prev.next
        print("Palindrome")
