from LinkedList import Node, LinkedList

def locate_middle_node(linked_list):
    """
    Locates the middle node of the given linked list.

    Parameters:
    linked_list: The linked list to operate on.

    Returns:
    Node: The middle node.
    """
    slow_cursor = linked_list.head
    fast_cursor = linked_list.head
    while fast_cursor and fast_cursor.next_node:
        slow_cursor = slow_cursor.next_node
        fast_cursor = fast_cursor.next_node.next_node
    return slow_cursor

def reverse_second_half(middle_node):
    """
    Reverses the second half of the linked list starting from the given middle node.

    Parameters:
    middle_node: The middle node of the linked list.

    Returns:
    Node: The head node of the reversed second half.
    """
    current = middle_node
    previous = None
    next_node = None
    while current:
        next_node = current.next_node
        current.next_node = previous
        previous = current
        current = next_node
    return previous
