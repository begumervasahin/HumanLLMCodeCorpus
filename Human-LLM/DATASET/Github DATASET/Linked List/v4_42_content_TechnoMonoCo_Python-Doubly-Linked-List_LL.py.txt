class LinkedListNode:
    def __init__(self, data, post=None, pred=None):
        self.data = data
        self.post = post
        self.pred = pred

    def remove(self, data):
        # Removing a node with given data from the linked list
        if self.pred is None:
            if self.data == data:
                # If the node to be removed is the first node
                if self.post is None:
                    # If there's only one node in the list
                    self.data = None
                    return 0
                # Otherwise, update the data and post pointer of the first node
                hold = self.post
                self.data = self.post.data
                self.post = self.post.post
                del hold
                return 0
            elif self.post is None:
                # If the node to be removed is the last node
                return -1
            else:
                # If the node to be removed is in the middle
                return self.post.remove(data)
        else:
            # If the current node is not the first node
            if self.post is None:
                # If the node to be removed is the last node
                if self.data == data:
                    # If the last node is the one to be removed
                    hold = self
                    temp = self.pred
                    temp.post = None
                    del hold
                    return 0
                else:
                    return -1
            else:
                # If the node to be removed is in the middle or not the last one
                if self.data == data:
                    # If the node to be removed is found
                    hold = self
                    temp = self.pred
                    temp.post = self.post
                    del hold
                    return 0
                else:
                    return self.post.remove(data)

    def insert(self, data):
        # Inserting a new node with the given data
        if data < self.data:
            # If the data is less than the current node's data, insert before it
            hold = LinkedListNode(data, self, None)
            self = hold
        elif self.post is None:
            # If the current node is the last one, insert after it
            hold = LinkedListNode(data, None, self)
            self.post = hold
        else:
            # Otherwise, recursively insert
            self.post.insert(data)

    def contain(self, s):
        # Helper method to recursively print the contents of the linked list
        if self.data is None:
            print(s)
        else:
            s = s + ", " + str(self.data)
            if self.post is None:
                print(s)
            else:
                self.post.contain(s)

    def contains(self):
        # Method to print the contents of the linked list
        if self.post is None:
            if self.data is None:
                print("Empty.")
            else:
                print(self.data)
        else:
            self.post.contain(str(self.data))
