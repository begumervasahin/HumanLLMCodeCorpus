class HashTable:
    def __init__(self):
        # Initialize the hash table with 10000 empty slots
        self.table = [None] * 10000

    def store(self, string):
        """Stores a string in the hash table."""
        # Calculate the hash value for the string
        hash_value = self.calculate_hash_value(string)
        # If the slot is empty, create a new list and store the string
        if self.table[hash_value] is None:
            self.table[hash_value] = [string]
        # If the slot is not empty, append the string to the existing list
        else:
            self.table[hash_value].append(string)

    def lookup(self, string):
        """Returns the hash value if the string is already in the hash table. Returns -1 otherwise."""
        # Calculate the hash value for the string
        hash_value = self.calculate_hash_value(string)
        # Check if the slot is not empty and the string is in the list
        if self.table[hash_value] and string in self.table[hash_value]:
            return hash_value
        # If the string is not found, return -1
        return -1

    def calculate_hash_value(self, string):
        """Helper function to calculate a hash value from a string."""
        # Check if the string is not empty and has at least two characters
        if string and len(string) >= 2:
            # Calculate the hash value based on the ASCII values of the first two characters
            return ord(string[0]) * 100 + ord(string[1])
        # If the string is empty or too short, return -1
        return -1

# Test the HashTable class
hash_table = HashTable()
print(hash_table.calculate_hash_value('UDACITY'))
print(hash_table.lookup('UDACITY'))
hash_table.store('UDACITY')
print(hash_table.lookup('UDACITY'))
hash_table.store('UDACIOUS')
print(hash_table.lookup('UDACIOUS'))
