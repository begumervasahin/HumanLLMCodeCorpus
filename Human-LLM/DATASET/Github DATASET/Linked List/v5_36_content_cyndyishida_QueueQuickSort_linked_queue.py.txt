class Node:
    """A lightweight class for storing a singly linked node."""

    __slots__ = 'val', 'next'

    def __init__(self, val, next_node=None):
        """Initialize a Node with a value and a reference to the next node.

        Args:
            val: The value to be stored in the node.
            next_node: Reference to the next node (default is None).
        """
        self.val = val
        self.next = next_node

    def __lt__(self, other):
        """Less than comparison between nodes based on their values."""
        return self.val < other.val

    def __le__(self, other):
        """Less than or equal comparison between nodes based on their values."""
        return self.val <= other.val


class LinkedQueue:
    """FIFO queue implementation using a singly linked list for storage."""

    def __init__(self):
        """Create an empty queue."""
        self.head = None
        self.tail = None
        self.size = 0

    def __str__(self):
        """Return a string representation of the queue."""
        values = [str(node.val) for node in self]
        return ", ".join(values)

    __repr__ = __str__

    def __len__(self):
        """Return the number of elements in the queue."""
        return self.size

    def is_empty(self):
        """Check if the queue is empty."""
        return self.size == 0

    def dequeue(self):
        """Remove and return the first element of the queue (FIFO)."""
        if self.is_empty():
            raise IndexError("Queue is empty")
        answer = self.head.val
        self.head = self.head.next
        self.size -= 1
        if self.is_empty():
            self.tail = None
        return answer

    def enqueue(self, element):
        """Add an element to the back of the queue."""
        newest = Node(element)
        if self.is_empty():
            self.head = newest
        else:
            self.tail.next = newest
        self.tail = newest
        self.size += 1

    def __getitem__(self, index):
        """Get the element at the specified index."""
        current = self.head
        for _ in range(index):
            current = current.next
        return current

    def __setitem__(self, index, val):
        """Set the value of the element at the specified index."""
        current = self.head
        for _ in range(index):
            current = current.next
        current.val = val
