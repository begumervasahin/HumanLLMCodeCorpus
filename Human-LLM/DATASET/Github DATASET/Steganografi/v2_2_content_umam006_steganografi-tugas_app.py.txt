# Import necessary libraries
from LSBSteg import LSBSteg  # Assuming LSBSteg class is implemented in the LSBSteg module
import cv2  # OpenCV library for image processing
from flask import Flask, render_template, request  # Flask for creating web application
import numpy as np  # NumPy for array manipulation

# Create a Flask application
app = Flask(__name__, instance_relative_config=True)

# Configure the Flask application
app.config.update(
    DEBUG=True,  # Enable debug mode
    SECRET_KEY='secret_xxx'  # Secret key for session security
)

# Route for the homepage
@app.route('/')
def index():
    return render_template("index.html")  # Render the HTML template for the homepage

# Route for encoding text into an image
@app.route('/api/encode', methods=['POST'])
def encode():
    if request.method == 'POST':  # Check if the request method is POST
        file = request.files['file_gambar']  # Get the uploaded image file
        text = request.form['text']  # Get the text to be encoded
        img = np.fromstring(file.read(), np.uint8)  # Convert the image file into a NumPy array
        steg = LSBSteg(cv2.imdecode(img, cv2.IMREAD_UNCHANGED))  # Create an LSBSteg object
        img_encoded = steg.encode_text(text)  # Encode the text into the image
        cv2.imwrite("my_new_image.png", img_encoded)  # Save the encoded image
        return render_template("index.html", pesan="berhasil menyisipkan")  # Render the homepage with success message

# Route for decoding text from an image
@app.route('/api/decode', methods=['POST'])
def decode():
    file = request.files['file_gambar']  # Get the uploaded image file
    img = np.fromstring(file.read(), np.uint8)  # Convert the image file into a NumPy array
    im = cv2.imdecode(img, cv2.IMREAD_UNCHANGED)  # Decode the image
    steg = LSBSteg(im)  # Create an LSBSteg object
    pesan = steg.decode_text().decode("utf-8")  # Decode the text from the image
    print(pesan)  # Print the decoded message
    return render_template("index.html", pesan=pesan)  # Render the homepage with the decoded message

# Main block to run the Flask application
if __name__ == '__main__':
    app.run()  # Run the Flask application
