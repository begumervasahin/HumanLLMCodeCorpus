# Import necessary modules
import os
import sys

# Check if this script is being run directly
if __name__ == "__main__":
    # Set default Django settings module
    os.environ.setdefault("DJANGO_SETTINGS_MODULE", "root.settings")

    try:
        # Try to import Django's management commands
        from django.core.management import execute_from_command_line
    except ImportError as exc:
        # If Django is not found, raise an ImportError with instructions
        raise ImportError(
            "Couldn't import Django. Please ensure it's installed and "
            "available on your PYTHONPATH environment variable. Also, check if you "
            "forgot to activate a virtual environment."
        ) from exc

    # Execute Django management commands from the command line arguments
    execute_from_command_line(sys.argv)
