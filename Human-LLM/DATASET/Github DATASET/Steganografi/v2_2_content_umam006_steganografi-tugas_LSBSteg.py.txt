import cv2
import numpy as np
import argparse

# Exception class for steganography-related errors
class SteganographyException(Exception):
    pass

# Class for handling LSB steganography
class LSBSteg:
    def __init__(self, im):
        self.image = im
        self.height, self.width, self.nbchannels = im.shape
        self.size = self.width * self.height
        self.maskONEValues = [1, 2, 4, 8, 16, 32, 64, 128]
        self.maskZEROValues = [254, 253, 251, 247, 239, 223, 191, 127]
        self.curwidth = 0
        self.curheight = 0
        self.curchan = 0
        self.maskONE = self.maskONEValues.pop(0)
        self.maskZERO = self.maskZEROValues.pop(0)

    # Function to embed binary data into the image
    def put_binary_value(self, bits):
        for bit in bits:
            pixel_value = list(self.image[self.curheight, self.curwidth])

            # Embed the bit into the LSB of the pixel value
            if int(bit) == 1:
                pixel_value[self.curchan] |= self.maskONE
            else:
                pixel_value[self.curchan] &= self.maskZERO

            # Update the pixel value in the image
            self.image[self.curheight, self.curwidth] = tuple(pixel_value)
            self._move_to_next_slot()

    # Move to the next available slot for embedding
    def _move_to_next_slot(self):
        if self.curchan == self.nbchannels - 1:
            self.curchan = 0
            if self.curwidth == self.width - 1:
                self.curwidth = 0
                if self.curheight == self.height - 1:
                    self.curheight = 0
                    if self.maskONE == 128:
                        raise SteganographyException("No available slot remaining (image filled)")
                    else:
                        self.maskONE = self.maskONEValues.pop(0)
                        self.maskZERO = self.maskZEROValues.pop(0)
                else:
                    self.curheight += 1
            else:
                self.curwidth += 1
        else:
            self.curchan += 1

    # Read a single bit from the image
    def read_bit(self):
        pixel_value = self.image[self.curheight, self.curwidth][self.curchan]
        bit_value = int(pixel_value) & self.maskONE
        self._move_to_next_slot()
        return "1" if bit_value > 0 else "0"

    # Read a byte (8 bits) from the image
    def read_byte(self):
        return self.read_bits(8)

    # Read specified number of bits from the image
    def read_bits(self, nb):
        bits = ""
        for _ in range(nb):
            bits += self.read_bit()
        return bits

    # Embed text message into the image
    def encode_text(self, txt):
        txt_length = len(txt)
        txt_length_bin = bin(txt_length)[2:].zfill(16)  # Convert text length to binary (16 bits)
        self.put_binary_value(txt_length_bin)  # Embed text length into the image
        for char in txt:
            char_bin = bin(ord(char))[2:].zfill(8)  # Convert character to binary (8 bits)
            self.put_binary_value(char_bin)  # Embed each character into the image
        return self.image

    # Extract text message from the image
    def decode_text(self):
        txt_length_bin = self.read_bits(16)  # Read the embedded text length in binary (16 bits)
        txt_length = int(txt_length_bin, 2)  # Convert binary text length to integer
        decoded_text = ""
        for _ in range(txt_length):
            char_bin = self.read_byte()  # Read each character in binary (8 bits)
            decoded_text += chr(int(char_bin, 2))  # Convert binary to character and append to the decoded text
        return decoded_text

    # Embed an image into another image
    def encode_image(self, imtohide):
        # Get dimensions of the image to be embedded
        emb_height, emb_width, emb_channels = imtohide.shape

        # Check if carrier image is large enough to hold the embedded image
        if self.width * self.height * self.nbchannels < emb_width * emb_height * emb_channels:
            raise SteganographyException("Carrier image not big enough to hold all the data for steganography")

        # Embed dimensions of the image to be embedded
        self.put_binary_value(bin(emb_width)[2:].zfill(16))  # Width of the embedded image
        self.put_binary_value(bin(emb_height)[2:].zfill(16))  # Height of the embedded image

        # Embed pixel values of the image to be embedded
        for i in range(emb_height):
            for j in range(emb_width):
                for chan in range(emb_channels):
                    val_bin = bin(imtohide[i, j][chan])[2:].zfill(8)  # Convert pixel value to binary (8 bits)
                    self.put_binary_value(val_bin)  # Embed each pixel value into the carrier image
        return self.image

    # Extract an image from another image
    def decode_image(self):
        emb_width = int(self.read_bits(16), 2)  # Read embedded image width (16 bits)
        emb_height = int(self.read_bits(16), 2)  # Read embedded image height (16 bits)
        extracted_image = np.zeros((emb_height, emb_width, 3), np.uint8)  # Initialize extracted image

        # Extract pixel values of the embedded image
        for i in range(emb_height):
            for j in range(emb_width):
                for chan in range(extracted_image.shape[2]):
                    pixel_value_bin = self.read_byte()  # Read pixel value in binary (8 bits)
                    extracted_image[i, j][chan] = int(pixel_value_bin, 2)  # Set pixel value in extracted image
        return extracted_image

    # Embed binary data into the image
    def encode_binary(self, data):
        data_length = len(data)
        if self.width * self.height * self.nbchannels < data_length + 64:
            raise SteganographyException("Carrier image not big enough to hold all the data for steganography")
        self.put_binary_value(bin(data_length)[2:].zfill(64))  # Embed data length into the image (64 bits)
        for byte in data:
            byte_bin = bin(byte)[2:].zfill(8)  # Convert byte to binary (8 bits)
            self.put_binary_value(byte_bin)  # Embed each byte of data into the image
        return self.image

    # Extract binary data from the image
    def decode_binary(self):
        data_length_bin = self.read_bits(64)  # Read embedded data length in binary (64 bits)
        data_length = int(data_length_bin, 2)
