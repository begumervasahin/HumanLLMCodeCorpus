import argparse
import math
import struct
import wave
from scipy.io import wavfile
import numpy as np
from numpy import fft

# Constants
SAMPLE_RATE = 44100
NYQUIST_RATE = SAMPLE_RATE / 2.0
FFT_LENGTH = 512

def calculate_lowpass_coefs(cutoff):
    """
    Calculate lowpass filter coefficients.

    Args:
    - cutoff (float): Cutoff frequency for the lowpass filter.

    Returns:
    - impulse_response (list): Impulse response of the lowpass filter.
    """
    cutoff /= (NYQUIST_RATE / (FFT_LENGTH / 2.0))
    mask = []
    negatives = []
    l = FFT_LENGTH // 2
    
    # Generate lowpass filter mask
    for f in range(0, l + 1):
        rampdown = 1.0
        if f > cutoff:
            rampdown = 0
        mask.append(rampdown)
        if 0 < f < l:
            negatives.append(rampdown)
    
    # Reverse and combine the mask
    negatives.reverse()
    mask = mask + negatives
    
    # Compute impulse response
    impulse_response = fft.ifft(mask).real.tolist()
    left = impulse_response[:FFT_LENGTH // 2]
    right = impulse_response[FFT_LENGTH // 2:]
    impulse_response = right + left
    
    # Apply Hann window to impulse response
    b = FFT_LENGTH // 2
    for n in range(0, b):
        impulse_response[n] *= (n + 0.0) / b
    for n in range(b + 1, FFT_LENGTH):
        impulse_response[n] *= (FFT_LENGTH - n + 0.0) / b
        
    return impulse_response

def apply_lowpass_filter(original, cutoff):
    """
    Apply lowpass filter to the input signal.

    Args:
    - original (numpy.ndarray): Original signal.
    - cutoff (float): Cutoff frequency for the lowpass filter.

    Returns:
    - filtered_signal (numpy.ndarray): Signal after applying lowpass filter.
    """
    coefs = calculate_lowpass_coefs(cutoff)
    return np.convolve(original, coefs)

def convert_to_single_channel(input_file, apply_filter):
    """
    Convert multi-channel audio to single channel.

    Args:
    - input_file (str): Path to the input audio file.
    - apply_filter (bool): Whether to apply lowpass filter.

    Returns:
    - None
    """
    rate, audio = wavfile.read(input_file)
    
    try:
        num_channels = len(audio[0])
        mono_channel_frame = np.array([0] * len(audio[..., 0])).astype(np.float)
        
        for i in range(num_channels):
            mono_channel_frame += 1 / num_channels * audio[..., i].astype(np.float)
        
        if apply_filter:
            mono_channel_frame = 2 * apply_lowpass_filter(1 / 2 * mono_channel_frame, 4500)
        
        output_file = 'aux' + input_file
        wavfile.write(output_file, rate, mono_channel_frame.astype(np.int16))
        print('Step - Successful')
    
    except Exception as e:
        print(f'Exception occurred: {e}')
        
        if apply_filter:
            audio = 2 * apply_lowpass_filter(1 / 2 * audio, 4500)
        
        output_file = 'aux' + input_file
        wavfile.write(output_file, rate, audio.astype(np.int16))
        print('Step - Successful, but encountered exception')

def steganography_method_1(input_audio, input_text, output_file):
    """
    Perform steganography using method 1.

    Args:
    - input_audio (str): Path to the input audio file.
    - input_text (str): Path to the input text file.
    - output_file (str): Path to the output stego audio file.

    Returns:
    - None
    """
    song = wave.open(input_audio, mode='rb')
    frame_bytes = bytearray(list(song.readframes(song.getnframes())))
    
    with open(input_text, 'r') as file:
        string = file.read().replace('\n', '')
    
    string = string + int((len(frame_bytes) - (len(string) * 8 * 8)) / 8) * ' '
    bits = list(map(int, ''.join([bin(ord(i)).lstrip('0b').rjust(8, '0') for i in string])))
    
    if len(bits) < song.getnframes():
        for i, bit in enumerate(bits):
            frame_bytes[i] = (frame_bytes[i] & 254) | bit
        frame_modified = bytes(frame_bytes)
        
        with wave.open(output_file, 'wb') as fd:
            fd.setparams(song.getparams())
            fd.writeframes(frame_modified)
        
        song.close()
    else:
        print("Bits overflow for steganography, but still embedding part of the message in your audio carrier!")
        contor = 0
        
        for i, bit in enumerate(bits):
            if contor < song.getnframes():
                frame_bytes[i] = (frame_bytes[i] & 254) | bit
                contor += 1
            else:
                break
        
        frame_modified = bytes(frame_bytes)
        
        with wave.open(output_file, 'wb') as fd:
            fd.setparams(song.getparams())
            fd.writeframes(frame_modified)
        
        song.close()

def steganography_method_2(input_audio1, input_audio2, output_file):
    """
    Perform steganography using method 2.

    Args:
    - input_audio1 (str): Path to the input audio file 1.
    - input_audio2 (str): Path to the input audio file 2.
    - output_file (str): Path to the output stego audio file.

    Returns:
    - None
    """
    baseband1 = wave.open('aux' + input_audio1, "r")
    baseband2 = wave.open('aux' + input_audio2, "r")
    amsc = wave.open(output_file, "w")
    
    for f in [amsc]:
        f.setnchannels(1)
        f.setsampwidth(2)
        f.setframerate(44100)
    
    contor = 0
    
    for n in range(0, baseband1.getnframes()):
        base = (struct.unpack('h', baseband1.readframes(1))[0] / 32768.0) / 2
        
        if contor < baseband2.getnframes():
            encodFrame = struct.unpack('h', baseband2.readframes(1))[0] / 32768.0
            carrier_sample = math.cos(22050.0 * (contor / 44100.0) * math.pi * 2)
            base += encodFrame * carrier_sample / 4
            contor += 1
        
        amsc.writeframes(struct.pack('h', int(base * 32767)))

def steganography_method_3(input_audio1, input_audio2, output_file):
    """
    Perform steganography using method 3.

    Args:
    - input_audio1 (str): Path to the input audio file 1.
    - input_audio2 (str): Path to the input audio file 2.
    - output_file (str): Path to the output stego audio file.

    Returns:
    - None
    """
    rate1, audio1 = wavfile.read(input_audio1)
    rate2, audio2 = wavfile.read('aux' + input_audio2)
    left = audio1[..., 0].copy()
    right = audio1[..., 1].copy()
    left[0] = right[0] = len(audio2) / 1000.0
    left[1] = right[1] = len(audio2) % 1000.0
    contor = 4
    add = int((len(left) / len(audio2) - 1))
    
    if add >= 6:
        print("OK")
        index = 0
        
        while index < len(audio2):
            aux = int(np.abs(audio2[index]))
            left[contor + 5] = np.sign(left[contor + 5]) * ((int(np.abs(left[contor + 5]) / 10) * 10) + 1 + np.sign(audio2[index]))
            right[contor + 4] = np.sign(right[contor + 4]) * ((int(np.abs(right[contor + 4]) / 10) * 10) + int(np.abs(aux) % 10))
            aux = int(aux / 10)
            left[contor + 3] = np.sign(left[contor + 3]) * ((int(np.abs(left[contor + 3]) / 10) * 10) + int(np.abs(aux) % 10))
            aux = int(aux / 10)
            right[contor + 2] = np.sign(right[contor + 2]) * ((int(np.abs(right[contor + 2]) / 10) * 10) + int(np.abs(aux) % 10))
            aux = int(aux / 10)
            left[contor + 1] = np.sign(left[contor + 1]) * ((int(np.abs(left[contor + 1]) / 10) * 10) + int(np.abs(aux) % 10))
            aux = int(aux / 10)
            right[contor] = np.sign(right[contor]) * ((int(np.abs(right[contor]) / 10) * 10) + int(np.abs(aux) % 10))
            contor += add
            index += 1
        
        audiox = np.column_stack((left, right)).astype(np.int16)
        wavfile.write(output_file, rate1, audiox)
    else:
        print('Your carrier should be min 6x longer than your message for this kind of steganography')

def parse_arguments():
    """
    Parse command line arguments.

    Returns:
    - Tuple containing parsed arguments.
    """
    parser = argparse.ArgumentParser()
    parser.add_argument("INPUT1", help="Name of the wave file")
    parser.add_argument("INPUT2", help="Name of the wave or txt file")
    parser.add_argument("-t", "--text", help="Input text.", action='store_true')
    parser.add_argument("-s", "--sound", help="Input sound for AM modulation.", action='store_true')
    parser.add_argument("-c", "--sCrypto", help="Input sound for crypto-steganography.", action='store_true')
    parser.add_argument("-o", "--output", help="Name of the output wav file. Default value: out.wav).")
    args = parser.parse_args()
    
    text_flag = True
    sound_flag = False
    sCrypto_flag = False
    output = "out.wav"
    operation = "Text"
    count = 0
    
    if args.text:
        text_flag = args.text
        count += 1
    if args.sound:
        sound_flag = args.sound
        operation = "Sound"
        count += 1
        text_flag = False
    if args.sCrypto:
        sCrypto_flag = args.sCrypto
        operation = "Crypto"
        count += 1
        text_flag = False
    if args.output:
        output = args.output
        
    print('Input file1: %s' % args.INPUT1)
    print('Input file2: %s' % args.INPUT2)
    print('Operation: %s' % operation)
    print('Output: %s' % output)
    
    if count != 1:
        print('Error: Exactly one operation should be selected!')
        text_flag = sound_flag = sCrypto_flag = False
        
    return (args.INPUT1, args.INPUT2, output, text_flag, sound_flag, sCrypto_flag)

def main():
    """
    Main function to execute the selected operation.
    """
    args = parse_arguments()
    
    if args[3]:
        steganography_method_1(args[0], args[1], args[2])
    elif args[4]:
        convert_to_single_channel(args[1], True)
        convert_to_single_channel(args[0], False)
        steganography_method_2(args[0], args[1], args[2])
    elif args[5]:
        convert_to_single_channel(args[1], False)
        steganography_method_3(args[0], args[1], args[2])

if __name__ == '__main__':
    main()
