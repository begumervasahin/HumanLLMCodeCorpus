def heap_sort(A, verbose=0, desc=0):
    # Replace the import statement and the implementation of the swap function with the actual ones
    import common as c

    def heapify(i, n):
        parent = i
        ch_1 = i * 2 + 1
        ch_2 = ch_1 + 1
        while (ch_1 < n and A[ch_1] > A[parent]):
            parent = ch_1
        while (ch_2 < n and A[ch_2] > A[parent]):
            parent = ch_2
        if verbose == 2:
            print("  sub:", i, n, " :: ", A)
        if i != parent:
            A[i], A[parent] = c.swap(A[i], A[parent])  # Replace with the actual swap function
            heapify(parent, n - 1)

    def h_sort():
        n = len(A)
        for i in range((n // 2) - 1, -1, -1):
            heapify(i, n)
            if verbose:
                print("iter")
        for i in range(n - 1, 0, -1):
            A[0], A[i] = c.swap(A[0], A[i])  # Replace with the actual swap function
            heapify(0, i)
            if verbose:
                print("iter")

    h_sort()
    if desc:
        A = A[::-1]
    return A
