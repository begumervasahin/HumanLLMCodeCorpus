def quicksort(lista):
    """
    Sorts a list using the quicksort algorithm.

    Args:
        lista (list): The list to be sorted.

    Returns:
        list: The sorted list.
    """
    # Base case: if the list has 1 or fewer elements, it is already sorted
    if len(lista) <= 1:
        return lista

    pivot = lista[0]
    # Divide the list into three sublists: less than, equal to, and greater than the pivot
    equal = [x for x in lista if x == pivot]
    less = [x for x in lista if x < pivot]
    greater = [x for x in lista if x > pivot]

    # Recursively sort the less and greater sublists and concatenate them with the equal sublist
    return quicksort(less) + equal + quicksort(greater)


def selection_sort(v):
    """
    Sorts a list using the selection sort algorithm.

    Args:
        v (list): The list to be sorted.

    Returns:
        list: The sorted list.
    """
    sorted_list = []
    # Repeat the process until the original list is empty
    while v:
        min_value = min(v)  # Find the minimum value in the remaining unsorted list
        sorted_list.append(min_value)  # Append the minimum value to the sorted list
        v.remove(min_value)  # Remove the minimum value from the original list
    return sorted_list


def merge(left, right):
    """
    Merges two sorted lists into one sorted list.

    Args:
        left (list): The left sorted list.
        right (list): The right sorted list.

    Returns:
        list: The merged sorted list.
    """
    merged = []
    i, j = 0, 0
    # Merge the two lists by comparing elements from each list
    while i < len(left) and j < len(right):
        if left[i] <= right[j]:
            merged.append(left[i])
            i += 1
        else:
            merged.append(right[j])
            j += 1
    # Append any remaining elements from the left and right lists
    merged += left[i:]
    merged += right[j:]
    return merged


def mergesort(v):
    """
    Sorts a list using the merge sort algorithm.

    Args:
        v (list): The list to be sorted.

    Returns:
        list: The sorted list.
    """
    # Base case: if the list has 1 or fewer elements, it is already sorted
    if len(v) <= 1:
        return v
    else:
        middle = len(v) // 2  # Find the middle index of the list
        # Recursively split the list into halves and merge them
        left = mergesort(v[:middle])
        right = mergesort(v[middle:])
        return merge(left, right)
