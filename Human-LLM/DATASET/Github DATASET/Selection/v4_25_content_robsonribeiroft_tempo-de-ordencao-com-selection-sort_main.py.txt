import random
import timeit
import matplotlib.pyplot as plt

# Global variables
lista = []
num_elementos = [100, 1000, 3000, 6000, 9000, 12000, 15000, 18000, 21000, 24000]
tempo = []
tempo_ordena = []

# Function to generate random numbers
def gerar_aleatorios(tamanho):
    """
    Generate random numbers and store them in the list.

    Args:
    tamanho (int): Number of random elements to generate.

    Returns:
    list: List containing random numbers.
    """
    random.seed()
    global lista
    lista = []
    i = 0
    while i < tamanho:
        x = random.randint(1, 10 * tamanho)
        if x not in lista:
            lista.append(x)
            i += 1
    return lista

# Function to perform selection sort
def selection_sort(tamanho):
    """
    Sort the list using selection sort algorithm.

    Args:
    tamanho (int): Number of elements in the list.

    Returns:
    list: Sorted list.
    """
    for i in range(len(lista)):
        menor = i
        for j in range(i + 1, len(lista)):
            if lista[j] < lista[menor]:
                menor = j
        lista[i], lista[menor] = lista[menor], lista[i]
    return lista

# Generate random numbers and measure generation time
for tamanho in num_elementos:
    tempo.append(timeit.timeit("gerar_aleatorios({})".format(tamanho), setup="from __main__ import gerar_aleatorios", number=1))
    # Sort the generated list and measure sorting time
    tempo_ordena.append(timeit.timeit("selection_sort({})".format(tamanho), setup="from __main__ import selection_sort", number=1))
    print(tamanho)

# Plotting the results
plt.plot(num_elementos, tempo, '*-', label='Tempo de Geração')
plt.plot(num_elementos, tempo_ordena, 'o-', label='Tempo de Ordenação')
plt.ylabel('Tempo(s)')
plt.xlabel('Quantidade de elementos')
plt.legend()
plt.show()
