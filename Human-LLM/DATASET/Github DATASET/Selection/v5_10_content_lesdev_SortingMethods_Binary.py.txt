import tkinter as tk
import tkinter.simpledialog

def binary_search(arr, target):
    left, right = 0, len(arr) - 1
    while left <= right:
        mid = (left + right) // 2
        if arr[mid] == target:
            return mid
        elif arr[mid] < target:
            left = mid + 1
        else:
            right = mid - 1
    return -1

def search():
    try:
        data_count = int(entry_num.get())
        numbers = []
        target = tkinter.simpledialog.askinteger("", "Enter the number to search for")
        for _ in range(data_count):
            number = tkinter.simpledialog.askinteger("", "Enter numbers in order")
            numbers.append(number)
        numbers_str = str(numbers)
        label_numbers.config(text=numbers_str)

        result_index = binary_search(numbers, target)
        result_text = "The number is at position: %s" % (result_index) if result_index != -1 else "Number not found"
        label_result.config(text=result_text)
    except ValueError:
        label_result.config(text="Please enter a valid number.")

app = tk.Tk()
app.title("Binary Search")
app.geometry('250x250')
app.configure(bg='SkyBlue2')

label_prompt = tk.Label(app, text="How many numbers will you enter?", font='Helvetica 12', bg='SkyBlue2')
label_prompt.pack()

entry_num = tk.Entry(app, width=8)
entry_num.pack()

button_search = tk.Button(app, text="Search", command=search)
button_search.pack(pady=(10, 0))

label_numbers = tk.Label(app, text="", bg='SkyBlue2')
label_numbers.pack(pady=(10, 0))

label_result = tk.Label(app, text="", bg='SkyBlue2')
label_result.pack(pady=(10, 0))

app.mainloop()
