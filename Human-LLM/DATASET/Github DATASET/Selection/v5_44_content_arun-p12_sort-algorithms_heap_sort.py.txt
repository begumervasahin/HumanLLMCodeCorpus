import common as c  # Assuming 'common' module contains the 'swap' function

def heap_sort(A, verbose=0, desc=0):
    """Sorts the input array using Heap Sort algorithm."""
    
    def heapify(i, n):
        """Heapifies the subtree rooted at index i."""
        parent = i
        left_child = i * 2 + 1
        right_child = left_child + 1
        
        # Find the index of the largest element among parent and its children
        if left_child < n and A[left_child] > A[parent]:
            parent = left_child
        if right_child < n and A[right_child] > A[parent]:
            parent = right_child
        
        # Swap parent with the largest child if needed and recursively heapify
        if i != parent:
            A[i], A[parent] = c.swap(A[i], A[parent])
            heapify(parent, n)
        
        if verbose == 2:
            print("  sub:", i, n, " :: ", A)

    def build_max_heap():
        """Builds the max heap from the input array."""
        n = len(A)
        # Start from the last non-leaf node and heapify all subtrees
        for i in range((n // 2) - 1, -1, -1):
            heapify(i, n)
            if verbose:
                print("iter")

    def sort_heap():
        """Sorts the heap by repeatedly swapping the root with the last element and heapifying."""
        n = len(A)
        for i in range(n - 1, 0, -1):
            A[0], A[i] = c.swap(A[0], A[i])
            heapify(0, i)
            if verbose:
                print("iter")

    build_max_heap()
    sort_heap()
    
    if desc:
        A = A[::-1]  # Reverse the array if descending order is requested
    
    return A
