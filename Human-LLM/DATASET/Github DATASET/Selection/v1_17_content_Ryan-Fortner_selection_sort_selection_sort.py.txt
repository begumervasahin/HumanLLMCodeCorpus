my_arr = [8, 5, 2, 6, 9, 3, 1, 4, 0, 7]

def selection_sort(my_arr):
    for i in range(len(my_arr)):
        min_idx = i
        for j in range(i + 1, len(my_arr)):
            if my_arr[min_idx] > my_arr[j]:
                min_idx = j
        temp = my_arr[i]
        my_arr[i] = my_arr[min_idx]
        my_arr[min_idx] = temp
    return my_arr

print(selection_sort(my_arr))
