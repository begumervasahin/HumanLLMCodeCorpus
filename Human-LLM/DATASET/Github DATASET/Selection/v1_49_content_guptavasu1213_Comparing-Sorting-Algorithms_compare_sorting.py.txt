import ctypes
import random
import time

class SelectionSort:
    def __init__(self, array):
        self.array = array
        self.array_accesses = 0
        self.extra_memory = 0
        self.recursive_calls = 0
        self.selection_sort()

    def selection_sort(self):
        for i in range(len(self.array)-1, 0, -1):
            max_position = 0
            for location in range(1, i+1):
                self.array_accesses += 2
                if self.array[location] > self.array[max_position]:
                    max_position = location
            temp = self.array[i]
            self.array[i] = self.array[max_position]
            self.array[max_position] = temp
            self.array_accesses += 4

class QuickSort:
    def __init__(self, array, start, end):
        self.array = array
        self.start = start
        self.end = end
        self.extra_memory = 0
        self.array_accesses = 0
        self.recursive_calls = -1
        self.quick_sort(self.array, self.start, self.end)

    def quick_sort(self, array, start, end):
        self.recursive_calls += 1
        if start < end:
            pivot = self.partition(array, start, end)
            self.quick_sort(array, start, pivot-1)
            self.quick_sort(array, pivot+1, end)

    def partition(self, array, first, last):
        big = first + 1
        small = last
        pivot = array[first]
        self.array_accesses += 1
        while big <= small:
            while big <= last and array[big] <= pivot:
                self.array_accesses += 1
                big += 1
            while array[small] > pivot:
                self.array_accesses += 1
                small -= 1
            if big < small:
                temp1 = array[small]
                array[small] = array[big]
                array[big] = temp1
                self.array_accesses += 4
        temp2 = array[first]
        array[first] = array[small]
        array[small] = temp2
        self.array_accesses += 4
        return small

class MergeSort:
    def __init__(self, array, array_accesses=0):
        self.array = array
        self.extra_memory = 0
        self.array_accesses = 0
        self.recursive_calls = -1
        self.array = self.merge_sort(self.array)

    def merge_sort(self, array):
        self.recursive_calls += 1
        n = len(array)
        if n <= 1:
            return array
        left_array = (n//2 * ctypes.py_object)()
        self.extra_memory += n//2
        for index in range(n//2):
            left_array[index] = array[index]
            self.array_accesses += 2
        self.extra_memory += n-(n//2)
        right_array = ((n-(n//2)) * ctypes.py_object)()
        for index in range(n//2, n):
            right_array[index-(n//2)] = array[index]
            self.array_accesses += 2
        list1 = self.merge_sort(left_array)
        list2 = self.merge_sort(right_array)
        array = self.merge(list1, list2)
        return array

    def merge(self, l1, l2):
        self.extra_memory += len(l1)+len(l2)
        final_array = ((len(l1)+len(l2)) * ctypes.py_object)()
        ind = 0
        i = 0
        j = 0
        while i < len(l1) and j < len(l2):
            self.array_accesses += 2
            if l1[i] < l2[j]:
                final_array[ind] = l1[i]
                self.array_accesses += 2
                i += 1
                ind += 1
            else:
                final_array[ind] = l2[j]
                self.array_accesses += 2
                j += 1
                ind += 1
        while i < len(l1):
            final_array[ind] = l1[i]
            self.array_accesses += 2
            i += 1
            ind += 1
        while j < len(l2):
            final_array[ind] = l2[j]
            self.array_accesses += 2
            j += 1
            ind += 1
        return final_array

class HeapSort:
    def __init__(self, array):
        self.array = array
        self.extra_memory = 0
        self.recursive_calls = 0
        self.array_accesses = 0
        self.size = len(self.array)
        i = (self.size-2)//2
        while i >= 0:
            self.max_heap(i)
            i -= 1
        self.heap_sort()

    def max_heap(self, index):
        largest = index
        left = 2*index+1
        right = 2*index+2
        self.array_accesses += 2
        if left < self.size and self.array[left] > self.array[largest]:
            largest = left
        self.array_accesses += 2
        if right < self.size and self.array[right] > self.array[largest]:
            largest = right
        if not largest == index:
            temp = self.array[largest]
            self.array[largest] = self.array[index]
            self.array[index] = temp
            self.array_accesses += 4
            self.recursive_calls += 1
            self.max_heap(largest)

    def heap_sort(self):
        while self.size > 1:
            temp = self.array[0]
            self.array[0] = self.array[self.size-1]
            self.array[self.size-1] = temp
            self.array_accesses += 4
            self.size -= 1
            self.max_heap(0)

def create_array(given_array_size):
    numbers_in_array = (given_array_size * ctypes.py_object)()
    for value in range(1, given_array_size+1):
        numbers_in_array[value-1] = value
    return numbers_in_array

def shuffle_array(array):
    for element in range(len(array)-1, 0, -1):
        random_index = random.randint(0, element)
        temp = array[element]
        array[element] = array[random_index]
        array[random_index] = temp
    return array
