def generate_random_list():
    """
    Generate a random list of integers.
    """
    return [randint(1, 10000) for _ in range(2000)]

def measure_sorting_time(sort_func, data):
    """
    Measure the time taken to sort the data using the specified sorting function.
    """
    start_time = time.time()
    sort_func(data.copy())  # Copy to prevent modifying the original list
    end_time = time.time()
    return end_time - start_time
