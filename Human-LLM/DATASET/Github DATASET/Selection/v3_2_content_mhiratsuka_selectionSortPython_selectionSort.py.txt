# Define the list
my_list = [0, 2, 100, 0, 120, 11]

# Perform selection sort
for i in range(len(my_list) - 1):
    # Initialize variables to track the maximum value and its index
    max_value = my_list[i]
    max_index = i

    # Find the maximum value and its index in the unsorted portion of the list
    for j in range(i + 1, len(my_list)):
        if my_list[j] > max_value:
            max_value = my_list[j]
            max_index = j

    # Swap elements if necessary
    if max_index != i:
        my_list[i], my_list[max_index] = my_list[max_index], my_list[i]

# Print the sorted list
print(my_list)
