import random

def generate_array(size):
    """
    Generate an array of random integers with the given size.

    Args:
    size (int): The size of the array.

    Returns:
    list: The generated array.
    """
    return [random.randint(-size, size) for _ in range(size)]

def generate_payload():
    """
    Generate a payload containing arrays of various sizes.

    Returns:
    list: The payload containing arrays of different sizes.
    """
    sizes = [10, 100, 1000, 10000, 100000, 1000000, 10000000]
    return [generate_array(size) for size in sizes]
