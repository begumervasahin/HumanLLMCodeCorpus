import argparse
from quickSort import quickSort
from bubbleSort import bubbleSort
from selectionSort import selectionSort
from mergeSort import mergeSort

# Dictionary mapping algorithm names to their corresponding class names
class_names = {'bubble': 'bubbleSort', 'selection': 'selectionSort', 'merge': 'mergeSort', 'quick': 'quickSort'}

# Function to display a list
def display_list(lst):
    lst = [str(i) for i in lst]
    if len(lst) == 0:
        print('| |')
    else:
        output = '|'
        for i in lst:
            output = output + ' ' + i + ' |'
        print(output)
    input("Press Enter to continue...")

# Main function
def main():
    trace_mode = False
    parser = argparse.ArgumentParser()
    parser.add_argument("input_list", nargs='+', type=int, help="List of integers separated by space")
    parser.add_argument("alg", type=str, help="bubble, selection, merge, or quick")
    parser.add_argument("-t", "--trace", action="store_true", help="Print the sorting algorithm step-by-step")
    args = parser.parse_args()

    if args.trace:
        trace_mode = True

    # Check if the provided algorithm name is valid
    if args.alg not in class_names.keys():
        print('\nInvalid algorithm name')
        parser.print_help()
        return

    # Display the input list
    print('Input: ' + str(args.input_list))

    # Instantiate the sorting object based on the selected algorithm
    class_name = class_names[args.alg]
    sorting_object = globals()[class_name](args.input_list, trace_mode)

    # Sort the input list using the selected algorithm and display the result
    print('Result: ' + str(sorting_object.sort()))

if __name__ == '__main__':
    main()
