import os
import sys
import pygame.mixer as mixer

# Define paths for input, save, and alternative directories
input_directory = "/home/error-404/Desktop/songs"
save_directory = "/home/error-404/Desktop/dance"
alternative_directory = "/home/error-404/Desktop/others"

# Get list of audio files in the input directory
audio_files = [file for file in os.listdir(input_directory) if os.path.isfile(os.path.join(input_directory, file))]

# Initialize Pygame mixer
mixer.init()

# Flag to track pause/unpause state
paused = False

def play_audio(file_path, start_time=0):
    """Play audio file with optional starting time."""
    mixer.music.load(file_path)
    mixer.music.play(0, start_time)

def move_and_stop(file_path, target_directory):
    """Move the audio file to the target directory and stop playback."""
    mixer.music.stop()
    os.rename(file_path, os.path.join(target_directory, os.path.basename(file_path)))

# Loop through each audio file in the input directory
for file_name in audio_files:
    print("Now playing:", file_name)
    file_path = os.path.join(input_directory, file_name)
    play_audio(file_path, 40)  # Start playing from 40 seconds

    # Loop to handle user input while audio is playing
    while True:
        user_input = input()

        # Seek to a specific position (multiplied by 10 seconds)
        if user_input.isdigit():
            play_audio(file_path, int(user_input) * 10)

        # Stop playback and move the file to the alternative directory
        elif user_input == '':
            move_and_stop(file_path, alternative_directory)
            break

        # Stop playback and move the file to the save directory
        elif user_input == '/':
            move_and_stop(file_path, save_directory)
            break

        # Quit the program
        elif user_input == 'q':
            sys.exit(0)

        # Pause/unpause the playback
        elif user_input == 'p':
            if paused:
                mixer.music.unpause()
                paused = False
            else:
                mixer.music.pause()
                paused = True
