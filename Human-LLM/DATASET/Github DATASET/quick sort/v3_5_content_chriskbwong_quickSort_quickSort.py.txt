def partition(arr, low, high):
    """
    This function takes the last element as pivot, places the pivot element at its correct position in the sorted
    array, and places all smaller (smaller than pivot) to the left of the pivot and all greater elements to the right of the pivot.
    """
    i = low - 1
    pivot = arr[high]
    for j in range(low, high):
        if arr[j] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1

def quick_sort_iterative(arr, low, high):
    """
    The main function that implements QuickSort iteratively.
    arr[] --> Array to be sorted,
    low  --> Starting index,
    high  --> Ending index
    """
    size = high - low + 1
    stack = [(low, high)]
    while stack:
        low, high = stack.pop()
        p = partition(arr, low, high)
        if p - 1 > low:
            stack.append((low, p - 1))
        if p + 1 < high:
            stack.append((p + 1, high))

# Example usage
arr = [4, 3, 5, 2, 1, 3, 2, 3]
quick_sort_iterative(arr, 0, len(arr) - 1)
print("Sorted array is:")
print(*arr)
