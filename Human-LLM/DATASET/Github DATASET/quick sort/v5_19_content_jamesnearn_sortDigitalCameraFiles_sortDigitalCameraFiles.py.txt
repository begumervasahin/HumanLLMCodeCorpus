import getopt
import os
import sys
import time

def sort_files(directory_path, separator):
    """
    Sorts files with specific extensions in the given directory.

    Args:
        directory_path (str): The path of the directory to sort.
        separator (str): The separator to use in the destination path.
    """
    extensions = (".jpg", ".mp4", ".mts", ".mov")
    for root, _, filenames in os.walk(directory_path):
        for filename in filenames:
            if filename.lower().endswith(extensions):
                parse_file(root, filename, separator)

def parse_file(directory_path, filename, separator):
    """
    Parses a file and moves it to the destination directory.

    Args:
        directory_path (str): The directory path where the file is located.
        filename (str): The name of the file to parse.
        separator (str): The separator to use in the destination path.
    """
    full_file_path = os.path.join(directory_path, filename)
    new_path = get_destination_path(full_file_path, separator)
    new_path = os.path.join(directory_path, new_path)
    os.makedirs(new_path, exist_ok=True)
    os.rename(full_file_path, os.path.join(new_path, filename))

def get_destination_path(full_file_path, separator):
    """
    Generates the destination path for the file based on its modification time.

    Args:
        full_file_path (str): The full path of the file.
        separator (str): The separator to use in the destination path.

    Returns:
        str: The destination path for the file.
    """
    epoch_sec = os.path.getmtime(full_file_path)
    new_path = time.strftime('%Y' + separator + '%Y %m %d', time.localtime(epoch_sec))
    return new_path

def main(argv):
    """
    Main function to parse command line arguments and initiate sorting of files.

    Args:
        argv (list): The list of command line arguments.
    """
    separator = "\\"
    path = ""
    try:
        opts, _ = getopt.getopt(argv, "hi:o:", ["separator=", "path="])
    except getopt.GetoptError:
        print('Usage: SortFiles.py --separator <value> --path <path>')
        sys.exit(2)
    for opt, arg in opts:
        if opt == '--help':
            print('Usage: SortFiles.py --separator <value> --path <path>')
            sys.exit()
        elif opt == "--separator":
            separator = arg
        elif opt == "--path":
            path = arg
    sort_files(path, separator)

if __name__ == "__main__":
    if len(sys.argv) > 1 and sys.argv[1] == "runtests":
        print('Running tests')
    else:
        main(sys.argv[1:])
