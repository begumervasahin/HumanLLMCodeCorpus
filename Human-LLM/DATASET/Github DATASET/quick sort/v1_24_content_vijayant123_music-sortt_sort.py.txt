import os
import sys
from os import listdir
from os.path import isfile, join
import pygame.mixer as mixer

# Define paths for input, save, and alternative directories
mypath = "/home/error-404/Desktop/songs"
savepath = "/home/error-404/Desktop/dance"
altpath = "/home/error-404/Desktop/others"

# Get list of audio files in the input directory
onlyfiles = [f for f in listdir(mypath) if isfile(join(mypath, f))]

# Initialize Pygame mixer
mixer.init()

# Flag to track pause/unpause state
flag = False

# Loop through each audio file in the input directory
for f in onlyfiles:
    print(f)
    mixer.music.load(join(mypath, f))  # Load audio file
    mixer.music.play(0, 40)  # Start playing from 40 seconds
    
    # Loop to handle user input while audio is playing
    while True:
        a = input()
        
        # Seek to a specific position (multiplied by 10 seconds)
        if a.isdigit():
            mixer.music.rewind()
            mixer.music.play(0, int(a) * 10)
        
        # Stop playback and move the file to the alternative directory
        elif a == '':
            mixer.music.stop()
            os.rename(join(mypath, f), join(altpath, f))
            break
        
        # Stop playback and move the file to the save directory
        elif a == '/':
            mixer.music.stop()
            os.rename(join(mypath, f), join(savepath, f))
            break
        
        # Quit the program
        elif a == 'q':
            sys.exit(0)
        
        # Pause/unpause the playback
        elif a == 'p':
            if flag:
                mixer.music.unpause()
                flag = False
            else:
                mixer.music.pause()
                flag = True
