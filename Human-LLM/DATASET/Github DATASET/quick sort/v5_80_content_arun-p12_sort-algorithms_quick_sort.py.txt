import common as c

def quick_sort(A, verbose=0, desc=0):
    """
    Sorts the given list using the Quick Sort algorithm.

    Args:
    - A: The list to be sorted.
    - verbose: Verbosity level for printing intermediate steps.
               0: No printing, 1: Print iteration steps, 2: Print sub-list details.
    - desc: Flag to sort the list in descending order.

    Returns:
    - A: The sorted list.
    """

    def partition(lb, ub):
        """
        Partitions the list around a pivot element.

        Args:
        - lb: Lower bound index of the sub-list.
        - ub: Upper bound index of the sub-list.

        Returns:
        - The index of the pivot element.
        """
        pivot, start, end = A[lb], lb, ub
        while start < end:
            while A[start] <= pivot and start < ub:
                start += 1
            while A[end] > pivot and end > lb:
                end -= 1
            if start < end:
                A[start], A[end] = c.swap(A[start], A[end])
            if verbose == 2:
                print("  sub:", pivot, start, end, " :: ", A)
        A[lb], A[end] = c.swap(A[lb], A[end])
        return end

    def quick_sort_recursive(x, y):
        """
        Recursive function to perform Quick Sort.

        Args:
        - x: Start index of the sub-list.
        - y: End index of the sub-list.
        """
        if x < y:
            loc = partition(x, y)
            if verbose:
                print("iter :", loc, x, y, " :: ", A)
            quick_sort_recursive(x, loc - 1)
            quick_sort_recursive(loc + 1, y)

    quick_sort_recursive(0, len(A) - 1)
    if desc:
        A = A[::-1]
    return A
