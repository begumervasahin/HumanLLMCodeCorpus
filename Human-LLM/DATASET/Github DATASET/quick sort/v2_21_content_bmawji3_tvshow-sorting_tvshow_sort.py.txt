# Define an empty list to hold TV shows
tv_shows = []

# Read the unsorted TV show list from the file
with open("tvshow_list_unsorted.txt", "r") as file:
    # Iterate over each line in the file
    for line in file:
        # Split each line into show name and season number
        show_name, season_number = line.strip().split('_')
        # Append the tuple (show name, season number) to the list
        tv_shows.append((show_name, season_number))

# Sort the TV show list alphabetically by show name, then by season number
tv_shows.sort(key=lambda x: (x[1], x[0]))

# Write the sorted TV show list to a new file
with open('tvshow_list_sorted.txt', 'w') as output_file:
    # Iterate over each tuple in the sorted list
    for show in tv_shows:
        # Write each tuple to the file, converting it to a string
        output_file.write(f"{show}\n")
