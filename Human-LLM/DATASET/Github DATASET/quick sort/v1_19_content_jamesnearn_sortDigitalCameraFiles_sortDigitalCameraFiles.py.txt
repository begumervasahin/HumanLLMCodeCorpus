import os
import time
import shutil
import sys
import argparse

def sort_files(given_dir, separator):
    for (dirpath, dirnames, filenames) in os.walk(given_dir):
        for filename in filenames:
            if (filename.lower().endswith(".jpg") or filename.lower().endswith(".mp4") or filename.lower().endswith(".mts") or filename.lower().endswith(".mov")):
                parse_file(dirpath, filename, separator)

def parse_file(given_directory_path, given_filename, separator):
    given_full_filepath = os.path.join(given_directory_path, given_filename)
    print(given_full_filepath)
    new_path = get_destination_path(given_full_filepath, separator)
    new_path = os.path.join(given_directory_path, new_path)
    print(new_path)
    if not os.path.exists(new_path):
        os.makedirs(new_path)
    print(new_path + separator + given_filename)
    os.rename(given_full_filepath, os.path.join(new_path, given_filename))

def get_destination_path(given_full_filepath, separator):
    epoch_sec = os.path.getmtime(given_full_filepath)
    converted_date = time.ctime(epoch_sec)
    new_path = time.strftime('%Y' + separator + '%Y %m %d', time.localtime(epoch_sec))
    return new_path

def main(args):
    parser = argparse.ArgumentParser(description="Sort files based on modification date")
    parser.add_argument("--separator", default="\\", help="Separator for directory structure")
    parser.add_argument("--path", required=True, help="Path to the directory containing files")
    args = parser.parse_args(args)

    print('Separator = ', args.separator)
    print('Path = ', args.path)
    sort_files(args.path, args.separator)

if __name__ == "__main__":
    main(sys.argv[1:])
