def quick_sort_lomuto(arr):
    return _quick_sort_lomuto_helper(arr, 0, len(arr) - 1)

def _quick_sort_lomuto_helper(arr, start, end):
    if start < end:
        pivot = _lomuto_partition(arr, start, end)
        _quick_sort_lomuto_helper(arr, start, pivot - 1)
        _quick_sort_lomuto_helper(arr, pivot + 1, end)
    return arr

def _lomuto_partition(arr, start, end):
    pivot_value = arr[end]
    i = start - 1
    for j in range(start, end):
        if arr[j] <= pivot_value:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i + 1], arr[end] = arr[end], arr[i + 1]
    return i + 1

def quick_sort_hoare(arr):
    return _quick_sort_hoare_helper(arr, 0, len(arr) - 1)

def _quick_sort_hoare_helper(arr, start, end):
    if start < end:
        pivot = _hoare_partition(arr, start, end)
        _quick_sort_hoare_helper(arr, start, pivot)
        _quick_sort_hoare_helper(arr, pivot + 1, end)
    return arr

def _hoare_partition(arr, start, end):
    pivot_value = arr[start]
    i, j = start - 1, end + 1
    while True:
        i += 1
        while arr[i] < pivot_value:
            i += 1
        j -= 1
        while arr[j] > pivot_value:
            j -= 1
        if i >= j:
            return j
        arr[i], arr[j] = arr[j], arr[i]

arr = [15, 4, 68, 24, 75, 16, 42]

sorted_hoare = quick_sort_hoare(arr.copy())
print("Sorted using Hoare's partition scheme:", sorted_hoare)

sorted_lomuto = quick_sort_lomuto(arr.copy())
print("Sorted using Lomuto's partition scheme:", sorted_lomuto)
