# Define the list to hold TV shows
tv_list = []

# Read the unsorted TV show list from the file
with open("tvshow_list_unsorted.txt", "r") as myfile:
    for line in myfile:
        # Split each line into show name and season number
        temp_split = line.split('_')
        # Append the tuple (show name, season number) to the list
        tv_list.append((temp_split[0], temp_split[1].strip('\n')))

# Sort the TV show list alphabetically by show name, then by season number
tv_list.sort(key=lambda x: (x[1], x[0]))

# Write the sorted TV show list to a new file
with open('tvshow_list_sorted.txt', 'w') as f:
    for item in tv_list:
        # Write each tuple to the file
        f.write(str(item) + '\n')
