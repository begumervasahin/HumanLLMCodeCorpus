def read_tv_show_list(file_path):
    """
    Read the unsorted TV show list from the file.

    Args:
        file_path (str): The path to the file containing the unsorted TV show list.

    Returns:
        list: A list of tuples containing TV show titles and genres.
    """
    tv_list = []
    with open(file_path, "r") as file:
        for line in file:
            title, genre = line.strip().split('_')
            tv_list.append((title, genre))
    return tv_list

def sort_tv_show_list(tv_list):
    """
    Sort the TV show list first by genre and then by title.

    Args:
        tv_list (list): The unsorted TV show list.

    Returns:
        list: The sorted TV show list.
    """
    return sorted(tv_list, key=lambda x: (x[1], x[0]))

def write_tv_show_list(tv_list, file_path):
    """
    Write the sorted TV show list to a file.

    Args:
        tv_list (list): The sorted TV show list.
        file_path (str): The path to the file to write the sorted list.
    """
    with open(file_path, 'w') as file:
        for title, genre in tv_list:
            file.write(f"{title}_{genre}\n")

def main():
    unsorted_file_path = "tvshow_list_unsorted.txt"
    sorted_file_path = "tvshow_list_sorted.txt"
    
    tv_list = read_tv_show_list(unsorted_file_path)
    sorted_tv_list = sort_tv_show_list(tv_list)
    write_tv_show_list(sorted_tv_list, sorted_file_path)

if __name__ == "__main__":
    main()
