def quick_sort(arr, left, right):
    """Sorts an array using the quicksort algorithm."""
    if left < right:
        pivot = arr[left]  # Choose the first element as the pivot
        partition_index = left  # Initialize the partition index
        # Partitioning step
        for i in range(left + 1, right):
            if arr[i] < pivot:
                partition_index += 1
                arr[partition_index], arr[i] = arr[i], arr[partition_index]
        # Swap the pivot with the element at the partition index
        arr[left], arr[partition_index] = arr[partition_index], arr[left]
        # Recursive calls for the left and right sublists
        quick_sort(arr, left, partition_index)
        quick_sort(arr, partition_index + 1, right)

# Test the quick_sort function
arr = [3, 6, 8, 1, 5, 2, 7, 4]
quick_sort(arr, 0, len(arr))
