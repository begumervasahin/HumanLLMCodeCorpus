class UninformedSearchAlgorithm:
    """
    Represents an uninformed search algorithm.

    Attributes:
        listNodes (list): List of nodes in the graph.
        searched (str): The node to be searched.
        queue (list): Queue used for the search algorithm.
    """

    def __init__(self, listNodes, searched):
        """
        Initializes an instance of the UninformedSearchAlgorithm class.

        Args:
            listNodes (list): List of nodes in the graph.
            searched (str): The node to be searched.
        """
        self.listNodes = listNodes
        self.searched = searched
        self.create_queue()

    def create_queue(self):
        """Creates the initial queue."""
        self.queue = [self.listNodes[0]]

    def read_queue(self):
        """Removes and returns the first element from the queue."""
        return self.queue.pop(0)

    def get_queue_len(self):
        """Returns the length of the queue."""
        return len(self.queue)

    def get_node_from_list(self, name):
        """Returns the node with the given name from the list of nodes."""
        for node in self.listNodes:
            if node.name == name:
                return node

    def match_searched(self, node_name):
        """
        Raises an exception if the given node name matches the searched node.

        Args:
            node_name (str): The name of the node to be checked.
        """
        if node_name == self.searched:
            raise Exception("City found: %s" % node_name)

        Inserts the child nodes of the given node into the queue.

        """
