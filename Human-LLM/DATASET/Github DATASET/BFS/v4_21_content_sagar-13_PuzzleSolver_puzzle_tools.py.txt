from collections import deque


class PuzzleSolver:
    """
    A class for solving puzzles using BFS and DFS algorithms.
    """

    def __init__(self):
        """Initialize the PuzzleSolver object."""
        pass

    def depth_first_solve(self, puzzle):
        """
        Return a path from the initial puzzle configuration to a PuzzleNode containing
        a solution, with each child containing an extension of the puzzle in its parent.
        Return None if a solution is not possible.

        Args:
            puzzle (Puzzle): The initial puzzle configuration.

        Returns:
            PuzzleNode or None: The solution PuzzleNode if found, otherwise None.
        """

        visited = set()
        stack = deque()
        stack.append(PuzzleNode(puzzle))

        while stack:
            element = stack.pop()

            for config in element.puzzle.extensions():
                if str(config) in visited:
                    continue

                if config.is_solved():
                    return PuzzleNode(config)

                if config.fail_fast() or len(config.extensions()) == 0:
                    continue

                a = PuzzleNode(config)
                visited.add(str(a.puzzle))
                stack.append(a)
        return None

    def breadth_first_solve(self, puzzle):
        """
        Return a path from the initial puzzle configuration to a PuzzleNode containing
        a solution, with each child PuzzleNode containing an extension of the puzzle in its parent.
        Return None if a solution is not possible.

        Args:
            puzzle (Puzzle): The initial puzzle configuration.

        Returns:
            PuzzleNode or None: The solution PuzzleNode if found, otherwise None.
        """
        visited = set()
        queue = deque()
        queue.append(PuzzleNode(puzzle))

        while queue:
            element = queue.popleft()

            for child in element.puzzle.extensions():
                if str(child) in visited:
                    continue
                if child.is_solved():
                    return PuzzleNode(child)
                if child.fail_fast() or len(child.extensions()) == 0:
                    continue

                a = PuzzleNode(child)
                visited.add(str(a.puzzle))
                queue.append(a)
        return None


class PuzzleNode:
    """
    A Puzzle configuration that refers to other configurations that it can be extended to.
    """

    def __init__(self, puzzle=None, children=None, parent=None):
        """
        Create a new puzzle node with the given configuration, children, and parent.

        Args:
            puzzle (Puzzle): The puzzle configuration.
            children (list[PuzzleNode]): List of children nodes.
            parent (PuzzleNode): Parent node.
        """
        self.puzzle = puzzle
        self.parent = parent
        self.children = children[:] if children is not None else []
        self.visited = False

    def __eq__(self, other):
        """
        Return whether PuzzleNode self is equivalent to other.

        Args:
            other (PuzzleNode | Any): The object to compare.

        Returns:
            bool: True if equivalent, False otherwise.
        """
        return (type(self) == type(other) and
                self.puzzle == other.puzzle and
                all(x in self.children for x in other.children) and
                all(x in other.children for x in self.children))

    def __str__(self):
        """
        Return a human-readable string representing PuzzleNode self.
        """
        return "{}\n\n{}".format(self.puzzle, "\n".join(str(x) for x in self.children))
