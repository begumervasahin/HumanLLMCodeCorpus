# Importing necessary libraries
import pytest
import networkx as nx

# Importing the function to be tested from max_flow module
from max_flow import flow_with_demands


# Function to compute the net flow into each node
def divergence(flow):
    """
    Computes the total flow into each node according to the given flow dict.

    Args:
        flow: the flow dict recording flow between nodes.

    Returns:
        A dict of the net flow into each node.
    """
    div = {k: 0 for k in flow.keys()}
    for node in flow.keys():
        forward_flow = sum(flow[node].values())
        backward_flow = sum(flow[i][node] for i in flow[node].keys())
        div[node] = backward_flow - forward_flow
    return div


# Test function for max flow calculation
def test_max_flow():
    # Creating a graph from the contiguous USA data
    G = nx.Graph()
    usa = open('contiguous-usa.dat')
    for i, line in enumerate(usa):
        s1, s2 = line.strip().split()
        G.add_edge(s1, s2)
    # Setting demands for each state
    for state in G.nodes():
        if state != 'CA':
            G.node[state]['demand'] = 1
    G.node['CA']['demand'] = -48
    # Converting the graph to a directed graph
    G = nx.DiGraph(G)
    # Assigning uniform capacity to all edges
    uniform_capacity = 16
    for (s1, s2) in G.edges():
        G.edge[s1][s2]['capacity'] = uniform_capacity
    # Computing flow with demands using the function to be tested
    flow = flow_with_demands(G)
    # Computing net flow into each node
    div = divergence(flow)
    # Assertion to check if the net flow satisfies the demands of each node
    assert all(div[n] == G.node[n]['demand'] for n in G.nodes())


# Allowing the test function to be run if this script is executed directly
if __name__ == '__main__':
    test_max_flow()
