import random

class Graph:
    """
    Represents a graph data structure.
    """

    def __init__(self, is_undirected=False):
        """
        Initializes an instance of the Graph class.

        Parameters:
        - is_undirected: A boolean indicating whether the graph is undirected.
        """
        self._adjacency_list = {}
        self._is_weighted = False
        self._is_undirected = is_undirected

    def vertices(self):
        """Returns a list of vertices in the graph."""
        return list(self._adjacency_list.keys())

    def edges(self):
        """Returns a list of edges in the graph."""
        visited = set()
        edge_list = []
        for v in self.vertices():
            new_edges = [(v, u, self.weight((v, u))) for u in self.adjacent(v)]
            for edge in new_edges:
                u, v, weight = edge
                if (u, v) not in visited and (v, u) not in visited:
                    edge_list.append(edge)
                    visited.add((u, v))
        return edge_list

    def adjacent(self, vertex):
        """Returns a list of adjacent vertices for a given vertex."""
        return list(self._adjacency_list.get(vertex, {}).keys())

    def is_adjacent(self, origin, vertex):
        """Checks if a vertex is adjacent to another vertex."""
        return vertex in self._adjacency_list.get(origin, {})

    def add_vertex(self, vertex):
        """Adds a vertex to the graph."""
        if vertex not in self._adjacency_list:
            self._adjacency_list[vertex] = {}

    def add_edge(self, edge, weight=1):
        """
        Adds an edge to the graph.

        Parameters:
        - edge: A tuple representing the edge (vertex1, vertex2).
        - weight: An optional parameter specifying the weight of the edge.
        """
        x, y = edge
        self._adjacency_list.setdefault(x, {})[y] = weight
        if self.is_undirected():
            self._adjacency_list.setdefault(y, {})[x] = weight
        self._is_weighted = self._is_weighted or weight != 1

    def weight(self, edge):
        """Returns the weight of the specified edge."""
        x, y = edge
        return self._adjacency_list.get(x, {}).get(y, 0)

    def random_vertex(self):
        """Returns a random vertex from the graph."""
        vertex_list = self.vertices()
        return random.choice(vertex_list) if vertex_list else None

    def is_weighted(self):
        """Checks if the graph has weighted edges."""
        return self._is_weighted

    def is_undirected(self):
        """Checks if the graph is undirected."""
        return self._is_undirected

# Example usage:
if __name__ == "__main__":
    # Create a graph object
    graph = Graph()

    # Add vertices
    graph.add_vertex('A')
    graph.add_vertex('B')
    graph.add_vertex('C')

    # Add weighted edges
    graph.add_edge(('A', 'B'), 3)
    graph.add_edge(('B', 'C'), 5)
    graph.add_edge(('A', 'C'), 2)

    # Print vertices and edges
    print("Vertices:", graph.vertices())
    print("Edges:", graph.edges())
