def breadth_first_search(graph, start, goal):
    """
    Perform Breadth-First Search (BFS) on the given graph to find a path from start to goal.

    Parameters:
    graph (dict): The graph in the form of adjacency list.
    start (str): The starting node.
    goal (str): The goal node.

    Returns:
    list or None: A list representing the path from start to goal if found, else None.
    """
    queue = [[start]]  # Initialize the queue with the start node
    visited = set()  # Initialize a set to keep track of visited nodes

    while queue:
        # Dequeue the first path from the queue
        path = queue.pop(0)
        # Get the current node from the path
        current_node = path[-1]

        # If the current node is the goal node, return the path
        if current_node == goal:
            return path

        # If the current node has not been visited
        if current_node not in visited:
            # Iterate through the neighbors of the current node
            for neighbor in graph.get(current_node, []):
                # Create a new path by copying the current path
                new_path = list(path)
                # Append the neighbor to the new path
                new_path.append(neighbor)
                # Enqueue the new path
                queue.append(new_path)
            # Mark the current node as visited
            visited.add(current_node)

    # Print message if the goal node is not found
    print("Path not found")
    return None

# Define the graph
graph = {
    'A': {'B', 'C'},
    'B': {'A', 'H', 'J'},
    'C': {'A', 'D', 'G'},
    'D': {'C', 'E'},
    'E': {'D', 'F'},
    'F': {'E', 'G', 'K', 'L'},
    'G': {'C', 'F'},
    'H': {'B', 'I'},
    'I': {'H', 'J', 'K'},
    'J': {'B', 'I'},
    'K': {'F', 'I', 'L'},
    'L': {'F', 'K'}
}

# Example usage:
start_node = 'A'
goal_node = 'L'
result = breadth_first_search(graph, start_node, goal_node)
if result:
    print("Path found:", result)
else:
    print("Path not found")
