from collections import deque
from enum import Enum


class Node(Enum):
    """
    Enumeration representing the state of a grid node.
    """
    BLACK = 5
    WHITE = 2
    EMPTY = 0


class Grid:
    """
    Represents a grid for the Go game.
    """

    def __init__(self, size, node_pairs):
        """
        Initializes a grid with the given size and node pairs.

        Args:
            size (int): The size of the grid.
            node_pairs (list): List of tuples containing node coordinates and their state.
        """
        self.board = [[Node.EMPTY for _ in range(size)] for _ in range(size)]
        self.size = size
        for location, node_state in node_pairs:
            self.set_node_state(location, node_state)

    def get_node_state(self, location):
        """
        Retrieves the state of the node at the given location in the grid.

        Args:
            location (tuple): The location coordinates.

        Returns:
            Node: The state of the node at the specified location.
        """
        return self.board[location[0]][location[1]]

    def set_node_state(self, location, state):
        """
        Sets the state of the node at the given location in the grid.

        Args:
            location (tuple): The location coordinates.
            state (Node): The state to set.
        """
        self.board[location[0]][location[1]] = state

    def is_captured(self, location):
        """
        Checks if the black node at the given location is captured.

        Args:
            location (tuple): The location coordinates.

        Returns:
            bool: True if the black node is captured, False otherwise.
        """
        if self.get_node_state(location) == Node.WHITE:
            return False
        if self.is_within_bounds(location):
            queue = deque()
            queue.appendleft(location)
            visited = [[False for _ in range(self.size)] for _ in range(self.size)]
            visited[location[0]][location[1]] = True
            while queue:
                current = queue.popleft()
                if self.is_surrounded(queue, visited, current):
                    continue
                else:
                    return False
            return True
        else:
            return False

    def is_within_bounds(self, location):
        """
        Checks if the location is within the grid boundaries.

        Args:
            location (tuple): The location coordinates.

        Returns:
            bool: True if the location is within the grid boundaries, False otherwise.
        """
        row, col = location
        return 0 <= row < self.size and 0 <= col < self.size

    def is_surrounded(self, queue, visited, current):
        """
        Checks if the current node is surrounded by white nodes.

        Args:
            queue (deque): The queue for BFS.
            visited (list): List to track visited nodes.
            current (tuple): The current node coordinates.

        Returns:
            bool: True if the current node is surrounded, False otherwise.
        """
        row, col = current
        neighbors = [(row, col + 1), (row, col - 1), (row + 1, col), (row - 1, col)]
        for neighbor_row, neighbor_col in neighbors:
            if self.is_within_bounds((neighbor_row, neighbor_col)):
                if not visited[neighbor_row][neighbor_col]:
                    neighbor_state = self.get_node_state((neighbor_row, neighbor_col))
                    if neighbor_state == Node.WHITE:
                        visited[neighbor_row][neighbor_col] = True
                        continue
                    elif neighbor_state == Node.EMPTY:
                        return False
                    elif neighbor_state == Node.BLACK:
                        queue.append((neighbor_row, neighbor_col))
            else:
                continue
        return True
