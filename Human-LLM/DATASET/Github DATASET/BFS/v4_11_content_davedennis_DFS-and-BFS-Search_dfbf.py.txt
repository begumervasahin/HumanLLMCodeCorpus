import sys

def read_graph(file_name, debug=False):
    """
    Read a file containing graph information into a dictionary.
    Each line has a nodeName followed by its adjacent nodeNames.
    """
    graph = {}
    with open(file_name) as file:
        for line in file:
            parts = line.strip().split(" ")
            if debug:
                print("parts:", parts, "length:", len(parts))
            if parts == ['']:
                continue  # Skip empty lines
            graph[parts[0]] = ('white', parts[1:])
    return graph

def dump_graph(graph):
    print("Dumping graph: nodeName (color, [adjacent list]) ")
    for node, data in graph.items():
        print(node, data)

def bfs(graph, root):
    queue = []
    queue.append(root)
    graph[root] = ("black", graph[root][1], 0)

    if len(graph[root][1]) == 0:
        return []

    result = [(root, 0)]
    while queue:
        node = queue.pop(0)
        for adj_node in graph[node][1]:
            if graph[adj_node][0] == "white":
                graph[adj_node] = ('grey', graph[adj_node][1], graph[node][2] + 1)
                result.append((adj_node, graph[adj_node][2]))
                queue.append(adj_node)
        graph[node] = ("black", graph[node][1])
    return result

def paint_white(graph):
    """
    Paint all graph nodes white.
    """
    for node in graph:
        graph[node] = ('white', graph[node][1])

def dfs(root):
    visited = set()
    def dfs_helper(node):
        visited.add(node)
        graph[node] = ("grey", graph[node][1])
        for adj_node in graph[node][1]:
            if adj_node not in visited:
                dfs_helper(adj_node)
            elif graph[adj_node][0] == "grey":
                print("Cycle in {}".format(adj_node))
        graph[node] = ("black", graph[node][1])

    dfs_helper(root)

if __name__ == "__main__":
    debug = len(sys.argv) > 3
    graph = read_graph(sys.argv[1], debug)
    root = sys.argv[2]
    
    if debug:
        dump_graph(graph)
        
    print("Root key:", root)
    graph[root] = ('black', graph[root][1])
    
    bfs_result = bfs(graph, root)
    print("BFS:")
    print(bfs_result)
    
    if debug:
        dump_graph(graph)
        
    paint_white(graph)
    
    if debug:
        dump_graph(graph)
        
    print("DFS:")
    dfs(root)
    
    if debug:
        dump_graph(graph)
