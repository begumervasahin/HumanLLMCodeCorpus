class BFS:
    def breadth_first_search(self, graph, source, terminal, family):
        # Initialize a list to keep track of visited vertices
        visited = [False] * len(graph)
        
        # Initialize a queue for BFS
        queue = []
        
        # Add the source vertex to the queue
        queue.append(source)
        
        # Mark the source vertex as visited
        visited[source] = True

        # Perform BFS
        while queue:
            # Get the vertex at the front of the queue
            current_vertex = queue.pop(0)

            # Iterate through the neighbors of the current vertex
            for neighbor_idx, edge_weight in enumerate(graph[current_vertex]):
                # Check if there is an edge and the neighbor hasn't been visited
                if edge_weight > 0 and not visited[neighbor_idx]:
                    # Add the neighbor to the queue
                    queue.append(neighbor_idx)
                    
                    # Mark the neighbor as visited
                    visited[neighbor_idx] = True
                    
                    # Record the parent vertex of the neighbor
                    family[neighbor_idx] = current_vertex

        # Check if the terminal vertex is reachable
        if visited[terminal]:
            return True
        else:
            return False
