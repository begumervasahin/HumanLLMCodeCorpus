from collections import deque
from enum import Enum


class Node(Enum):
    """
    Enumeration representing the state of a grid node.
    """
    BLACK = 5
    WHITE = 2
    EMPTY = 0


class Grid:
    """
    Represents a grid for the Go game.
    """

    def __init__(self, size, node_pairs):
        """
        Initializes a grid with the given size and node pairs.

        Args:
            size (int): The size of the grid.
            node_pairs (list): List of tuples containing node coordinates and their state.
        """
        self.board = [[Node.EMPTY for _ in range(size)] for _ in range(size)]
        self.size = size
        for node in node_pairs:
            self.set_val(self.board, node[0], node[1])

    def get(self, arr, location):
        """
        Retrieves the value at the given location in the grid.

        Args:
            arr (list): The grid array.
            location (tuple): The location coordinates.

        Returns:
            Node: The value at the specified location.
        """
        return arr[location[0]][location[1]]

    def set_val(self, arr, location, value):
        """
        Sets the value at the given location in the grid.

        Args:
            arr (list): The grid array.
            location (tuple): The location coordinates.
            value (Node): The value to set.
        """
        arr[location[0]][location[1]] = value

    def is_captured(self, location):
        """
        Checks if the black node at the given location is captured.

        Args:
            location (tuple): The location coordinates.

        Returns:
            bool: True if the black node is captured, False otherwise.
        """
        if self.get(self.board, location) == Node.WHITE:
            return False
        if self.isnt_out(location):
            queue = deque()
            queue.appendleft(location)
            visited = [[False for _ in range(self.size)] for _ in range(self.size)]
            self.set_val(visited, location, True)
            while len(queue) != 0:
                curr = queue.popleft()
                self.set_val(visited, curr, True)
                if self.is_surrounded(queue, visited, curr):
                    continue
                else:
                    return False
            return True
        else:
            return False

    def isnt_out(self, location):
        """
        Checks if the location is within the grid boundaries.

        Args:
            location (tuple): The location coordinates.

        Returns:
            bool: True if the location is within the grid boundaries, False otherwise.
        """
        return 0 <= location[0] < self.size and 0 <= location[1] < self.size

    def is_surrounded(self, q, visited, curr):
        """
        Checks if the current node is surrounded by white nodes.

        Args:
            q (deque): The queue for BFS.
            visited (list): List to track visited nodes.
            curr (tuple): The current node coordinates.

        Returns:
            bool: True if the current node is surrounded, False otherwise.
        """
        neighbors = [[curr[0], curr[1] + 1], [curr[0], curr[1] - 1],
                     [curr[0] + 1, curr[1]], [curr[0] - 1, curr[1]]]
        for loc in neighbors:
            if self.isnt_out(loc):
                if not self.get(visited, loc):
                    if self.get(self.board, loc) == Node.WHITE:
                        self.set_val(visited, loc, True)
                        continue
                    elif self.get(self.board, loc) == Node.EMPTY:
                        return False
                    elif self.get(self.board, loc) == Node.BLACK:
                        q.append(loc)
            else:
                continue
        return True

                        ([2, 3], Node.BLACK),
                        ])
