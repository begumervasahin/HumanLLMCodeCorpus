class Stack:
    """
    Stack data structure implementation.
    """

    def __init__(self):
        """
        Initializes an empty stack.
        """
        self.list = []  # List to store stack elements
        self.state = set()  # Set to keep track of unique states

    def push(self, item):
        """
        Pushes an item onto the top of the stack.
        """
        self.list.append(item)  # Add item to the stack
        self.state.add(tuple(item.list))  # Add state to the set

    def pop(self):
        """
        Removes and returns the item at the top of the stack in LIFO order.
        """
        if not self.isEmpty():
            tos = self.list[-1]  # Get the top item from the stack
            del self.list[-1]  # Remove the top item
            self.state.remove(tuple(tos.list))  # Remove state from set
            return tos
        else:
            return 0

    def isEmpty(self):
        """
        Checks if the stack is empty.
        """
        return len(self.list) == 0

    def clear_stack(self):
        """
        Clears the stack.
        """
        self.list.clear()  # Clear the list
        self.state.clear()  # Clear the set
