import collections

# Function to read wrestler data from the input file
def read_input(file_name):
    wrestlers = {}
    with open(file_name, "r") as f:
        lines = f.readlines()
        num_players = int(lines[0])
        num_rivalries = int(lines[num_players + 1])
        for line in lines[num_players + 2:]:
            wrestler1, wrestler2 = line.strip().split()
            wrestlers.setdefault(wrestler1, []).append(wrestler2)
            wrestlers.setdefault(wrestler2, []).append(wrestler1)
    return wrestlers

# Function to determine if it's possible to categorize wrestlers as baby faces and heels
def is_possible(wrestlers):
    baby_faces = []
    heels = []
    queue = collections.deque()

    # BFS to assign baby faces and heels
    for wrestler in wrestlers:
        if wrestler not in baby_faces and wrestler not in heels:
            baby_faces.append(wrestler)
            queue.append(wrestler)

            while queue:
                current_wrestler = queue.popleft()
                for rival in wrestlers[current_wrestler]:
                    if rival not in baby_faces and rival not in heels:
                        heels.append(rival)
                        queue.append(rival)
                    elif rival in baby_faces:
                        return False, [], []

    return True, baby_faces, heels

# Function to print the results
def print_results(possible, baby_faces, heels):
    if possible:
        print("Yes, it's possible")
        print("Baby Faces:", baby_faces)
        print("Heels:", heels)
    else:
        print("No, it's not possible")

# Main function
if __name__ == "__main__":
    file_name = "wrestler.txt"  # Default input file name
    wrestlers = read_input(file_name)
    possible, baby_faces, heels = is_possible(wrestlers)
    print_results(possible, baby_faces, heels)
