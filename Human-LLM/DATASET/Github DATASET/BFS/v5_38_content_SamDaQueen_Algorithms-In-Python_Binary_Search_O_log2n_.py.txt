
def binary_search(input_array, value):
    """
    Perform binary search to find the index of the given value in the sorted input_array.

    Parameters:
    input_array (list): A sorted list of integers.
    value (int): The value to search for in the input_array.

    Returns:
    int: The index of the value in the input_array if found, otherwise -1.
    """
    low = 0
    high = len(input_array) - 1  # Adjusted high index to last valid index
    while low <= high:
        mid = (low + high) // 2
        if input_array[mid] == value:
            return mid
        elif input_array[mid] < value:
            low = mid + 1
        else:
            high = mid - 1
    return -1

# Test cases
test_list = [1, 3, 9, 11, 15, 19, 29]
test_val1 = 25
test_val2 = 15
print(binary_search(test_list, test_val1))  # Output: -1 (not found)
print(binary_search(test_list, test_val2))  # Output: 4 (found at index 4)
