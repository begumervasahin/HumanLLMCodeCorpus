import collections

# Function to perform breadth-first search on a grid
def bfs(grid, start):
    queue = collections.deque([[start]])  # Initialize queue with start position as a list
    seen = set([start])  # Set to store visited positions
    while queue:
        path = queue.popleft()  # Dequeue the first path from the queue
        x, y = path[-1]  # Get the last position in the path
        if grid[y][x] == goal:  # Check if the current position is the goal
            return path  # Return the path if the goal is found
        # Explore neighboring positions
        for x2, y2 in ((x+1, y), (x-1, y), (x, y+1), (x, y-1)):
            # Check if the neighboring position is within the grid and not a wall
            if 0 <= x2 < collumns and 0 <= y2 < rows and grid[y2][x2] != wall and (x2, y2) not in seen:
                queue.append(path + [(x2, y2)])  # Add the new path to the queue
                seen.add((x2, y2))  # Mark the neighboring position as visited

# Define constants for grid symbols and dimensions
wall, clear, goal = "#", ".", "*"
collumns, rows = 10, 5  # One more than the grid because it's indexed from 0

# Define the grid layout
grid = [
    "..........",
    "...#*..##.",
    "..##...#..",
    ".....###..",
    ".........."
]

# Example usage:
start = (0, 0)  # Start position
print(bfs(grid, start))  # Perform BFS from the start position
