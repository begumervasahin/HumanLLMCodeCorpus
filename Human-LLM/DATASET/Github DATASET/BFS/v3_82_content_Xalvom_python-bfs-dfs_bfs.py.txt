from collections import deque

def breadth_first_search(graph, start, goal):
    """
    Performs Breadth-First Search (BFS) on a given graph.

    Parameters:
    - graph: The graph represented as a dictionary.
    - start: The starting node for the search.
    - goal: The target node to reach.

    Returns:
    - The path from the start node to the goal node if found, otherwise None.
    """
    queue = deque([[start]])  # Initialize the queue with the starting node
    visited = set()           # Set to keep track of visited nodes

    while queue:
        path = queue.popleft()     # Dequeue the first path from the queue
        state = path[-1]           # Get the current state from the path

        if state == goal:
            return path            # If the goal is reached, return the path

        elif state not in visited:
            for neighbor in graph.get(state, []):
                new_path = path + [neighbor]
                queue.append(new_path)  # Enqueue new paths with neighboring nodes

            visited.add(state)  # Mark the current state as visited

    print("Tidak ditemukan")  # If the goal is not reached, print a message
    return None

# Define the graph
graph = {
    'A': {'B'},
    'B': {'A', 'C'},
    'C': {'B', 'D', 'H', 'I'},
    'D': {'C', 'E', 'F', 'H', 'M'},
    'E': {'D'},
    'F': {'D', 'G', 'M'},
    'G': {'F', 'H'},
    'H': {'C', 'D', 'G', 'L'},
    'I': {'C', 'J', 'K'},
    'J': {'I'},
    'K': {'I', 'L'},
    'L': {'H', 'K'},
    'M': {'D', 'F'}
}

print("********* BFS ***********")
start = input("Input Awal: ")
goal = input("Input Tujuan: ")
print("**************************")
print()

# Perform BFS
result = breadth_first_search(graph, start, goal)
if result:
    print("Jalur BFS:", result)
else:
    print("Tidak ditemukan")
