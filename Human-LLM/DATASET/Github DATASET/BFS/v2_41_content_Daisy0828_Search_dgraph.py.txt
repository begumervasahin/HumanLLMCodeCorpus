# Define a generic search problem class
class SearchProblem:
    # Method to get the start state of the problem
    def get_start_state(self):
        raise NotImplementedError

    # Method to check if a given state is a goal state
    def is_goal_state(self, state):
        raise NotImplementedError

    # Method to get successors of a given state
    def get_successors(self, state):
        raise NotImplementedError


# Define a class for representing a directed graph, which is a type of search problem
class DirectedGraph(SearchProblem):
    # Constructor to initialize the directed graph with a matrix representation,
    # a set of goal indices, and an optional start state
    def __init__(self, matrix, goal_indices, start_state=0):
        """
        matrix - the matrix representation of the directed graph
        goal_indices - a Python set of the indices of the states that are goal states.
        start_state - the index of the start state. 0 by default.
        """
        self.matrix = matrix
        self.goal_indices = goal_indices
        self.start_state = start_state

    # Method to get the start state of the graph
    def get_start_state(self):
        return self.start_state

    # Method to check if a given state is a goal state
    def is_goal_state(self, state):
        return state in self.goal_indices

    # Method to get successors of a given state
    def get_successors(self, state):
        row = self.matrix[state]
        successors = {}
        index = 0
        for cost in row:
            if cost is not None:
                successors[index] = cost
            index += 1
        return successors
