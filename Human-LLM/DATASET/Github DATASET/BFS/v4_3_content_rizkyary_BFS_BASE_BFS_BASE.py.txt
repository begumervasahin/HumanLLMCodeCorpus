# Define the graph
graph = {
    'A': set(['B', 'C']),
    'B': set(['A', 'H', 'J']),
    'C': set(['A', 'D', 'G']),
    'D': set(['C', 'E']),
    'E': set(['D', 'F']),
    'F': set(['E', 'G', 'K', 'L']),
    'G': set(['C', 'F']),
    'H': set(['B', 'I']),
    'I': set(['H', 'J', 'K']),
    'J': set(['B', 'I']),
    'K': set(['F', 'I', 'L']),
    'L': set(['F', 'K'])
}

def bfs(graph, start, goal):
    """
    Perform Breadth-First Search (BFS) on the given graph to find a path from start to goal.

    Parameters:
    graph (dict): The graph in the form of adjacency list.
    start (str): The starting node.
    goal (str): The goal node.

    Returns:
    list or None: A list representing the path from start to goal if found, else None.
    """
    queue = [[start]]  # Initialize the queue with the start node
    visited = set()  # Initialize a set to keep track of visited nodes

    while queue:
        path = queue.pop(0)  # Dequeue the first path from the queue
        state = path[-1]  # Get the current node from the path

        if state == goal:  # If the current node is the goal node, return the path
            return path

        if state not in visited:  # If the current node has not been visited
            for neighbor in graph.get(state, []):  # Iterate through the neighbors of the current node
                new_path = list(path)  # Create a new path by copying the current path
                new_path.append(neighbor)  # Append the neighbor to the new path
                queue.append(new_path)  # Enqueue the new path
            visited.add(state)  # Mark the current node as visited

    print("Tidak ditemukan")  # Print message if the goal node is not found
    return None

# Example usage:
start_node = 'A'
goal_node = 'L'
result = bfs(graph, start_node, goal_node)
print("Path found:", result) if result else print("Path not found")
