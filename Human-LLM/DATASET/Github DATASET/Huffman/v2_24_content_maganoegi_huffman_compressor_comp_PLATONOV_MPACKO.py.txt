from tqdm import tqdm
import os
import time
from huffman_module import *
from comp_lib import *

def main():
    # Clear the console screen
    os.system('cls' if os.name == 'nt' else 'clear')

    # Print header
    hepia_print()
    print("\n")

    # Define tqdm progress bar format
    bar_format = '{l_bar}{bar}| {n_fmt}/{total_fmt}'

    # Create a tqdm progress bar with a total of 9 steps
    with tqdm(total=9, ncols=80, bar_format=bar_format) as pbar:
        # Read user input for input and output file names
        filenames = read_user_input()

        # Update progress bar for opening file
        update_bar(pbar, 'Open File  ')
        file_2_read = filenames[0]
        file_2_write = filenames[1]

        # Convert file to byte array
        byte_array = bytes_2_array(file_2_read)

        # Update progress bar for file to array conversion
        update_bar(pbar, 'File->Array')

        # Calculate the hash value of the original file
        HashVal = hashOriginal(file_2_read)

        # Update progress bar for hashing
        update_bar(pbar, 'Hashing    ')

        # Determine frequencies of each byte in the array
        freq_dict = determine_frequencies(byte_array)

        # Update progress bar for frequency determination
        update_bar(pbar, 'Frequencies')

        # Apply Huffman encoding to generate a compressed list
        compressed_list = huffman(freq_dict)

        # Update progress bar for Huffman encoding
        update_bar(pbar, 'Huffman    ')

        # Translate the byte array using the compressed list
        byte_string = translate(byte_array, compressed_list)

        # Update progress bar for translation
        update_bar(pbar, 'Translation')

        # Finalize the byte string
        final_byte_array = finalize(byte_string)

        # Update progress bar for array to file conversion
        update_bar(pbar, 'Array->File')

        # Write the compressed dictionary to a JSON file
        write_2_json(compressed_list, "dict.json")

        # Update progress bar for writing dictionary to file
        update_bar(pbar, 'dict.json  ')

        # Write the final compressed byte array to the output file
        write_2_binary(final_byte_array, file_2_write)

        # Update progress bar for completion
        update_bar(pbar, 'Done       ')

        # Add a short delay for display purposes
        time.sleep(0.1)

        # Close the progress bar
        pbar.close()

        # Display information about the compression
        compressor_display(freq_dict, compressed_list, file_2_read, file_2_write, byte_array, HashVal)

if __name__ == "__main__":
    main()
