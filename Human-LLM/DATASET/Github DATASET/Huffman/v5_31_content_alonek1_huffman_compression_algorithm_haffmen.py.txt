import queue
from collections import Counter

def read_text(filename):
    """
    Read text from a file.

    Args:
        filename (str): The name of the file to read.

    Returns:
        str: The text read from the file.
    """
    with open(filename, 'r') as file:
        return file.read()

def calculate_frequencies(text):
    """
    Calculate the frequencies of characters in the text.

    Args:
        text (str): The input text.

    Returns:
        dict: A dictionary mapping characters to their frequencies.
    """
    return Counter(text)

def calculate_probabilities(frequencies):
    """
    Calculate the probabilities of characters based on their frequencies.

    Args:
        frequencies (dict): A dictionary mapping characters to their frequencies.

    Returns:
        dict: A dictionary mapping characters to their probabilities.
    """
    total_sum = sum(frequencies.values())
    return {char: freq / total_sum for char, freq in frequencies.items()}

class HuffmanNode:
    """
    Represents a node in the Huffman tree.
    """

    def __init__(self, left=None, right=None, root=None):
        """
        Initialize a HuffmanNode object.

        Args:
            left (HuffmanNode): The left child node.
            right (HuffmanNode): The right child node.
            root (HuffmanNode): The root node.
        """
        self.left = left
        self.right = right
        self.root = root

    def children(self):
        """
        Get the children nodes of the current node.

        Returns:
            tuple: A tuple containing the left and right child nodes.
        """
        return self.left, self.right

def create_tree(frequencies):
    """
    Create a Huffman tree based on character frequencies.

    Args:
        frequencies (dict): A dictionary mapping characters to their frequencies.

    Returns:
        HuffmanNode: The root node of the Huffman tree.
    """
    priority_queue = queue.PriorityQueue()
    for char, freq in frequencies.items():
        priority_queue.put((freq, HuffmanNode(None, None, char)))

    while priority_queue.qsize() > 1:
        left, right = priority_queue.get(), priority_queue.get()
        node = HuffmanNode(left, right)
        priority_queue.put((left[0] + right[0], node))

    return priority_queue.get()

def walk_tree(node, prefix="", code={}):
    """
    Recursively traverse the Huffman tree to generate codes for characters.

    Args:
        node (HuffmanNode): The current node being traversed.
        prefix (str): The binary prefix code.
        code (dict): A dictionary to store the Huffman codes for characters.

    Returns:
        dict: The dictionary containing the Huffman codes for characters.
    """
    if isinstance(node[1].left[1], HuffmanNode):
        walk_tree(node[1].left, prefix + "0", code)
    else:
        code[node[1].left[1]] = prefix + "0"

    if isinstance(node[1].right[1], HuffmanNode):
        walk_tree(node[1].right, prefix + "1", code)
    else:
        code[node[1].right[1]] = prefix + "1"

    return code

def encode_text(text, code):
    """
    Encode the text using Huffman codes.

    Args:
        text (str): The text to be encoded.
        code (dict): A dictionary containing Huffman codes for characters.

    Returns:
        str: The encoded text.
    """
    return ''.join(code[char] for char in text)

if __name__ == '__main__':
    # Read text from file
    text = read_text('rand_file')

    # Calculate character frequencies and probabilities
    frequencies = calculate_frequencies(text)
    probabilities = calculate_probabilities(frequencies)

    # Create Huffman tree
    huffman_tree = create_tree(probabilities)

    # Generate Huffman codes
    huffman_codes = walk_tree(huffman_tree)

    # Display character frequencies and codes
    for char, freq in sorted(probabilities.items(), key=lambda x: x[1], reverse=True):
        print(char, '{:.10f}'.format(freq), huffman_codes[char])

    # Encode the text using Huffman codes
    encoded_text = encode_text(text, huffman_codes)

    # Print the encoded string
    print(encoded_text)
