def huffman_decode(text, root):
    """
    Decode the binary code using the provided Huffman tree.

    Args:
        text (str): The binary code to decode.
        root: The root node of the Huffman tree.

    Returns:
        str: The decoded text.
    """
    decoded_text = ""
    current_node = root

    for bit in text:
        if not is_internal_node(current_node):
            decoded_text += current_node.val
            current_node = root

        if bit == "0":
            current_node = current_node.left
        elif bit == "1":
            current_node = current_node.right

    if not is_internal_node(current_node):
        decoded_text += current_node.val

    return decoded_text

def is_internal_node(node):
    """
    Check if a node is an internal node (non-leaf) in the Huffman tree.

    Args:
        node: The node to check.

    Returns:
        bool: True if the node is an internal node, False otherwise.
    """
    return node.left is not None or node.right is not None
