# Huffman Encoding Program

def read_file(filename):
    """Reads content from a file."""
    with open(filename, 'r') as file:
        return file.read()

def extract_letters_and_frequencies(contents):
    """Extracts unique letters and their frequencies from the content."""
    letters_with_frequency = {}
    for letter in contents:
        letters_with_frequency[letter] = letters_with_frequency.get(letter, 0) + 1
    return letters_with_frequency

def build_huffman_tree(letters_with_frequency):
    """Builds the Huffman tree."""
    nodes = [[weight, [char, '']] for char, weight in letters_with_frequency.items()]
    while len(nodes) > 1:
        nodes.sort()
        left_child = nodes.pop(0)
        right_child = nodes.pop(0)
        combined_node = [left_child[0] + right_child[0], [left_child, right_child]]
        nodes.append(combined_node)
    return nodes[0]

def generate_codes(huffman_tree):
    """Generates binary codes for each letter based on the Huffman tree."""
    codes = {}

    def traverse_tree(node, code=''):
        if len(node[1]) == 2:
            traverse_tree(node[1][0], code + '0')
            traverse_tree(node[1][1], code + '1')
        else:
            codes[node[1][0]] = code

    traverse_tree(huffman_tree)
    return codes

def display_codes_and_ascii(codes):
    """Displays binary codes for each character along with their corresponding ASCII values."""
    print('Binary codes are:')
    print('Character\tBinary Huffman\tBinary ASCII')
    for char, code in codes.items():
        print(f'\'{char}\'\t\t{code}\t\t{bin(ord(char))[2:]}')

def huffman_encode(contents, codes):
    """Encodes the content using Huffman codes."""
    bitstring = ''.join(codes[char] for char in contents)
    return bitstring

def main():
    print('Huffman Encoding Program')
    filename = input('Enter the name of a text file to open: ') + '.txt'

    contents = read_file(filename)
    letters_with_frequency = extract_letters_and_frequencies(contents)
    huffman_tree = build_huffman_tree(letters_with_frequency)
    codes = generate_codes(huffman_tree)

    display_codes_and_ascii(codes)

    encoded_content = huffman_encode(contents, codes)

    print(f"The contents of the file are: \'{contents}\'")
    print(f'The Huffman encoded contents are: \'{encoded_content}\'')

if __name__ == "__main__":
    main()
