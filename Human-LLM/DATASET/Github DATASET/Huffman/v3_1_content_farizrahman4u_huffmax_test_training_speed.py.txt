import time
import numpy as np
from keras.layers import Dense, Input
from keras.models import Sequential, Model
from huffmax import Huffmax

def generate_random_data(nb_samples, input_dim, nb_classes):
    """
    Generate random input data and labels.
    """
    X = np.random.random((nb_samples, input_dim))
    Y_huffmax = np.random.randint(0, nb_classes, size=(nb_samples, 1))
    Y_softmax = np.zeros((nb_samples, nb_classes))

    for i in range(nb_samples):
        random_index = np.random.randint(0, nb_classes)
        Y_softmax[i, random_index] = 1

    return X, Y_huffmax, Y_softmax

def train_softmax_model(X, Y_softmax, input_dim, nb_classes, batch_size):
    """
    Define and train the softmax model.
    """
    softmax_model = Sequential()
    softmax_model.add(Dense(input_dim=input_dim, units=nb_classes, activation='softmax'))
    softmax_model.compile(loss='mse', optimizer='sgd')
    softmax_model.predict(X[:1])

    start_time = time.time()
    softmax_model.fit(X, Y_softmax, batch_size=batch_size)
    end_time = time.time()

    training_time = end_time - start_time
    del softmax_model  # Free memory
    return training_time

def train_huffmax_model(X, Y_huffmax, input_dim, nb_classes, batch_size, mode):
    """
    Define and train the Huffmax model.
    """
    vector = Input((input_dim,))
    target_class = Input((1,))
    probability = Huffmax(nb_classes, verbose=True, mode=mode)([vector, target_class])

    huffmax_model = Model(inputs=[vector, target_class], outputs=probability)
    huffmax_model.compile(loss='mse', optimizer='sgd')
    huffmax_model.predict([X[:1], Y_huffmax[:1]])

    start_time = time.time()
    huffmax_model.fit([X, Y_huffmax], np.ones((len(X), 1)), batch_size=batch_size)
    end_time = time.time()

    training_time = end_time - start_time
    return training_time

def main():
    # Define constants
    batch_size = 32
    input_dim = 100
    nb_classes = 100000
    nb_samples = 10000
    times = {}

    # Generate random input data and labels
    X, Y_huffmax, Y_softmax = generate_random_data(nb_samples, input_dim, nb_classes)

    # Train the softmax model
    softmax_training_time = train_softmax_model(X, Y_softmax, input_dim, nb_classes, batch_size)
    times['Softmax'] = softmax_training_time

    # Train the Huffmax model for both modes
    for mode in [0, 1]:
        huffmax_training_time = train_huffmax_model(X, Y_huffmax, input_dim, nb_classes, batch_size, mode)
        times[f'Huffmax (mode {mode})'] = huffmax_training_time

    # Print training times for each model
    for key, value in times.items():
        print(f'{key}: {value}')

if __name__ == "__main__":
    main()
