from minheap import MinHeap
from collections import Counter

# Define classes to represent tree nodes
class TreeLeafEndMessage:
    pass

class TreeLeaf:
    def __init__(self, value):
        self.value = value

class TreeBranch:
    def __init__(self, left, right):
        self.left = left
        self.right = right

# Function to build the Huffman tree
def build_huffman_tree(freq_table):
    trees = MinHeap()
    trees.add(1, TreeLeafEndMessage())  # Add a special end-of-message leaf to the heap
    # Add each symbol and its frequency to the heap as TreeLeaf nodes
    for symbol, freq in freq_table.items():
        trees.add(freq, TreeLeaf(symbol))
    # Merge nodes from the heap to build the Huffman tree
    while len(trees) > 1:
        rfreq, right = trees.pop_min()
        lfreq, left = trees.pop_min()
        trees.add(lfreq + rfreq, TreeBranch(left, right))
    # Return the root of the Huffman tree
    totalfreq, tree = trees.pop_min()
    return tree

# Function to decode a bitstream using the Huffman tree
def decode_huffman(tree, bitreader):
    while True:
        # If we reach the end of a message, return None
        if isinstance(tree, TreeLeafEndMessage):
            return None
        # If we reach a leaf node, return the symbol value
        elif isinstance(tree, TreeLeaf):
            return tree.value
        # If we reach an internal node, traverse left or right based on the next bit in the bitstream
        elif isinstance(tree, TreeBranch):
            if bitreader.readbit() == 0:
                tree = tree.left
            else:
                tree = tree.right
        else:
            raise TypeError('{} is not a tree type'.format(type(tree)))

# Function to create an encoding table from the Huffman tree
def build_encoding_table(huffman_tree):
    table = {}
    # Helper function to recursively traverse the tree and build the encoding table
    def traverse_tree(node, path):
        if isinstance(node, TreeLeafEndMessage):
            table[None] = path
        elif isinstance(node, TreeLeaf):
            table[node.value] = path
        elif isinstance(node, TreeBranch):
            traverse_tree(node.left, path + (False,))
            traverse_tree(node.right, path + (True,))
        else:
            raise TypeError('{} is not a tree type'.format(type(node)))
    traverse_tree(huffman_tree, ())
    return table

# Function to create a frequency table from a stream of bytes
def build_freq_table(stream):
    freqs = Counter()
    buffer = bytearray(512)  # Buffer to read bytes from the stream
    # Read bytes from the stream and update the frequency table
    while True:
        count = stream.readinto(buffer)
        freqs.update(buffer[:count])
        if count < len(buffer):
            break
    return freqs
