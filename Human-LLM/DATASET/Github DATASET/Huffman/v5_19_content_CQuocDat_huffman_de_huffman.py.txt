import numpy as np
import cv2

class Node:
    def __init__(self):
        self.gray_value = None
        self.is_leaf = None
        self.node_id = None
        self.left = None
        self.right = None

def build_tree(parent_node, bit_array, bit_index, index):
    if bit_index == len(bit_array) - 1:
        # Leaf node
        child_node = parent_node.left if bit_array[bit_index] == 0 else parent_node.right
        if child_node is None:
            new_node = Node()
            new_node.node_id = bit_array[bit_index]
            child_node = parent_node.left if bit_array[bit_index] == 0 else parent_node.right
            child_node = new_node
            new_node.gray_value = index
            new_node.is_leaf = True
    else:
        # Internal node
        child_node = parent_node.left if bit_array[bit_index] == 0 else parent_node.right
        if child_node is None:
            new_node = Node()
            new_node.node_id = bit_array[bit_index]
            child_node = new_node
            new_node.is_leaf = False
            build_tree(child_node, bit_array, bit_index + 1, index)
        else:
            build_tree(child_node, bit_array, bit_index + 1, index)

def find_leaf_nodes(code, node, index):
    if node.is_leaf:
        leaf_values.append(node.gray_value)
        leaf_indices.append(index)
    else:
        if code[index] == '0':
            find_leaf_nodes(code, node.left, index + 1)
        elif code[index] == '1':
            find_leaf_nodes(code, node.right, index + 1)

def read_tree_and_image_data():
    tree_array = []
    for line in dictionary_file:
        code_list = [int(bit) for bit in line.split(' ')[1].rstrip()]
        tree_array.append(code_list)

    width, height = tree_array.pop(), tree_array.pop()
    return tree_array, width, height

def decode_image(image_file, root_node):
    decoded_image = []
    for line in image_file:
        code_list = list(line.split()[0])
        current_index = 0
        leaf_values = []

        while current_index < len(code_list) - 1:
            find_leaf_nodes(code_list, root_node, current_index)
            current_index = leaf_indices[-1]

        decoded_image.append(leaf_values)

    return np.array(decoded_image, dtype=np.uint8)

def main():
    global leaf_values, leaf_indices
    leaf_values, leaf_indices = [], []

    tree_array, width, height = read_tree_and_image_data()

    root_node = Node()
    for i, code_list in enumerate(tree_array):
        build_tree(root_node, code_list, 0, i)

    decoded_image = decode_image(image_file, root_node)

    print("Image Dimensions:", height, width)

    cv2.imshow('DECOMPRESS', decoded_image)
    cv2.imwrite('meo-decompress.png', decoded_image)
    cv2.waitKey(0)
    cv2.destroyAllWindows()

if __name__ == "__main__":
    dictionary_file = open('dictionary_meo.txt', 'r')
    image_file = open('imageinBit_meo.txt', 'r')
    main()
