class FileManager:
    def load_text(self, filename):
        """
        Load the content of a text file.
        """
        text = ""
        with open(filename, 'r') as file:
            text = file.read()
        return text

    def save_text(self, text, filename):
        """
        Save text to a file.
        """
        with open(filename, 'w') as file:
            file.write(text)

    def encode_and_save(self, filename, text, codes):
        """
        Encode text using the provided codes and save it to a file.
        """
        binary_string = ''.join(codes[char] for char in text)
        encoded_text = self._binary_string_to_text(binary_string)
        with open(filename, 'w') as file:
            file.write(encoded_text)

    def load_codes(self, filename):
        """
        Load character codes from a file.
        """
        with open(filename, "r") as file:
            text = file.read()

        codes = {}
        pairs = text.split(",")
        pairs = [pair.strip() for pair in pairs]

        for pair in pairs:
            if pair:
                char, code = pair.split("-")
                codes[code] = char

        return codes

    def _binary_string_to_text(self, binary_string):
        """
        Convert a binary string to text, adding padding if necessary.
        """
        text = ''
        for i in range(0, len(binary_string) - 8, 8):
            text += chr(int(binary_string[i:i + 8], 2))

        if len(binary_string) % 8 != 0:
            text += "~~~" + binary_string[len(binary_string) - 1 - len(binary_string) % 8:len(binary_string)]

        return text

# Example usage:
file_manager = FileManager()
codes = file_manager.load_codes("codes.txt")
print(codes)

text = file_manager.load_text("input.txt")
print(text)

file_manager.encode_and_save("output.txt", text, codes)
