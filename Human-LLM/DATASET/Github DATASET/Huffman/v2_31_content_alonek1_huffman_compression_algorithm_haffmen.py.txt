from collections import Counter
import queue

# Read the text from the file
text = open('rand_file').read()

# Count the occurrences of each character in the text
count_ = Counter(text)

# Calculate the probabilities of each character
sum_ = sum(count_.values())
probabilities = {char: count_[char] / sum_ for char in count_}

class HuffmanNode:
    """
    Represents a node in the Huffman tree.
    """

    def __init__(self, left=None, right=None, root=None):
        self.left = left
        self.right = right
        self.root = root

def create_tree(frequencies):
    """
    Creates a Huffman tree from the given character frequencies.
    """
    pq = queue.PriorityQueue()
    for value in frequencies:
        pq.put(value)
    while pq.qsize() > 1:
        left, right = pq.get(), pq.get()
        node = HuffmanNode(left, right)
        pq.put((left[0] + right[0], node))
    return pq.get()

# Convert character probabilities to (probability, character) tuples
frequencies = [(probabilities[char], char) for char in probabilities]

# Build the Huffman tree
root_node = create_tree(frequencies)

def walk_tree(node, prefix="", code={}):
    """
    Traverse the Huffman tree and assign binary codes to characters.
    """
    if isinstance(node[1].left[1], HuffmanNode):
        walk_tree(node[1].left, prefix + "0", code)
    else:
        code[node[1].left[1]] = prefix + "0"
    if isinstance(node[1].right[1], HuffmanNode):
        walk_tree(node[1].right, prefix + "1", code)
    else:
        code[node[1].right[1]] = prefix + "1"
    return code

# Generate Huffman codes for each character
huffman_code = walk_tree(root_node)

# Print character frequencies and their corresponding Huffman codes
for char, probability in sorted(probabilities.items(), key=lambda x: x[1], reverse=True):
    print(char, '{:6.10f}'.format(probability), huffman_code[char])

# Encode the text using the generated Huffman codes
encoded_text = "".join(huffman_code[char] for char in text)

print(encoded_text)
