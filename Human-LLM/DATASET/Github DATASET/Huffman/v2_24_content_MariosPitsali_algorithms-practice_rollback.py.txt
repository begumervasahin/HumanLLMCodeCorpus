import sqlite3
import datetime
import pytz

# Establish a connection to the SQLite database
db = sqlite3.connect("accounts.sqlite", detect_types=sqlite3.PARSE_DECLTYPES)

# Create the accounts and history tables if they do not exist
db.execute("CREATE TABLE IF NOT EXISTS accounts(name TEXT PRIMARY KEY NOT NULL, balance INTEGER NOT NULL)")
db.execute("CREATE TABLE IF NOT EXISTS history (time TIMESTAMP NOT NULL, account TEXT NOT NULL, amount INTEGER NOT NULL, PRIMARY KEY(time, account))") 
db.execute("CREATE VIEW IF NOT EXISTS localhistory AS SELECT strftime('%Y-%m-%d %H:%M:%f', history.time, 'localtime') AS localtime, history.account, history.amount FROM history ORDER BY history.time")

class Account(object):
    """
    Represents a bank account.
    """

    def __init__(self, name: str, opening_balance: int = 0):
        """
        Initializes a new Account object with the given name and opening balance.
        """
        cursor = db.execute("SELECT name, balance FROM accounts WHERE (name = ?)", (name,))
        row = cursor.fetchone()
        if row is not None:
            self.name, self._balance = row
            print("Retrieved balance for {}".format(self.name), end="")
        else:
            self.name = name
            self._balance = opening_balance / 100
            cursor.execute("INSERT INTO accounts VALUES (?, ?)", (name, opening_balance))
            cursor.connection.commit()
            print("Account created for {0.name}".format(self))
        self.show_balance()

    def show_balance(self):
        """
        Prints the current balance of the account.
        """
        print("Balance on account {0.name} is {0._balance}".format(self))

    def current_time(self):
        """
        Returns the current time in UTC timezone.
        """
        return datetime.datetime.now(pytz.timezone('UTC'))

    def deposit(self, amount: int) -> float:
        """
        Deposits the specified amount into the account and updates the balance.
        Returns the new balance.
        """
        if amount > 0.0:
            self._save_update(amount)
            print("{:.2f} deposited".format(amount / 100))
        return self._balance

    def withdraw(self, amount: int) -> float:
        """
        Withdraws the specified amount from the account and updates the balance.
        Returns the new balance.
        """
        if self._balance > amount > 0.0:
            self._save_update(-amount)
            print("{:.2f} withdrawn".format(amount / 100))
            return self._balance
        else:
            print("Amount must be more than zero and less than your account balance.")
            return self._balance

    def _save_update(self, amount):
        """
        Updates the balance and saves the transaction to the history table.
        """
        new_balance = self._balance + amount
        deposit_time = self.current_time()
        db.execute("UPDATE accounts SET balance = ? WHERE (name = ?)", (new_balance, self.name))
        db.execute("INSERT INTO history VALUES (?, ?, ?)", (deposit_time, self.name, amount))
        db.commit()
        self._balance = new_balance
