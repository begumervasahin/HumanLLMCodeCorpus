def huffman_decode(text, root):
    """Decode the code word using the provided Huffman tree."""
    result = ""
    current_node = root
    
    for bit in text:
        if not is_leaf(current_node) and bit == "0":
            current_node = current_node.left
        elif not is_leaf(current_node) and bit == "1":
            current_node = current_node.right
        
        if is_leaf(current_node):
            result += current_node.val
            current_node = root
    
    return result

def is_leaf(node):
    """Check if a node is a leaf in the Huffman tree."""
    return node.left is None and node.right is None
