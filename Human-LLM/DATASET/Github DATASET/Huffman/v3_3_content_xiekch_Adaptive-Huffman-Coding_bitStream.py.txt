class BitStream:
    def __init__(self, file_name, mode):
        """
        Initialize the BitStream object with the specified file name and mode.
        """
        self.file = open(file_name, mode)
        self.word = 0
        self.pos = -1 if mode[0] == 'r' else 7
        self.mode = mode

    def read(self, size):
        """
        Read binary data from the file.
        """
        ret = ''
        for _ in range(size):
            if self.pos == -1:
                self.word = ord(self.file.read(1)) if (data := self.file.read(1)) else ''
                self.pos = 7
            ret += '1' if self.word & (1 << self.pos) else '0'
            self.pos -= 1
        return ret

    def write(self, string):
        """
        Write binary data to the file.
        """
        for char in string:
            if char == '0':
                self.word <<= 1
            elif char == '1':
                self.word = (self.word << 1) + 1
            else:
                continue
            self.pos -= 1
            if self.pos == -1:
                self.flush()

    def close(self):
        """
        Close the file.
        """
        if self.mode[0] == 'w':
            self.flush()
        self.file.close()

    def flush(self):
        """
        Flush the buffer to the file.
        """
        if self.pos != 7:
            self.word <<= (self.pos + 1)
            self.file.write(self.word.to_bytes(1, byteorder='little'))
        self.word = 0
        self.pos = 7
