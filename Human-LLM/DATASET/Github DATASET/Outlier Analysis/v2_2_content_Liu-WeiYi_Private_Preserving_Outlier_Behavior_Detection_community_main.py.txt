import sys
import os
import glob
import json
import pickle
import time
import networkx as nx
import matplotlib.pyplot as plt
from statistics import mean
import Abnormal_Scenario_Generation as ASG

def main(normal_path, abnormal_path, synthetic_type):
    """
    Main function to analyze normal files and compute analysis time statistics.
    
    Parameters:
        normal_path (str): Path to directory containing normal files.
        abnormal_path (str): Path to directory containing abnormal files.
        synthetic_type (str): Type of synthetic data.
    """
    # Fetching list of normal and abnormal files
    normal_files = glob.glob(normal_path + '/*.gml')
    abnormal_files = glob.glob(abnormal_path + '/*.gml')
    
    # List to store analysis time for each file
    analysis_time = []

    # Iterating over each normal file for analysis
    for file in normal_files:
        start_time = time.time()  # Recording start time for analysis
        print('Analyzing file:', file)
        
        # Reading network from GML file
        network = nx.read_gml(file)
        
        # Detecting outliers in the network
        score_set, score_com = ASG.Detection_Outliers(network)
        
        # Calculating time taken for analysis
        time_interval = time.time() - start_time
        
        # Adding analysis time to the list
        analysis_time.append(time_interval)

    # Computing statistics for analysis time
    min_time = min(analysis_time)
    max_time = max(analysis_time)
    avg_time = mean(analysis_time)
    sum_time = sum(analysis_time)

    # Displaying analysis time statistics
    print('Min time:', min_time)
    print('Avg time:', avg_time)
    print('Max time:', max_time)
    print('Sum time:', sum_time)

if __name__ == "__main__":
    # Example paths and synthetic type
    normal_path = 'Normal_Example'
    abnormal_path = 'Abnormal_Example'
    synthetic_type = "Nodes"
    
    # Calling the main function
    main(normal_path, abnormal_path, synthetic_type)
