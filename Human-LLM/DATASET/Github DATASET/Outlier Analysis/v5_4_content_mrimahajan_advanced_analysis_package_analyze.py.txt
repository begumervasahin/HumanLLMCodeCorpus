import pandas as pd
import numpy as np
import statsmodels.api as sm
from statsmodels.formula.api import ols
from scipy.stats import skew, kurtosis, chisquare
import matplotlib.pyplot as plt

def numerical_categorical_division(df):
    numerical = []
    categorical = []
    
    for col in df.columns:
        if df[col].dtype in ['int64', 'float64']:
            numerical.append(col)
        else:
            categorical.append(col)
    
    return numerical, categorical

def exploratory_data_description(data, dependent_variable=None, regression=True, 
                                  percentile=[.01, .05, .1, .5, .9, .95, .99], cv=[2, 3]):
    numerical, categorical = numerical_categorical_division(data)
    
    df_desc = data.describe().transpose()
    df_desc['Variable'] = df_desc.index
    df_desc.reset_index(drop=True, inplace=True)
    df_desc.drop('count', axis=1, inplace=True)
    
    df_desc['Skewness'] = df_desc['Variable'].apply(lambda x: skew(data[x].dropna()))
    df_desc['Kurtosis'] = df_desc['Variable'].apply(lambda x: kurtosis(data[x].dropna(), fisher=False))
    
    for pct in percentile:
        df_desc[f'p{int(pct*100)}'] = df_desc['Variable'].apply(lambda x: data[x].quantile(pct))
    
    for dev in cv:
        df_desc[f'mean-{int(dev)}sigma'] = df_desc['mean'] - dev * df_desc['std']
        df_desc[f'mean+{int(dev)}sigma'] = df_desc['mean'] + dev * df_desc['std']
    
    df_desc['Type'] = 'Numeric'
    
    df_categorical = pd.DataFrame(columns=df_desc.columns)
    df_categorical['Variable'] = categorical
    df_categorical['Type'] = 'Categorical'

    for col in categorical:
        cum_pct = data[col].value_counts(normalize=True, ascending=True, dropna=False).cumsum()
        min_val, max_val = cum_pct.index[0], cum_pct.index[-1]
        df_cat = cum_pct.reset_index()
        df_cat.columns = ['Categories', 'Cumulative_Pct']
        df_categorical.loc[df_categorical['Variable'] == col, 'Min'] = min_val
        df_categorical.loc[df_categorical['Variable'] == col, 'Max'] = max_val
        for pct in percentile:
            df_categorical.loc[df_categorical['Variable'] == col, f'p{int(pct*100)}'] = df_cat.loc[df_cat['Cumulative_Pct'] >= pct, 'Categories'].iloc[0]

    edd = pd.concat([df_desc, df_categorical])
    edd['Count'] = edd['Variable'].apply(lambda x: data[x].count())
    edd['Missing'] = data.shape[0] - edd['Count']
    edd['Missing_Rate'] = (edd['Missing'] / data.shape[0]) * 100
    edd['Unique'] = edd['Variable'].apply(lambda x: data[x].nunique())

    columns_order = ['Variable', 'Type', 'Count', 'Missing', 'Missing_Rate', 'Unique', 'std', 
                     'Skewness', 'Kurtosis', 'mean', 'Min'] + \
                    [f'mean-{int(dev)}sigma' for dev in cv] + [f'p{int(pct*100)}' for pct in percentile] + \
                    [f'mean+{int(dev)}sigma' for dev in cv] + ['Max']
    
    edd = edd[columns_order]

    if dependent_variable:
        edd['Correlation/P_Value'] = np.nan
        if regression:
            corr_matrix = data.corr()
            for col in numerical:
                edd.loc[edd['Variable'] == col, 'Correlation/P_Value'] = corr_matrix.loc[col, dependent_variable]
            for col in categorical:
                mod = ols(f'{dependent_variable} ~ {col}', data=data).fit()
                aov_table = sm.stats.anova_lm(mod, type=2)
                edd.loc[edd['Variable'] == col, 'Correlation/P_Value'] = aov_table.loc[col, 'PR(>F)']
        else:
            for col in numerical:
                mod = ols(f'{col} ~ {dependent_variable}', data=data).fit()
                aov_table = sm.stats.anova_lm(mod, type=2)
                edd.loc[edd['Variable'] == col, 'Correlation/P_Value'] = aov_table.loc[dependent_variable, 'PR(>F)']
            for col in categorical:
                contingency_table = pd.crosstab(data[dependent_variable], data[col], dropna=False)
                edd.loc[edd['Variable'] == col, 'Correlation/P_Value'] = chisquare(np.ravel(contingency_table))[1]

    edd.reset_index(drop=True, inplace=True)
    return edd

def graphical_analysis(data, dependent_variable, path='', regression=True):
    numerical, categorical = numerical_categorical_division(data)
    if regression:
        for col in numerical:
            if col != dependent_variable:
                data.plot(x=col, y=dependent_variable)
                plt.savefig(path + col + '.png', dpi=1000)
                plt.close()
        for col in categorical:
            if col != dependent_variable:
                data.boxplot(column=dependent_variable, by=col)
                plt.savefig(path + col + '.png', dpi=1000)
                plt.close()
    else:
        for col in numerical:
            if col != dependent_variable:
                data.boxplot(column=col, by=dependent_variable)
                plt.savefig(path + col + '.png', dpi=1000)
                plt.close()
        for col in categorical:
            if col != dependent_variable:
                contingency_table = pd.crosstab(data[dependent_variable], data[col], dropna=False)
                contingency_table = (contingency_table / contingency_table.sum()) * 100
                contingency_table.plot(kind='bar', stacked=True)
                plt.savefig(path + col + '.png', dpi=1000)
                plt.close()
