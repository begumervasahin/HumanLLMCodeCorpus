from scipy.sparse import csr_matrix
import numpy as np

class TransactionEncoder:
    """
    Encoder class for transaction data in Python lists.
    """

    def __init__(self):
        """
        Initialize TransactionEncoder.
        """
        self.columns_ = None
        self.columns_mapping_ = None

    def fit(self, transactions):
        """
        Learn unique column names from transaction DataFrame.

        Parameters:
        - transactions: list of lists
            A list of lists where each inner list represents a transaction.
        """
        unique_items = set(item for transaction in transactions for item in transaction)
        self.columns_ = sorted(unique_items)
        self.columns_mapping_ = {item: idx for idx, item in enumerate(self.columns_)}
        return self

    def transform(self, transactions, sparse=False):
        """
        Transform transactions into a one-hot encoded NumPy array or CSR matrix.

        Parameters:
        - transactions: list of lists
            A list of lists where each inner list represents a transaction.
        - sparse: bool (default=False)
            If True, transform will return a Compressed Sparse Row matrix.

        Returns:
        - array: NumPy array or CSR matrix
            The one-hot encoded boolean array of the input transactions.
        """
        if sparse:
            indptr = [0]
            indices = []
            for transaction in transactions:
                for item in set(transaction):
                    indices.append(self.columns_mapping_[item])
                indptr.append(len(indices))
            data = [True] * len(indices)
            array = csr_matrix((data, indices, indptr), dtype=bool)
        else:
            array = np.zeros((len(transactions), len(self.columns_)), dtype=bool)
            for row_idx, transaction in enumerate(transactions):
                for item in transaction:
                    array[row_idx, self.columns_mapping_[item]] = True
        return array

    def inverse_transform(self, array):
        """
        Transform an encoded NumPy array back into transactions.

        Parameters:
        - array: NumPy array
            The one-hot encoded boolean array of the input transactions.

        Returns:
        - list of lists: A list of lists representing transactions.
        """
        return [[self.columns_[idx] for idx, cell in enumerate(row) if cell] for row in array]

    def fit_transform(self, transactions, sparse=False):
        """
        Fit a TransactionEncoder encoder and transform a dataset.

        Parameters:
        - transactions: list of lists
            A list of lists where each inner list represents a transaction.
        - sparse: bool (default=False)
            If True, transform will return a Compressed Sparse Row matrix.

        Returns:
        - array: NumPy array or CSR matrix
            The one-hot encoded boolean array of the input transactions.
        """
        return self.fit(transactions).transform(transactions, sparse=sparse)
