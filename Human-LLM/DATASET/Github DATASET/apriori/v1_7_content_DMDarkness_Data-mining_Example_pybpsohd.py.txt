# Import necessary modules
import pybpsohd as pybp
import re

# Define a function to read the transactional dataset from a file
def read_dataset(file_name):
    transactions = []
    with open(file_name, 'r') as file:
        for line in file:
            # Split the line by space or newline and extract integers
            items = re.findall(r'\d+', line)
            if items:
                transactions.append([int(item) for item in items])
    return transactions

if __name__ == '__main__':
    # Specify the file containing the transactional dataset
    dataset_file = "kosarak.dat"

    # Read the transactional dataset from the file
    dataset = read_dataset(dataset_file)

    # Mine long patterns from the dataset
    # Set minimum support to 0.001% with length 10
    # Use a population size of 30 and run for 30 generations
    # Set the interia weight to 0.5 and both accelerations to 1
    patterns = pybp.getFP(dataset, 0.00001, 30, 30, 0.5, 1, 1, 10)

    # Display the mined patterns
    for pattern in patterns:
        print(pattern)
