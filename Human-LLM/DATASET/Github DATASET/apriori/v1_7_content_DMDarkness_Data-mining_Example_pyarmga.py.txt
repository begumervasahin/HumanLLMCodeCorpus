import pyarmga as ga
import re

def read_dataset(file_name):
    """
    Read the transactional dataset from the given file and return a list of transactions.
    Each transaction is represented as a list of integers.
    """
    transactions = []
    with open(file_name, 'r') as f:
        for line in f:
            items = re.findall(r'\d+', line)  # Extract integers from the line using regular expression
            if items:
                transactions.append([int(item) for item in items])
    return transactions

if __name__ == '__main__':
    # Read the transactional dataset from the file 'kosarak.dat'
    dataset = read_dataset("kosarak.dat")
    
    # Mine long association rules from the dataset
    # Set minimum support to 0.7, minimum confidence to 1
    # Set population size to 30, number of generations to 30
    # Set mutation probability to 0.25, selection probability to 1, crossover probability to 1
    # Set rule length to 10
    # It may take some time
    association_rules = ga.getAR(dataset, 0.7, 1, 30, 30, 0.25, 1, 1, 10)
    
    # Print the mined association rules
    for rule in association_rules:
        print(rule)
