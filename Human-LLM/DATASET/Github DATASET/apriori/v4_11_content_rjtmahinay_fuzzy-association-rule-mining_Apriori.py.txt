from itertools import chain, combinations


# Open the data from a .csv file
def open_data(filename):
    with open(filename, 'r') as file:
        for line in file:
            line = line.strip().rstrip(',')
            row = frozenset(line.split(','))
            yield row


# Extract itemsets and transactions from the data
def itemset_from_data(data):
    itemset = set()
    transaction_list = []

    for row in data:
        transaction_list.append(frozenset(row))
        for item in row:
            if item:
                itemset.add(frozenset([item]))

    return itemset, transaction_list


# Calculate support for itemsets
def itemset_support(transaction_list, itemset, min_support=0):
    len_transaction_list = len(transaction_list)
    support_dict = {
        item: float(sum(1 for row in transaction_list if item.issubset(row))) / len_transaction_list
        for item in itemset
    }

    return {item: support for item, support in support_dict.items() if support >= min_support}


# Generate frequent itemsets
def frequent_itemset(transaction_list, candidate_itemset, min_support):
    frequent_itemset = {}
    k = 1

    while True:
        if k > 1:
            candidate_itemset = join_set(last_frequent_itemset, k)

        current_frequent_itemset = itemset_support(transaction_list, candidate_itemset, min_support)

        if not current_frequent_itemset:
            break

        frequent_itemset.update(current_frequent_itemset)
        last_frequent_itemset = current_frequent_itemset
        k += 1

    return frequent_itemset


# Join itemsets to generate candidates
def join_set(itemset, k):
    return {i.union(j) for i in itemset for j in itemset if len(i.union(j)) == k}


# Generate subsets of itemsets
def subsets(itemset):
    return chain(*[combinations(itemset, i + 1) for i in range(len(itemset))])


# Generate association rules
def get_rules(frequent_itemset, min_confidence, min_lift):
    rules = []

    for item, support in frequent_itemset.items():
        if len(item) > 1:
            for antecedent in subsets(item):
                consequent = item.difference(antecedent)

                if consequent:
                    antecedent = frozenset(antecedent)
                    antecedent_support = frequent_itemset[antecedent]
                    antecedent_consequent_support = frequent_itemset[item]
                    confidence = antecedent_consequent_support / antecedent_support
                    lift = confidence / (antecedent_support * frequent_itemset[consequent])

                    if confidence >= min_confidence and lift >= min_lift:
                        rules.append((antecedent, consequent, confidence, lift))

    return rules


# Mine frequent itemsets and association rules
def generate_itemsets_rules(data, min_support, min_confidence, min_lift):
    csv_data = open_data(data)
    itemset, transaction_list = itemset_from_data(csv_data)
    frequent_itemset = frequent_itemset(transaction_list, itemset, min_support)
    rules = get_rules(frequent_itemset, min_confidence, min_lift)
    return rules


# Print the generated rules
def print_result(rules):
    print('--Rules--')

    for antecedent, consequent, confidence, lift in sorted(rules, key=lambda x: x[0]):
        print(f'RULES: {tuple(antecedent)} => {tuple(consequent)} : {round(confidence, 5)} : {round(lift, 3)}')


# Store the generated rules and frequent itemsets
def store_result(rules, frequent_itemset):
    antecedents = []
    consequents = []
    confidences = []
    lifts = []

    for antecedent, consequent, confidence, lift in sorted(rules, key=lambda x: x[0]):
        antecedents.append(tuple(antecedent))
        consequents.append(tuple(consequent))
        confidences.append(round(confidence, 4))
        lifts.append(round(lift, 3))

    return antecedents, consequents, confidences, lifts


# Mine rules and itemsets from the dataset
def mine(csv, default_support=0.014, default_confidence=0.9, default_lift=1):
    data = open_data(csv)
    rules, itemset = generate_itemsets_rules(data, default_support, default_confidence, default_lift)
    return store_result(rules, itemset)
