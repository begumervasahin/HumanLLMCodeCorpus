import pandas as pd
from apyori import apriori

def market_basket_analysis(data_file, country='France', min_support=0.05, min_confidence=0.2, min_lift=3, min_length=3):
    """
    Perform Market Basket Analysis using the Apriori algorithm.

    Args:
        data_file (str): Path to the data file (Excel format).
        country (str): Country to filter the data for (default is 'France').
        min_support (float): Minimum support threshold for generating frequent itemsets (default is 0.05).
        min_confidence (float): Minimum confidence threshold for generating association rules (default is 0.2).
        min_lift (float): Minimum lift threshold for generating association rules (default is 3).
        min_length (int): Minimum length of the association rules (default is 3).

    Returns:
        pandas DataFrame: Contains the extracted association rules.
    """
    # Read data from the file
    data = pd.read_excel(data_file)

    # Filter data for the specified country
    data_country = data[data['Country'] == country]

    # Extract necessary columns
    main_data = data_country[['InvoiceNo', 'Description']]

    # Group items by InvoiceNo
    transactions = main_data.groupby('InvoiceNo')['Description'].apply(list).values.tolist()

    # Build association rules using Apriori Algorithm
    association_rules = apriori(transactions, min_support=min_support, min_confidence=min_confidence,
                                min_lift=min_lift, min_length=min_length)

    # Extract association rules into a DataFrame
    rules_df = pd.DataFrame(columns=['Base_item', 'Add_item', 'Confidence', 'Lift'])

    for rule in association_rules:
        base_item = rule.ordered_statistics[0].items_base
        add_item = rule.ordered_statistics[0].items_add
        confidence = rule.ordered_statistics[0].confidence
        lift = rule.ordered_statistics[0].lift

        rules_df = rules_df.append({'Base_item': list(base_item), 'Add_item': list(add_item),
                                    'Confidence': confidence, 'Lift': lift}, ignore_index=True)

    return rules_df

# Example usage
if __name__ == "__main__":
    # Specify the path to the data file
    data_file_path = 'D:\\ML-Class\\Data Sets\\Apriori\\Online Retail.xlsx'

    # Perform Market Basket Analysis for France
    rules_df = market_basket_analysis(data_file_path, country='France')

    print(rules_df)
