#!/usr/bin/env python
import os
import sys
from django.core.management import execute_from_command_line

if __name__ == '__main__':
    # Set the Django settings module
    os.environ.setdefault('DJANGO_SETTINGS_MODULE', 'it_sales.settings')

    try:
        # Import the execute_from_command_line function from the Django management module
        from django.core.management import execute_from_command_line
    except ImportError as exc:
        # Handle ImportError
        error_message = (
            "Couldn't import Django. Make sure it's installed and "
            "available on your PYTHONPATH environment variable. Did you "
            "forget to activate a virtual environment?"
        )
        raise ImportError(error_message) from exc

    # Execute Django management commands from the command line
    execute_from_command_line(sys.argv)
