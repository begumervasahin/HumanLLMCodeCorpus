def read_CSV(filename):
    """
    Read a CSV file and return a list of items and the count.

    Args:
    - filename: The name of the CSV file to read.

    Returns:
    - n_itemlist: A list containing the items read from the CSV file.
    - count: The count of items read from the CSV file.
    """
    try:
        with open(filename, "r") as file:
            reader = csv.reader(file)
            count = 0
            for row in reader:
                n_itemlist.append(row)
                count += 1
            print("CREATED LIST ITEMS WITH DUPLICATES: \n", count, len(n_itemlist), n_itemlist)
            return n_itemlist, count
    except FileNotFoundError:
        print(f"File {filename} not found.")
        return None, 0

def compute_minimum_support_count(count):
    """
    Compute the minimum support count.

    Args:
    - count: The count of items.

    Returns:
    - minsup: The minimum support.
    - supcount: The support count.
    """
    minsup = math.exp(-0.4 * count - 0.2) + 0.2
    supcount = minsup * count / 100
    return minsup, supcount

def merge_lists(n_itemlist):
    """
    Merge lists of lists into a single list.

    Args:
    - n_itemlist: A list of lists.

    Returns:
    - a: A single list containing all elements.
    """
    for sublist in n_itemlist:
        a.extend(sublist)
    print("ONLY ONE LIST with duplicates: \n", len(a), a)
    return a

def remove_duplicates(seq, idfun=None):
    """
    Remove duplicates from a list while preserving order.

    Args:
    - seq: The input list.
    - idfun: A function to determine uniqueness.

    Returns:
    - result: A list with duplicates removed.
    """
    if idfun is None:
        def idfun(x): return x
    seen = {}
    for item in seq:
        marker = idfun(item)
        if marker in seen:
            continue
        seen[marker] = 1
        result.append(item)
    print("DUPLICATES REMOVED LIST :\n", len(result), result)
    return result

def create_dictionary(result):
    """
    Create a dictionary with unique items as keys and their indices as values.

    Args:
    - result: The list of unique items.

    Returns:
    - test_dict: The created dictionary.
    """
    test_dict = dict(zip(result, range(1, len(result) + 1)))
    print('DICTIONARY = ', test_dict)
    for key in sorted(test_dict.keys()):
        print("%s: %s" % (key, test_dict[key]))
    return test_dict

def convert_to_binary(Mapper_list, result):
    """
    Convert mapped transactions to binary format.

    Args:
    - Mapper_list: The list of mapped items.
    - result: The list of unique items.

    Returns:
    - Transaction_list: A list of binary transactions.
    """
    for sublist in Mapper_list:
        t_list = [0] * len(result)
        for item_index in sublist:
            t_list[item_index - 1] = 1
        Transaction_list.append(t_list)
    print("Transaction_list:\n", Transaction_list)
    return Transaction_list

    """
