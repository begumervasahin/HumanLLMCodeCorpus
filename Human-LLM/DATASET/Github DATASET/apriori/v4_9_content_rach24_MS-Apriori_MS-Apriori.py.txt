from ast import literal_eval
import operator
from itertools import combinations

def string_converter(item_list):
    return ','.join(map(str, item_list))

def initialize_pass(min_support, item_set, transactions):
    item_counts = {}
    frequent_items = []

    for item in item_set:
        count = sum(1 for transaction in transactions if item in transaction)
        item_counts[item] = count

    total_transactions = len(transactions)
    first_item = False

    for item in item_set:
        if not first_item:
            if (item_counts[item] / total_transactions) >= min_support[item]:
                frequent_items.append(item)
                l1 = item
                first_item = True

        if first_item:
            if (item_counts[item] / total_transactions) >= min_support[l1]:
                if l1 != item:
                    frequent_items.append(item)

    return frequent_items, item_counts

def sort_itemset(item_set, min_support):
    sorted_itemset = sorted(min_support.items(), key=operator.itemgetter(1))
    sorted_items = [item for item, _ in sorted_itemset]
    return sorted_items

def generate_freq_1_itemset(L, min_support, item_counts, total_transactions, must_have_items):
    freq_1_itemset = []

    if not must_have_items:
        for item in L:
            if (item_counts[item] / total_transactions) >= min_support[item]:
                freq_1_itemset.append([item])
    else:
        for item in L:
            if (item_counts[item] / total_transactions) >= min_support[item] and item in must_have_items:
                freq_1_itemset.append([item])

    return freq_1_itemset

def generate_level2_candidate_set(L, support_difference_constraint, min_support, item_counts, total_transactions):
    level2_candidates = []

    for item1 in L:
        if (item_counts[item1] / total_transactions) >= min_support[item1]:
            index = L.index(item1) + 1
            for item2 in L[index:]:
                if ((item_counts[item2] / total_transactions) >= min_support[item1] and
                        abs((item_counts[item2] / total_transactions) - (item_counts[item1] / total_transactions)) <= support_difference_constraint):
                    level2_candidates.append([item1, item2])

    return level2_candidates

def generate_ms_candidate_set(prev_freq_set, support_difference_constraint, min_support, item_counts, total_transactions):
    candidate_set = []

    for i in range(len(prev_freq_set) - 1):
        for j in range(i + 1, len(prev_freq_set)):
            f1, f2 = prev_freq_set[i], prev_freq_set[j]

            if (f1[:-1] == f2[:-1] and min_support[f2[-1]] >= min_support[f1[-1]] and
                    (abs((item_counts[f1[-1]] / total_transactions) - (item_counts[f2[-1]] / total_transactions)) <= support_difference_constraint)):
                c = f1 + [f2[-1]]
                remove_item = False
                subset = [list(k) for k in combinations(c, len(c) - 1)]

                for s in subset:
                    slist = list(s)
                    if (c[0] in slist) or min_support[c[1]] == min_support[c[0]]:
                        if slist not in prev_freq_set:
                            remove_item = True

                if not remove_item:
                    candidate_set.append(c)

    return candidate_set

        else:
            fileF.write(f"Frequent {k}-itemsets\n\n")
            for itemset in freq_set:
                c_count = string_converter(itemset)
                t_count = c_count + '-' + string_converter(itemset[1:])
                fileF.write(f"\t{item_counts[c_count]} : {str(itemset).replace('[', '{').replace(']', '}')}\n")
                fileF.write(f"Tailcount = {tail_count[t_count]}\n")
            fileF.write(f"\n\tTotal number of frequent {k}-itemsets = {len(freq_set)}\n\n")

def get_transaction_data(filename):
    transactions = []

    with open(filename) as f:
        for line in f:
            formatted_line = line.replace("<", "").replace(">", "").replace("}{", "},{")
            if formatted_line[0] != '{':
                formatted_line = '{' + formatted_line
            view = list(literal_eval(formatted_line))

            if isinstance(view[0], set):
                transactions.extend([list(map(int, x)) for x in view])
            else:
                transactions.append(list(map(int, view)))

    return transactions

def get_parameters(filename='parameterfile.txt'):
    min_support = {}
    items = []
    cannot_be_together = []
    must_have = []

    with open(filename) as f:
        for line in f:
            line = line.rstrip('\n')

            if "MIS" in line:
                item, support = map(float, line.split("MIS(")[1].split(") = "))
                min_support[int(item)] = support
                items.append(str(int(item)))

            elif "SDC" in line:
                support_difference_constraint = float(line.split("SDC = ")[1])

            elif "cannot_be_together" in line:
                cannot_be_together = list(literal_eval(line.split("cannot_be_together: ")[1]))

            elif "must-have" in line:
                must_have = [int(item) for item in line.split("must-have:")[1].split("or")]

    return min_support, items, support_difference_constraint, cannot_be_together, must_have

def ms_apriori(transactions, min_support, items, support_difference_constraint, must_have, cannot_be_together):
    item_set = sort_itemset(items, min_support)
    frequent_items, item_counts = initialize_pass(min_support, item_set, transactions)
    freq_1_item_set = generate_freq_1_itemset(frequent_items, min_support, item_counts, len(transactions), must_have)
    print_frequent_set(freq_1_item_set, 1, item_counts)

    freq_item_set_empty = False
    k = 1
    freq_set = []

    while not freq_item_set_empty:
        k += 1
        c_count = dict()
        tail_count = dict()

        if k == 2:
            candidate_set = generate_level2_candidate_set(frequent_items, support_difference_constraint, min_support, item_counts, len(transactions))
        else:
            candidate_set = generate_ms_candidate_set(freq_set, support_difference_constraint, min_support, item_counts, len(transactions))

        for c in candidate_set:
            c_child = string_converter(c)
            c_count[c_child] = 0
            c_t_child = c_child + '-' + string_converter(c[1:])
            tail_count[c_t_child] = 0

        for t in transactions:
            tt = set(t)
            for c in candidate_set:
                c_child = string_converter(c)
                c_t_child = c_child + '-' + string_converter(c[1:])
                cc = set(c)

                if cc.issubset(tt):
                    c_count[c_child] += 1

                ccc = set(c[1:])

                if ccc.issubset(tt):
                    tail_count[c_t_child] += 1

        freq_set = []
        prune_freq_set = []

        for c in candidate_set:
            string_c = string_converter(c)

            if (c_count[string_c] / len(transactions)) >= min_support[c[0]]:
                put_in_list = True
                must_have_in_list = False
                c_set = set(c)
                freq_set.append(c)

                for x in cannot_be_together:
                    x_set = set(x)

                    if x_set.issubset(c_set):
                        put_in_list = False

                for y in must_have:
                    if y in c:
                        must_have_in_list = True

                if not must_have or must_have_in_list:
                    prune_freq_set.append(c)

        if not freq_set:
            freq_item_set_empty = True

        if not freq_item_set_empty and len(prune_freq_set) >= 1:
            print_frequent_set(prune_freq_set, k, c_count, tail_count)
