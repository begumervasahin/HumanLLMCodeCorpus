from ast import literal_eval
import operator
from itertools import combinations

def string_converter(list_l):
    string_s = ','.join(map(str, list_l))
    return string_s

def init_pass(MS, M, T):
    Icount = {}
    L = []

    for x in M:
        count = 0
        for i in range(len(T)):
            if x in T[i]:
                count += 1
        Icount[x] = count

    n = len(T)
    first_item = False

    for x in M:
        if not first_item:
            if (Icount[x] / n) >= MS[x]:
                L.append(x)
                l1 = x
                first_item = True

        if first_item:
            if (Icount[x] / n) >= MS[l1]:
                if l1 != x:
                    L.append(x)

    return L, Icount

def sort_itemset(I, MS):
    M = []
    MIS = sorted(MS.items(), key=operator.itemgetter(1))

    for i in range(len(MIS)):
        val1, val2 = MIS[i]
        M.append(val1)

    for i in range(len(M)):
        M[i] = (M[i])

    return M

def freq_1_itemset(L, MS, Icount, n, must_have):
    F1 = []

    if not must_have:
        for x in L:
            if (Icount[x] / n) >= MS[x]:
                view = [x]
                F1.append(view)
    else:
        for x in L:
            if (Icount[x] / n) >= MS[x] and x in must_have:
                view = [x]
                F1.append(view)

    return F1

def level2_candidate_gen(L, SDC, MS, Icount, n):
    C2 = []

    for l in L:
        if (Icount[l] / n) >= MS[l]:
            ind = L.index(l) + 1
            for h in L[ind:]:
                if ((Icount[h] / n) >= MS[l]) and (abs((Icount[h] / n) - (Icount[l] / n)) <= SDC):
                    view = [l, h]
                    C2.append(view)

    return C2

def MS_candidate_gen(F, SDC, MS, Icount, n):
    C = []

    for i in range(len(F) - 1):
        for j in range(i + 1, len(F)):
            f1, f2 = F[i], F[j]

            if (f1[:-1] == f2[:-1] and MS[f2[-1]] >= MS[f1[-1]]
                    and (abs((Icount[f1[-1]] / n) - (Icount[f2[-1]] / n)) <= SDC)):
                c = f1 + [f2[-1]]
                remove_item = False
                subset = [list(k) for k in combinations(c, len(c) - 1)]

                for s in subset:
                    slist = list(s)
                    if (c[0] in slist) or MS[c[1]] == MS[c[0]]:
                        if slist not in F:
                            remove_item = True

                if not remove_item:
                    C.append(c)

    return C

def print_freq_set(FreqSet, k, Icount, TailCount=0):
    fileF = open(outputfile, 'a+')

    if k == 1:
        fileF.write(f"Frequent {k}-itemsets\n\n")

        for f in FreqSet:
            fileF.write(f"\t{Icount[f[0]]} : {str(f).replace('[', '{').replace(']', '}')}\n")

        fileF.write(f"\n\tTotal number of frequent {k}-itemsets = {len(FreqSet)}\n\n")
    else:
        fileF.write(f"Frequent {k}-itemsets\n\n")

        for f in FreqSet:
            c_count = string_converter(f)
            t_count = c_count + '-' + string_converter(f[1:])

            fileF.write(f"\t{Icount[c_count]} : {str(f).replace('[', '{').replace(']', '}')}\n")
            fileF.write(f"Tailcount = {TailCount[t_count]}\n")

        fileF.write(f"\n\tTotal number of frequent {k}-itemsets = {len(FreqSet)}\n\n")

    fileF.close()

def get_transaction_data(filename):
    T = []

    with open(filename) as f:
        for line in f:
            W = line.replace("<", "")
            W = W.replace(">", "")
            W = W.replace("}{", "},{")

            if W[0] != '{':
                W = '{' + W

            view = list(literal_eval(W))

            if isinstance(view[0], set):
                T.extend([list(map(int, x)) for x in view])
            else:
                T.append(list(map(int, view)))

    return T

def get_parameters(filename='parameterfile.txt'):
    MS = {}
    I = []
    cannot_be_together = []
    must_have = []

    with open(filename) as f:
        for line in f:
            line = line.rstrip('\n')

            if "MIS" in line:
                a, b = line.split("MIS(")
                a, b = b.split(") = ")
                MS[int(a)] = float(b)
                I.append(a)

            elif "SDC" in line:
                a, b = line.split("SDC = ")
                SDC = float(b)

            elif "cannot_be_together" in line:
                a, b = line.split("cannot_be_together: ")
                view = list(literal_eval(b))

                if isinstance(view[0], set):
                    cannot_be_together.extend(list(map(list, view)))
                else:
                    cannot_be_together.append(list(view))

            elif "must-have" in line:
                a, b = line.split("must-have:")
                must_have = [int(aa) for aa in b.split("or")]

    return MS, I, SDC, cannot_be_together, must_have

def MS_Apriori(T, MS, I, SDC, must_have, cannot_be_together):
    M = sort_itemset(I, MS)
    L, Icount = init_pass(MS, M, T)
    F1 = freq_1_itemset(L, MS, Icount, len(T), must_have)
    print_freq_set(F1, 1, Icount)
    freq_item_set_empty = False
    k = 1
    FreqSet = []

    while not freq_item_set_empty:
        k += 1
        cCount = dict()
        cTailCount = dict()

        if k == 2:
            C = level2_candidate_gen(L, SDC, MS, Icount, len(T))
        else:
            C = MS_candidate_gen(FreqSet, SDC, MS, Icount, len(T))

        for c in C:
            c_child = string_converter(c)
            cCount[c_child] = 0
            c_Tchild = c_child + '-' + string_converter(c[1:])
            cTailCount[c_Tchild] = 0

        for t in T:
            tt = set(t)
            for c in C:
                c_child = string_converter(c)
                c_Tchild = c_child + '-' + string_converter(c[1:])
                cc = set(c)

                if cc.issubset(tt):
                    cCount[c_child] += 1

                ccc = set(c[1:])

                if ccc.issubset(tt):
                    cTailCount[c_Tchild] += 1

        FreqSet = []
        PruneFreqSet = []

        for c in C:
            stringC = string_converter(c)

            if (cCount[stringC] / len(T)) >= MS[c[0]]:
                put_in_list = True
                must_have_in_list = False
                cset = set(c)
                FreqSet.append(c)

                for x in cannot_be_together:
                    xset = set(x)

                    if xset.issubset(cset):
                        put_in_list = False

                for y in must_have:
                    if y in c:
                        must_have_in_list = True

                if not must_have or must_have_in_list:
                    PruneFreqSet.append(c)

        if not FreqSet:
            freq_item_set_empty = True

        if not freq_item_set_empty and len(PruneFreqSet) >= 1:
            print_freq_set(PruneFreqSet, k, cCount, cTailCount)

# Main function to execute the program
def main(inputdata, parameters):
    global outputfile
    print("*********WELCOME**********")
    print("MS-Apriori Implementation")
    print("Transaction Details")
    f = open(outputfile, 'w')
    T = get_transaction_data(inputdata)
    MS, I, SDC, cannot_be_together, must_have = get_parameters(parameters)
    MS_Apriori(T, MS, I, SDC, must_have, cannot_be_together)
    f.close()
