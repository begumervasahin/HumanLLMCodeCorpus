from time import time
import matplotlib.pyplot as plt
import data
import tsne

def main():
    # Record start time
    time00 = time()
    time0 = time()
    
    # Print status message
    print("Loading data...")

    # Load MNIST dataset
    filename_train_images = "MNIST/train-images.idx3-ubyte"
    filename_train_labels = "MNIST/train-labels.idx1-ubyte"
    images_train, labels_train = data.read_MNIST(filename_train_images, filename_train_labels)

    # Limit data to 1000 samples
    n_samples = 1000
    images_train = images_train[:n_samples] / 255.
    labels_train = labels_train[:n_samples]
    
    # Perform PCA for dimensionality reduction
    images_train = tsne.get_pca_proj(images_train, 30)

    # Print status message
    print("Data loading completed. Time elapsed: {:.2f} s".format(time() - time0))

    # Set animation flag
    anim = False

    if anim:
        # Import matplotlib and set backend for animation
        import matplotlib
        matplotlib.use("Agg")

    # Initialize t-SNE visualization object
    vis = tsne.TSNE(max_iter=1000)
    
    # Fit t-SNE model to the data
    vis.fit(images_train, animate=anim, labels=labels_train, anim_file="tsne_movie.mp4")

    # Print total execution time
    print("Total execution time: {:.2f} s".format(time() - time00))

    # If animation is not enabled, plot 2D embedding
    if not anim:
        fig, ax = plt.subplots()
        vis.plot_embedding2D(labels_train, ax)
        plt.show()

if __name__ == "__main__":
    main()
