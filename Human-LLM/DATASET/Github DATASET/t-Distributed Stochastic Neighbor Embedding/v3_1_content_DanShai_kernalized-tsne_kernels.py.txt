# Import necessary modules
from kernels import Kernels
from Bio import SeqIO

def load_dna_sequences(fasta_file):
    """
    Load DNA sequences from a FASTA file.

    Args:
        fasta_file (str): Path to the FASTA file.

    Returns:
        list: List of DNA sequences.
    """
    sequences = []
    for record in SeqIO.parse(fasta_file, "fasta"):
        sequences.append(record.seq)
    return sequences

def main():
    # Load DNA sequences from a FASTA file
    fasta_file = "DNASequences.fasta"
    sequences = load_dna_sequences(fasta_file)

    # Initialize Kernels class with DNA sequences
    kernel = Kernels(sequences)

    # Process data using PCA kernel
    pca_result = kernel.process_data()

    # Print the PCA result
    print("PCA result:")
    print(pca_result)

if __name__ == "__main__":
    main()
