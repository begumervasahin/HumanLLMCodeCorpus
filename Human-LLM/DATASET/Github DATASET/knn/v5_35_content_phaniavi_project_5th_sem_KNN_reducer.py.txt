import sys

def main():
    # Initialize variables
    distances = []
    estimate_value = 0.0
    k = 5

    # Read input from stdin
    for line in sys.stdin:
        line = line.strip()
        elements = line.split()
        distances.append(elements)

    # Sort the distances
    distances.sort()

    # Calculate the estimate value
    for i in range(k):
        try:
            estimate_value += float(distances[i][1])
        except ValueError:
            # Skip if the value cannot be converted to float
            continue

    # Print the average estimate value
    print(estimate_value / k)

if __name__ == "__main__":
    main()
