import csv

def transform_csv(input_filename, output_filename):
    # Open input and output files
    with open(input_filename, 'r') as input_file, open(output_filename, 'w') as output_file:
        # Create CSV reader and writer objects
        reader = csv.DictReader(input_file)
        writer = csv.writer(output_file)

        # Write header to output file
        writer.writerow(['Lon', 'Lat', 'Text_General_Code'])

        # Process each row in the input CSV file
        for row in reader:
            # Extract relevant fields from the row
            lon = row['Lon']
            lat = row['Lat']
            code = row['Text_General_Code']
            # Write transformed row to the output CSV file
            writer.writerow([lon, lat, code])

def main():
    input_filename = 'input.csv'
    output_filename = 'output.csv'
    transform_csv(input_filename, output_filename)

if __name__ == "__main__":
    main()
