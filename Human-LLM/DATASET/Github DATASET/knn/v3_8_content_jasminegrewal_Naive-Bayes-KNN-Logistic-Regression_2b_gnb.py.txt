import csv
import random
import math

def divide_by_class(data):
    """
    Divides the dataset into separate lists based on class labels.
    """
    class_data = {}
    for instance in data:
        label = instance[-1]
        if label not in class_data:
            class_data[label] = []
        class_data[label].append(instance)
    return class_data

def mean(numbers):
    """
    Calculates the mean of a list of numbers.
    """
    return sum(numbers) / len(numbers)

def stdev(numbers):
    """
    Calculates the standard deviation of a list of numbers.
    """
    avg = mean(numbers)
    variance = sum((x - avg) ** 2 for x in numbers) / (len(numbers) - 1)
    return math.sqrt(variance)

def calculate_parameters(data):
    """
    Calculates the mean and standard deviation for each feature in the dataset.
    """
    parameters = [(mean(feature), stdev(feature)) for feature in zip(*data)]
    del parameters[-1]
    return parameters

def calculate_probability(x, mean, stdev):
    """
    Calculates the probability density function value.
    """
    exponent = math.exp(-(math.pow(x - mean, 2) / (2 * math.pow(stdev, 2))))
    return (1 / (math.sqrt(2 * math.pi) * stdev)) * exponent

def calculate_class_probabilities(parameters, sample):
    """
    Calculates the probabilities of the sample belonging to each class.
    """
    probabilities = {}
    for class_label, params in parameters.items():
        probabilities[class_label] = 1
        for i in range(len(params)):
            mean, stdev = params[i]
            x = sample[i]
            prob_density = calculate_probability(x, mean, stdev)
            probabilities[class_label] *= prob_density
    return probabilities

def predict_class(parameters, sample):
    """
    Predicts the class label for the given sample.
    """
    probabilities = calculate_class_probabilities(parameters, sample)
    best_label, best_prob = None, -1
    for class_label, probability in probabilities.items():
        if best_label is None or probability > best_prob:
            best_prob = probability
            best_label = class_label
    return best_label

def get_predictions(parameters, test_set):
    """
    Gets predictions for all samples in the test set.
    """
    result = predict_class(parameters, test_set)
    return result

def main():
    given_data = []
    with open('data.csv', 'r') as csvfile:
        reader = csv.reader(csvfile)
        data = list(reader)
        for row in data:
            for i in range(4):
                if i == 3:
                    row[i] = 1 if row[i] == 'M' else 2
                row[i] = float(row[i])
            given_data.append(row)
    parameters = divide_by_class(given_data)
    print('Summary by class value: {0}'.format(parameters))
    test_set = [float(input('Enter data for prediction: ')) for _ in range(4)]
    prediction = get_predictions(parameters, test_set)
    print('Predicted class:', 'M' if prediction == 1 else 'W')

if __name__ == "__main__":
    main()
