import numpy as np
import matplotlib.pyplot as plt

class SupportVectorMachine:
    def __init__(self):
        self.colors = {1: 'r', -1: 'b'}
        self.fig = plt.figure()
        self.ax = self.fig.add_subplot(1, 1, 1)

    def train(self, data):
        self.data = data
        self.max_feature_value = float('-inf')
        self.min_feature_value = float('inf')

        # Find the maximum and minimum feature values
        for y in self.data:
            for features in self.data[y]:
                self.max_feature_value = max(self.max_feature_value, np.max(features))
                self.min_feature_value = min(self.min_feature_value, np.min(features))

        # Optimization step sizes and ranges
        step_sizes = [self.max_feature_value * 0.1, self.max_feature_value * 0.01, self.max_feature_value * 0.001]
        b_range_multiple = 5
        b_multiple = 5
        latest_optimum = self.max_feature_value * 10

        # Iterate over step sizes
        for step in step_sizes:
            w = np.array([latest_optimum, latest_optimum])
            optimized = False
            while not optimized:
                for b in np.arange(-1 * self.max_feature_value * b_range_multiple, self.max_feature_value * b_range_multiple, step * b_multiple):      
                    for transformation in [[1, 1], [-1, 1], [-1, -1], [1, -1]]:
                        w_t = w * transformation
                        if self.check_all_data(w_t, b):
                            self.w = w_t
                            self.b = b
                            optimized = True
                            break
                    if optimized:
                        break
                if w[0] < 0:
                    optimized = True
                else:
                    w -= step

    def check_all_data(self, w, b):
        for i in self.data:
            for x in self.data[i]:
                y = i
                if not y * (np.dot(w, x) + b) >= 1:
                    return False
        return True

    def predict(self, features):
        classification = np.sign(np.dot(features, self.w) + self.b)
        if classification != 0:
            self.ax.scatter(features[0], features[1], s=300, marker='*', c=self.colors[classification])
        return classification

    def visualize(self):
        for i in self.data:
            for x in self.data[i]:
                self.ax.scatter(x[0], x[1], s=50, c=self.colors[i])

        self.plot_hyperplane()

        plt.show()

    def plot_hyperplane(self):
        def hyperplane(x, w, b, v):
            return (-w[0] * x - b + v) / w[1]

        data_range = (self.min_feature_value * 0.9, self.max_feature_value * 1.1)
        hyp_x_min = data_range[0]
        hyp_x_max = data_range[1]

        # Positive support vector hyperplane
        psv1 = hyperplane(hyp_x_min, self.w, self.b, 1)
        psv2 = hyperplane(hyp_x_max, self.w, self.b, 1)
        self.ax.plot([hyp_x_min, hyp_x_max], [psv1, psv2], 'k')

        # Negative support vector hyperplane
        nsv1 = hyperplane(hyp_x_min, self.w, self.b, -1)
        nsv2 = hyperplane(hyp_x_max, self.w, self.b, -1)
        self.ax.plot([hyp_x_min, hyp_x_max], [nsv1, nsv2], 'k')

        # Decision boundary hyperplane
        db1 = hyperplane(hyp_x_min, self.w, self.b, 0)
        db2 = hyperplane(hyp_x_max, self.w, self.b, 0)
        self.ax.plot([hyp_x_min, hyp_x_max], [db1, db2], 'y--')

if __name__ == '__main__':
    data_set = {-1: np.array([[1, 7], [2, 8], [3, 8]]),
                1: np.array([[5, 1], [6, -1], [7, 3]])}
    svm = SupportVectorMachine()
    svm.train(data_set)
    for predict_feature in [[0, 10], [2, 6], [1, 3], [4, 3], [5.5, 7.5], [8, 3]]:
        print(svm.predict(predict_feature))
    svm.visualize()
