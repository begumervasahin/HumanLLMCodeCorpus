import numpy as np
import operator

# Example of basic operations
def basic_operations_example():
    a = 1
    b = 2
    c = a + b
    print("Result of basic operations example:", c)

# Example of using numpy array and shape
def numpy_array_example():
    b = np.array([6, 7, 8])
    c = b.shape[0]
    print("Shape of numpy array:", c)

# Example of summing elements in an array
def sum_elements_example():
    a = np.sum([[0, 1, 2], [2, 1, 3]])
    print("Sum of elements in array:", a)

# Example of summing elements along an axis
def sum_elements_axis_example():
    a = np.sum([[0, 1, 2], [2, 1, 3]], axis=0)
    print("Sum of elements along axis 0:", a)

# Example of summing elements along a different axis
def sum_elements_different_axis_example():
    a = np.sum([[8, 1, 2], [2, 1, 3], [0, 0, 0]], axis=1)
    print("Sum of elements along axis 1:", a)

# Example of using argsort to get indices that would sort an array
def argsort_example():
    a = np.array([5, 3, 1, 2, 4])
    d = np.argsort(a)
    print("Indices that would sort the array:", d)

# Example of sorting a dictionary by values
def sort_dictionary_example():
    class_count = {"A": 6, "B": 1, "C": 0, "D": 2}
    sorted_class_count = sorted(class_count.items(), key=operator.itemgetter(1), reverse=True)
    print("Sorted dictionary by values:", sorted_class_count)

# Example of converting a 2D array to a vector
def convert_to_vector_example():
    f = np.array([[5, 8], [1, 2]])
    my_vector = f.flatten()
    print("Flattened array to vector:", my_vector)

# Example of accessing elements in a numpy array
def access_elements_example():
    my_vector = np.array([[1, 2, 3, 4]])
    print("Accessing elements in a numpy array:", my_vector[0, 1])

# Run examples
basic_operations_example()
numpy_array_example()
sum_elements_example()
sum_elements_axis_example()
sum_elements_different_axis_example()
argsort_example()
sort_dictionary_example()
convert_to_vector_example()
access_elements_example()
