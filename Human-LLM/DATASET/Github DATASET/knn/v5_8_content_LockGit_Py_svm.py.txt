import numpy as np
import matplotlib.pyplot as plt

class SupportVectorMachine:
    def __init__(self, visualize=True):
        self.visualize = visualize
        self.colors = {1: 'red', -1: 'blue'}
        if self.visualize:
            self.fig = plt.figure()
            self.ax = self.fig.add_subplot(1, 1, 1)

    def train(self, data):
        self.data = data
        max_feature_value = float('-inf')
        min_feature_value = float('inf')
        for y in self.data:
            for features in self.data[y]:
                for feature in features:
                    max_feature_value = max(max_feature_value, feature)
                    min_feature_value = min(min_feature_value, feature)

        step_sizes = [max_feature_value * 0.1, max_feature_value * 0.01, max_feature_value * 0.001]
        b_range_multiple = 5
        b_multiple = 5
        latest_optimum = max_feature_value * 10
        opt_dict = {}

        for step in step_sizes:
            w = np.array([latest_optimum, latest_optimum])
            optimized = False
            while not optimized:
                for b in np.arange(-1 * max_feature_value * b_range_multiple, max_feature_value * b_range_multiple,
                                    step * b_multiple):
                    for transformation in [(1, 1), (-1, 1), (-1, -1), (1, -1)]:
                        w_t = w * transformation
                        found_option = True
                        for i in self.data:
                            for x in self.data[i]:
                                y = i
                                if not y * (np.dot(w_t, x) + b) >= 1:
                                    found_option = False
                                    break
                        if found_option:
                            opt_dict[np.linalg.norm(w_t)] = [w_t, b]
                if w[0] < 0:
                    optimized = True
                else:
                    w = w - step

        norms = sorted([n for n in opt_dict])
        opt_choice = opt_dict[norms[0]]
        self.w = opt_choice[0]
        self.b = opt_choice[1]

    def predict(self, features):
        classification = np.sign(np.dot(features, self.w) + self.b)
        if classification != 0 and self.visualize:
            self.ax.scatter(features[0], features[1], s=300, marker='*', c=self.colors[classification])
        return classification

    def visualize(self):
        if self.visualize:
            for i in self.data:
                for x in self.data[i]:
                    self.ax.scatter(x[0], x[1], s=50, c=self.colors[i])

            def hyperplane(x, w, b, v):
                return (-w[0] * x - b + v) / w[1]

            data_range = (min([x[0] for x in self.data[-1]] + [x[0] for x in self.data[1]]),
                          max([x[0] for x in self.data[-1]] + [x[0] for x in self.data[1]]))
            hyp_x_min = data_range[0]
            hyp_x_man = data_range[1]
            psv1 = hyperplane(hyp_x_min, self.w, self.b, 1)
            psv2 = hyperplane(hyp_x_man, self.w, self.b, 1)
            self.ax.plot([hyp_x_min, hyp_x_man], [psv1, psv2], c=self.colors[1])
            nsv1 = hyperplane(hyp_x_min, self.w, self.b, -1)
            nsv2 = hyperplane(hyp_x_man, self.w, self.b, -1)
            self.ax.plot([hyp_x_min, hyp_x_man], [nsv1, nsv2], c=self.colors[-1])
            db1 = hyperplane(hyp_x_min, self.w, self.b, 0)
            db2 = hyperplane(hyp_x_man, self.w, self.b, 0)
            self.ax.plot([hyp_x_min, hyp_x_man], [db1, db2], 'y--')
            plt.show()

if __name__ == '__main__':
    data_set = {-1: np.array([[1, 7], [2, 8], [3, 8]]),
                1: np.array([[5, 1], [6, -1], [7, 3]])}
    svm = SupportVectorMachine()
    svm.train(data_set)
    for predict_feature in [[0, 10], [2, 6], [1, 3], [4, 3], [5.5, 7.5], [8, 3]]:
        print(svm.predict(predict_feature))
    svm.visualize()
