import heapq
import numpy as np
from sklearn.metrics.pairwise import pairwise_distances

class CosineKNN:
    def __init__(self, n_neighbors=5):
        self.n_neighbors = n_neighbors

    def fit(self, X, y):
        # Convert input data to numpy arrays
        self.data = np.array(X)
        self.train = np.array(y)

    def predict(self, X):
        # Initialize variables
        simtree = []
        dist_dic = {}
        dic = {}
        near_labels = []

        # Convert input X to a numpy array and reshape it
        X = np.array([X])
        X = X.reshape(-1, 1)

        # Calculate pairwise cosine distances between X and training data
        for data in self.data:
            sim = pairwise_distances(X, np.array(data).reshape(1, -1), metric='cosine')
            dist_dic[float(sim)] = data
            heapq.heappush(simtree, sim)

        # Select the nearest neighbors based on n_neighbors parameter
        for i in range(self.n_neighbors):
            min_sim = heapq.heappop(simtree)
            vector = dist_dic[float(min_sim)]
            for index, item in enumerate(self.data):
                if (item == vector).all():
                    break
            near_labels.append(self.train[index])

        # Count occurrences of each label among nearest neighbors
        for label in near_labels:
            dic[label] = 0

        for label in near_labels:
            dic[label] += 1

        # Determine the predicted label based on the majority among nearest neighbors
        y_pred = self.value_to_key(dic, max(dic.values()))
        return y_pred

    def value_to_key(self, dic, value):
        # Convert dictionary value to corresponding key
        for key, val in zip(dic.keys(), dic.values()):
            if val == value:
                return key

if __name__ == '__main__':
    # Example usage of CosineKNN class
    I = CosineKNN(n_neighbors=2)
    X = [[0.5, 0.86], [0.86, 0.5], [-0.5, -0.86], [-0.86, -0.5]]
    y = [1, 1, 0, 0]
    I.fit(X, y)
    print('Prediction for [0.7, 0.7]:', I.predict([0.7, 0.7]))
    print('Prediction for [-0.7, -0.7]:', I.predict([-0.7, -0.7]))
