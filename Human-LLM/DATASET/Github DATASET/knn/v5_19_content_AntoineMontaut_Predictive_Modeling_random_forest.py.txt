import pandas as pd
import matplotlib.pyplot as plt
import re
from sklearn.ensemble import RandomForestClassifier
from sklearn.metrics import accuracy_score, precision_score, recall_score, f1_score, confusion_matrix

# Define activity lists and dictionary
activity_list = ['Walking', 'Walking upstairs', 'Walking downstairs',
                 'Sitting', 'Standing', 'Laying']
activity_list_abr = ['Walk', 'Walk up', 'Walk down', 'Sit', 'Stand', 'Lay']
activity_dict = {'1': 'Walking', '2': 'Walking upstairs', '3': 'Walking downstairs',
                 '4': 'Sitting', '5': 'Standing', '6': 'Laying'}

# Import and clean feature names
def clean_feature_name(feature):
    feature = re.sub(r'[-(),]', '_', feature)
    feature = feature.replace('()', '').replace('__', '_').replace('BodyBody', 'Body').replace('Body', '')
    if feature.endswith('_'):
        feature = feature[:-1]
    return feature

with open('./UCI HAR Dataset/features.txt', 'r') as file:
    features = [clean_feature_name(line.split()[1]) for line in file]

# Import and clean training data
X_train = pd.read_csv('./UCI HAR Dataset/train/X_train.txt', delim_whitespace=True, names=features)
y_train = pd.read_csv('./UCI HAR Dataset/train/y_train.txt', header=None, names=['activity_cat'])
y_train['activity'] = y_train['activity_cat'].map(activity_dict)

# Preview plots function
def preview_plots():
    X_train['activity'] = y_train['activity']
    plt.figure(figsize=(10, 6))
    for i, act in enumerate(activity_list, start=1):
        plt.subplot(2, 3, i)
        plt.hist(X_train[X_train['activity'] == act]['fAccMag_mean'])
        plt.xlim([-1.0, 1.0])
        plt.title(act)
        if i == 1:
            plt.ylabel('Dynamic Activity')
        if i == 4:
            plt.ylabel('Static Activity')
    plt.show()
    X_train.drop('activity', axis=1, inplace=True)

# Feature selection
def filter_features(X_train):
    flags_out = ['angle', 'band', 'arCoeff', 'Mag']
    flags_in = ['mean', 'std', 'skewness', 'kurtosis']
    to_keep = [col for col in X_train.columns if not any(flag in col for flag in flags_out) and any(flag in col for flag in flags_in)]
    return X_train[to_keep]

X_train = filter_features(X_train)

# Train Random Forest on training data
print('\n\n-----DOMAIN KNOWLEDGE APPROACH-----\n')
clf = RandomForestClassifier(n_estimators=50)
clf.fit(X_train, y_train['activity'])
score_train = accuracy_score(y_train['activity'], clf.predict(X_train))
print('Random Forest accuracy on training data:', round(score_train, 5))
feat_imp = pd.Series(clf.feature_importances_, index=X_train.columns).sort_values(ascending=False)
print('Most important features:')
print(feat_imp[:10])

# Import and preprocess testing data
X_test = pd.read_csv('./UCI HAR Dataset/test/X_test.txt', delim_whitespace=True, names=features)
X_test = filter_features(X_test)
y_test = pd.read_csv('./UCI HAR Dataset/test/y_test.txt', header=None, names=['activity_cat'])
y_test['activity'] = y_test['activity_cat'].map(activity_dict)

# Apply Random Forest model to testing data
prediction = clf.predict(X_test)
score_test = accuracy_score(y_test['activity'], prediction)

# Evaluation metrics
precision_test = precision_score(y_test['activity'], prediction, average='micro')
recall_test = recall_score(y_test['activity'], prediction, average='micro')
f1_test = f1_score(y_test['activity'], prediction, average='micro')

# Print scores
print('\nRandom Forest scores on test data (global):')
print('\tAccuracy:', round(score_test, 5))
print('\tPrecision:', round(precision_test, 5))
print('\tRecall:', round(recall_test, 5))
print('\tF1:', round(f1_test, 5))

# Confusion matrix
conf_matrix = pd.DataFrame(confusion_matrix(y_test['activity'], prediction), columns=activity_list_abr, index=activity_list_abr)
print('\nConfusion matrix:')
print(conf_matrix)

# Black box approach
print('\n\n-----BLACK BOX APPROACH-----\n')

# Train Random Forest without domain knowledge
clf_bb = RandomForestClassifier(n_estimators=50)
clf_bb.fit(X_train, y_train['activity'])
score_train_bb = accuracy_score(y_train['activity'], clf_bb.predict(X_train))
print('Random Forest accuracy on training data (black box):', round(score_train_bb, 5))
feat_imp_bb = pd.Series(clf_bb.feature_importances_, index=X_train.columns).sort_values(ascending=False)
print('Most important features (black box):')
print(feat_imp_bb[:10])

# Apply black box Random Forest model to testing data
prediction_bb = clf_bb.predict(X_test)
score_test_bb = accuracy_score(y_test['activity'], prediction_bb)

# Evaluation metrics for black box approach
precision_test_bb = precision_score(y_test['activity'], prediction_bb, average='micro')
recall_test_bb = recall_score(y_test['activity'], prediction_bb, average='micro')
f1_test_bb = f1_score(y_test['activity'], prediction_bb, average='micro')

# Print scores for black box approach
print('\nRandom Forest scores on test data (global) - Black Box:')
print('\tAccuracy:', round(score_test_bb, 5))
print('\tPrecision:', round(precision_test_bb, 5))
print('\tRecall:', round(recall_test_bb, 5))
print('\tF1:', round(f1_test_bb, 5))

# Confusion matrix for black box approach
conf_matrix_bb = pd.DataFrame(confusion_matrix(y_test['activity'], prediction_bb), columns=activity_list_abr, index=activity_list_abr)
print('\nConfusion matrix (Black Box):')
print(conf_matrix_bb)
