import numpy as np

def KNNClassifier(unlabeled_data, dataset, labels, k):
    # Calculate distances
    distances = np.sqrt(np.sum((np.tile(unlabeled_data, (dataset.shape[0], 1)) - dataset) ** 2, axis=1))
    # Sort distances and get indices
    sorted_dist_index = np.argsort(distances)
    # Count votes for each class
    class_count = {}
    for i in range(k):
        vote_label = labels[sorted_dist_index[i]]
        class_count[vote_label] = class_count.get(vote_label, 0) + 1
    # Determine the class with the maximum votes
    max_index = max(class_count, key=class_count.get)
    return max_index

def file_to_matrix(filename, dim):
    data = np.loadtxt(filename)
    return data[:, :dim], data[:, -1].astype(int)

def auto_normalize(dataset):
    min_value = dataset.min(0)
    max_value = dataset.max(0)
    ranges = max_value - min_value
    norm_dataset = (dataset - min_value) / ranges
    return norm_dataset, ranges, min_value

def dating_class_test(filename, dim, rate=0.1, k=3):
    dating_data, dating_labels = file_to_matrix(filename, dim)
    norm_data, _, _ = auto_normalize(dating_data)
    test_vector_count = int(norm_data.shape[0] * rate)
    error_count = 0.0
    for i in range(test_vector_count):
        classified_result = KNNClassifier(norm_data[i], norm_data[test_vector_count:], dating_labels[test_vector_count:], k)
        print(f"Result from KNN Classifier is: {classified_result}, the actual class is: {dating_labels[i]}")
        if classified_result != dating_labels[i]:
            error_count += 1
    print(f"The total error rate is: {error_count / test_vector_count}")

def classify_people(filename, dim):
    results = ['not at all', 'in small doses', 'in large doses']
    percent_time_game = float(input('Percentage of time spent playing video games? '))
    fre_fly_miles_earn = float(input('Frequent flier miles earned per year? '))
    ice_consumed = float(input('Liters of ice cream consumed per year? '))
    dating_data, dating_labels = file_to_matrix(filename, dim)
    norm_data, ranges, min_value = auto_normalize(dating_data)
    input_array = np.array([percent_time_game, fre_fly_miles_earn, ice_consumed])
    classified_result = KNNClassifier((input_array - min_value) / ranges, norm_data, dating_labels, 3)
    print(f'You will probably like this person: {results[classified_result - 1]}')

if __name__ == '__main__':
    # Test the KNN classifier with the dating dataset
    dating_class_test('datingTestSet2.txt', dim=3)

    # Classify people based on their input attributes
    classify_people('datingTestSet2.txt', dim=3)
