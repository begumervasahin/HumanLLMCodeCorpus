import sys
import re
import nltk
from nltk.corpus import opinion_lexicon

# Reload sys to set the default encoding
reload(sys)
sys.setdefaultencoding("ISO-8859-1")

# Initialize a flag and a dictionary to store sentiment words
flag = 0
sentiment_words = {}

# Create a dictionary of sentiment words from the opinion lexicon
for w in nltk.corpus.opinion_lexicon.words():
    if flag == 0:
        sentiment_words[w] = flag
        if re.search("zombie", w):
            flag = 1
    else:
        sentiment_words[w] = flag

# Open the test file and output file
test_file = open("test.list", 'r')
out_file = open("baseline_out.txt", 'w')

# Initialize a line counter
line_cnt = 1

# Iterate through each tweet in the test file
for tweet in test_file:
    words = tweet.split(" ")
    neg = 0
    pos = 0
    # Count positive and negative sentiment words
    for word in words:
        try:
            if sentiment_words[word.lower()] == 1:
                pos += 1
            elif sentiment_words[word.lower()] == 0:
                neg += 1
        except KeyError:
            pass
    # Write the sentiment of the tweet to the output file
    if pos > neg:
        out_file.write(str(line_cnt) + " positive\n")
    elif pos < neg:
        out_file.write(str(line_cnt) + " negative\n")
    else:
        out_file.write(str(line_cnt) + " neutral\n")
    line_cnt += 1

# Close the files
test_file.close()
out_file.close()
