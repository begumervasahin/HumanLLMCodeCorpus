# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures

# Training data - diameters of pizzas and their corresponding prices
X_train = [[6], [8], [10], [14], [18]]
y_train = [[7], [9], [13], [17.5], [18]]

# Testing data - diameters of pizzas and their corresponding prices
x_test = [[6], [8], [11], [16]]
y_test = [[8], [12], [15], [18]]

# Creating a linear regression model
regressor = LinearRegression()

# Fitting the linear regression model to the training data
regressor.fit(X_train, y_train)

# Generating a range of diameters for plotting purposes
xx = np.linspace(0, 26, 100)

# Predicting prices using the linear regression model
yy = regressor.predict(xx.reshape(xx.shape[0], 1))

# Plotting the linear regression line
plt.plot(xx, yy)

# Creating a quadratic features transformer
quadratic_featurizer = PolynomialFeatures(degree=2)

# Transforming the training data to include quadratic features
X_train_quadratic = quadratic_featurizer.fit_transform(X_train)

# Transforming the testing data to include quadratic features
X_test_quadratic = quadratic_featurizer.transform(x_test)

# Creating a new linear regression model with quadratic features
regressor_quadratic = LinearRegression()

# Fitting the quadratic regression model to the training data
regressor_quadratic.fit(X_train_quadratic, y_train)

# Transforming the range of diameters to include quadratic features for plotting
xx_quadratic = quadratic_featurizer.transform(xx.reshape(xx.shape[0], 1))

# Plotting the quadratic regression line
plt.plot(xx, regressor_quadratic.predict(xx_quadratic), c='r', linestyle='--')

# Adding title and labels to the plot
plt.title('Pizza price regressed on diameter')
plt.xlabel('Diameter in inches')
plt.ylabel('Price in dollars')

# Setting axis limits
plt.axis([0, 25, 0, 25])

# Displaying grid
plt.grid(True)

# Plotting the training data points
plt.scatter(X_train, y_train)

# Displaying the plot
plt.show()

# Printing the transformed training and testing data for quadratic regression
print("Transformed training data with quadratic features:")
print(X_train_quadratic)
print("\nTransformed testing data with quadratic features:")
print(X_test_quadratic)
