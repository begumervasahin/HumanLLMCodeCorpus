import numpy as np
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures

# Training data: diameter of fish tank
x_train = [[12], [18], [20], [25], [27]]
# Corresponding prices of fish tank
y_train = [[14], [15], [16], [20], [27]]

# Test data for evaluation
x_test = [[12], [20], [45], [47]]
y_test = [[14], [16], [50], [52]]

# Simple Linear Regression
regressor = LinearRegression()
regressor.fit(x_train, y_train)

# Generating points for plotting the linear regression line
xx = np.linspace(0, 28, 110)
yy = regressor.predict(xx.reshape(xx.shape[0], 1))

# Plotting the linear regression line
plt.plot(xx, yy)

# Polynomial Regression with quadratic features
quadratic_featurizer = PolynomialFeatures(degree=2)
x_train_quadratic = quadratic_featurizer.fit_transform(x_train)
x_test_quadratic = quadratic_featurizer.transform(x_test)

# Fitting quadratic regression model
regressor_quadratic = LinearRegression()
regressor_quadratic.fit(x_train_quadratic, y_train)

# Generating points for plotting the quadratic regression line
xx_quadratic = quadratic_featurizer.transform(xx.reshape(xx.shape[0], 1))

# Plotting the quadratic regression line
plt.plot(xx, regressor_quadratic.predict(xx_quadratic), c='r', linestyle='--')     

# Plotting the training data points
plt.scatter(x_train, y_train)

# Adding title and labels
plt.title('Fish Tank price regressed on diameter')
plt.xlabel('Diameter in inches')
plt.ylabel('Price in dollars')

# Setting the axis limits
plt.axis([0, 30, 0, 30])

# Adding grid
plt.grid(True)

# Displaying the plot
plt.show()

# Printing out the data
print("Training data (linear):", x_train)
print("Training data (quadratic):", x_train_quadratic)
print("Test data (linear):", x_test)
print("Test data (quadratic):", x_test_quadratic)
