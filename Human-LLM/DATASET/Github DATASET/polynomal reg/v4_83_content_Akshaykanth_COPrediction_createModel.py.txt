# Importing necessary libraries
import pandas as pd
from sklearn.svm import SVR
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import PolynomialFeatures
from sklearn.linear_model import LinearRegression

# Printing a message indicating data preprocessing
print("Data Pre Processing")

# Reading data from the CSV file
data = pd.read_csv("coEmission.csv")

# Separating features (x) and target variable (y)
x = data.iloc[:, 0:3].values
y = data.iloc[:, 3].values

# Splitting the data into training and testing sets
x_train, x_test, y_train, y_test = train_test_split(x, y, test_size=0.2, random_state=0)

# Printing a message indicating model creation
print("Model Creating...!")

# Creating polynomial features
poly_reg = PolynomialFeatures(degree=4)
x_poly = poly_reg.fit_transform(x)

# Creating and training the polynomial regression model
pol_reg = LinearRegression()
print("Training Started..!")
pol_reg.fit(x_poly, y)
print("Training Ended")

# Loop for continuous user interaction
while True:
    # Prompting user for input
    print("\nEnter data for CO emission prediction:")
    year = int(input("Year (after 2005): ")) - 2005
    temp = int(input("Temperature: "))
    fuel = int(input("Fuel type: "))
    
    # Constructing parameter array for prediction
    params = [temp, fuel, year]
    
    # Predicting CO emission and printing the result
    predicted_emission = pol_reg.predict(poly_reg.fit_transform([params]))[0]
    print("Predicted CO emission:", predicted_emission, "grams")
