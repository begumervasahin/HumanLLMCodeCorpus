import matplotlib.pyplot as plt
import numpy as np
import pandas as pd
import time

# Record the start time
startTime = time.time()

# Read data from CSV file
data = pd.read_csv("housePriceDataset.csv")

# Extract features and labels
squareMeter = data["squareMeter"]
price = data["price"]

# Convert to numpy arrays
x = np.array(squareMeter)
y = np.array(price)

# Fit polynomial regression model
a, b, c, d = np.polyfit(x, y, 3)

# Generate points for the polynomial curve
z = np.arange(150)

# Plot the original data and the polynomial curve
plt.scatter(x, y)
plt.plot(z, a*(z**3) + b*(z**2) + c*z + d)
plt.show()

# Record the time after plotting
pause1Time = time.time()

# Ask for square meter input
m2Guess = float(input("Please, enter square meter (m2): "))

# Record the time after the first input
continue1Time = time.time()

# Calculate house price guess
housePriceGuess = a*(m2Guess**3) + b*(m2Guess**2) + c*m2Guess + d
print("House Price Guess => $" + str(housePriceGuess))

# Record the time after printing the guess
pause2Time = time.time()

# Ask for house price input
housePrice = float(input("Please, enter house price ($): "))

# Record the time after the second input
continue2Time = time.time()

# Search for square meter corresponding to the given house price
for i in np.arange(0, 1000, 0.001):
    guessPrice = a*(i**3) + b*(i**2) + c*i + d
    if(guessPrice < housePrice):
        continue
    else:
        print("Square Meter (m2) Guess => " + str(i) + "m2")
        break

# Calculate and print the total time taken
print("{} Seconds".format((time.time() - continue2Time) + (pause2Time - continue1Time) + (pause1Time - startTime)))
