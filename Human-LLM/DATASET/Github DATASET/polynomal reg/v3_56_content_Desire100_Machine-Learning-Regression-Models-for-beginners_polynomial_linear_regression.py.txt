import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures

# Load the dataset
dataset = pd.read_csv('Position_Salaries.csv')
X = dataset['Level'].values.reshape(-1, 1)  # Feature: Level
y = dataset['Salary'].values  # Target: Salary

# Fit a linear regression model
linear_model = LinearRegression()
linear_model.fit(X, y)

# Fit a polynomial regression model of degree 4
polynomial_features = PolynomialFeatures(degree=4)
X_poly = polynomial_features.fit_transform(X)
polynomial_model = LinearRegression()
polynomial_model.fit(X_poly, y)

# Visualize the linear regression results
plt.scatter(X, y, color='red', label='Actual data')
plt.plot(X, linear_model.predict(X), color='blue', label='Linear Regression')
plt.title('Truth or Bluff (Linear Regression)')
plt.xlabel('Position Level')
plt.ylabel('Salary')
plt.legend()
plt.show()

# Visualize the polynomial regression results
plt.scatter(X, y, color='red', label='Actual data')
plt.plot(X, polynomial_model.predict(X_poly), color='blue', label='Polynomial Regression')
plt.title('Truth or Bluff (Polynomial Regression)')
plt.xlabel('Position Level')
plt.ylabel('Salary')
plt.legend()
plt.show()
