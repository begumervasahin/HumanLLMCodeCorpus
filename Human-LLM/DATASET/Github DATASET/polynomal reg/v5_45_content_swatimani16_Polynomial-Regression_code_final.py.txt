import numpy as np
import matplotlib.pyplot as plt

# Generate sample data
num_points = 1000
num_samples = 10
x_samples = np.linspace(0, 1, num_samples)
noise = np.random.normal(0, 0.3, num_samples)
t_samples = np.sin(2 * np.pi * x_samples) + noise

# Define functions

def predict_y(x, weights, degree):
    # Calculate predicted y values using polynomial equation
    polynomial_terms = np.array([weights[i] * (x ** i) for i in range(degree + 1)])
    return np.sum(polynomial_terms)

def calculate_weights(x, t, degree):
    # Solve for weights using linear regression
    A = np.zeros((degree + 1, degree + 1))
    for i in range(degree + 1):
        for j in range(degree + 1):
            A[i, j] = (x ** (i + j)).sum()
    T = np.array([((x ** i) * t).sum() for i in range(degree + 1)])
    return np.linalg.solve(A, T)

def compute_design_matrix(x, degree):
    # Compute design matrix for given degree
    return x[:, None] ** np.arange(degree + 1)

# Try different polynomial degrees and plot the results
for degree in [0, 1, 3, 9]:
    # Calculate weights
    weights = calculate_weights(x_samples, t_samples, degree)
    # Predict y values for sample points
    y_estimates = [predict_y(i, weights, degree) for i in x_samples]
    # Plotting
    plt.plot(x_samples, y_estimates, 'r-', label=f'Degree {degree}')  # Estimated curve
    plt.plot(x_samples, t_samples, 'bo', label='Sample Points')  # Sample points
    plt.plot(x_samples, np.sin(2 * np.pi * x_samples), 'g-', label='True Sine Curve')  # True sine curve
    plt.legend()
    plt.show()

# Perform regularization and plot the results
degree = 9
regularization_param = 1
design_matrix = compute_design_matrix(x_samples, degree)
S_0 = design_matrix.T.dot(design_matrix) + regularization_param * np.eye(degree + 1)
y_0 = t_samples.dot(design_matrix)
coefficients = np.linalg.solve(S_0, y_0)[::-1]
fitted_polynomial = np.poly1d(coefficients)
x_smooth = np.linspace(0, 1, num_points)
# Plotting
plt.plot(x_samples, t_samples, 'bo', label='Sample Points')  # Sample points
plt.plot(x_smooth, np.sin(2 * np.pi * x_smooth), 'g-', label='True Sine Curve')  # True sine curve
plt.plot(x_smooth, fitted_polynomial(x_smooth), 'r-', label='Fitted Polynomial')  # Fitted polynomial
plt.legend()
plt.show()
