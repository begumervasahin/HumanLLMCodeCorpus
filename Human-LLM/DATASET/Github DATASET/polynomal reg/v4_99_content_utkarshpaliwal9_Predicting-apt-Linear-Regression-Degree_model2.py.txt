# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
import pandas as pd

# Reading the dataset
dataset = pd.read_csv('Position_Salaries.csv')

# Extracting features and target variable
X = dataset.iloc[:, 1:2].values
y = dataset.iloc[:, 2].values

# Importing required modules for polynomial regression
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures

# Setting initial degree and variables for finding the appropriate degree
degree = 1
min_mse = float('inf')
best_degree = 1

# Looping through degrees from 1 to 10 to find the best degree for polynomial regression
while degree <= 10:
    # Creating polynomial features
    poly_reg = PolynomialFeatures(degree=degree)
    X_poly_temp = poly_reg.fit_transform(X)
    
    # Fitting polynomial regression model
    poly_reg.fit(X_poly_temp, y)
    lin_reg = LinearRegression()
    lin_reg.fit(X_poly_temp, y)
    
    # Predicting the target variable
    y_pred = lin_reg.predict(X_poly_temp)
    
    # Calculating Mean Squared Error (MSE)
    mse = np.mean((y_pred - y) ** 2)
    
    # Updating the best degree if current MSE is smaller
    if mse < min_mse:
        min_mse = mse
        best_degree = degree
    
    # Incrementing the degree
    degree += 1

# Applying polynomial regression with the best degree
poly_reg = PolynomialFeatures(degree=best_degree)
X_poly = poly_reg.fit_transform(X)
poly_reg.fit(X_poly, y)
lin_reg = LinearRegression()
lin_reg.fit(X_poly, y)

# Visualizing the results
plt.scatter(X, y, color='red')
plt.plot(X, lin_reg.predict(poly_reg.fit_transform(X)), color='blue')
plt.title('Truth or Bluff (Polynomial Regression)')
plt.xlabel('Position level')
plt.ylabel('Salary')
plt.show()
