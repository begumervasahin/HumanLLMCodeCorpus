# Import necessary libraries
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures

# Load the dataset from a CSV file
data = pd.read_csv("datasets/dollar.csv")

# Extract 'Day' as feature (X) and 'Price' as target variable (Y), reshaping them for sklearn compatibility
days = np.array(data["Day"]).reshape(-1, 1)
prices = np.array(data["Price"]).reshape(-1, 1)

# Visualize the original data points
plt.figure(figsize=(10, 6))  # Set the figure size for better visibility
plt.scatter(days, prices, color='blue', label='Actual data')
plt.xlabel("Day")
plt.ylabel("Price")
plt.title("Dollar Price Trend")

# Apply a polynomial transformation to the feature set
degree_of_polynomial = 2  # Degree of the polynomial for the transformation
poly_features = PolynomialFeatures(degree=degree_of_polynomial)
days_poly = poly_features.fit_transform(days)

# Perform linear regression on the transformed features
linear_regression_model = LinearRegression()
linear_regression_model.fit(days_poly, prices)

# Generate predictions using the model for a smooth curve
prices_predicted = linear_regression_model.predict(days_poly)

# Plot the polynomial regression curve
plt.plot(days, prices_predicted, color="red", label="Polynomial Regression")

# Add a legend to the plot
plt.legend()

# Display the plot
plt.show()
