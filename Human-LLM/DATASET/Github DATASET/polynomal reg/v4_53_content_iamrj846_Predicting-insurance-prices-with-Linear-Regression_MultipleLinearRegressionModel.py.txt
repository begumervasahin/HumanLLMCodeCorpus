# Importing necessary libraries
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import LabelEncoder, OneHotEncoder
from sklearn.model_selection import train_test_split
from sklearn.linear_model import LinearRegression
from sklearn.metrics import r2_score
import statsmodels.formula.api as sm

# Loading the dataset
dataset = pd.read_csv('insurance.csv')

# Separating features (X) and target variable (y)
X = dataset.iloc[:, :5].values
y = dataset.iloc[:, 6].values

# Encoding categorical features
labelencoder_X = LabelEncoder()
X[:, 1] = labelencoder_X.fit_transform(X[:, 1])
X[:, 4] = labelencoder_X.fit_transform(X[:, 4])

# One-hot encoding categorical features
onehotencoder = OneHotEncoder(categorical_features=[1, 4])
X = onehotencoder.fit_transform(X).toarray()
X = X[:, [1, 3, 4, 5, 6]]

# Adding a column of ones to the features (for the intercept term)
X = np.append(arr=np.ones((1338, 1)).astype(int), values=X, axis=1)

# Performing backward elimination using Ordinary Least Squares (OLS)
X_opt = X[:, [0, 2, 3, 4]]
regressor_OLS = sm.OLS(endog=y, exog=X_opt).fit()
regressor_OLS.summary()

# Splitting the dataset into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(X_opt, y, test_size=0.2, random_state=0)

# Fitting the linear regression model to the training data
regressor = LinearRegression()
regressor.fit(X_train, y_train)

# Predicting on the training set and calculating R-squared
y_predTrain = regressor.predict(X_train)
r2_Score_Train = r2_score(y_train, y_predTrain) * 100

# Predicting on the test set and calculating R-squared
y_predTest = regressor.predict(X_test)
r2_Score_Test = r2_score(y_test, y_predTest) * 100
