# Import necessary libraries
import matplotlib.pyplot as plt
import numpy as np
from sklearn.metrics import r2_score

# Set a seed for reproducibility
np.random.seed(2)

# Generate synthetic data for page speeds and purchase amounts
pageSpeeds = np.random.normal(3.0, 1.0, 100)
purchaseAmount = np.random.normal(50.0, 30.0, 100) / pageSpeeds

# Scatter plot of page speeds vs. purchase amounts
plt.scatter(pageSpeeds, purchaseAmount)
plt.xlabel("Page Speeds")
plt.ylabel("Purchase Amount")
plt.title("Page Speeds vs. Purchase Amount")
plt.show()

# Split data into training and testing sets
trainX = pageSpeeds[:80]
testX = pageSpeeds[80:]
trainY = purchaseAmount[:80]
testY = purchaseAmount[80:]

# Scatter plot of training and testing data
plt.scatter(trainX, trainY, label="Training Data")
plt.scatter(testX, testY, label="Testing Data")
plt.xlabel("Page Speeds")
plt.ylabel("Purchase Amount")
plt.legend()
plt.title("Training and Testing Data")
plt.show()

# Fit a 7th degree polynomial to the training data
p4 = np.poly1d(np.polyfit(trainX, trainY, 7))

# Generate x values for plotting the polynomial curve
xp = np.linspace(0, 7, 100)

# Plot the polynomial curve along with training data
plt.scatter(trainX, trainY, label="Training Data")
plt.plot(xp, p4(xp), c='b', label="Polynomial Fit")
plt.xlabel("Page Speeds")
plt.ylabel("Purchase Amount")
plt.legend()
plt.title("Polynomial Fit on Training Data")
plt.show()

# Plot the polynomial curve along with testing data
plt.scatter(testX, testY, label="Testing Data")
plt.plot(xp, p4(xp), c='b', label="Polynomial Fit")
plt.xlabel("Page Speeds")
plt.ylabel("Purchase Amount")
plt.legend()
plt.title("Polynomial Fit on Testing Data")
plt.show()

# Calculate R-squared score for testing data
r2_test = r2_score(testY, p4(testX))
print("R-squared score for testing data:", r2_test)

# Calculate R-squared score for training data
r2_train = r2_score(trainY, p4(trainX))
print("R-squared score for training data:", r2_train)
