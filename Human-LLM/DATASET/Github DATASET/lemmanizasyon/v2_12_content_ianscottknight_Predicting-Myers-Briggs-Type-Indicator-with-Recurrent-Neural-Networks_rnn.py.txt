import os
import numpy as np
import pandas as pd
import csv
import random
import pickle
from sklearn.model_selection import KFold
from sklearn.metrics import confusion_matrix, accuracy_score
import tensorflow as tf
from keras.models import Sequential, load_model
from keras.layers import Embedding, LSTM, Dense
from keras.optimizers import Adam
from keras.preprocessing import sequence, text
from keras.wrappers.scikit_learn import KerasClassifier
from nltk import word_tokenize
from nltk.stem import WordNetLemmatizer
from nltk.corpus import stopwords

MODELS_DIR = "models"
DATA_DIR = "data"
GLOVE_PATH = os.path.join(DATA_DIR, "glove.6B.50d.txt")
DIMENSIONS = ["IE", "NS", "FT", "PJ"]
MODEL_BATCH_SIZE = 128
TOP_WORDS = 2500
MAX_POST_LENGTH = 40
EMBEDDING_VECTOR_LENGTH = 50
LEARNING_RATE = 0.01
DROPOUT = 0.1
NUM_EPOCHS = 1
CROSS_VALIDATION = False
SAMPLE = True
WORD_CLOUD = True
SAVE_MODEL = True

def load_data(dimension):
    x_train, y_train, x_test, y_test = [], [], [], []

    def load_csv(file_path, label):
        with open(file_path, "r") as f:
            reader = csv.reader(f)
            for row in reader:
                for post in row:
                    x_train.append(post)
                    y_train.append(label)

    load_csv(os.path.join(DATA_DIR, f"train_{dimension[0]}.csv"), 0)
    load_csv(os.path.join(DATA_DIR, f"train_{dimension[1]}.csv"), 1)
    load_csv(os.path.join(DATA_DIR, f"test_{dimension[0]}.csv"), 0)
    load_csv(os.path.join(DATA_DIR, f"test_{dimension[1]}.csv"), 1)

    return x_train, y_train, x_test, y_test

def lemmatize(x, MBTI_TYPES, stop_words, lemmatizer):
    lemmatized = []
    for post in x:
        temp = post.lower()
        for mbti_type in MBTI_TYPES:
            mbti_type = mbti_type.lower()
            temp = temp.replace(" " + mbti_type, "")
        temp = " ".join([lemmatizer.lemmatize(word) for word in temp.split(" ") if word not in stop_words])
        lemmatized.append(temp)
    return np.array(lemmatized)

def preprocess(x, tokenizer, TOP_WORDS, MAX_POST_LENGTH):
    tokenized = tokenizer.texts_to_sequences(x)
    return sequence.pad_sequences(tokenized, maxlen=MAX_POST_LENGTH)

def build_model(TOP_WORDS, EMBEDDING_VECTOR_LENGTH, MAX_POST_LENGTH, DROPOUT, LEARNING_RATE):
    model = Sequential()
    model.add(Embedding(TOP_WORDS, EMBEDDING_VECTOR_LENGTH, input_length=MAX_POST_LENGTH, mask_zero=True, trainable=True))
    model.add(LSTM(EMBEDDING_VECTOR_LENGTH, dropout=DROPOUT, recurrent_dropout=DROPOUT, activation="sigmoid", kernel_initializer="zeros"))
    model.add(Dense(1, activation="sigmoid"))
    optimizer = Adam(lr=LEARNING_RATE, beta_1=0.9, beta_2=0.999, epsilon=1e-8)
    model.compile(loss="binary_crossentropy", optimizer=optimizer, metrics=["accuracy"])
    return model

def main():
    for k in range(len(DIMENSIONS)):
        x_train, y_train, x_test, y_test = load_data(DIMENSIONS[k])

        MBTI_TYPES = ["INFJ", "ENTP", "INTP", "INTJ", "ENTJ", "ENFJ", "INFP", "ENFP", "ISFP", "ISTP", "ISFJ", "ISTJ", "ESTP", "ESFP", "ESTJ", "ESFJ"]
        stop_words = stopwords.words("english")
        lemmatizer = WordNetLemmatizer()
        tokenizer = text.Tokenizer(num_words=TOP_WORDS, filters="")
        tokenizer.fit_on_texts(x_train + x_test)

        x_train = lemmatize(x_train, MBTI_TYPES, stop_words, lemmatizer)
        x_test = lemmatize(x_test, MBTI_TYPES, stop_words, lemmatizer)

        df = pd.DataFrame(data={"x": x_train, "y": y_train})
        df = df.sample(frac=1).reset_index(drop=True)

        if SAMPLE:
            df = df.head(10000)

        embeddings_index = dict()
        with open(GLOVE_PATH) as f:
            for line in f:
                values = line.split()
                word = values[0]
                embeddings_index[word] = np.asarray(values[1:], dtype="float32")

        print(f"Loaded {len(embeddings_index)} word vectors.")

        embedding_matrix = np.zeros((TOP_WORDS, EMBEDDING_VECTOR_LENGTH))
        for word, i in tokenizer.word_index.items():
            if i < TOP_WORDS:
                embedding_vector = embeddings_index.get(word)
                if embedding_vector is not None:
                    embedding_matrix[i] = embedding_vector

        with tf.device("/gpu:0"):
            model = build_model(TOP_WORDS, EMBEDDING_VECTOR_LENGTH, MAX_POST_LENGTH, DROPOUT, LEARNING_RATE)
            print(model.summary())

            if CROSS_VALIDATION:
                k_fold = KFold(n_splits=6)
                scores_k = []
                confusion_k = np.array([[0, 0], [0, 0]])

                for train_indices, test_indices in k_fold:
                    x_train_k = df.iloc[train_indices]["x"].values
                    y_train_k = df.iloc[train_indices]["y"].values
                    x_test_k = df.iloc[test_indices]["x"].values
                    y_test_k = df.iloc[test_indices]["y"].values

                    model.fit(preprocess(x_train_k, tokenizer, TOP_WORDS, MAX_POST_LENGTH), y_train_k, epochs=NUM_EPOCHS, batch_size=MODEL_BATCH_SIZE)

                    predictions_k = model.predict_classes(preprocess(x_test_k, tokenizer, TOP_WORDS, MAX_POST_LENGTH))
                    confusion_k += confusion_matrix(y_test_k, predictions_k)
                    score_k = accuracy_score(y_test_k, predictions_k)
                    scores_k.append(score_k)

                with open(os.path.join(DATA_DIR, f"rnn_cross_validation_{DIMENSIONS[k]}.txt"), "w") as f:
                    f.write(f"*** {DIMENSIONS[k][0]}/{DIMENSIONS[k][1]} TRAINING SET CROSS VALIDATION (POSTS) ***\n")
                    f.write(f"Total posts classified: {len(x_train)}\n")
                    f.write(f"Accuracy: {sum(scores_k) / len(scores_k)}\n")
                    f.write("Confusion matrix: \n")
                    f.write(np.array2string(confusion_k, separator=", "))

            model.fit(preprocess(df["x"].values, tokenizer, TOP_WORDS, MAX_POST_LENGTH), df["y"].values, epochs=NUM_EPOCHS, batch_size=MODEL_BATCH_SIZE)

            predictions = model.predict_classes(preprocess(x_test, tokenizer, TOP_WORDS, MAX_POST_LENGTH))
            confusion = confusion_matrix(y_test, predictions)
            score = accuracy_score(y_test, predictions)

            with open(os.path.join(MODELS_DIR, f"rnn_accuracy_{DIMENSIONS[k]}.txt"), "w") as f:
                f.write(f"*** {DIMENSIONS[k][0]}/{DIMENSIONS[k][1]} TEST SET CLASSIFICATION (POSTS) ***\n")
                f.write(f"Total posts classified: {len(x_test)}\n")
                f.write(f"Accuracy: {score}\n")
                f.write("Confusion matrix: \n")
                f.write(np.array2string(confusion, separator=", "))

            print(f"\nWrote training / test results for {DIMENSIONS[k]} here: {os.path.join(MODELS_DIR, 'rnn_accuracy_{DIMENSIONS[k]}.txt')}\n")

            if WORD_CLOUD:
                NUM_EXTREME_EXAMPLES = 500
                probs = model.predict_proba(preprocess(x_test, tokenizer, TOP_WORDS, MAX_POST_LENGTH))
                scores = []
                indices = []

                for i, prob in enumerate(probs, 0):
                    scores.append(prob[0])
                    indices.append(i)

                sorted_probs = sorted(zip(scores, indices))
                min_prob_indices = sorted_probs[:NUM_EXTREME_EXAMPLES]

                max_prob_indices = sorted_probs[-NUM_EXTREME_EXAMPLES:]

                with open(os.path.join(DATA_DIR, f"extreme_examples_{DIMENSIONS[k][0]}.txt"), "w") as f:
                    for prob, i in min_prob_indices:
                        f.write(x_test[i] + "\n")
                        f.write("\n")

                with open(os.path.join(DATA_DIR, f"extreme_examples_{DIMENSIONS[k][1]}.txt"), "w") as f:
                    for prob, i in max_prob_indices:
                        f.write(x_test[i] + "\n")
                        f.write("\n")

            model.save(os.path.join(MODELS_DIR, f"rnn_model_{DIMENSIONS[k]}.h5"))

            with open(os.path.join(MODELS_DIR, f"rnn_tokenizer_{DIMENSIONS[k]}.pkl"), "wb") as f:
                pickle.dump(tokenizer, f, protocol=pickle.HIGHEST_PROTOCOL)

if __name__ == "__main__":
    main()
