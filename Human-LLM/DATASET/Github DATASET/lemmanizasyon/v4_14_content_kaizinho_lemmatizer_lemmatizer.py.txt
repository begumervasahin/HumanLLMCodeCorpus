import re

# Function to lemmatize a sentence based on a word list
def lemmatize(sentence, word_list):
    lemm_sentence = ""
    for word in sentence.split():
        lemm_sentence += word_list.get(word, word) + " "
    return lemm_sentence.strip()

# Function to extract words from raw data file and save to a JSON file
def extract_words(raw_data_file, filename="data/swedish_words.json"):
    # Regular expression patterns for different word types
    pattern1 = r"^(===(Verb|Substantiv|Adjektiv|Adverb|Konjuktion|Pronomen|Räkneord|Interjektion|Konjugation)===)$"
    pattern2 = r"^'''([a-zA-ZåäöÅÄÖ]+)'''$"
    pattern3 = r"^(.*?)$"
    
    # Initialize variables
    data = ""
    matching_strings = []

    # Open the raw data file for reading and the JSON file for writing
    with open(raw_data_file, "r") as input_file:
        with open(filename, "w") as output_file:
            output_file.write("{\r\n")  # Write the JSON file opening brace
            # Iterate through each line in the raw data file
            for line in input_file:
                # Check for matches with each pattern
                match1 = re.search(pattern1, line)
                match2 = re.search(pattern2, line)
                match3 = re.search(pattern3, line)
                
                # Handling the matching strings based on their lengths
                if len(matching_strings) == 0:
                    if match1:
                        matching_strings.append(match1.group())
                elif len(matching_strings) == 1:
                    if match2:
                        matching_strings.append(match2.group(1))
                    else:
                        matching_strings = []
                elif len(matching_strings) == 2:
                    if match3:
                        # Construct the data string to be written to the JSON file
                        data += '\t"' + matching_strings[1] + '": "' + match3.group(1) + '",\r\n'
                    matching_strings = []
            # Write the constructed data to the JSON file
            output_file.write(data.rstrip(',\r\n'))
            output_file.write("\r\n}\r\n")  # Write the JSON file closing brace
