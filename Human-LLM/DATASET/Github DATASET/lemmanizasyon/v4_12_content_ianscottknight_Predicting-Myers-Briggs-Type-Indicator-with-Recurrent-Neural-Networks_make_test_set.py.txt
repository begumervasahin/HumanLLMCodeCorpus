import os
import pandas as pd
import csv
import re

DATA_DIR = "data"
MBTI_CLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_clean.csv")
DIMENSIONS = ("IE", "NS", "TF", "PJ")

# Read the CSV file into a pandas DataFrame
df = pd.read_csv(MBTI_CLEAN_CSV_PATH)

# Iterate over each personality dimension
for dimension in DIMENSIONS:
    letter_1, letter_2 = dimension
    
    # Iterate over each letter in the dimension (e.g., "I" and "E" for "IE")
    for letter in [letter_1, letter_2]:
        posts = []

        # Iterate over each row in the DataFrame
        for index, row in df.iterrows():
            # Check if the letter is in the personality type
            if letter in row["type"]:
                # Split posts into individual posts
                hundred_posts = row["posts"].split("|||") 
                
                # Iterate over each post
                for post in hundred_posts:
                    # Skip posts with URLs, empty posts, posts with no alphabetic characters
                    if ("http" in post) or (post == "") or (post is None) or (not re.search("[a-zA-Z]", post)):
                        continue
                    posts.append(post)

        # Create a CSV file for the selected letter
        test_csv_path = os.path.join(DATA_DIR, f"test_{letter}.csv")
        with open(test_csv_path, "w") as f:
            writer = csv.writer(f)
            
            # Write each post to the CSV file
            for post in posts:
                writer.writerow([post])
