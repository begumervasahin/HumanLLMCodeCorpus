import json
from jsonrpc import ServerProxy, JsonRpc20, TransportTcpIp
from pprint import pprint

class StanfordNLP:
    def __init__(self):
        # Establishing connection with CoreNLP server
        self.server = ServerProxy(JsonRpc20(), TransportTcpIp(addr=("127.0.0.1", 8080)))

    def parse(self, text):
        # Parsing the input text using CoreNLP server
        return json.loads(self.server.parse(text))

# Creating an instance of StanfordNLP class
nlp = StanfordNLP()

# Parsing a sample text and printing the parse structures
result = nlp.parse("He couldn't fully explain why, but ventured that ``people have a summer rally in July and August and then it's just profit-taking.''")
pprint(result)
