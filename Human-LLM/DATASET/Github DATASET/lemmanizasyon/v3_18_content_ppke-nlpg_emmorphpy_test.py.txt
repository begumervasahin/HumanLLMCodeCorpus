import emmorphpy

# Initialize EmMorphPy
emmorph = emmorphpy.EmMorphPy()

# Define the file path
file_path = 'test/test_words.txt'

# Open the file containing words to process
with open(file_path, 'r') as file:
    # Iterate through each line (word) in the file
    for line in file:
        # Remove leading/trailing whitespace from the word
        word = line.strip()
        
        # Apply stemming to the word and get the results
        stem_results = emmorph.dstem(word, out_mode=list)
        
        # Check if stemming results are available
        if stem_results:
            # Iterate through each result for the word
            for result in stem_results:
                # Check if the result has the expected format
                if len(result) == 5:
                    # Print the stemmed form along with additional information
                    print(word, result[2], result[0], result[1], sep='\t')
                else:
                    # Print if the result format is unexpected
                    print(word, '<unknown>', sep='\t')
        else:
            # Print if no stemming results are available for the word
            print(word, '<unknown>', sep='\t')
        
        # Separate each word's results with an empty line for clarity
        print()
