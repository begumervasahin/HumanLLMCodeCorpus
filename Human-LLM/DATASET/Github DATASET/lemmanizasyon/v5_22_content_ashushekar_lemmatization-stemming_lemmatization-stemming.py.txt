# Importing necessary libraries
import spacy
import nltk

# Loading English language model for spaCy
english_nlp = spacy.load('en')

# Initializing Porter Stemmer from NLTK
porter_stemmer = nltk.stem.PorterStemmer()

# Define a function to compare lemmatization and stemming
def compare_normalization(text):
    # Tokenizing the input text using spaCy
    doc = english_nlp(text)
    
    # Printing lemmatized tokens
    print("Lemmatization:")
    lemmatized_tokens = [token.lemma_ for token in doc]
    print(lemmatized_tokens)
    
    # Printing stemmed tokens
    print("Stemming:")
    stemmed_tokens = [porter_stemmer.stem(token.norm_.lower()) for token in doc]
    print(stemmed_tokens)

# Example usage
compare_normalization("Our meeting today was worse than yesterday, "
                      "I'm scared of meeting the clients tomorrow")
