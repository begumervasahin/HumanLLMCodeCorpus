# Importing necessary modules
import os
import sys

# Ensuring that Django settings module is properly set
if __name__ == "__main__":
    os.environ.setdefault("DJANGO_SETTINGS_MODULE", "webservice.settings")
    
    # Trying to execute Django management commands
    try:
        from django.core.management import execute_from_command_line
    except ImportError:
        # Handling case where Django is not installed
        try:
            import django
        except ImportError:
            # Raising error if Django is not installed or not found in PYTHONPATH
            raise ImportError(
                "Couldn't import Django. Are you sure it's installed and "
                "available on your PYTHONPATH environment variable? Did you "       
                "forget to activate a virtual environment?"
            )
        raise
    
    # Executing Django management commands from the command line arguments
    execute_from_command_line(sys.argv)
