def check(root, suffix, guess, action):
    if action == "unsuz yumusamasi":
        return len(suffix) > 0 and suffix[0] in ["a", "e", "ý", "i", "o", "ö", "u", "ü"] and check_suffix_validation(suffix)[0]
    if action == "unlu daralmasi":
        if (guess == "demek" or guess == "yemek") and check_suffix_validation(suffix)[0]:
            return True
        if suffix.startswith("yor"):
            last_vowel = next((letter for letter in reversed(guess[:-3]) if letter in ["a", "e", "ý", "i", "o", "ö", "u", "ü"]), None)
            if last_vowel in ["a", "e"] and check_suffix_validation(suffix)[0]:
                return True
        return False
    if action == "fiil" or action == "olumsuzluk eki":
        return check_suffix_validation(suffix)[0] and not ((root.endswith("la") or root.endswith("le")) and suffix.startswith("r"))
    if action == "unlu dusmesi":
        count = sum(1 for letter in guess if letter in ["a", "e", "ý", "i", "o", "ö", "u", "ü"])
        last_vowel = next((letter for letter in reversed(guess) if letter in ["a", "e", "ý", "i", "o", "ö", "u", "ü"]), None)
        if check_suffix_validation(suffix)[0] and count == 2 and (last_vowel in ["ý", "i", "u", "ü"]) and (
                len(suffix) > 0 and suffix[0] in ["a", "e", "ý", "i", "o", "ö", "u", "ü"]):
            if last_vowel == "ý":
                return suffix[0] in ["a", "ý"]
            elif last_vowel == "i":
                return suffix[0] in ["e", "i"]
            elif last_vowel == "u":
                return suffix[0] in ["a", "u"]
            elif last_vowel == "ü":
                return suffix[0] in ["e", "ü"]
        return False
    return True

def find_position(word, revised_dict):
    result_list = []

    if "'" in word:
        result_list.append([word[:word.index("'")] + "_1", "tirnaksiz", word])

    mid_list = []
    for i in range(len(word)):
        guess = word[:len(word) - i]
        suffix = word[len(word) - i:]
        ct = 1
        while guess + "_" + str(ct) in revised_dict:
            if check(guess, suffix, revised_dict[guess + "_" + str(ct)][1], revised_dict[guess + "_" + str(ct)][0]):
                guess_list = revised_dict[guess + "_" + str(ct)]
                while guess_list[0] not in ["kok", "fiil", "olumsuzluk"]:
                    guess_list = revised_dict[guess_list[1]]
                mid_list.append([guess_list[1], revised_dict[guess + "_" + str(ct)][0], guess + "_" + str(ct)])
            ct += 1

    temp_list = []
    for kel in mid_list:
        kelime_kok = kel[0][:kel[0].index("_")]
        kelime_len = len(kelime_kok)
        if kelime_kok.endswith("mak") or kelime_kok.endswith("mek"):
            kelime_len -= 3
        not_inserted = True
        for index in range(len(temp_list)):
            temp_kelime = temp_list[index]
            temp_kelime_kok = temp_kelime[0][:temp_kelime[0].index("_")]
            temp_len = len(temp_kelime_kok)
            if temp_kelime_kok.endswith("mak") or temp_kelime_kok.endswith("mek"):
                temp_len -= 3
            if kelime_len > temp_len:
                temp_list.insert(index, kel)
                not_inserted = False
        if not_inserted:
            temp_list.append(kel)

    result = result_list + temp_list

    if len(result) == 0:
        result.append([word + "_1", "çaresiz", word + "_1"])

    return result

def load_revised_dict():
    try:
        with open('revisedDict.pkl', 'rb') as f:
            return pickle.load(f)
    except IOError:
        print("Please run trainLexicon.py to generate revisedDict.pkl file")
        sys.exit(1)

def main():
    if len(sys.argv) < 2:
        print("Please provide a word as a system argument")
        sys.exit(1)

    word = sys.argv[1]
    print("Possible lemmas for", word, "in ranked order:")
    revised_dict = load_revised_dict()
    findings = find_position(word.lower(), revised_dict)

    for finding in findings:
        print(finding[0])

if __name__ == "__main__":
    main()
