import os
import collections
import pandas as pd
import csv

# Define file paths
DATA_DIR = "data"
MBTI_RAW_CSV_PATH = os.path.join(DATA_DIR, "mbti_1.csv")
MBTI_CLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_clean.csv")
MBTI_UNCLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_unclean.csv")

# Define the frequencies of MBTI types
MBTI_TO_FREQUENCY_DICT = {
    "ISTJ": 0.11, "ISFJ": 0.09, "INFJ": 0.04, "INTJ": 0.05,
    "ISTP": 0.05, "ISFP": 0.05, "INFP": 0.06, "INTP": 0.06,
    "ESTP": 0.04, "ESFP": 0.04, "ENFP": 0.08, "ENTP": 0.06,
    "ESTJ": 0.08, "ESFJ": 0.09, "ENFJ": 0.05, "ENTJ": 0.05
}

# Read the raw data from the CSV file
df = pd.read_csv(MBTI_RAW_CSV_PATH)

# Count occurrences of each MBTI type
counts = collections.defaultdict(int)
for mbti in df["type"]:
    counts[mbti] += 1

# Determine the MBTI type with the fewest occurrences
limiting_type = None
min_size = float("infinity")
for mbti in counts.keys():
    size = counts[mbti] / MBTI_TO_FREQUENCY_DICT[mbti]
    if size < min_size:
        min_size = size
        limiting_type = mbti

# Group the data by MBTI type
mbti_groups = collections.defaultdict(list)
for index, row in df.iterrows():
    mbti_groups[row["type"]].append(row)

# Prepare data for the unclean CSV file
unclean_data = []
for mbti, data_list in mbti_groups.items():
    slice_index = int(round(min_size * MBTI_TO_FREQUENCY_DICT[mbti]))
    unclean_data.extend(data_list[slice_index:])

# Write unclean data to CSV
with open(MBTI_UNCLEAN_CSV_PATH, "w") as f:
    writer = csv.writer(f)
    writer.writerow(["type", "posts"])
    for row in unclean_data:
        writer.writerow([row["type"], row["posts"]])

# Prepare data for the clean CSV file
clean_data = []
for mbti in MBTI_TO_FREQUENCY_DICT.keys():
    data_list = mbti_groups[mbti]
    clean_data.extend(data_list[:int(round(min_size * MBTI_TO_FREQUENCY_DICT[mbti]))])

# Write clean data to CSV
with open(MBTI_CLEAN_CSV_PATH, "w") as f:
    writer = csv.writer(f)
    writer.writerow(["type", "posts"])
    for row in clean_data:
        writer.writerow([row["type"], row["posts"]])
