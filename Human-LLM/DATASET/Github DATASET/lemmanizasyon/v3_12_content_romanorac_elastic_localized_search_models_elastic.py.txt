from elasticsearch_dsl import Document, analyzer, String

# Define a custom analyzer
custom_analyzer = analyzer(
    'lemmagen_sl',
    type='custom',
    tokenizer="uax_url_email",
    filter=["lowercase"]
)

# Define a custom document class inheriting from elasticsearch_dsl's Document class
class MyDocument(Document):
    """
    Represents a document type in Elasticsearch.
    """
    # Define a field named 'title' with the custom analyzer
    title = String(analyzer=custom_analyzer)

# Example usage
if __name__ == "__main__":
    # Create an instance of the custom document
    document = MyDocument()

    # Set the value for the 'title' field
    document.title = "Example Document Title"

    # Access and print the value of the 'title' field
    print("Title:", document.title)

    # To index the document in Elasticsearch, you would typically use the `save` method.
    # For example:
    # document.save()
