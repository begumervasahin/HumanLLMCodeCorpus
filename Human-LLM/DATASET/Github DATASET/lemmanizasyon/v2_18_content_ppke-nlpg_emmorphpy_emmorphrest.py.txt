# Import necessary functions
from xtsv import pipeline_rest_api, singleton_store_factory

# Define configurations for EmMorphPy tools
em_morph_stem = (
    'emmorphpy.emmorphpy',     # Module containing EmMorphPy
    'EmMorphPy',                # Class name
    'stem',                     # Method for stemming
    (),                         # No additional arguments for method
    {
        'task': 'stem',                     # Task to perform: stem
        'source_fields': {'form'},          # Fields to process
        'target_fields': ['anas']           # Target fields for storing results
    }
)
em_morph_analyze = (
    'emmorphpy.emmorphpy',
    'EmMorphPy',
    'analyze',
    (),
    {
        'task': 'analyze',
        'source_fields': {'form'},
        'target_fields': ['anas']
    }
)
em_morph_dstem = (
    'emmorphpy.emmorphpy',
    'EmMorphPy',
    'dstem',
    (),
    {
        'task': 'dstem',
        'source_fields': {'form'},
        'target_fields': ['anas']
    }
)

# List of EmMorphPy tools
tools = [
    (em_morph_stem, ('stem',)),
    (em_morph_analyze, ('analyze',)),
    (em_morph_dstem, ('dstem',))
]

# Configuration for the pipeline REST API
api_config = {
    'conll_comments': False,                            # Include CONLL comments or not
    'form_title': 'emMorph demo',                       # Title for the demo form
    'form_type': 'radio',                               # Type of form elements
    'doc_link': 'https://github.com/ppke-nlpg/emmorphpy' # Link to documentation
}

# Create the pipeline REST API instance
app = pipeline_rest_api(
    'emMorph',                      # API name
    tools,                          # Tools to include
    {},                             # Additional settings (empty in this case)
    **api_config,                   # Configuration settings
    singleton_store=singleton_store_factory()  # Singleton store factory
)

# Execute the API if this script is run directly
if __name__ == '__main__':
    app.run(debug=False)
