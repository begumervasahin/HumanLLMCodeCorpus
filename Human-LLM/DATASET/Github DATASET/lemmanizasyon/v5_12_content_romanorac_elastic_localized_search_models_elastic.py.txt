# Importing necessary components from Elasticsearch DSL
from elasticsearch_dsl.document import DocType
from elasticsearch_dsl import analyzer, String

# Defining a custom analyzer called lemmagen_sl
lemmagen_sl_analyzer = analyzer(
    'lemmagen_sl', 
    type='custom',
    tokenizer="uax_url_email",
    filter=["lowercase"]
)

# Defining a Document class that represents a type in Elasticsearch
class Document(DocType):
    """
    Represents a document type in Elasticsearch.
    """
    # Defining a field called 'title' with lemmagen_sl analyzer applied
    title = String(analyzer=lemmagen_sl_analyzer)
