import nltk
from nltk.corpus import wordnet as wn

def get_hyponyms(noun):
    list_of_synsets = wn.synsets(noun)
    filtered_synsets = [synset for synset in list_of_synsets if noun in str(synset)]
    hypo = lambda s: s.hyponyms()
    var_out = [list(synset.closure(hypo, depth=1)) for synset in filtered_synsets]
    return var_out

def get_hypernyms(noun):
    list_of_synsets = wn.synsets(noun)
    strings_of_synsets = [str(synset) for synset in list_of_synsets if noun in str(synset)]
    hyper = lambda s: s.hypernyms()
    var_out = [list(wn.synset(synset).closure(hyper, depth=8)) for synset in strings_of_synsets]
    return strings_of_synsets, var_out

def univ_pos_changer(stringa):
    if stringa == "NOUN":
        return "n"
    elif stringa == "VERB":
        return "v"
    elif stringa == "ADJ":
        return "a"
    elif stringa == "ADV":
        return "r"
    else:
        return None

def lemmatizer(tupla):
    if wn.morphy(tupla[0], univ_pos_changer(tupla[1])) is None:
        return tupla[0], tupla[1]
    else:
        return wn.morphy(tupla[0], univ_pos_changer(tupla[1])), tupla[1]

def tag_me(corpus):
    return [[lemmatizer(x) for x in linea] for linea in corpus]

# Example usage:
noun = "car"
hyponyms = get_hyponyms(noun)
hypernyms = get_hypernyms(noun)
print("Hyponyms:")
print(hyponyms)
print("\nHypernyms:")
print(hypernyms)
