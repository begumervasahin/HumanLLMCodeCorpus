import pandas as pd
from fbprophet import Prophet
from sklearn.metrics import mean_squared_error
from math import sqrt
from datetime import datetime, date

# Function to calculate revenue
def calculate_revenue(passtype, duration, time):
    per_thirty = 1.75
    if time < date(year=2018, month=7, day=12):
        per_thirty = 3.5
    
    qty = int(duration / 30)
    extra = duration % 30
    if extra != 0:
        qty += 1
    if passtype in ['Monthly Pass', 'Annual Pass', 'One Day Pass', 'Flex Pass']:
        qty -= 1
    if qty < 0:
        qty = 0
    return per_thirty * qty

# Read data from Excel file
df = pd.read_excel('C:/Users/tusha/Desktop/newtable.xlsx', sheet_name='newtable')

# Convert start_time and end_time to datetime format
df['start_time'] = pd.to_datetime(df['start_time'], format="%d/%m/%Y %I:%M:%S %p")
df['end_time'] = pd.to_datetime(df['end_time'], format="%d/%m/%Y %I:%M:%S %p")

# Calculate revenue
df['revenue'] = df.apply(lambda row: calculate_revenue(row['passholder_type'], row['trip_duration'], row['start_time'].date()), axis=1)

# Pivot table to aggregate data
df_pivot = pd.pivot_table(df[['revenue', 'start_time']], aggfunc='sum', index=df['start_time'].dt.date, columns=['revenue'], fill_value=0)
df_pivot.index = pd.to_datetime(df_pivot.index)

# Initialize empty DataFrame for output and lists for storing MSE
df_output = pd.DataFrame()
mse_train = []
mse_test = []

# Function to prepare data for Prophet
def prepare_prophet_data(df, column_num=0):
    df_prophet = pd.DataFrame()
    df_prophet['ds'] = df.index
    df_prophet['y'] = list(df.iloc[:, column_num])
    return df_prophet

# Function to resample, split data into train and test sets, and perform Prophet forecasting
def perform_prophet_forecasting(df, sample_freq='D', train_proportion=0.80):
    df_resampled = df.resample(sample_freq).sum()
    rem_dates = ['2016-08-14', '2018-10-16', '2017-03-26', '2017-05-11', '2017-08-13', '2017-10-08',
                 '2017-12-10', '2018-04-22', '2018-06-24', '2018-09-30', '2018-12-02']
    df_resampled['temp'] = df_resampled.index
    df_resampled = df_resampled[~df_resampled['temp'].isin(rem_dates)]
    df_resampled.drop('temp', axis=1, inplace=True)
    
    train_size = int(train_proportion * len(df_resampled))
    df_train = df_resampled.iloc[:train_size]
    df_test = df_resampled.iloc[train_size:]
    
    df_train_prophet = prepare_prophet_data(df_train)
    df_test_prophet = prepare_prophet_data(df_test)
    
    return df_train_prophet, df_test_prophet

# Loop over columns for forecasting
for i, column in enumerate(df_pivot.columns):
    # Prepare data for Prophet
    train_data, test_data = perform_prophet_forecasting(df_pivot)

    # Initialize and fit Prophet model
    model = Prophet(yearly_seasonality=True, weekly_seasonality=True, daily_seasonality=True)
    model.fit(train_data)

    # Make future dataframe and predict
    future_dataframe = model.make_future_dataframe(periods=len(test_data), freq='D')
    forecast = model.predict(future_dataframe)
    df_output[column] = forecast['yhat']

    # Calculate and store MSE for train and test sets
    y_actual_test = test_data['y']
    y_predicted_test = df_output[column].tail(len(test_data))
    mse_test.append(sqrt(mean_squared_error(y_actual_test, y_predicted_test)))

    y_actual_train = train_data['y']
    y_predicted_train = df_output[column].iloc[:len(train_data)]
    mse_train.append(sqrt(mean_squared_error(y_actual_train, y_predicted_train)))

df_output['ds'] = forecast['ds']

# Remove specified dates from df_pivot
rem_dates = ['2016-08-14', '2018-10-16', '2017-03-26', '2017-05-11', '2017-08-13', '2017-10-08',
             '2017-12-10', '2018-04-22', '2018-06-24', '2018-09-30', '2018-12-02']
df_pivot['ds'] = df_pivot.index
df_pivot = df_pivot[~df_pivot['ds'].isin(rem_dates)]
df_pivot = df_pivot.reset_index(drop=True)

mse_train_df = pd.DataFrame(mse_train)
mse_test_df = pd.DataFrame(mse_test)

# Write to Excel file
with pd.ExcelWriter('outputY_Passholder.xlsx') as writer:
    df_pivot.to_excel(writer, sheet_name='actual(YWD)')
    df_output.to_excel(writer, sheet_name='predicted(YWD)')
    mse_train_df.to_excel(writer, sheet_name='mse_train(YWD)')
    mse_test_df.to_excel(writer, sheet_name='mse_test(YWD)')
