import statistics

def pivot_position(given_list, first, last):
    """
    Finds the pivot position for the quicksort algorithm.

    Args:
        given_list (list): The list of integers to be sorted.
        first (int): The starting index of the sublist.
        last (int): The ending index of the sublist.

    Returns:
        int: The pivot position.
    """
    # Determine pivot value using median-of-three method
    low = given_list[first]
    high = given_list[last]
    mid = (first + last) // 2
    pivot_val = statistics.median([low, given_list[mid], high])

    # Determine pivot index
    if pivot_val == low:
        pivot_index = first
    elif pivot_val == high:
        pivot_index = last
    else:
        pivot_index = mid

    # Swap pivot with last element
    given_list[last], given_list[pivot_index] = given_list[pivot_index], given_list[last]
    pivot = given_list[last]
    left = first
    right = last - 1

    # Partition the sublist around the pivot
    while True:
        while left <= right and given_list[left] <= pivot:
            left += 1
        while left <= right and given_list[right] >= pivot:
            right -= 1
        if right < left:
            break
        else:
            given_list[left], given_list[right] = given_list[right], given_list[left]   

    # Move pivot to its final position
    given_list[first], given_list[right] = given_list[right], given_list[first]
    return right

def quicksort(given_list, first, last):
    """
    Sorts a list of integers using the quicksort algorithm.

    Args:
        given_list (list): The list of integers to be sorted.
        first (int): The starting index of the sublist.
        last (int): The ending index of the sublist.
    """
    if first < last:
        # Partition and recursively sort sublists
        p = pivot_position(given_list, first, last)
        quicksort(given_list, first, p - 1)
        quicksort(given_list, p + 1, last)
