def quicksort_Spivot(qPArr, start, end):
    # Calculate the size of the array segment
    size = end - start + 1
    # Initialize a stack for storing indices
    stack = [0] * size
    top = -1

    # Initialize the stack with start and end indices
    top = top + 1
    stack[top] = start
    top = top + 1
    stack[top] = end

    # Loop until the stack is empty
    while top >= 0:
        # Pop start and end indices from the stack
        end = stack[top]
        top = top - 1
        start = stack[top]
        top = top - 1

        # Initialize pivot index and pivot value
        i = start - 1
        pv = qPArr[end]

        # Partition the array segment around the pivot value
        for j in range(start, end):
            if qPArr[j] <= pv:
                i = i + 1
                qPArr[i], qPArr[j] = qPArr[j], qPArr[i]
        qPArr[i + 1], qPArr[end] = qPArr[end], qPArr[i + 1]
        pv = i + 1

        # Determine which subarray to process next based on pivot position
        if abs(start - pv) < abs(pv - end):
            if pv + 1 < end:
                top = top + 1
                stack[top] = pv + 1
                top = top + 1
                stack[top] = end
            if pv - 1 > start:
                top = top + 1
                stack[top] = start
                top = top + 1
                stack[top] = pv - 1
        else:
            if pv - 1 > start:
                top = top + 1
                stack[top] = start
                top = top + 1
                stack[top] = pv - 1
            if pv + 1 < end:
                top = top + 1
                stack[top] = pv + 1
                top = top + 1
                stack[top] = end

def gq(arrGQ):
    # Perform QuickSort using pivot selection logic
    quicksort_Spivot(arrGQ, 0, len(arrGQ) - 1)
    print("---GQ---")

    # Find the maximum length of the decreasing subsequence
    for x in range(len(arrGQ) - 1, -1, -1):
        if len(arrGQ) - x == arrGQ[x]:
            print(arrGQ[x])
            break
        elif len(arrGQ) - x >= arrGQ[x]:
            print(len(arrGQ) - x - 1)
            break
        elif x == 0:
            print(len(arrGQ))

def main():
    print("Enter value Of n:")
    n = int(input().strip())

    print("Enter array in format [1, 2, 3]:")
    arr = list(map(int, input().strip()[1:-1].strip().split(',')))

    if len(arr) == n:
        gq(arr)
    else:
        print("Error: N and array size don't match. Exiting Program.")

    main()
