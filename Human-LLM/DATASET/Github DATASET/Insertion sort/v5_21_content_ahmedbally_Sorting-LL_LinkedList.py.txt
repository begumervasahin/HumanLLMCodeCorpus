from Node import Node

class LinkedList:
    def __init__(self):
        self.head = None
        self.tail = None
        self.size = 0

    def is_empty(self):
        """Check if the linked list is empty."""
        return self.head is None

    def append(self, item):
        """Add a new node with the given item at the beginning of the linked list."""
        new_node = Node(item)
        new_node.next = self.head
        if self.head is None:
            self.tail = new_node
        self.head = new_node
        self.size += 1

    def append_sorted(self, item):
        """Add a new node with the given item in sorted order."""
        new_node = Node(item)
        new_node.next = None
        current = self.tail
        previous = None
        while current is not None:
            if current.data > item:
                break
            else:
                previous = current
                current = current.next
        if previous is None:
            new_node.next = self.tail
            self.tail = new_node
        else:
            new_node.next = current
            previous.next = new_node

    def length(self):
        """Return the number of nodes in the linked list."""
        current = self.head
        count = 0
        while current is not None:
            count += 1
            current = current.next
        return count

    def search(self, item):
        """Search for the given item in the linked list."""
        current = self.head
        while current is not None:
            if current.data == item:
                return True
            current = current.next
        return False

    def remove(self, item):
        """Remove the node with the given item from the linked list."""
        current = self.head
        previous = None
        found = False
        while not found:
            if current.data == item:
                found = True
            else:
                previous = current
                current = current.next
        if previous is None:
            self.head = current.next
        else:
            previous.next = current.next
        self.size -= 1
