import statistics

def pivot_position(arr, first, last):
    """
    Finds the pivot position for the quicksort algorithm using the median-of-three method.

    Args:
        arr (list): The list of integers to be sorted.
        first (int): The starting index of the sublist.
        last (int): The ending index of the sublist.

    Returns:
        int: The pivot position.
    """
    low = arr[first]
    high = arr[last]
    mid = (first + last) // 2
    pivot_val = statistics.median([low, arr[mid], high])

    if pivot_val == low:
        pivot_index = first
    elif pivot_val == high:
        pivot_index = last
    else:
        pivot_index = mid

    # Swap pivot with last element
    arr[last], arr[pivot_index] = arr[pivot_index], arr[last]
    pivot = arr[last]
    left = first
    right = last - 1

    while True:
        while left <= right and arr[left] <= pivot:
            left += 1
        while left <= right and arr[right] >= pivot:
            right -= 1
        if right < left:
            break
        else:
            arr[left], arr[right] = arr[right], arr[left]

    # Move pivot to its final position
    arr[first], arr[right] = arr[right], arr[first]
    return right

def quicksort(arr, first, last):
    """
    Sorts a list of integers using the quicksort algorithm.

    Args:
        arr (list): The list of integers to be sorted.
        first (int): The starting index of the sublist.
        last (int): The ending index of the sublist.
    """
    if first < last:
        # Partition and recursively sort sublists
        p = pivot_position(arr, first, last)
        quicksort(arr, first, p - 1)
        quicksort(arr, p + 1, last)
