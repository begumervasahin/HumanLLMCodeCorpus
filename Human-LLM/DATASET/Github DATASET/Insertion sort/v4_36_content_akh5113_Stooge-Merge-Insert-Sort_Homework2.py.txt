# Read data from file
nums = []
with open("data.txt", "r") as file:
    for line in file:
        # Split each line by spaces and convert elements to integers
        var = line.split(" ")
        nums.append(list(map(int, var[1:])))

print("Stooge Sort")

# Get the number of arrays
array_size = len(nums)

# Print unsorted arrays
print("The unsorted arrays are: ")
for array in nums:
    print(array)

# Function to perform Stooge Sort
def stooge_sort(array, start, end):
    num_elms = end - start + 1
    if num_elms == 2:
        # Swap elements if necessary
        if array[start] > array[end]:
            array[start], array[end] = array[end], array[start]
    elif num_elms > 2:
        # Divide the array into three parts
        third = num_elms // 3
        # Recursively sort the first 2/3 elements
        stooge_sort(array, start, end - third)
        # Recursively sort the last 2/3 elements
        stooge_sort(array, start + third, end)
        # Recursively sort the first 2/3 elements again
        stooge_sort(array, start, end - third)

# Sort arrays using Stooge Sort
for array in nums:
    stooge_sort(array, 0, len(array) - 1)

# Print sorted arrays
print("The sorted arrays are: ")
for array in nums:
    print(array)

# Write sorted arrays to file
with open("stooge.out", "w") as file_out:
    for array in nums:
        # Write each array to a line in the file
        file_out.write(" ".join(map(str, array)) + "\n")
