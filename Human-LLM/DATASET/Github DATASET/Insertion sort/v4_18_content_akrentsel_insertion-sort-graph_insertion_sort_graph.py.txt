import matplotlib.pyplot as plt
import matplotlib.animation as anim
import random

def plot_cont(generator):
    # Get initial values
    y_values, red_index = next(generator)
    max_frames = len(y_values) ** 2  # Set maximum number of frames
    fig = plt.figure()
    ax = fig.add_subplot(1, 1, 1)

    def update(i):
        # Get next values
        y_values, red_index = next(generator)
        red_bar = [0] * len(y_values)
        red_bar[red_index] = y_values[red_index]
        x = range(len(y_values))
        ax.clear()
        # Plot bars
        ax.bar(x, y_values, width=1)
        ax.bar(x, red_bar, width=1, color="red")
        print('Iteration:', i)

    # Create animation
    animation = anim.FuncAnimation(fig, update, frames=max_frames, repeat=False)
    plt.show()

def insertion_sort_step_generator(lst):
    # Ensure list length is valid
    assert len(lst) > 1, "Length of list must be greater than 1 to sort"
    current = 1

    # Swap function
    def swap(pos1, pos2):
        nonlocal lst
        lst[pos1], lst[pos2] = lst[pos2], lst[pos1]

    # Sorting algorithm
    while current < len(lst):
        temp = current
        while current > 0 and lst[current] < lst[current - 1]:
            swap(current, current - 1)
            current -= 1
            yield lst, current  # Yield current state
        if current != temp:
            current = temp + 1
        else:
            current += 1
            yield lst, current  # Yield current state
    yield lst, len(lst) - 1  # Final state

number_of_elements = 25
random_list = [random.randrange(1, 200) for _ in range(number_of_elements)]
plot_cont(insertion_sort_step_generator(random_list))
