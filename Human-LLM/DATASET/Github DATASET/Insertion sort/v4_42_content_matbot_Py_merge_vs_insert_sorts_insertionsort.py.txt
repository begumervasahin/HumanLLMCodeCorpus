# Reads unsorted lists of integers from a file, named 'data.txt', in the current directory,
# sorts the lists and writes the sorted lists to a new file, named 'insertionsort.txt'.

def insertion_sort(array):
    """
    Sorts an array of integers using the insertion sort algorithm.

    Args:
        array (list): The unsorted array of integers.

    Returns:
        list: The sorted array.
    """
    for i in range(1, len(array)):
        index = i - 1
        value = array[i]
        while index >= 0 and array[index] > value:
            array[index + 1] = array[index]
            index -= 1
        array[index + 1] = value

if __name__ == "__main__":
    number_lists = []

    # Read unsorted lists of integers from the file
    with open("data.txt", "r") as input_file:
        for line in input_file:
            data_list = []
            data = line.split()
            data.pop(0)  # Remove the first element (list size)
            for number in data:
                data_list.append(int(number))
            number_lists.append(data_list)

    # Sort each list using insertion sort
    for x in number_lists:
        insertion_sort(x)

    # Write sorted lists to a new file
    with open("insertionsort.txt", "w+") as output_file:
        for x in number_lists:
            for y in x:
                output_file.write("%i " % y)
            output_file.write("\n")
