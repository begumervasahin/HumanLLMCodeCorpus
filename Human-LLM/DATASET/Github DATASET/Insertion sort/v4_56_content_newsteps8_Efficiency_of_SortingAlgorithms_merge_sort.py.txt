import time

# Global variable to count the number of comparisons
count = 0

def merge_sort(arr):
    """
    Sorts a list of integers using the merge sort algorithm.

    Args:
        arr (list): The list of integers to be sorted.
    """
    if len(arr) > 1:
        # Divide the array into two halves
        mid = len(arr) // 2
        left_half = arr[:mid]
        right_half = arr[mid:]

        # Recursively sort the left and right halves
        merge_sort(left_half)
        merge_sort(right_half)

        # Merge the sorted halves
        merge(arr, left_half, right_half)

def merge(arr, left_half, right_half):
    """
    Merges two sorted sublists into a single sorted list.

    Args:
        arr (list): The list to be merged into.
        left_half (list): The left sorted sublist.
        right_half (list): The right sorted sublist.
    """
    global count
    i = j = k = 0

    while i < len(left_half) and j < len(right_half):
        count += 1
        if left_half[i] < right_half[j]:
            arr[k] = left_half[i]
            i += 1
        else:
            arr[k] = right_half[j]
            j += 1
        k += 1

    while i < len(left_half):
        count += 1
        arr[k] = left_half[i]
        i += 1
        k += 1

    while j < len(right_half):
        count += 1
        arr[k] = right_half[j]
        j += 1
        k += 1

def print_list(arr):
    """
    Prints the elements of a list.

    Args:
        arr (list): The list to be printed.
    """
    for num in arr:
        print(num, end=" ")
    print()

if __name__ == '__main__':
    # Start time
    start_time = time.time()

    # Read input from file and convert to list of integers
    with open("case") as file:
        arr = list(map(int, file.read().split()))

    # Perform merge sort
    merge_sort(arr)

    # Print sorted array and total number of comparisons
    print("Sorted array:")
    print_list(arr)
    print("Total comparisons:", count)

    # End time
    end_time = time.time()
    print("Execution time:", end_time - start_time)
