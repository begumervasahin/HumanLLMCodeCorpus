import random

counter = 0

def merge_sort(lst):
    global counter
    if len(lst) <= 1:
        counter += 1
        return lst
    mid = len(lst) // 2
    left = merge_sort(lst[:mid])
    right = merge_sort(lst[mid:])
    return merge(left, right)

def merge(left, right):
    global counter
    merged = []
    i = 0
    j = 0
    while i < len(left) and j < len(right):
        if left[i] < right[j]:
            counter += 1
            merged.append(left[i])
            i += 1
        else:
            merged.append(right[j])
            j += 1
    counter += 1
    merged += left[i:]
    merged += right[j:]
    return merged

def test_sorting(words):
    global counter
    counter = 0
    sorted_words = merge_sort(words)
    return sorted_words, counter

def main():
    lines = [line.rstrip('\n') for line in open('words.txt')]
    
    sizes_to_test = [10, 30, 100, 300, 1000, 3000, 10000, len(lines)]
    
    for size in sizes_to_test:
        print('-------------')
        print(f'Testing on {size} words')
        words_to_test = lines[:size]
        
        sorted_words, comparisons = test_sorting(words_to_test)
        print(f'{size} words sorted')
        print('Sorted words:', sorted_words)
        print('Number of comparisons:', comparisons)
        
        reversed_words, comparisons = test_sorting(list(reversed(words_to_test)))
        print(f'{size} words reversed')
        print('Sorted words:', reversed_words)
        print('Number of comparisons:', comparisons)
        
        random.shuffle(words_to_test)
        shuffled_words, comparisons = test_sorting(words_to_test)
        print(f'{size} words random')
        print('Sorted words:', shuffled_words)
        print('Number of comparisons:', comparisons)
        print('-------------')
        print()

if __name__ == "__main__":
    main()
