def binary_search(arr, value, start, end):
    """
    Perform binary search to find the correct position to insert a value into a sorted array.

    Parameters:
    arr (list): The sorted array.
    value (int): The value to be inserted.
    start (int): The starting index of the search range.
    end (int): The ending index of the search range.

    Returns:
    int: The index where the value should be inserted.
    """
    if start == end:
        # If there's only one element left in the search range
        if arr[start] > value:
            return start  # Insert before this element
        else:
            return start + 1  # Insert after this element

    if start > end:
        return start  # Insert at this index

    mid = (start + end) // 2
    if arr[mid] < value:
        # Search in the right half
        return binary_search(arr, value, mid + 1, end)
    elif arr[mid] > value:
        # Search in the left half
        return binary_search(arr, value, start, mid - 1)
    else:
        # Element already exists at mid
        return mid

def insertion_sort(arr):
    """
    Perform insertion sort on an array.

    Parameters:
    arr (list): The unsorted array.

    Returns:
    list: The sorted array.
    """
    for i in range(1, len(arr)):
        value_to_insert = arr[i]
        # Find the correct position to insert value_to_insert using binary search
        insertion_index = binary_search(arr, value_to_insert, 0, i - 1)
        # Move elements to make space for value_to_insert at insertion_index
        arr = arr[:insertion_index] + [value_to_insert] + arr[insertion_index:i] + arr[i + 1:]

    return arr

    sorted_array = insertion_sort([37, 23, 0, 17, 12, 72, 31, 46, 100, 88, 54])
