import timeit
import random

# Function to copy elements from one array to another
def copy_array(src, src_pos, dest, dest_pos, length):
    for i in range(length):
        dest[i + dest_pos] = src[i + src_pos]

# Function to perform merge sort
def merge_sort(array):
    # Base case: if the array has 1 or fewer elements, it's already sorted
    if len(array) <= 1:
        return array

    # Split the array into two halves
    mid = len(array) // 2
    left = array[:mid]
    right = array[mid:]

    # Recursively sort each half
    merge_sort(left)
    merge_sort(right)

    # Merge the sorted halves
    merge(left, right, array)

    return array

# Function to merge two sorted arrays
def merge(left, right, array):
    i = j = k = 0

    # Merge the elements from the two arrays
    while i < len(left) and j < len(right):
        if left[i] < right[j]:
            array[k] = left[i]
            i += 1
        else:
            array[k] = right[j]
            j += 1
        k += 1

    # Copy any remaining elements from the left array
    copy_array(left, i, array, k, len(left) - i)

    # Copy any remaining elements from the right array
    copy_array(right, j, array, k, len(right) - j)

# Generate a list of random numbers
numbers = [7, 4, 8, 3, 1, 12, 9, 14]
for _ in range(5000):
    numbers.append(random.randint(1, 99999))

# Measure the time taken to perform merge sort
start_time = timeit.default_timer()
merge_sort(numbers)
elapsed_time = timeit.default_timer() - start_time

# Print the elapsed time and the sorted list
print("Merge Sort: " + str(elapsed_time))
print(numbers)
