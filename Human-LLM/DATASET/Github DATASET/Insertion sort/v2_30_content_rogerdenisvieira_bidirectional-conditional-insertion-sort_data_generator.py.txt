# First off, let's bring in the tools we need: numpy for the number crunching, and csv for the file handling.
import numpy as np
import csv

# Setting the stage with some constants - we're looking at 7000 test samples.
TEST_SIZE = 7000
DATA_FILENAME = 'distributions.csv'  # The name of our output file.

def write_data(values):
    """Given a list of distribution values, this writes them neatly into a CSV file."""
    # Our data categories
    fieldnames = ['uniform', 'binomial', 'poisson', 'normal']
    
    # Let's start writing!
    print('Writing data...')
    with open(DATA_FILENAME, 'w', newline='') as output_file:
        csv_writer = csv.DictWriter(output_file, fieldnames=fieldnames)
        csv_writer.writeheader()
        csv_writer.writerows(values)
    
    # A quick message to let us know all went well.
    print('Data written successfully.')

def generate_distributions(size):
    """Generates a random sample for several distributions, given a sample size."""
    # Rolling the dice for each distribution
    uniform_dist = np.random.uniform(-1, 1, size)
    binomial_dist = np.random.binomial(1000, 0.5, size)
    poisson_dist = np.random.poisson(1000, size)
    normal_dist = np.random.normal(0, 0.1, size)
    
    # Time to bundle up our data into a neat list of dictionaries
    distributions_list = []
    for i in range(size):
        distributions_list.append({
            'uniform': uniform_dist[i],
            'binomial': binomial_dist[i],
            'poisson': poisson_dist[i],
            'normal': normal_dist[i]
        })
    
    return distributions_list

def main():
    """The heart of our script - where everything comes together."""
    # Generating the data...
    distributions_list = generate_distributions(TEST_SIZE)
    # ...and writing it down.
    write_data(distributions_list)

# The script's starting point.
if __name__ == "__main__":
    main()
