import common as c  # Assuming the 'common' module contains the 'minimum' and 'maximum' functions

def radix_sort(arr, verbose=0, desc=0):
    # Calculate the number of passes needed for Radix Sort
    def passes(number, base):
        count = 0
        while number:
            count += 1
            number //= base
        return count

    # Extract a positional digit from a number
    def positional_digit(number, position, base=10):
        count = 0
        while True:
            count += 1
            remainder = number % base
            number //= base
            if count > position:
                break
        return remainder

    # Distribute numbers into buckets based on a positional digit
    def bucketize(position, base):
        n = len(arr)
        buckets = [[] for _ in range(n)]
        for i in range(n):
            digit = positional_digit(arr[i], position, base)
            buckets[digit].append(arr[i])
            if verbose == 2:
                print("    Bucket:", i, " :: ", buckets)
        return [buckets[x][y] for x in range(len(buckets)) for y in range(len(buckets[x]))]

    base = 10  # Base for numbers (e.g., 10 for decimal)

    # Adjust array to make it non-negative
    smallest = c.minimum(arr)[0]
    if smallest:
        arr = [x - smallest for x in arr]

    # Find the largest number and calculate the number of passes needed
    largest = c.maximum(arr)[0]
    iter_count = passes(largest, base)

    # Perform Radix Sort
    for i in range(iter_count):
        arr = bucketize(i, base)
        if verbose:
            print("Iteration", i + 1, ":", arr)

    # Adjust array back to its original form
    if smallest:
        arr = [x + smallest for x in arr]

    # Optionally reverse the sorted array
    if desc:
        arr = arr[::-1]

    return arr

# Example usage:
if __name__ == "__main__":
    # Example array
    array = [170, 45, 75, 90, 802, 24, 2, 66]
    print("Original Array:", array)

    # Perform Radix Sort
    sorted_array = radix_sort(array, verbose=1, desc=0)

    # Output sorted array
    print("Sorted Array:", sorted_array)
