def print_string_properties(input_string):
    # Convert the string to a list of characters
    char_list = list(input_string)

    # Select characters with even indexes
    even_char_list = [char for index, char in enumerate(char_list) if index % 2 == 0]   
    even_string = "".join(even_char_list)

    # Print the input string and characters with even indexes
    print("The input string is:", input_string)
    print("The characters that have even indexes are:", even_string)

    # Print the string in reverse order
    print("The string in reverse order is:", input_string[::-1])

    # Check if the string is a palindrome
    if input_string == input_string[::-1]:
        print("The string is a palindrome: True")
    else:
        print("The string is a palindrome: False")

def print_set(start, end):
    num_list = []
    # Generate numbers between start and end-1
    for i in range(start + 1, end):
        # Check if the number is divisible by 7 but not by 5
        if i % 7 == 0 and i % 5 != 0:
            num_list.append(i)

    # Convert the list to a set
    result_set = set(num_list)
    print(result_set)

# Test the functions
print_string_properties("11411")
print_set(11, 21)
