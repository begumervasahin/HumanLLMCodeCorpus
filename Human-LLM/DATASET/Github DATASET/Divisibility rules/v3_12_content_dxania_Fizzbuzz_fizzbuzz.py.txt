def get_input():
    # Get input for the first list from the user
    list1 = input_list("first")

    # Get input for the second list from the user
    list2 = input_list("second")

    # Calculate and print Fizzbuzz result
    print_fizzbuzz_result(list1, list2)

def input_list(list_name):
    """Prompt the user to enter a list of elements separated by commas."""
    input_str = input(f"Enter the {list_name} list (separate elements with commas): ")
    elements = [element.strip() for element in input_str.split(",")]
    print(f"{list_name.capitalize()} list:", elements)
    print(f"{list_name.capitalize()} list is {len(elements)} elements long")
    return elements

def print_fizzbuzz_result(list1, list2):
    """Calculate and print the FizzBuzz result based on the total length of both lists."""
    total_length = len(list1) + len(list2)
    result = fizzbuzz(total_length)
    print("FizzBuzz result:", result)

def fizzbuzz(n):
    """Return 'Fizz' if n is divisible by 3, 'Buzz' if divisible by 5, 'FizzBuzz' if divisible by both, otherwise return n."""
    if n % 3 == 0 and n % 5 == 0:
        return 'FizzBuzz'
    elif n % 3 == 0:
        return 'Fizz'
    elif n % 5 == 0:
        return 'Buzz'
    else:
        return n

if __name__ == "__main__":
