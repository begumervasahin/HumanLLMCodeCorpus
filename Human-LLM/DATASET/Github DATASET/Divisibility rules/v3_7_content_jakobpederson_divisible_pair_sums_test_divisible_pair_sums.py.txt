import unittest
from divisible_pair_sums import DivisiblePairSums
from collections import namedtuple

# Define a namedtuple for test data
Data = namedtuple('Data', ['n', 'k', 'ar'])

# Define test data instances
TEST_DATA = [
    Data(n=6, k=3, ar=[1, 3, 2, 6, 1, 2]),
    Data(n=10, k=3, ar=[29, 97, 52, 86, 27, 89, 77, 19, 99, 96]),
    Data(n=100, k=22, ar=[
        43, 95, 51, 55, 40, 86, 65, 81, 51, 20, 47, 50, 65, 53, 23, 78,
        75, 75, 47, 73, 25, 27, 14, 8, 26, 58, 95, 28, 3, 23, 48, 69,
        26, 3, 73, 52, 34, 7, 40, 33, 56, 98, 71, 29, 70, 71, 28, 12,
        18, 49, 19, 25, 2, 18, 15, 41, 51, 42, 46, 19, 98, 56, 54, 98,
        72, 25, 16, 49, 34, 99, 48, 93, 64, 44, 50, 91, 44, 17, 63, 27,
        3, 65, 75, 19, 68, 30, 43, 37, 72, 54, 82, 92, 37, 52, 72, 62,
        3, 88, 82, 71
    ])
]

class DivisiblePairSumsTest(unittest.TestCase):
    def test_answer_question(self):
        for data in TEST_DATA:
            # Instantiate DivisiblePairSums objects with test data instances
            divisible_pair_sums = DivisiblePairSums(data.n, data.k, data.ar)

            # Assert expected results for answer_question method
            expected_result = self.get_expected_result(data)
            self.assertEqual(expected_result, divisible_pair_sums.answer_question())

    def get_expected_result(self, data):
        # Calculate expected result manually for each test case
        if data.n == 6 and data.k == 3:
            return 5
        elif data.n == 10 and data.k == 3:
            return 15
        elif data.n == 100 and data.k == 22:
            return 216
        else:
            return "Error: unknown test case"

    def test_validate_input(self):
        # Test validate_input method with invalid input data
        invalid_inputs = [
            (0, 3, [1, 2, 3]),
            (101, 3, [1, 2, 3]),
            (2, 0, [1, 2, 3]),
            (2, 101, [1, 2, 3]),
            (2, 1, []),
            (2, 1, [x for x in range(0, 102)])
        ]

        for n, k, ar in invalid_inputs:
            divisible_pair_sums = DivisiblePairSums(n, k, ar)
            self.assertEqual("Error: invalid input", divisible_pair_sums.answer_question())

if __name__ == '__main__':
    unittest.main()
