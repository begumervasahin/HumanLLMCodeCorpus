def load(filename="input.txt"):
    # Open the input file and read its contents
    with open(filename, 'r') as file:
        lines = file.readlines()

    # Extract values for fizz, buzz, and end from the first line of the file
    fizz, buzz, end = map(int, lines[0].split())

    # Loop through the range from 1 to end
    for i in range(1, end + 1):
        # Check divisibility by fizz and buzz
        if i % fizz == 0 and i % buzz == 0:
            print("FB")
        # Check divisibility by fizz only
        elif i % fizz == 0:
            print("F")
        # Check divisibility by buzz only
        elif i % buzz == 0:
            print("B")
        # If not divisible by either, print the number itself
        else:
            print(i)

# Load and process the input file, assuming 'input.txt' exists in the same directory
load()
