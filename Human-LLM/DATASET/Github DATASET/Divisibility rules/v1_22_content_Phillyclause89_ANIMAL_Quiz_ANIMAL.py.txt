def number_checker():
    print("Enter a number or 'q' to quit:")
    user_input = input().strip().lower()  # Remove leading/trailing whitespaces and convert to lowercase

    if user_input in ('q', 'quit'):
        print("Goodbye!")
    else:
        try:
            n = int(user_input)
            if n % 32 == 0:
                print("ANIMAL")
            elif n % 16 == 0:
                print("ANIM")
            elif n % 8 == 0:
                print("AN")
            else:
                print("Try a different number next time.")
            number_checker()
        except ValueError:
            print("Invalid input. Please enter a valid number or 'q' to quit.")
            number_checker()

# Start the number checker
number_checker()
