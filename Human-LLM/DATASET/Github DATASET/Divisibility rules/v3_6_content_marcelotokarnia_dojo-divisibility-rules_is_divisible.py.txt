# Define a generator function to generate the alternating pattern for the divisibility rule
def generate_divisibility_rule(m):
    index = 0
    while True:
        yield (-1) ** index  # Yield alternating 1 and -1
        index += 1

# Define a function to check if a number is divisible by another number using a specific divisibility rule
def is_divisible_by_rule(number, divisor):
    # Create an instance of the divisibility rule generator
    divisibility_rule = generate_divisibility_rule(divisor)

    # Initialize the total sum to track the intermediate result
    running_total = 0

    # Convert the number to a string and reverse it for processing digit by digit
    number_str_reversed = str(number)[::-1]

    # Iterate through each digit of the reversed number
    for digit in number_str_reversed:
        # Get the next multiplier from the divisibility rule generator
        multiplier = next(divisibility_rule)

        # Multiply the digit by the multiplier and add it to the running total
        running_total += int(digit) * multiplier

        # If the absolute value of the running total exceeds 3 times the divisor, reduce it to a manageable range
        if abs(running_total) > 3 * divisor:
            running_total %= 3 * divisor

    # Check if the final running total is divisible by the divisor and return True or False accordingly
    return running_total % divisor == 0

# Example usage:
number_to_check = 123456  # The number to check for divisibility
divisor_to_check = 7      # The divisor to check against
result = is_divisible_by_rule(number_to_check, divisor_to_check)  # Check if number is divisible by divisor using the defined rule
print(result)  # Output: True or False depending on whether number is divisible by divisor
