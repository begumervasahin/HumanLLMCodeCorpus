import sys
import datetime
import itertools

def parse_date(year, month, day):
    """
    Parses the date and returns a datetime.date object if valid, otherwise returns False.
    """
    try:
        # Convert 2-digit years to full 4-digit years
        year = int(year) + 2000 if int(year) < 1000 else int(year)
        return datetime.date(int(year), int(month), int(day))
    except ValueError:
        return False

def main():
    """
    Main function to read dates from a file and print the earliest valid date.
    """
    try:
        filename = sys.argv[1]
    except IndexError:
        print("Missing file name!")
        exit()

    # Open the file and process each line
    with open(filename) as file:
        for line in [l.rstrip('\n') for l in file]:
            parts = line.split('/')  # Split the line into date parts
            dates = []

            # Generate all permutations of date parts
            for (year, month, day) in itertools.permutations(parts, 3):
                date = parse_date(year, month, day)
                if date and 2000 <= date.year < 3000:
                    dates.append(date)

            # Print the earliest valid date if found, otherwise print the line with an error message
            if dates:
                print(min(dates))
            else:
                print(f"{line} is illegal")

if __name__ == '__main__':
    main()
