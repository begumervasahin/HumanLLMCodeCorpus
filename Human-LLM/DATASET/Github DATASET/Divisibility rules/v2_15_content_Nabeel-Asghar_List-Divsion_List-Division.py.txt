def division_function(list_a, list_b, threshold):
    """
    Finds numbers from list_a that are divisible by at least 'threshold' numbers from list_b.

    Args:
        list_a (list): List of numbers.
        list_b (list): List of numbers to divide by.
        threshold (int): Minimum count of divisors required for a number in list_a to be included in the result.

    Returns:
        set: Set of numbers from list_a that meet the criteria.
    """
    result = []
    for num_a in list_a:
        count = 0
        for num_b in list_b:
            if num_a % num_b == 0:
                count += 1
        if count >= threshold:
            result.append(num_a)
    return set(result)

def main():
    # Input lists from the user
    input_list_a = input("Enter a list of numbers separated by a space: ")
    list_a = list(map(int, input_list_a.split()))

    input_list_b = input("Enter another list of numbers separated by a space: ")
    list_b = list(map(int, input_list_b.split()))

    # Calculate the threshold as half the length of list_b
    threshold = len(list_b) // 2

    # Call the division function and print the result
    result = division_function(list_a, list_b, threshold)
    print(result)

    # Wait for user input before exiting
    input("Press Enter to exit.")

if __name__ == "__main__":
    main()
