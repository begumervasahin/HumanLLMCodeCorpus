# Define a function to check divisibility by 2, 3, 5, or 7 and provide results
def check_divisibility(number):
    results = ""  # Initialize an empty string to store results

    # Check divisibility by 2
    if number % 2 == 0:
        results += "The number is divisible by 2.\n"

    # Check divisibility by 3
    if number % 3 == 0:
        results += "The number is divisible by 3.\n"

    # Check divisibility by 5
    if number % 5 == 0:
        results += "The number is divisible by 5.\n"

    # Check divisibility by 7
    if number % 7 == 0:
        results += "The number is divisible by 7.\n"

    # If none of the conditions are met, add a message indicating not divisible by 2, 3, 5, or 7
    if not results:
        results += "The number is not divisible by 2, 3, 5, or 7.\n"

    return results

# Get input from the user
number = int(input("Please enter a number of your choice:\n"))

# Call the function to check divisibility and print the results
print(check_divisibility(number))
