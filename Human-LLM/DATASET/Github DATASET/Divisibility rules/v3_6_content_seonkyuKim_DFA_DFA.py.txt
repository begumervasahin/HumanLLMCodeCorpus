def dfa_accepts_binary_string_ending_with_00(input_string):
    """
    DFA to accept binary strings ending with '00'
    """
    # Define states of the DFA
    states = {'q0', 'q1', 'q2', 'q3', 'q4'}

    # Define transition function for the DFA
    transitions = {
        ('q0', '0'): 'q0',
        ('q0', '1'): 'q1',
        ('q1', '0'): 'q2',
        ('q1', '1'): 'q3',
        ('q2', '0'): 'q4',
        ('q2', '1'): 'q0',
        ('q3', '0'): 'q1',
        ('q3', '1'): 'q2',
        ('q4', '0'): 'q3',
        ('q4', '1'): 'q4'
    }

    # Define initial and final states of the DFA
    initial_state = 'q0'
    final_state = 'q0'

    # Initialize current state to the initial state
    current_state = initial_state

    # Process each character in the input string
    for char in input_string:
        # Get the next state based on the current state and input character
        current_state = transitions.get((current_state, char), 'invalid')

        # If the transition is invalid, the string is not accepted
        if current_state == 'invalid':
            return False

    # Check if the final state is the accepting state
    return current_state == final_state

def is_binary_string(input_string):
    # Check if the input string consists only of '0' and '1'
    return set(input_string).issubset({'0', '1'})

if __name__ == "__main__":
    description = "Input a binary string. Enter 'exit' to quit.\n"

    while True:
        input_string = input(description)

        if input_string.lower() == 'exit':
            break

        if not is_binary_string(input_string):
            print('Error: Input value is not a binary string.\n')
            continue

        is_accepted = dfa_accepts_binary_string_ending_with_00(input_string)

        if is_accepted:
            print('Accepted\n')
        else:
            print('Not Accepted\n')
