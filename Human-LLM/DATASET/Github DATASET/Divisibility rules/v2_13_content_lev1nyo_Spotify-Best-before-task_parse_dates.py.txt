import sys
import datetime
import itertools

def parse_date(year, month, day):
    try:
        # Convert 2-digit years to full 4-digit years
        year = int(year) + 2000 if int(year) < 1000 else int(year)
        return datetime.date(int(year), int(month), int(day))
    except ValueError:
        return False

def main():
    try:
        # Get the filename from command line arguments
        filename = sys.argv[1]
    except IndexError:
        print("Missing file name!")
        return

    # Open the file and process each line
    with open(filename) as file:
        for line in file:
            # Remove newline character from the end of the line
            line = line.rstrip('\n')
            # Split the line into parts using '/'
            parts = line.split('/')
            dates = []
            # Generate all permutations of date parts
            for (year, month, day) in itertools.permutations(parts, 3):
                # Try to parse the date and check if it's valid
                date = parse_date(year, month, day)
                if date and 2000 <= date.year < 3000:
                    dates.append(date)
            # If valid dates are found, print the earliest one
            if dates:
                print(min(dates))
            else:
                print(f"{line} is illegal")

if __name__ == '__main__':
    main()
