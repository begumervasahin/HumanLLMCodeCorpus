# Import the divisibility rule generator from a module named divisibility_rule
from divisibility_rule import div_rule

# Define a function to check if a number is divisible by another number using a specific divisibility rule
def is_divisible_by_rule(n, m):
    # Create an instance of the divisibility rule generator using the divisor m
    divisibility_rule = div_rule(m)

    # Initialize the sum variable to keep track of the intermediate result
    total_sum = 0

    # Convert the number n to a string and reverse it for processing digit by digit
    n = n[::-1]

    # Iterate through each digit of the reversed number n
    for digit in n:
        # Get the next multiplier from the divisibility rule generator
        multiplier = next(divisibility_rule)

        # Multiply the digit by the multiplier and add it to the total sum
        total_sum += int(digit) * multiplier

        # Check if the absolute value of the total sum exceeds 3 times the divisor m
        if abs(total_sum) > 3 * m:
            # Adjust the total sum to a manageable range
            total_sum %= 3 * m

    # Check if the final total sum is divisible by the divisor m and return True or False accordingly
    return total_sum % m == 0

# Example usage:
number_to_check = 123456  # The number to check for divisibility
divisor_to_check = 7      # The divisor to check against
result = is_divisible_by_rule(number_to_check, divisor_to_check)  # Check if number is divisible by divisor using the defined rule
print(result)  # Output: True or False depending on whether number is divisible by divisor
