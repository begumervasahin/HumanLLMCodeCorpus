def fizz_buzz(number):
    '''
    This function returns 'Fizz', 'Buzz', 'FizzBuzz', or the number itself based on the input.
    '''
    # Check if the number is divisible by both 3 and 5
    if number % 5 == 0 and number % 3 == 0:
        return 'FizzBuzz'
    # Check if the number is divisible by 3 only
    elif number % 3 == 0:
        return 'Fizz'
    # Check if the number is divisible by 5 only
    elif number % 5 == 0:
        return 'Buzz'
    # If none of the above conditions are met, return the number itself
    else:
        return number

if __name__ == "__main__":
    try:
        # Prompt the user to enter a number
        num = int(input("Enter a number: "))
        # Get the result from the fizz_buzz function
        result = fizz_buzz(num)
        # Print the result
        print(result)
    except ValueError:
        # Handle invalid input
        print("Invalid input. Please enter an integer.")
