def check_divisibility_by_seven(number):
    # Convert the number to a string for manipulation
    number_str = str(number)

    # Iterate until the number becomes a two-digit number or reaches zero
    while len(number_str) != 2 and number != 0:
        # Print the current number
        print(f"Current number: {number_str}")

        # Extract the prefix of the number
        prefix = number_str[:-1]

        # Extract the last digit of the number
        last_digit = int(number_str[-1])

        # Update the number by subtracting twice the last digit from the prefix
        number_str = str(int(prefix) - (2 * last_digit))

        # Update the number
        number = int(number_str)

    # Check if the resulting number is divisible by 7
    if number % 7 == 0:
        print(f"Number {number_str} is divisible by 7!")
    else:
        print(f"Number {number_str} is not divisible by 7.")

# Initialize the number to check
number_to_check = 1946

# Call the function to check divisibility by seven
check_divisibility_by_seven(number_to_check)
