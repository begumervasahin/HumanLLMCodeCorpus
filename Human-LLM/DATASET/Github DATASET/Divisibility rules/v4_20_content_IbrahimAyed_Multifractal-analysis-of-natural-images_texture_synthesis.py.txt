import numpy as np
import matplotlib.pyplot as plt

# Define functions f1 to f7
def f1(x, y):
    '''Defines a circular region with radius 0.5.'''
    return ((x ** 2 + y ** 2) < 0.25)

def f2(x, y):
    '''Defines a square region with side length 1.'''
    return (np.abs(x) < 0.5) * (np.abs(y) < 0.5)

def f3(x, y):
    '''Defines a cosine function with elliptical contours.'''
    a = 1.0
    b = 10.0
    M = np.sqrt((x / a) ** 2 + (y / b) ** 2)
    res = np.zeros_like(M)
    select = M < 0.5
    res[select] = np.cos(np.pi * M[select])
    return res

def f4(x, y):
    '''Defines a cosine squared function with elliptical contours.'''
    a = 1.0
    b = 1.0
    M = np.sqrt((x / a) ** 2 + (y / b) ** 2)
    res = np.zeros_like(M)
    select = M < 0.5
    res[select] = np.cos(np.pi * M[select]) ** 2
    return res

def f5(x, y):
    '''Defines a cosine squared function with exponential scaling along M.'''
    a = 1.0
    b = 1.0
    M = np.sqrt((x / a) ** 2 + (y / b) ** 2)
    res = np.zeros_like(M)
    select = M < 0.5
    res[select] = np.cos(np.pi * np.exp(M[select])) ** 2
    return res

def f6(x, y):
    '''Defines a cosine squared function with exponential double scaling along M.'''
    a = 10.0
    b = 1.0
    M = np.sqrt((x / a) ** 2 + (y / b) ** 2)
    res = np.zeros_like(M)
    select = M < 0.5
    res[select] = np.cos(np.pi * np.exp(np.exp(M[select]))) ** 2
    return res

def f7(x, y):
    '''Defines an exponential decay function with elliptical contours.'''
    a = 1.0
    b = 1.0
    M = np.sqrt((x / a) ** 2 + (y / b) ** 2)
    res = np.zeros_like(M)
    select = M < 0.5
    res[select] = np.exp(-1 / (1 - (2 * M[select]) ** 4))
    return res

def generate_intensity_map(f):
    '''Generates an intensity map using the given function.'''
    n = 256
    r_min = 1.0 / n
    X_max = 1.0
    Y_max = 1.0
    u_max = (1.0 / (r_min ** 2) - 1)
    lbda = (2.0 / np.pi) * u_max * (X_max + 1) * (Y_max + 1)

    # Generate random parameters
    N = np.random.poisson(lbda)
    ui = u_max * np.random.uniform(size=N)
    ri = 1.0 / np.sqrt(u_max - ui)
    xi = -0.5 + np.random.uniform(size=N) * (X_max + 1)
    yi = -0.5 + np.random.uniform(size=N) * (Y_max + 1)
    sigma = np.sqrt(0.08)
    mu = -0.5 * 0.008
    log_wi = np.random.randn(N) * sigma + mu

    # Generate the grid for plotting
    x = np.linspace(0, X_max, n)
    y = np.linspace(0, Y_max, n)
    X, Y = np.meshgrid(x, y)

    # Initialize the intensity map
    log_Q = np.zeros((n, n))

    # Generate the intensity map
    for i in range(N):
        tmp = f((X - xi[i]) / ri[i], (Y - yi[i]) / ri[i])
        slct = (tmp != 0)
        log_Q[slct] += tmp[slct] * log_wi[i]

    # Normalize the intensity map
    C = r_min ** (np.exp(mu + 0.5 * sigma ** 2) - 1)
    Q = np.exp(log_Q)
    return Q / C

def fourier_domain_integration(Qr, alpha):
    '''Performs Fourier domain integration on the given intensity map.'''
    n = Qr.shape[0]
    x = np.concatenate((np.arange(0, 1 + n / 2), np.arange(-n / 2 + 1, 0)))
    U, V = np.meshgrid(x, x)
    S = U ** 2 + V ** 2
    S[0, 0] = 1.0
    return np.real(np.fft.ifft2(np.fft.fft2(Qr) / (S ** alpha)))

if __name__ == "__main__":
    # Generate the intensity map and perform integration
    intensity_map = generate_intensity_map(f1)
    integrated_map = fourier_domain_integration(intensity_map, 0.05)

    # Display the result
    plt.figure()
    plt.imshow(integrated_map, cmap='gray')
    plt.show()
