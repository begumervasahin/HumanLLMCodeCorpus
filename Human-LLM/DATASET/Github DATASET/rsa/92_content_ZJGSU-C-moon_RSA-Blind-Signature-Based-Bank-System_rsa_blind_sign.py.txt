import hashlib
import gmpy2
import random
import time
def getprime(rs):
    p = gmpy2.mpz_urandomb(rs, 1024)
    while not gmpy2.is_prime(p):
        p = p + 1
    return p
def keygen():
    rs = gmpy2.random_state(int(time.time()))
    p = getprime(rs)
    q = getprime(rs)
    n = p * q
    e = 0x10001
    H = lambda m: int(hashlib.sha256(m).hexdigest(), 16)
    pk = [n, e, H]
    d = gmpy2.invert(e, (p - 1) * (q - 1))
    sk = d
    return pk, sk
def sign(pk, sk, m):
    n = pk[0]
    e = pk[1]
    H = pk[2]
    d = sk
    R = random.randrange(0, n)
    H_m = H(m) % n
    print 'Sign H(m) =', H_m
    R_e = gmpy2.powmod(R, e, n)
    M = R_e * H_m % n
    M_d = gmpy2.powmod(M, d, n)
    R_ = gmpy2.invert(R, n)
    sigma = gmpy2.mul(M_d, R_) % n
    return sigma
def verify(pk, sigma):
    n = pk[0]
    e = pk[1]
    print 'Verify H(m) =', gmpy2.powmod(sigma, e, n)
if __name__ == '__main__':
    m = raw_input('Please input your message:')
    pk, sk = keygen()
    sigma = sign(pk, sk, m)
    print 'Sigma =', sigma
    verify(pk, sigma)
