def read_estimated_data(file_path):
    """Reads 'g' and 'b' values from the estimated data file."""
    with open(file_path) as file:
        lines = file.readlines()
        g = float(lines[-2].strip().split("=")[1])
        b = float(lines[-1].strip().split("=")[1])
    return g, b

def estimate_real(g, b, x):
    """Estimates real time based on 'g', 'b', and bit size 'x'."""
    return b * (g ** x)

def estimate_fastest(x):
    """Estimates fastest time for given bit size 'x'."""
    return (210 * (x ** 2) + 1060 * x - 1410) * (10 ** -9)

def estimate_slowest(x):
    """Estimates slowest time for given bit size 'x'."""
    return (2650 * (x ** 3) + 1060 * x - 1410) * (10 ** -9)

def main():

    HIGHEST_BIT_SIZE = 2048

    # Read 'g' and 'b' values from the estimated data file
    g, b = read_estimated_data(RESULT_FILE_PATH)

    # Calculate time estimates for different bit sizes
