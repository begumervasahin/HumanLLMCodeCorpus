import math
import time

def main():
    # Prompt user to enter the upper limit for checking prime numbers
    upper_limit = int(input("Please enter a number to check prime numbers up to: "))
    
    # Record the start time of the prime number generation process
    start_time = time.time()
    
    # Load previously found prime numbers from the file 'output.txt'
    primes = load_primes_from_file("output.txt")
    
    # Initialize variables for tracking the last prime number found
    last_prime = primes[-1] if primes else 2
    
    # Generate new prime numbers up to the specified upper limit
    for num in range(last_prime + 1, upper_limit + 1):
        if is_prime(num, primes):
            primes.append(num)
    
    # Write the updated list of prime numbers back to the file 'output.txt'
    save_primes_to_file("output.txt", primes)
    
    # Calculate and display the time taken for the prime number generation process
    end_time = time.time()
    print(f"Prime number generation completed in {end_time - start_time:.2f} seconds.")

def load_primes_from_file(filename):
    """Reads prime numbers from a file and returns them as a list of integers."""
    try:
        with open(filename, 'r') as file:
            return [int(line.strip()) for line in file]
    except FileNotFoundError:
        return []  # Return an empty list if the file does not exist

def save_primes_to_file(filename, primes):
    """Writes the list of prime numbers to a file, one per line."""
    with open(filename, 'w') as file:
        for prime in primes:
            file.write(f"{prime}\n")

def is_prime(number, primes):
    """Checks if a number is prime using a list of known primes."""
    for prime in primes:
        if prime > math.sqrt(number):
            break
        if number % prime == 0:
            return False
    return True

if __name__ == "__main__":
    main()
