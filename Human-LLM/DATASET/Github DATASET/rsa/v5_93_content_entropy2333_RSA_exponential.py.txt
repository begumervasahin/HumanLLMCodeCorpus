def exp_mod_montgomery(base, exponent, modulus):
    # Convert exponent to binary and reverse it
    binary_exponent = bin(exponent)[2:][::-1]
    # Calculate the length of the binary exponent
    r = len(binary_exponent)
    
    # Array to store powers of base modulo n
    base_powers = [base]
    previous_base = base
    
    # Calculate powers of base modulo n up to exponent length
    for _ in range(r - 1):
        next_base = (previous_base * previous_base) % modulus
        base_powers.append(next_base)
        previous_base = next_base
    
    # Perform Montgomery multiplication to compute base^exponent mod modulus
    result = montgomery_multiply(base_powers, binary_exponent, modulus)
    return result % modulus

def montgomery_multiply(array, binary_exponent, modulus):
    result = 1
    # Iterate over the array of base powers and corresponding binary exponent digits
    for index, base_power in enumerate(array):
        # If the binary exponent digit is 0, skip multiplication
        if int(binary_exponent[index]) == 0:
            continue
        # Perform Montgomery multiplication
        result *= base_power
        result = result % modulus
    return result

# Example usage
print(exp_mod_montgomery(2, 6, 10000000))
