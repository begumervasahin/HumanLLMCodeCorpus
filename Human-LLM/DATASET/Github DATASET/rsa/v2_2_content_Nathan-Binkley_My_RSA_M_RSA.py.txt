from ecdsa.numbertheory import inverse_mod as modinv
import sys

def main():
    # Initialize the public exponent
    e = 65537
    
    # Prompt user to input two prime numbers
    try:
        p = int(input("Enter a prime number: "))
        q = int(input("Enter another prime number (ensure their sum is at least 26 for alphabet encoding): "))
        
        # Ensure the sum of the primes is sufficient for alphabet encoding
        if p + q < 26:
            raise ValueError("The sum of the input values should be at least 26.")
    
    # Handle input errors
    except ValueError as ve:
        print("Error:", ve)
        sys.exit(1)

    # Calculate modulus and private exponent
    n = p * q
    d = modinv(e, ((p-1)*(q-1)))

    # Get plaintext input from user
    plaintext = input("Enter the text you want to encrypt: ").lower()
    print("Your plaintext is:", plaintext)
    print("p:", p)
    print("q:", q)

    # Convert plaintext characters to numerical values
    plain_list_num = [ord(char) - 96 for char in plaintext if char.isalpha()]  # Assuming only lowercase alphabets are used
    print("Your plaintext list is:", plain_list_num)

    # Encrypt plaintext
    enc_list_num = [(i**e) % n for i in plain_list_num]
    print("Your encrypted values are:", enc_list_num)

    # Decrypt ciphertext
    end_list_num = [(i**d) % n for i in enc_list_num]
    print("Your decrypted values are:", end_list_num)

    # Convert decrypted numerical values back to characters
    ending = ''.join([chr(num + 96) for num in end_list_num])  # Convert back to characters assuming lowercase alphabets
    print("Plaintext (Decrypted) is:", ending)

if __name__ == "__main__":
    main()
