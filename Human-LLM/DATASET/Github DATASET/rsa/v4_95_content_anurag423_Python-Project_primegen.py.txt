def generate_prime_pair():
    import numpy as np

    # Prompt user for the number of bits
    bits = int(input('How many bits? '))

    # Calculate the maximum value based on the number of bits
    final = 2 ** bits

    # Generate a list of integers from 2 to final
    ints_to_final = list(range(2, final + 1))

    # Convert the list to a numpy array
    ints_to_final = np.array(ints_to_final)

    # Initialize index for looping through the list
    index = 0

    # Loop until all elements in the list are checked
    while index < len(ints_to_final):
        del_vect = []

        # Create an empty array to store elements to delete
        del_vect = np.array(del_vect)

        # Iterate through the list
        for i in range(0, len(ints_to_final)):
            if ints_to_final[i] == ints_to_final[index]:
                # Mark elements to keep
                del_vect = np.append(del_vect, 0)
            else:
                # Mark non-prime elements for deletion
                if ints_to_final[i] % ints_to_final[index] == 0:
                    del_vect = np.append(del_vect, ints_to_final[i])
                else:
                    del_vect = np.append(del_vect, 0)

        # Remove marked elements from the array
        ints_to_final = ints_to_final - del_vect
        ints_to_final = ints_to_final[ints_to_final != 0]

        # Move to the next index
        index = index + 1

    # Select two random prime numbers
    random1 = int(np.random.choice(ints_to_final, 1))
    random2 = int(np.random.choice(ints_to_final, 1))

    # Print the selected prime numbers (hidden)
    print(f'(Hidden) P Value: {random1}')
    print(f'(Hidden) Q Value: {random2}')

    return random1, random2
