def Decryption(Em, prk, pk):
    return (Em ** prk) % pk

def decrypt_message_from_file():
    l1 = []
    l2 = []
    l3 = []
    
    n = int(input("Enter Modulo: "))
    d = int(input("Enter Private Key: "))
    
    with open("Encrypted_Message.txt", "r") as f:
        stri = [line.rstrip('\n') for line in f]

    for x in stri:
        l1.append(x)
    for x in l1:
        l2.append(Decryption(int(x), d, n))
    for y in l2:
        l3.append(chr(y))
    Message = "".join(l3)
    print("Decrypted Message:", Message)
    
    with open("Decrypted_Message.txt", "w") as fw:
        for z in l3:
            fw.write(str(z))

def decrypt_file():
    l1 = []
    l2 = []
    
    n = int(input("Enter Modulo: "))
    d = int(input("Enter Private Key: "))
    
    with open("Encrypted_File.txt", "r") as f:
        fi = f.readlines()
    
    for y in fi:
        currentPlace = y[:-1]
        l1.append(int(currentPlace))
    
    for y in l1:
        l2.append(Decryption(int(y), d, n))
    
    with open("Decrypted_File.txt", "wb") as fi:
        for y in l2:
            fi.write(bytes((y,)))

def main():
    choose = input("Enter 1 to decrypt message from a file, or 2 to decrypt a file: ")
    
    if choose == '1':
        decrypt_message_from_file()
    elif choose == '2':
        decrypt_file()
    else:
        print("Invalid choice. Please enter either 1 or 2.")

if __name__ == "__main__":
    main()
