# Importing necessary classes from RSA-related modules
from rsa.crypt.crypt import Crypt
from rsa.decrypt.decrypt import Decrypt
from rsa.key_generator.key_generator import KeyGenerator

# Function to read bytes from a file
def read_file_bytes(file_path):
    with open(file_path, 'rb') as file:
        return file.read()

# Function to demonstrate encryption and decryption
def encrypt_decrypt_demo(data_bytes):
    # Generating RSA key pair
    key_generator = KeyGenerator()
    private_key, public_key = key_generator.generate()
    
    # Encrypting data using RSA public key
    crypter = Crypt(public_key)
    encrypted_bytes = crypter.crypt(list(data_bytes))  # Converting bytes to a list of integers for encryption
    
    # Decrypting data using RSA private key
    decrypter = Decrypt(private_key)
    decrypted_bytes = decrypter.decrypt(encrypted_bytes)  # Decrypting the list of integers
    
    # Printing the encrypted and decrypted data
    print("Encrypted Bytes:", encrypted_bytes)
    print("Decrypted Bytes:", bytes(decrypted_bytes))  # Converting decrypted integers back to bytes

# Function to find two large prime numbers
def find_two_large_primes():
    import math
    
    # Initializing an empty list to store prime numbers
    primes = []
    
    # Iterating over a range to find prime numbers
    for num in range(10_000, 2, -1):  # Starting from 10,000 and going backwards
        is_prime = True
        
        # Checking if the number is prime
        for i in range(2, int(math.sqrt(num)) + 1):  # Checking divisibility up to the square root of the number
            if num % i == 0:  # If the number is divisible by any other number, it's not prime
                is_prime = False
                break
        
        # If the number is prime, adding it to the list
        if is_prime:
            primes.append(num)
        
        # Exiting the loop if two prime numbers are found
        if len(primes) == 2:
            break
    
    # Printing the two prime numbers found
    print("Two Large Prime Numbers:", primes)

# Main execution
if __name__ == "__main__":
    # Reading bytes from a file
    input_bytes = read_file_bytes("files/input.txt")
    print("Input Bytes:", input_bytes)
    
    # Demonstrating encryption and decryption
    encrypt_decrypt_demo(input_bytes)
    
    # Finding two large prime numbers
    find_two_large_primes()
