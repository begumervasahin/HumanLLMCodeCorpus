import rsa
import gmpy2

class RSA:
    def __init__(self):
        pass

    def generate_prime(self, bits):
        return rsa.prime.getprime(bits)

    def calculate_e(self, phi):
        return phi - 1

    def generate_keys(self, bits):
        p = self.generate_prime(bits)
        q = self.generate_prime(bits)
        n = p * q
        phi = (p - 1) * (q - 1)
        e = self.calculate_e(phi)
        d = int(gmpy2.invert(e, phi).digits())
        return d, (n, e)

        decrypted_message = decrypted_message_bytes.decode('utf-8')
