import random

def get_odd_number(start, end):
    """Generate a random odd number in the range [start, end)."""
    rand_num = random.randrange(start, end)
    if rand_num % 2 == 0:
        rand_num += 1
    return rand_num

def is_composite(n, witness, exponent, remainder):
    """Check if a number is composite using the Miller-Rabin primality test."""
    result = pow(witness, exponent, n)
    if result == 1 or result == n - 1:
        return False
    for _ in range(int(remainder)):
        result = pow(result, 2, n)
        if result == n - 1:
            return False
    return True

def is_prime(n):
    """Check if a number is prime using the Miller-Rabin primality test."""
    exponent = n - 1
    remainder = 0
    while exponent % 2 == 0:
        exponent //= 2
        remainder += 1
    trials = (n - 1) // 4
    for _ in range(trials):
        witness = random.randint(2, n - 1)
        if is_composite(n, witness, exponent, remainder):
            return False
    return True

def generate_prime_number(start, end):
    """Generate a prime number in the range [start, end)."""
    while True:
        num = get_odd_number(start, end)
        if is_prime(num):
            return num

def gcd(a, b):
    """Calculate the greatest common divisor of two numbers."""
    while b:
        a, b = b, a % b
    return a

def multiplicative_inverse(e, phi):
    """Calculate the multiplicative inverse of e modulo phi."""
    x, y = 0, 1
    while phi:
        quotient = e // phi
        e, phi = phi, e % phi
        x, y = y, x - quotient * y
    return x % phi

# Example usage:
lower_bound = 10000
upper_bound = 100000

p = generate_prime_number(lower_bound, upper_bound)
q = generate_prime_number(lower_bound, upper_bound)
n = p * q
phi = (p - 1) * (q - 1)

e = random.randint(2, phi - 1)
while gcd(e, phi) != 1:
    e = random.randint(2, phi - 1)

d = multiplicative_inverse(e, phi)

print("p:", p)
print("q:", q)
print("n:", n)
print("phi:", phi)
print("e:", e)
print("d:", d)
