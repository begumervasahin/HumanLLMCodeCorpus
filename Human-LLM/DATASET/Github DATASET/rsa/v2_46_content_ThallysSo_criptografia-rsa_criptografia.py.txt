class Criptografia:
    def criptografia(self, mensagem, chave_publica, modulo):
        mensagem_criptografada = (mensagem ** chave_publica) % modulo
        return mensagem_criptografada

    def descriptografia(self, mensagem_criptografada, chave_privada, modulo):
        mensagem_descriptografada = (mensagem_criptografada ** chave_privada) % modulo
        return mensagem_descriptografada

    def encripta_mensagem(self):
        mensagem = input("Digite a mensagem: \t")
        while len(mensagem) > 128:
            print("Você ultrapassou o limite de caracteres.")
            mensagem = input("Digite a mensagem novamente: \t")

        print('=' * 5 + ' Digite as chaves públicas: ' + '=' * 5)
        chave_publica = int(input("Chave pública (e): \t"))
        modulo = int(input("Modulo (n): \t"))
        
        mensagem_criptografada = ''.join(chr(self.criptografia(ord(caractere), chave_publica, modulo)) for caractere in mensagem)
        
        print('Texto Criptografado: ', mensagem_criptografada, '\n')
        return mensagem_criptografada

    def decripta_mensagem(self, mensagem_criptografada):
        print('=' * 5 + ' Digite as chaves privadas: ' + '=' * 5)
        chave_privada = int(input("Chave privada (d): \t"))
        modulo = int(input("Modulo (n): \t"))

        mensagem_descriptografada = ''.join(chr(self.descriptografia(ord(caractere), chave_privada, modulo)) for caractere in mensagem_criptografada)
        
        print('Texto Simples: ', mensagem_descriptografada, '\n')
        return mensagem_descriptografada

# Example usage:
cripto = Criptografia()
mensagem_criptografada = cripto.encripta_mensagem()
cripto.decripta_mensagem(mensagem_criptografada)
