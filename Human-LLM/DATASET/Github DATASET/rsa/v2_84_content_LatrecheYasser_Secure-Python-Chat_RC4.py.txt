class RC4():
    def __init__(self):
        # Initialize the RC4 state with values from 0 to 255
        self.state = list(range(256))
    
    def shuffle(self, key):
        j = 0
        # Key-scheduling algorithm (KSA)
        for i in range(256):
            j = (j + self.state[i] + ord(key[i % len(key)])) % 256
            # Swap values at positions i and j
            self.state[i], self.state[j] = self.state[j], self.state[i]
    
    def Crypt(self, text):
        output = []  # Store the encrypted/decrypted characters
        i = j = 0
        # Pseudo-random generation algorithm (PRGA) for keystream generation
        for x in range(len(text)):
            i = (i + 1) % 256
            j = (self.state[i] + j) % 256
            # Swap values at positions i and j
            self.state[i], self.state[j] = self.state[j], self.state[i]
            # XOR the character with the keystream byte and append to output
            output.append(chr(ord(text[x]) ^ self.state[(self.state[i] + self.state[j]) % 256]))
        # Concatenate the characters in the output list to form the encrypted/decrypted text
        return ''.join(output)

if __name__ == "__main__":
    # Create two instances of the RC4 class
    rc4 = RC4()  # For encryption
    rc41 = RC4()  # For decryption
    
    # Prompt the user to enter a key
    key = input('Enter your key: ')
    
    # Initialize the RC4 states with the provided key
    rc4.shuffle(key)
    rc41.shuffle(key)
    
    # Encryption and decryption loop
    while True:
        # Prompt the user to enter a message
        message = input('Enter your message: ')
        
        # Encrypt the message using the first RC4 instance
        encrypted_message = rc4.Crypt(message)
        print('Encrypted message:', encrypted_message)
        
        # Decrypt the encrypted message using the second RC4 instance
        decrypted_message = rc41.Crypt(encrypted_message)
        print('Decrypted message:', decrypted_message)
