class Cryptography:
    def encrypt(self, message, public_key, modulus):
        cipher_text = (message ** public_key) % modulus
        return cipher_text

    def decrypt(self, cipher_text, private_key, modulus):
        plain_text = (cipher_text ** private_key) % modulus
        return plain_text

    def get_input_within_limit(self, prompt, limit):
        user_input = input(prompt)
        while len(user_input) > limit:
            print("You have exceeded the character limit.")
            user_input = input("Please enter the message again: \t")
        return user_input

    def get_public_keys(self):
        print('='*5 + ' Enter the public keys: ' + '='*5)
        public_key = int(input("Public key: \t"))
        modulus = int(input("Modulus: \t"))
        return public_key, modulus

    def get_private_keys(self):
        print('='*5 + ' Enter the private keys: ' + '='*5)
        private_key = int(input("Private key: \t"))
        modulus = int(input("Modulus: \t"))
        return private_key, modulus

    def encrypt_message(self):
        message = self.get_input_within_limit("Enter the message: \t", 128)
        public_key, modulus = self.get_public_keys()

        encrypted_message = ''.join(chr(self.encrypt(ord(char), public_key, modulus)) for char in message)
        print('Encrypted Text: ', encrypted_message, '\n')
        return encrypted_message

    def decrypt_message(self, encrypted_message):
        private_key, modulus = self.get_private_keys()

        decrypted_message = ''.join(chr(self.decrypt(ord(char), private_key, modulus)) for char in encrypted_message)
        return print('Decrypted Text: ', decrypted_message, '\n')
