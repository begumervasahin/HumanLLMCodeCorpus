import math
import time

# Ask user for input
n = input("Check prime numbers up to: ")
start = time.time()

# Convert input to integer
n = int(n)

# Initialize variables
x = 2
i = 0
primes = []
isprime = True
check = True

# Read existing primes from file
with open('output.txt', 'r') as rf:
    for line in rf:
        primes.append(int(line))
    x = int(primes[-1])

# Find primes up to n
while x <= n:
    if len(primes) > i:
        while primes[i] <= math.sqrt(x) and check:
            if x % primes[i] == 0:
                isprime = False
                check = False
            elif i + 1 < len(primes):
                i += 1
            else:
                check = False
        if isprime:
            primes.append(x)
    x += 1
    i = 0
    isprime = True
    check = True

# Write primes to file
with open('output.txt', 'w') as wf:
    for z in primes:
        wf.write(str(z) + "\n")

# Calculate and print execution time
end = time.time()
print(str(end - start) + " seconds to calculate")
