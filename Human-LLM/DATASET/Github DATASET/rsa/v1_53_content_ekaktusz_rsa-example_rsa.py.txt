import random
import decimal_string
import prime

def gcd(a, b):
    while b != 0:
        a, b = b, a % b
    return a

def coprime(a, b):
    return gcd(a, b) == 1

def extended_euclidean_alg(a, b):
    x, y = a, b
    t0, t1 = 0, 1
    r, t = b, 1
    while (x % y) != 0:
        q = x // y
        r = x % y
        t = (t0 - q*t1) % a
        x = y
        y = r
        t0 = t1
        t1 = t
    return t

def generate_keys():
    p, q = prime.generate_two_random_prime(512)
    n = p * q
    phi = (p-1)*(q-1)
    e = random.randrange(1, phi)
    while not coprime(e, phi):
        e = random.randrange(1, phi)
    d = extended_euclidean_alg(phi, e)
    return ((e, n), (d, n))

def decrypt(private_key, text_number):
    d, n = private_key
    decoded = pow(text_number, d, n)
    return decimal_string.integer_to_text(decoded)

# Example usage:
# Generating keys
public_key, private_key = generate_keys()

# Encrypting a message
message = "Hello, World!"
encrypted_message = encrypt(public_key, message)
print("Encrypted message:", encrypted_message)

# Decrypting the message
