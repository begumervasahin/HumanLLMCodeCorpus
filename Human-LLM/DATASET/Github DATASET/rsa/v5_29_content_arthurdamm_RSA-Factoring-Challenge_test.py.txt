# Import necessary modules
from sys import argv, exit

# Import lists of prime numbers from separate modules
from primes import primes
from primes2 import primes2
from primes3 import primes3

# Define a function to factorize a given number into prime factors
def factorize(num):
    # Iterate through each list of prime numbers
    for prime_list in (primes, primes2, primes3):
        for p in prime_list:
            # Check if the number is divisible by the current prime
            if num % p == 0:
                # Determine the smaller divisor
                smaller_divisor = min(p, num // p)
                # Print the factorization
                print(f"{num}={smaller_divisor}*{num // smaller_divisor}")
                return

# Main block
if __name__ == "__main__":
    # Check if the correct number of arguments is provided
    if len(argv) != 2:
        print("USAGE: factors <file>")
        exit(1)
    # Open the file provided in the command line argument
    with open(argv[1], 'r') as file:
        # Iterate through each line in the file
        for line in file:
            # Convert the line to an integer and factorize it
            factorize(int(line))
