import math
import numpy as np
from matplotlib import pyplot
from collections import Counter
import warnings

# Function to calculate k-nearest neighbors
def k_nearest_neighbors(data, predict, k=3):
    # Warning if k is smaller than the number of data points
    if len(data) >= k:
        warnings.warn("k is too small")
    
    # Calculate distances between data points and prediction point
    distances = []
    for group in data:
        for features in data[group]:
            euclidean_distance = np.linalg.norm(np.array(features) - np.array(predict))
            distances.append([euclidean_distance, group])
    
    # Sort distances and get the top k nearest groups
    sorted_distances = [i[1] for i in sorted(distances)]
    top_nearest = sorted_distances[:k]
    
    # Determine the most common group among the top k nearest neighbors
    group_res = Counter(top_nearest).most_common(1)[0][0]
    
    # Calculate confidence level based on the frequency of the most common group
    confidence = Counter(top_nearest).most_common(1)[0][1] * 1.0 / k
    
    return group_res, confidence

# Main program
if __name__ == '__main__':
    # Sample dataset with groups and corresponding features
    dataset = {'black': [[1, 2], [2, 3], [3, 1]], 'red': [[6, 5], [7, 7], [8, 6]]}    
    
    # New features to predict
    new_features = [3.5, 5.2]
    
    # Plotting existing data points
    for i in dataset:
        for ii in dataset[i]:
            pyplot.scatter(ii[0], ii[1], s=50, color=i)
    
    # Finding the group for the new features using k-nearest neighbors
    which_group, confidence = k_nearest_neighbors(dataset, new_features, k=3)      
    print(which_group, confidence)
    
    # Plotting the new features with the predicted group
    pyplot.scatter(new_features[0], new_features[1], s=300, color=which_group)    
    pyplot.show()
