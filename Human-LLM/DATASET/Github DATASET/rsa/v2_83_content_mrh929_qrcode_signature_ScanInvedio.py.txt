import cv2
import pyzbar.pyzbar as pyzbar

# Function to decode and display QR codes in an image
def decode_and_display_qr_codes(image):
    # Decode QR codes in the image
    decoded_barcodes = pyzbar.decode(image)
    
    # Iterate through each decoded barcode
    for barcode in decoded_barcodes:
        # Extract barcode bounding box coordinates
        x, y, w, h = barcode.rect
        
        # Draw a rectangle around the barcode
        cv2.rectangle(image, (x, y), (x + w, y + h), (255, 255, 255), 2)
        
        # Decode barcode data and type
        barcode_data = barcode.data.decode("utf-8")
        barcode_type = barcode.type
        
        # Create text to display the barcode data and type
        text = "{} ({})".format(barcode_data, barcode_type)
        
        # Draw text above the barcode
        cv2.putText(image, text, (x, y - 10), cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255, 255, 255), 2)
        
        # Return the image with the detected barcode and its data
        return image, barcode_data
    
    # If no barcode is detected, return the original image and empty data
    return image, ""

# Function to detect QR codes using the camera
def detect_qr_codes():
    # Initialize camera capture
    camera = cv2.VideoCapture(0)
    
    # Continuously capture video from the camera
    while True:
        # Read a frame from the camera
        ret, frame = camera.read()
        
        # Convert the frame to grayscale
        gray_frame = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY)
        
        # Decode and display QR codes in the frame
        image_with_qr, content = decode_and_display_qr_codes(gray_frame)
        
        # Display the frame with detected QR codes
        cv2.imshow("Camera", image_with_qr)
        
        # Check if a QR code is detected
        if content != '':
            break
        
        # Check for user input to quit the program
        if cv2.waitKey(1) & 0xFF == ord('q'):
            break
    
    # Release the camera and close all windows
    camera.release()
    cv2.destroyAllWindows()
    
    # Return the content of the detected QR code
    return content

if __name__ == '__main__':
    # Call the function to detect QR codes
    qr_code_content = detect_qr_codes()
    
    # Print the content of the detected QR code
    print("QR Code content:", qr_code_content)
