# Function to find the extended greatest common divisor
def extended_gcd(a, b):
    # Initializations
    x, y, u, v = 0, 1, 1, 0
    
    # Loop until 'a' becomes 0
    while a != 0:
        # Calculate quotient and remainder
        quotient, remainder = b // a, b % a
        
        # Update values
        m, n = x - u * quotient, y - v * quotient
        b, a, x, y, u, v = a, remainder, u, v, m, n
    
    # Return gcd and coefficients
    return b, x, y
