import math
import time

def main():
    n = int(input("Check prime numbers up to: "))  # Convert input directly to an integer
    start = time.time()
    
    primes = read_primes("output.txt")
    
    if primes:  # Check if the list is not empty
        last_prime = primes[-1]
    else:  # If no primes are found in the file, start from the beginning
        last_prime = 2
        primes = [2]
    
    for x in range(last_prime + 1, n + 1):
        if is_prime(x, primes):
            primes.append(x)
    
    write_primes("output.txt", primes)
    
    end = time.time()
    print(f"{end - start} seconds to calculate")

def read_primes(filename):
    """Reads prime numbers from a file and returns them as a list of integers."""
    try:
        with open(filename, 'r') as file:
            return [int(line.strip()) for line in file]
    except FileNotFoundError:
        return []  # Return an empty list if file does not exist

def write_primes(filename, primes):
    """Writes the list of prime numbers to a file, one per line."""
    with open(filename, 'w') as file:
        for prime in primes:
            file.write(f"{prime}\n")

def is_prime(number, primes):
    """Checks if a number is prime using a list of known primes."""
    for prime in primes:
        if prime > math.sqrt(number):
            break
        if number % prime == 0:
            return False
    return True

if __name__ == "__main__":
    main()
