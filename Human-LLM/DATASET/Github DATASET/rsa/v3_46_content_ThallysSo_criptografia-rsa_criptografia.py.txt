class Criptografia:
    def criptografar(self, mensagem, chave_publica, modulo):
        mensagem_criptografada = (mensagem ** chave_publica) % modulo
        return mensagem_criptografada

    def descriptografar(self, mensagem_criptografada, chave_privada, modulo):
        mensagem_descriptografada = (mensagem_criptografada ** chave_privada) % modulo
        return mensagem_descriptografada

    def solicitar_chaves_publicas(self):
        print('=' * 5 + ' Digite as chaves públicas: ' + '=' * 5)
        chave_publica = int(input("Chave pública (e): \t"))
        modulo = int(input("Módulo (n): \t"))
        return chave_publica, modulo

    def solicitar_chaves_privadas(self):
        print('=' * 5 + ' Digite as chaves privadas: ' + '=' * 5)
        chave_privada = int(input("Chave privada (d): \t"))
        modulo = int(input("Módulo (n): \t"))
        return chave_privada, modulo

    def encriptar_mensagem(self):
        mensagem = input("Digite a mensagem: \t")
        while len(mensagem) > 128:
            print("Você ultrapassou o limite de caracteres.")
            mensagem = input("Digite a mensagem novamente: \t")

        chave_publica, modulo = self.solicitar_chaves_publicas()

        mensagem_criptografada = ''.join(chr(self.criptografar(ord(caractere), chave_publica, modulo)) for caractere in mensagem)

        print('Texto Criptografado: ', mensagem_criptografada, '\n')
        return mensagem_criptografada

    def decriptar_mensagem(self, mensagem_criptografada):
        chave_privada, modulo = self.solicitar_chaves_privadas()

        mensagem_descriptografada = ''.join(chr(self.descriptografar(ord(caractere), chave_privada, modulo)) for caractere in mensagem_criptografada)

        print('Texto Simples: ', mensagem_descriptografada, '\n')
        return mensagem_descriptografada

# Example usage:
cripto = Criptografia()
mensagem_criptografada = cripto.encriptar_mensagem()
cripto.decriptar_mensagem(mensagem_criptografada)
