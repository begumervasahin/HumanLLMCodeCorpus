import fileOp
import additiveShares
import thresholdShares
import time

def refresh_shares():
    '''
    Refreshes all shares stored in the file "FadditiveShares".
    Each share represents a portion of a secret value, divided using additive sharing.
    '''
    additive_shares = fileOp.read_list("FadditiveShares")  # Read existing shares from file
    modulus_rsa = fileOp.read_list("FmodulusRSA")          # Read RSA modulus from file
    n = modulus_rsa[0]                                     # Get the RSA modulus value
    old_shares = additive_shares                           # Store old shares for reference
    num_shares = len(old_shares)                           # Get the number of shares
    new_shares = [0] * num_shares                          # Initialize a list to store new shares

    # Iterate through each old share and update it
    for i in range(num_shares):
        share_div = additiveShares.additive_sharing(old_shares[i], num_shares)  # Divide old share further
        new_shares = [(a + b) for a, b in zip(new_shares, share_div)]           # Add the divided shares

    fileOp.write_list("FadditiveShares", new_shares)   # Write the new shares back to the file
    print("Running")
    thresholdShares.threshold_additive_shares()       # Apply thresholding to the shares
    print("Done")
