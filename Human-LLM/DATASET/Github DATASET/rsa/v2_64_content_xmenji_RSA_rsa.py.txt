# Function to check if a number is prime
def is_prime(num):
    if num <= 1:
        return False
    for i in range(2, int(num**0.5) + 1):
        if num % i == 0:
            return False
    return True

# Function to find the greatest common divisor
def gcd(a, b):
    while b != 0:
        a, b = b, a % b
    return a

# Input prime numbers p and q
print("Enter 2 PRIME numbers for 'p' and 'q'.")
p = int(input("p: "))
q = int(input("q: "))

# Check if both p and q are prime
if not (is_prime(p) and is_prime(q)):
    print("Both 'p' and 'q' must be prime numbers.")
    exit()

# Compute n
n = p * q
print("\nn = " + str(n))

# Compute Euler's totient function phi
phi = (p - 1) * (q - 1)
print("phi = " + str(phi))

# Find an integer e such that 1 < e < phi and gcd(e, phi) = 1
e = 2
while gcd(e, phi) != 1:
    e += 1

print("e = " + str(e))
print("ENCRYPTION KEY= (" + str(e) + ", " + str(n) + ")")

# Find an integer d such that (d * e) % phi = 1
d = 10
while True:
    if (d * e) % phi == 1:
        break
    else:
        d += 1

print("d = " + str(d))
print("DECRYPTION KEY= (" + str(d) + ", " + str(n) + ")")

# Define the alphabet used for encryption
alphabet = ".abcdefghijklmnopqrstuvwxyz ABCDEFGHIJKLMNOPQRSTUVWXYZ"
letters = list(alphabet)

# Input the message to be encrypted
msg = input("\nEnter simple message (letters only): ")

# Encrypt the message
encryptedMsg = [(letters.index(x) ** e) % n for x in msg]
print("\nEncrypted message: ")
print(encryptedMsg)

# Decrypt the message
decryptedMsg = [letters[(x ** d) % n] for x in encryptedMsg]
print("\nDecrypted Message: ")
print(''.join(decryptedMsg))
