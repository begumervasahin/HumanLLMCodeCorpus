from cryptography.hazmat.backends import default_backend
from cryptography.hazmat.primitives import serialization, hashes
from cryptography.hazmat.primitives.asymmetric import padding

def load_private_key(encrypted_key_path, password):
    with open(encrypted_key_path, "rb") as key_file:
        encrypted_private_key = key_file.read()
        private_key = serialization.load_pem_private_key(
            encrypted_private_key,
            password=password.encode(),
            backend=default_backend()
        )
    return private_key

def convert_private_key_to_pem(private_key):
    pem_private_key = private_key.private_bytes(
        encoding=serialization.Encoding.PEM,
        format=serialization.PrivateFormat.TraditionalOpenSSL,
        encryption_algorithm=serialization.NoEncryption()
    )
    return pem_private_key.decode()

def decrypt_text(private_key, ciphertext_path):
    with open(ciphertext_path, 'r') as f:
        ciphertext = bytes.fromhex(f.read())

    plaintext = private_key.decrypt(
        ciphertext,
        padding.OAEP(
            mgf=padding.MGF1(algorithm=hashes.SHA256()),
            algorithm=hashes.SHA256(),
            label=None
        )
    )
    return plaintext.decode()

def main():
    encrypted_key_path = "private_key_encrypted.pem"
    ciphertext_path = "encoded_text.txt"
    password = 'In Crypto We Trust'

    private_key = load_private_key(encrypted_key_path, password)
    pem_private_key = convert_private_key_to_pem(private_key)
    decrypted_text = decrypt_text(private_key, ciphertext_path)

    print("Decrypted Private Key:")
    print(pem_private_key)

    print("\nDecrypted Text:")
    print(decrypted_text)

if __name__ == "__main__":
    main()
