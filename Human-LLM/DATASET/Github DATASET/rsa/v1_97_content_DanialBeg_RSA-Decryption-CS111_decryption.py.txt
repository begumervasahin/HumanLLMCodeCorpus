import string

def main():
    e = int(input("Please enter the exponent value in the public key: "))
    n = int(input("Please enter the n value in the public key: "))
    print(str(e) + " " + str(n))
    factors_list = factors(n)
    p = factors_list[1]
    q = factors_list[2]
    n_val = (p+1)*(q+1)
    decrypt_e = inv_modulo(e, n_val)
    input_array = [51, 12, 51, 39, 31, 21, 14, 10, 20, 17, 7, 25, 14, 26, 33, 52, 15, 7,
                   27, 51, 7, 49, 8, 15, 51, 7, 8, 25, 7, 25, 10, 49, 18, 52, 51, 7, 8, 
                   25, 7, 18, 26, 25, 25, 10, 27, 52, 51, 7, 27, 33, 21, 7, 20, 26, 21, 
                   7, 25, 10, 49, 18, 52, 51, 39]
    alphabet = list(string.ascii_lowercase)
    alphabet.append(' ')
    for i in input_array:
        char_val = (i ** decrypt_e) % n
        print(str(alphabet[char_val-2]), end="")
    print()

def inv_modulo(a, b):
    for num in range(1, b):
        if ((a * num) % b) == 1:
            return num
    return 1

def factors(r):
    list_factors = []
    i = 1
    while i <= r:
        if r % i == 0:
            list_factors.append(i)
        i = i + 1
    return list_factors

if __name__ == "__main__":
    main()
