# Importing necessary functions and libraries
from funcions import *
import numpy as np
import matplotlib.pyplot as plt
import timeit
from genera_claus import genera_claus

# Function to factorize N given e
def factoritza(N, e):
    # Obtain prime factors of N
    llista_factors = factors_primers(N)
    
    # If there are not exactly two prime factors, return zeros
    if len(llista_factors) != 2:
        return 0, 0, 0
    
    # Extracting prime factors
    p = llista_factors[0]
    q = llista_factors[1]
    
    # Calculating Euler's totient function
    phi_n = (p - 1) * (q - 1)
    
    # Calculating the modular inverse of e mod phi_n
    d = invers_modular(e, phi_n)
    
    return p, q, d

# Function to test the factorization function
def test():
    # Maximum number of bits
    max_bits = 25
    
    # Lists to store sizes and times
    sizes = []
    times = []
    
    # Loop through different sizes of keys
    for i in range(8, max_bits):
        # Generate keys
        n, e, d, real_p, real_q = genera_claus(i)
        
        # Measure time taken for factorization
        start_time = timeit.default_timer()
        p_found, q_found, d_found = factoritza(n, e)
        end_time = timeit.default_timer()
        
        # Record size and time
        sizes.append(i)
        times.append(end_time - start_time)
        
        # Display results
        print("Factorization of N completed.")
        print("Real value of p:", real_p)
        print("Real value of q:", real_q)
        print("Found value of p:", p_found)
        print("Found value of q:", q_found)
    
    # Plotting the time taken for factorization
    plt.plot(sizes, times, '-gD')
    plt.xlabel("Number of bits")
    plt.ylabel("Seconds")
    plt.show()

# Calling the test function
test()
