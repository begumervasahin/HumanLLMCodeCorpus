import hashlib
import gmpy2
import random
import time

# Function to generate a prime number
def get_prime(rs):
    p = gmpy2.mpz_urandomb(rs, 1024)
    while not gmpy2.is_prime(p):
        p += 1
    return p

# Key generation function
def key_generation():
    # Seed the random number generator
    rs = gmpy2.random_state(int(time.time()))
    
    # Generate two large prime numbers
    p = get_prime(rs)
    q = get_prime(rs)
    
    # Compute the modulus
    n = p * q
    
    # Public exponent
    e = 0x10001
    
    # Hash function
    H = lambda m: int(hashlib.sha256(m.encode()).hexdigest(), 16)
    
    # Public key
    public_key = [n, e, H]
    
    # Private exponent
    d = gmpy2.invert(e, (p - 1) * (q - 1))
    
    # Private key
    private_key = d
    
    return public_key, private_key

# Signing function
def sign_message(public_key, private_key, message):
    n = public_key[0]
    e = public_key[1]
    H = public_key[2]
    d = private_key
    
    # Generate a random number R
    R = random.randrange(0, n)
    
    # Hash the message
    H_m = H(message) % n
    
    # Calculate R^e mod n
    R_e = gmpy2.powmod(R, e, n)
    
    # Compute M = R^e * H(m) mod n
    M = R_e * H_m % n
    
    # Calculate M^d mod n
    M_d = gmpy2.powmod(M, d, n)
    
    # Compute R^-1 mod n
    R_inv = gmpy2.invert(R, n)
    
    # Calculate the signature
    signature = gmpy2.mul(M_d, R_inv) % n
    
    return signature

# Verification function
def verify_signature(public_key, signature):
    n = public_key[0]
    e = public_key[1]
    
    # Verify the signature
    verification_result = gmpy2.powmod(signature, e, n)
    
    return verification_result

if __name__ == '__main__':
    message = input('Please input your message: ')
    
    # Generate public-private key pair
    public_key, private_key = key_generation()
    
    # Sign the message
    signature = sign_message(public_key, private_key, message)
    print('Signature:', signature)
    
    # Verify the signature
    verification_result = verify_signature(public_key, signature)
    print('Verification Result:', verification_result)
