import random
import decimal_string
import prime

# Function to calculate the greatest common divisor
def gcd(a, b):
    while b != 0:
        a, b = b, a % b
    return a

# Function to check if two numbers are coprime
def coprime(a, b):
    return gcd(a, b) == 1

# Extended Euclidean Algorithm to find modular inverse
def extended_euclidean_alg(a, b):
    x, y = a, b
    t0, t1 = 0, 1
    r, t = b, 1
    
    while (x % y) != 0:
        q = x // y
        r = x % y
        t = (t0 - q * t1) % a
        x = y
        y = r
        t0 = t1
        t1 = t
    
    return t

# Function to generate public and private keys
def generate_keys():
    # Generate two random prime numbers
    p, q = prime.generate_two_random_prime(512)
    n = p * q
    phi = (p - 1) * (q - 1)
    
    # Choose a random integer e such that 1 < e < phi and e is coprime with phi
    e = random.randrange(1, phi)
    while not coprime(e, phi):
        e = random.randrange(1, phi)
    
    # Calculate the modular inverse of e modulo phi
    d = extended_euclidean_alg(phi, e)
    
    return ((e, n), (d, n))
