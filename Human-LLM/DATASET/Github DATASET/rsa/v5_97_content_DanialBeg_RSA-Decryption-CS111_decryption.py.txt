import string

def calculate_inverse_modulo(a, b):
    for num in range(1, b):
        if (a * num) % b == 1:
            return num
    return 1

def find_factors(number):
    factors_list = []
    for i in range(1, number + 1):
        if number % i == 0:
            factors_list.append(i)
    return factors_list

def main():
    print("RSA Decryption Program\n")

    # Input public key values
    exponent = int(input("Please enter the exponent value in the public key: "))
    n_value = int(input("Please enter the n value in the public key: "))
    print("Public key values: ", exponent, n_value)

    # Calculate factors of n
    factors_list = find_factors(n_value)
    p_value = factors_list[1]
    q_value = factors_list[2]

    # Calculate the totient value
    n_totient = (p_value + 1) * (q_value + 1)

    # Calculate the decryption exponent
    decrypt_exponent = calculate_inverse_modulo(exponent, n_totient)

    # Input array containing encrypted message
    encrypted_message = [51, 12, 51, 39, 31, 21, 14, 10, 20, 17, 7, 25, 14, 26, 33, 52, 15, 7,
                         27, 51, 7, 49, 8, 15, 51, 7, 8, 25, 7, 25, 10, 49, 18, 52, 51, 7, 8,
                         25, 7, 18, 26, 25, 25, 10, 27, 52, 51, 7, 27, 33, 21, 7, 20, 26, 21,
                         7, 25, 10, 49, 18, 52, 51, 39]

    # List of alphabets including space
    alphabet = list(string.ascii_lowercase)
    alphabet.append(' ')

    # Decrypt and print the message
    decrypted_message = ""
    for value in encrypted_message:
        decrypted_char = (value ** decrypt_exponent) % n_value
        decrypted_message += alphabet[decrypted_char - 2]

    print("\nDecrypted message:", decrypted_message)

if __name__ == "__main__":
    main()
