import math
import time

def main():
    upper_limit = get_upper_limit()  # Prompt user for the upper limit
    start_time = time.time()  # Record start time
    
    primes = load_primes_from_file("output.txt")  # Load previously found primes
    
    last_prime = get_last_prime(primes)  # Get the last prime number
    
    # Generate new prime numbers up to the specified upper limit
    generate_new_primes(last_prime, upper_limit, primes)
    
    save_primes_to_file("output.txt", primes)  # Save updated primes to file
    
    end_time = time.time()  # Record end time
    print_generation_time(start_time, end_time)  # Display generation time

def get_upper_limit():
    """Prompt user to enter the upper limit for prime number generation."""
    return int(input("Please enter the upper limit for prime number generation: "))

def load_primes_from_file(filename):
    """Read prime numbers from a file and return them as a list of integers."""
    try:
        with open(filename, 'r') as file:
            return [int(line.strip()) for line in file]
    except FileNotFoundError:
        return []

def get_last_prime(primes):
    """Get the last prime number from the list of primes."""
    return primes[-1] if primes else 2

def generate_new_primes(last_prime, upper_limit, primes):
    """Generate new prime numbers up to the specified upper limit."""
    for num in range(last_prime + 1, upper_limit + 1):
        if is_prime(num, primes):
            primes.append(num)

def is_prime(number, primes):
    """Check if a number is prime using a list of known primes."""
    for prime in primes:
        if prime > math.sqrt(number):
            break
        if number % prime == 0:
            return False
    return True

def save_primes_to_file(filename, primes):
    """Write the list of prime numbers to a file, one per line."""
    with open(filename, 'w') as file:
        for prime in primes:
            file.write(f"{prime}\n")

def print_generation_time(start_time, end_time):
    """Calculate and display the time taken for prime number generation."""
    generation_time = end_time - start_time
    print(f"Prime number generation completed in {generation_time:.2f} seconds.")

if __name__ == "__main__":
    main()
