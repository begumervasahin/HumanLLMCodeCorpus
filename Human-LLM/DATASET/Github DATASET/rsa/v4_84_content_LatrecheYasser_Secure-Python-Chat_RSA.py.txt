import rsa
import gmpy2

class RSA(object):
    # Method to generate two prime numbers of given bits length
    def get_p_q(self, bits):
        p = rsa.prime.getprime(bits)
        q = rsa.prime.getprime(bits)
        return p, q

    # Method to calculate the public exponent (e) using Euler's totient function
    def get_e(self, phi):
        return phi - 1

    # Method to generate RSA public and private keys
    def get_keys(self, bits):
        # Generate two prime numbers
        p, q = self.get_p_q(bits)
        n = p * q  # Calculate modulus (n)
        phi = (p - 1) * (q - 1)  # Calculate Euler's totient (phi)
        e = self.get_e(phi)  # Calculate public exponent (e)
        d = int(gmpy2.invert(e, phi).digits())  # Calculate private exponent (d)
        return d, (n, e)  # Return private exponent (d) and public key pair (n, e)

    # Method to encrypt a message
    def crypt(self, e, n, message):
        # Convert message to integer representation
        encoded_message = int.from_bytes(bytes(message.encode()), byteorder='little')
        # Encrypt message using RSA algorithm
        encrypted_message = pow(encoded_message, e, n)
        return encrypted_message

    # Method to decrypt an encrypted message
    def decrypt(self, d, n, message):
        # Decrypt message using RSA algorithm
        decrypted_message = pow(message, d, n)
        # Convert decrypted message from integer to bytes and then to string
        decoded_message = decrypted_message.to_bytes((decrypted_message.bit_length() // 8) + 1, byteorder='little').strip(b'\x00').decode("utf-8")
        return decoded_message

    # Input the desired number of bits for the keys
    bits = int(input('Enter the number of bits in the keys: '))
    
    # Create an instance of RSA class
    rsa_instance = RSA()

    d, (n, e) = rsa_instance.get_keys(bits)

    # Encrypt the message
    encrypted_message = rsa_instance.crypt(e, n, message)

    # Decrypt the message
    decrypted_message = rsa_instance.decrypt(d, n, encrypted_message)
    print('The decrypted message is:', decrypted_message)
