from math import inf
from itertools import product

def floyd_warshall(n, edge):
    # Initialize distance matrix and next node matrix
    dist = [[inf] * n for _ in range(n)]
    nxt = [[0] * n for _ in range(n)]

    # Set diagonal elements to 0
    for i in range(n):
        dist[i][i] = 0

    # Populate the distance matrix and next node matrix
    for u, v, w in edge:
        dist[u - 1][v - 1] = w
        nxt[u - 1][v - 1] = v - 1

    # Floyd-Warshall algorithm
    for k, i, j in product(range(n), repeat=3):
        sum_ik_kj = dist[i][k] + dist[k][j]
        if dist[i][j] > sum_ik_kj:
            dist[i][j] = sum_ik_kj
            nxt[i][j] = nxt[i][k]

    # Print the pairs, distance, and path
    print("    pair        dist     path")
    for i, j in product(range(n), repeat=2):
        if i != j:
            path = [i]
            while path[-1] != j:
                path.append(nxt[path[-1]][j])
            print("%3d -> %3d  %4d       %s"
                  % (i + 1, j + 1, dist[i][j],
                     ' -> '.join(str(p + 1) for p in path)))

if __name__ == '__main__':
    # Example usage
    floyd_warshall(4, [[1, 3, -2], [2, 1, 4], [2, 3, 3], [3, 4, 2], [4, 2, -1]])
