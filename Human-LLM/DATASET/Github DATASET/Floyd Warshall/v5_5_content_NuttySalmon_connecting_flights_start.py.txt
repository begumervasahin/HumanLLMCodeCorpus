from shortest import Database, ConnectingFlights, console

def main():
    # Initialize the database connection
    database = Database('localhost', 27017, "connecting_flight")

    # Initialize the ConnectingFlights object
    connecting_flights = ConnectingFlights(database)

    # Display the console menu and interact with the user
    console.menu(database, connecting_flights)

    # Clear the console screen after exiting the menu
    console.clear()

if __name__ == '__main__':
    # Start the application by calling the main function
    main()
