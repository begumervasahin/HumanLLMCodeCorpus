# Define the main function to compute shortest routes
def main(path_v1):
    # Import necessary functions from map_distance module
    from map_distance import main as get_edge_data
    
    # Retrieve edge data and the number of destinations from the input file
    edges, num_of_dest = get_edge_data(path_v1)
    
    # Filter edges to remove those with weights greater than or equal to 900
    edges_corrected = [(i, j, k) for i, j, k in edges if k < 900]
    
    # Define the Graph class
    class Graph:
        # Initialize the Graph class
        def __init__(self):
            self.vertices = {}
        
        # Add a vertex to the graph
        def add_vertex(self, key):
            vertex = Vertex(key)
            self.vertices[key] = vertex
        
        # Retrieve a vertex object by its key
        def get_vertex(self, key):
            return self.vertices[key]
        
        # Check if a vertex exists in the graph
        def __contains__(self, key):
            return key in self.vertices
        
        # Add an edge between two vertices with a given weight
        def add_edge(self, src_key, dest_key, weight=1):
            self.vertices[src_key].add_neighbour(self.vertices[dest_key], weight)
        
        # Check if there exists an edge between two vertices
        def does_edge_exist(self, src_key, dest_key):
            return self.vertices[src_key].does_it_point_to(self.vertices[dest_key])
        
        # Return the number of vertices in the graph
        def __len__(self):
            return len(self.vertices)
        
        # Return an iterator over the vertices of the graph
        def __iter__(self):
            return iter(self.vertices.values())
    
    # Define the Vertex class
    class Vertex:
        # Initialize the Vertex class
        def __init__(self, key):
            self.key = key
            self.points_to = {}
        
        # Return the key corresponding to the vertex object
        def get_key(self):
            return self.key
        
        # Add a neighboring vertex with a given edge weight
        def add_neighbour(self, dest, weight):
            self.points_to[dest] = weight
        
        # Return all vertices pointed to by this vertex
        def get_neighbours(self):
            return self.points_to.keys()
        
        # Return the weight of the edge to a specified destination vertex
        def get_weight(self, dest):
            return self.points_to[dest]
        
        # Check if this vertex points to a specified destination vertex
        def does_it_point_to(self, dest):
            return dest in self.points_to
    
    # Define the Floyd-Warshall algorithm for finding shortest paths
    def floyd_warshall(g):
        distance = {v: {u: float('inf') for u in g} for v in g}
        next_v = {v: {u: None for u in g} for v in g}
        
        for v in g:
            for n in v.get_neighbours():
                distance[v][n] = v.get_weight(n)
                next_v[v][n] = n
        
        for v in g:
            distance[v][v] = 0
            next_v[v][v] = None
        
        for p in g:
            for v in g:
                for w in g:
                    if distance[v][w] > distance[v][p] + distance[p][w]:
                        distance[v][w] = distance[v][p] + distance[p][w]
                        next_v[v][w] = next_v[v][p]
        
        return distance, next_v
    
    # Define a function to print the shortest path from one vertex to another
    def print_path(next_v, u, v):
        p = u
        path_list = []
        
        while next_v[p][v]:
            path_list.append(p.get_key())
            p = next_v[p][v]
        
        path_list.append(v.get_key())
        return path_list
    
    # Create an instance of the Graph class
    g = Graph()
    
    # Add vertices to the graph
    for i in range(num_of_dest):
        g.add_vertex(i + 1)
    
    # Add edges to the graph
    for i, j, k in edges_corrected:
        g.add_edge(i, j, k)
    
    # Apply the Floyd-Warshall algorithm to find shortest paths
    distance, next_v = floyd_warshall(g)
    
    # Initialize a list to store the shortest routes
    list_for_sketch = []
    
    # Iterate over all possible start and end vertices to find shortest paths
    for start in g:
        for end in g:
            if next_v[start][end]:
                path = print_path(next_v, start, end)
                list_for_sketch.append([[start.get_key(), end.get_key()], path, [distance[start][end]]])
    
    return list_for_sketch


if __name__ == "__main__":
    # Path to the text file containing edge data
    path_v1 = "path_to_your_text_file.txt"
    
    # Compute shortest routes and print the result
    shortest_routes = main(path_v1)
    print(shortest_routes)
