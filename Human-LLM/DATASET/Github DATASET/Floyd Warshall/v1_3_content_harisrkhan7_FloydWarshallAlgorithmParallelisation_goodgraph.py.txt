import random

# Define the list of number of vertices
num_vertices_list = [5, 10, 20, 50, 100, 200, 500, 750, 1000, 1250, 1500, 2000, 3000, 5000]

# Generate random graphs and save them to files
for num_vertices in num_vertices_list:
    # Generate a random adjacency matrix for the graph
    matrix = [[random.randint(1, 10) for _ in range(num_vertices)] for _ in range(num_vertices)]

    # Write the matrix to a file
    with open("randograph-{}.txt".format(num_vertices), "w") as f:
        # Write the number of vertices
        f.write("{}\n".format(num_vertices))
        
        # Write the edges with their weights
        for i in range(num_vertices):
            for j in [random.randint(0, num_vertices - 1) for _ in range(10)]:
                f.write("{} {} {}\n".format(i + 1, j + 1, matrix[i][j]))
