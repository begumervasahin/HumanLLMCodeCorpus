import numpy as np
import math

# Set options for numpy printing
np.set_printoptions(precision=2)
np.set_printoptions(suppress=True)

# Define infinity
infinity = math.inf

def get_adjacency_matrix():
    """
    Prompt the user to enter the adjacency matrix.

    Returns:
        tuple: A tuple containing the number of vertices and the adjacency matrix.      
    """
    num_vertices = int(input("Enter the number of vertices: "))
    print("Enter the values of the adjacency matrix row-wise:")
    adjacency_matrix = np.zeros((num_vertices, num_vertices))
    for i in range(num_vertices):
        for j in range(num_vertices):
            adjacency_matrix[i][j] = int(input())
    return num_vertices, adjacency_matrix

def floyd_warshall(adjacency_matrix, num_vertices):
    """
    Apply the Floyd-Warshall algorithm to find the shortest paths between all pairs of vertices.

    Args:
        adjacency_matrix (ndarray): The adjacency matrix representing the graph.
        num_vertices (int): The number of vertices in the graph.

    Returns:
        None. Prints the intermediate distance matrices.
    """
    print("\n\n")
    print("Initial Matrix (D[0]):\n ", adjacency_matrix)
    print("\n")

    for k in range(num_vertices):
        for i in range(num_vertices):
            for j in range(num_vertices):
                # Update the shortest path between vertices i and j through vertex k if applicable
                adjacency_matrix[i][j] = min(adjacency_matrix[i][j], adjacency_matrix[i][k] + adjacency_matrix[k][j])

        # Print the updated distance matrix after each iteration
        print(f"Matrix after iteration {k + 1} (D[{k + 1}]):\n ", adjacency_matrix)
        print("\n")

# Get the adjacency matrix from the user
num_vertices, adjacency_matrix = get_adjacency_matrix()

# Apply Floyd-Warshall algorithm
floyd_warshall(adjacency_matrix, num_vertices)
