def print_result(source, destination, path, distances):
    """
    Prints the shortest path and minimum cost from source to destination.
    
    Args:
    source (str): The source node.
    destination (str): The destination node.
    path (list): List containing the nodes in the shortest path.
    distances (dict): Dictionary containing the minimum distances to each node from the source.
    """
    print(f"{source} -> {destination}")
    final_path = " -> ".join(path)
    print("\nShortest path:\n", final_path)
    print(f"Minimum cost = {distances[destination]} km")

def dijkstra(graph, source, destination):
    """
    Finds the shortest path and minimum cost from source to destination using Dijkstra's algorithm.
    
    Args:
    graph (dict): The graph represented as a dictionary.
    source (str): The source node.
    destination (str): The destination node.
    
    Raises:
    ValueError: If source or destination node is not in the graph.
    """
    if source not in graph:
        raise ValueError('Source node is not in the graph')
    if destination not in graph:
        raise ValueError('Destination node is not in the graph')

    visited = []
    distances = {source: 0}
    predecessors = {}

    _dijkstra_recursive(graph, source, destination, visited, distances, predecessors)
    _print_result(source, destination, distances, predecessors)

def _dijkstra_recursive(graph, current_node, destination, visited, distances, predecessors):
    """
    Helper function for Dijkstra's algorithm to perform recursive traversal.
    
    Args:
    graph (dict): The graph represented as a dictionary.
    current_node (str): The current node being processed.
    destination (str): The destination node.
    visited (list): List of visited nodes.
    distances (dict): Dictionary containing the minimum distances to each node from the source.
    predecessors (dict): Dictionary containing predecessors of each node in the shortest path.
    """
    if current_node == destination:
        path = []
        pred = destination
        while pred is not None:
            path.append(pred)
            pred = predecessors.get(pred)
        return path[::-1]

    visited.append(current_node)

    for neighbor, weight in graph[current_node].items():
        if neighbor not in visited:
            new_distance = distances[current_node] + weight
            if new_distance < distances.get(neighbor, float('inf')):
                distances[neighbor] = new_distance
                predecessors[neighbor] = current_node

    unvisited = {k: distances.get(k, float('inf')) for k in graph if k not in visited}
    next_node = min(unvisited, key=unvisited.get)
    return _dijkstra_recursive(graph, next_node, destination, visited, distances, predecessors)

def _print_result(source, destination, distances, predecessors):
    """
    Helper function to print the result of Dijkstra's algorithm.
    
    Args:
    source (str): The source node.
    destination (str): The destination node.
    distances (dict): Dictionary containing the minimum distances to each node from the source.
    predecessors (dict): Dictionary containing predecessors of each node in the shortest path.
    """
    path = []
    pred = destination
    while pred is not None:
        path.append(pred)
        pred = predecessors.get(pred)
    print_result(source, destination, path[::-1], distances)
