from math import sqrt

# Function to calculate the Euclidean distance between two points
def calculate_distance(x1, y1, x2, y2):
    # Calculate squared differences in x and y coordinates
    delta_x_squared = (x1 - x2) ** 2
    delta_y_squared = (y1 - y2) ** 2
    
    # Calculate the distance by taking the square root of the sum of squared differences
    distance = sqrt(delta_x_squared + delta_y_squared)
    return distance

# Function to calculate heuristic values for each point based on the end point
def calculate_heuristic_values(end_point, coordinates):
    end_point_x, end_point_y = coordinates[end_point]  # Extract coordinates of the end point
    heuristic_values = []  # List to store heuristic values
    
    # Iterate over all points
    for point_x, point_y in coordinates:
        # Calculate Euclidean distance between the end point and current point
        distance = calculate_distance(end_point_x, end_point_y, point_x, point_y)
        heuristic_values.append(distance)  # Append the distance to the list
    return heuristic_values

# Function to calculate distances between connected points 
def calculate_connection_distances(coordinates, connection_list):   
    distances = []  # Initialize list for distances
    
    # Iterate over all points
    for i, connections in enumerate(connection_list):
        point_distances = []  # List to store distances from current point
        
        # Iterate over connections of each point
        for connection in connections:
            # Extract coordinates of connected points
            x1, y1 = coordinates[i]
            x2, y2 = coordinates[connection]
            
            # Calculate Euclidean distance between connected points
            distance = calculate_distance(x1, y1, x2, y2)  
            point_distances.append(distance)
        
        distances.append(point_distances)  # Append distances for current point to the main list
    return distances

# Example usage:
coordinates = [(0, 0), (3, 4), (6, 8)]
connection_list = [[1], [0, 2], [1]]
end_point = 0

# Print heuristic values and connection distances
print("Heuristic values:", calculate_heuristic_values(end_point, coordinates))
print("Connection distances:", calculate_connection_distances(coordinates, connection_list))
