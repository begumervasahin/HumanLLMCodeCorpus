from text import Text

class Conversation:
    def __init__(self, strings, position):
        self.position = position
        self.dialogue = self._split_strings_into_parts(strings)
        self.read = False
        self.current = 0

    def _split_strings_into_parts(self, strings):
        dialogue_parts = []
        current_part = []
        count = 0

        # Split strings into parts of 3 or less
        for string in strings:
            current_part.append(Text(string, scrollable=True))
            count += 1

            # Check if the part is complete or it's the last string
            if count >= 3 or string == strings[-1]:
                missing_texts = 3 - len(current_part)
                for _ in range(missing_texts):
                    current_part.append(Text('', scrollable=True))
                dialogue_parts.append(current_part)
                count = 0
                current_part = []

        return dialogue_parts

    def draw(self, surface, position=(0, 0)):
        count = 0

        # Draw the dialogue parts
        if self.dialogue:
            for text in self.dialogue[self.current]:
                text.draw(surface, (position[0], position[1] + count * 16))

                # Break if the text is not fully shown
                if text.current < len(text.text):
                    break
                else:
                    count += 1
