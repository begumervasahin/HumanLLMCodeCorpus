import collections

class Graph:
    def __init__(self):
        self.nodes = set()
        self.edges = collections.defaultdict(list)
        self.distances = {}

    def add_node(self, node):
        """
        Add a node to the graph.

        Args:
            node: The node to be added.
        """
        self.nodes.add(node)

    def add_edge(self, from_node, to_node, distance):
        """
        Add an edge between two nodes with a given distance.

        Args:
            from_node: The starting node of the edge.
            to_node: The ending node of the edge.
            distance: The distance between the nodes.
        """
        self.edges[from_node].append(to_node)
        self.edges[to_node].append(from_node)
        self.distances[(from_node, to_node)] = distance
        self.distances[(to_node, from_node)] = distance

def dijkstra(graph, start_node):
    """
    Perform Dijkstra's algorithm to find the shortest paths from a starting node to all other nodes.

    Args:
        graph (Graph): The graph object containing nodes, edges, and distances.
        start_node: The starting node for the algorithm.

    Returns:
        dict: Shortest distances from the starting node to all other nodes.
        dict: Predecessors of nodes in the shortest paths.
    """
    visited = {start_node: 0}  # Stores the shortest distances from the start node
    path = {}  # Stores the predecessors of nodes in the shortest paths
    unvisited_nodes = set(graph.nodes)  # Set of unvisited nodes

    while unvisited_nodes:
        max_node = None

        # Find the unvisited node with the maximum distance
        for node in unvisited_nodes:
            if node in visited:
                if max_node is None or visited[node] > visited[max_node]:
                    max_node = node

        # If all nodes are visited, exit the loop
        if max_node is None:
            break

        unvisited_nodes.remove(max_node)
        current_weight = visited[max_node]

        # Update distances and paths for neighbors of the current node
        for neighbor in graph.edges[max_node]:
            weight = current_weight + graph.distances[(max_node, neighbor)]
            if neighbor not in visited or weight > visited[neighbor]:
                visited[neighbor] = weight
                path[neighbor] = max_node

    return visited, path
