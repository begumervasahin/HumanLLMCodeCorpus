import networkx as nx
import matplotlib.pyplot as plt

def draw_graph_with_path(edges, path):
    """
    Draw a directed graph with weighted edges and highlight a specified path.

    Args:
        edges (list): List of tuples representing weighted edges [(source, target, weight), ...].
        path (list): List of tuples representing the path to highlight [(source, target), ...].
    """
    # Create a directed graph
    G = nx.DiGraph()

    # Add weighted edges to the graph
    for edge in edges:
        source, target, weight = edge
        G.add_weighted_edges_from([(source, target, weight)])

    # Define node colors
    node_values = {1: 1.0, 4: 0.5714285714285714, 8: 0.0}  # Example values
    node_colors = [node_values.get(node, 0.25) for node in G.nodes()]

    # Create a layout for the nodes
    pos = nx.spring_layout(G)

    # Draw nodes
    nx.draw_networkx_nodes(G, pos, cmap=plt.get_cmap('jet'), node_color=node_colors, node_size=500)

    # Draw edge labels
    edge_labels = {(u, v): d['weight'] for u, v, d in G.edges(data=True)}
    nx.draw_networkx_edge_labels(G, pos, edge_labels=edge_labels)

    # Draw node labels
    nx.draw_networkx_labels(G, pos)

    # Define edge colors
    red_edges = path
    edge_colors = ['black' if edge not in red_edges else 'red' for edge in G.edges()]

    # Draw edges
    nx.draw_networkx_edges(G, pos, edgelist=red_edges, edge_color='r', arrows=True)
    nx.draw_networkx_edges(G, pos, edgelist=[edge for edge in G.edges() if edge not in red_edges], arrows=False)

    # Show the plot
    plt.show()

# Example usage:
# edges = [('A', 'B', 1), ('B', 'C', 2), ('C', 'D', 3), ('D', 'E', 4)]
