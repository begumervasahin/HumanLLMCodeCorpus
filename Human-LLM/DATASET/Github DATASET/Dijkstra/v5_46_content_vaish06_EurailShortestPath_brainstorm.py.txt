from graph import Graph
from shortest_paths import shortest_path_lengths, shortest_path_tree
from linked_stack import LinkedStack

# Create a new instance of the Graph class to represent the rail system      
rail_system = Graph()

# Open the file containing rail routes
with open('eurail.txt', 'r') as eurail:
    cities = {}  # Dictionary to store cities
    routes = set()  # Set to store unique routes

    # Read each line in the file
    for line in eurail.readlines():
        # Extract source, destination, and duration information from each line
        source, destination, duration = line.split(',')
        hours, minutes = map(int, duration.split(':'))

        # Convert duration to minutes
        duration_minutes = hours * 60 + minutes

        # Add the route to the set of routes
        routes.add((source, destination, duration_minutes))     

        # Add cities to the cities dictionary if not already present
        cities.setdefault(source)
        cities.setdefault(destination)

    # Create vertices for each city and add them to the graph   
    for city in cities:
        cities[city] = rail_system.insert_vertex(city)

    # Insert edges for each route in the set of routes
    for route in routes:
        source, destination, duration = route
        rail_system.insert_edge(cities[source], cities[destination], duration)

# Find shortest paths from London to other cities
durations = shortest_path_lengths(rail_system, cities['London'])
paths = shortest_path_tree(rail_system, cities['London'], durations)

# Print shortest paths
print(paths)
