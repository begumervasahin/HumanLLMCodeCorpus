class Empty(Exception):
    """Exception raised when trying to perform an operation on an empty priority queue."""
    pass

class Item:
    """Represents an item with a key-value pair."""
    def __init__(self, key, value):
        self._key = key
        self._value = value

class PriorityQueueBase:
    """Abstract base class for priority queues."""
    def is_empty(self):
        """Checks if the priority queue is empty."""
        return len(self) == 0

class HeapPriorityQueue(PriorityQueueBase):
    """A min-oriented priority queue implemented with a binary heap."""
    def _parent(self, j):
        """Returns the index of the parent node."""
        return (j - 1) // 2

    def _left(self, j):
        """Returns the index of the left child node."""
        return 2 * j + 1

    def _right(self, j):
        """Returns the index of the right child node."""
        return 2 * j + 2

    def _has_left(self, j):
        """Checks if the node has a left child."""
        return self._left(j) < len(self._data)

    def _has_right(self, j):
        """Checks if the node has a right child."""
        return self._right(j) < len(self._data)

    def _swap(self, i, j):
        """Swaps elements at indices i and j."""
        self._data[i], self._data[j] = self._data[j], self._data[i]

    def _upheap(self, j):
        """Moves the node up to restore heap order."""
        parent = self._parent(j)
        if j > 0 and self._data[j]._key < self._data[parent]._key:
            self._swap(j, parent)
            self._upheap(parent)

    def _downheap(self, j):
        """Moves the node down to restore heap order."""
        if self._has_left(j):
            left = self._left(j)
            small_child = left
            if self._has_right(j):
                right = self._right(j)
                if self._data[right]._key < self._data[left]._key:
                    small_child = right
            if self._data[small_child]._key < self._data[j]._key:
                self._swap(j, small_child)
                self._downheap(small_child)

    def __init__(self):
        """Creates an empty Priority Queue."""
        self._data = []

    def __len__(self):
        """Returns the number of items in the priority queue."""
        return len(self._data)

    def add(self, key, value):
        """Adds a key-value pair to the priority queue."""
        self._data.append(Item(key, value))
        self._upheap(len(self._data) - 1)

    def min(self):
        """Returns the (key, value) tuple with the minimum key."""
        if self.is_empty():
            raise Empty('Priority queue is empty.')
        item = self._data[0]
        return (item._key, item._value)

    def remove_min(self):
        """Removes and returns the (key, value) tuple with the minimum key."""
        if self.is_empty():
            raise Empty('Priority queue is empty.')
        self._swap(0, len(self._data) - 1)
        item = self._data.pop()
        self._downheap(0)
        return (item._key, item._value)

# Example usage
pq = HeapPriorityQueue()
pq.add(4, 'Four')
pq.add(2, 'Two')
pq.add(1, 'One')
print(pq.min())  # Output: (1, 'One')
print(pq.remove_min())  # Output: (1, 'One')
print(len(pq))  # Output: 2
