from random import randint
from time import perf_counter
from multiprocessing import Process, JoinableQueue, current_process
from progressbar import ProgressBar
from dijkstra import Graph, dijkstra

# Function to generate a random graph
def generate_random_graph(nodes, edges_per_node, max_weight):
    graph = Graph()
    for node_index in range(1, nodes + 1):
        for _ in range(edges_per_node):
            weight = randint(1, max_weight)
            neighbour = node_index
            while neighbour == node_index:
                neighbour = randint(1, nodes)  # Ensure the neighbor is different from the current node
            graph.add_edge_with_cost(node_index, neighbour, weight)
    return graph

# Function to perform tests on randomly generated graphs
def test(queue, output):
    start = 1
    edges_per_node = 6
    max_weight = 100
    pid = current_process().pid

    # Loop until there are no more tasks in the queue
    while True:
        task = queue.get()
        if not task:
            break
        graph_size, graph_index = task
        end = graph_size

        # Generate a random graph and run Dijkstra's algorithm
        while True:
            graph = generate_random_graph(graph_size, edges_per_node, max_weight)
            try:
                start_time = perf_counter()
                moves, weights, steps = dijkstra(graph, start)
                end_time = perf_counter()
            except Exception as e:
                print(f"Exception occurred in PID {pid}: {e}")
                continue

            # Calculate elapsed time, shortest path, and its length
            elapsed = end_time - start_time
            path = []
            length = weights[end]
            current = end
            complete = False
            while current:
                path.append(current)
                if current == start:
                    complete = True
                    break
                current = moves[current]

            # If the path is incomplete, skip this test
            if not complete:
                continue

            # Prepare the result and put it into the output queue
            row = f"{graph_size},{graph_index},{elapsed},{steps},{length},{len(path)}"
            output.put(row)
            break

        # Mark the task as done
        queue.task_done()

    # Print a message when the process finishes
    print(f"PID {pid}: done")

# Function to write results to a CSV file
def writer(output):
    with open("results.csv", "w") as f, ProgressBar(max_value=5000) as bar:
        f.write("Graph size,Test index,Elapsed time,Steps,Path length,Path nodes\n")
        task_count = 0
        # Loop until there are no more results in the output queue
        while True:
            row = output.get()
            if not row:
                break
            f.write(row + "\n")
            task_count += 1
            bar.update(task_count)
            output.task_done()

# Main function to coordinate the processes
def main():
    queue = JoinableQueue()  # Queue for tasks
    output = JoinableQueue()  # Queue for results
    num_processes = 8
    processes = [Process(target=test, args=(queue, output)) for _ in range(num_processes)]  # Test processes
    write_proc = Process(target=writer, args=(output,))  # Writer process

    # Start test processes and the writer process
    for p in processes:
        p.start()
    write_proc.start()

    # Add tasks to the queue
    for size in range(100, 5100, 100):
        for index in range(100):  # Changed index range
            queue.put((size, index))

    # Wait for all tasks to be processed
    queue.join()

    # Add termination signals to the queue for test processes and the writer process
    for _ in range(num_processes):
        queue.put(None)
    output.put(None)

    # Wait for all processes to finish
    for p in processes:
        p.join()

# Entry point of the script
if __name__ == "__main__":
    main()
