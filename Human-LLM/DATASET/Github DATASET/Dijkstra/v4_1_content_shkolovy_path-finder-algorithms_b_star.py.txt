from queue import PriorityQueue
import grid_helper as gh
from pprint import pprint

def find_path_greedy(grid, start, end):
    """Find the path using the Greedy Best First Search algorithm."""
    priority_queue = PriorityQueue()
    priority_queue.put((0, start))
    came_from = {start: None}

    while not priority_queue.empty():
        current_pos = priority_queue.get()[1]
        if current_pos == end:
            break

        neighbors = gh.get_neighbors(grid, current_pos[0], current_pos[1])
        for neighbor in neighbors:
            if neighbor not in came_from:
                priority = gh.heuristic_distance(neighbor, end, type="manhattan")
                priority_queue.put((priority, neighbor))
                came_from[neighbor] = current_pos

    return came_from

def initialize_and_run_search():
    """Initialize the grid and run the Greedy Best First Search algorithm."""
    initial_grid = gh.generate_grid_empty()
    start, end = (0, 0), (8, 8)
    came_from = find_path_greedy(initial_grid, start, end)
    path = gh.find_path(start, end, came_from)
    gh.draw_path(path, initial_grid)
    pprint(initial_grid)

if __name__ == "__main__":
    initialize_and_run_search()
