import math

# Constants for grid elements
START_COL = "S"
END_COL = "E"
VISITED_COL = "x"
OBSTACLE_COL = "O"
PATH_COL = "@"

def generate_empty_grid(rows, cols):
    """Generate an empty grid with specified dimensions."""
    return [["." for _ in range(cols)] for _ in range(rows)]

def calculate_heuristic_distance(pos, end_pos, type="euclidean"):
    """Calculate the heuristic distance between two positions."""
    dx = abs(pos[0] - end_pos[0])
    dy = abs(pos[1] - end_pos[1])
    if type == "manhattan":
        return dx + dy  # Manhattan distance
    return math.sqrt(dx * dx + dy * dy)  # Euclidean distance

def find_path(start, end, came_from):
    """Reconstruct the shortest path from start to end point based on the 'came_from' dictionary."""
    path = [end]
    current = end
    while current != start:
        current = came_from[current]
        path.append(current)
    path.reverse()
    return path

def get_cell_cost(grid, pos):
    """Get the cost of a cell in the grid."""
    col_val = grid[pos[0]][pos[1]]
    return int(col_val) if col_val.isdigit() else 1

def get_valid_neighbors(grid, row, col):
    """Get valid neighboring cells in the grid."""
    height = len(grid)
    width = len(grid[0])
    neighbors = [(row + 1, col), (row, col - 1), (row - 1, col), (row, col + 1)]        
    if (row + col) % 2 == 0:
        neighbors.reverse()
    neighbors = filter(lambda t: (0 <= t[0] < height and 0 <= t[1] < width), neighbors) 
    neighbors = filter(lambda t: (grid[t[0]][t[1]] != OBSTACLE_COL), neighbors)
    return neighbors

def mark_path_on_grid(path, grid):
    """Mark the path on the grid."""
    for row, col in path:
        grid[row][col] = PATH_COL
    start_pos = path[0]
    end_pos = path[-1]
    grid[start_pos[0]][start_pos[1]] = START_COL
    grid[end_pos[0]][end_pos[1]] = END_COL
    return grid
