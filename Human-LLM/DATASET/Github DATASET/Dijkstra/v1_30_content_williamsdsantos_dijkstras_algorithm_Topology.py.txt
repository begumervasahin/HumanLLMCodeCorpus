class Topology:
    def __init__(self, filename):
        self.Nodes = []
        self.Links = []
        with open(filename, 'r') as TopologyData:
            next(TopologyData)  # Skip the first three lines (headers)
            for line in TopologyData:
                data = line.split()
                if len(data) == 0:  # Check for empty lines
                    break
                node_id = data[2]
                self.Nodes.append(node_id)
            next(TopologyData)  # Skip the next three lines (headers)
            for line in TopologyData:
                data = line.split()
                if len(data) == 0:  # Check for empty lines
                    break
                origin, destination, length = data[2:5]
                self.Links.append([origin, destination, length])

# Usage example:
topology = Topology('data/topologies/PacificBell')
print("Nodes:", topology.Nodes)
print("Links:", topology.Links)
