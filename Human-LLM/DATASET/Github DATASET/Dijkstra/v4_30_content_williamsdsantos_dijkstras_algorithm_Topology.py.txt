class Topology:
    def __init__(self):
        self.nodes = []  # List to store node IDs
        self.links = []  # List to store link information
        
        with open('data/topologies/PacificBell', 'r') as topology_data:
            # Skip the first three header lines
            for _ in range(3):
                next(topology_data)

            # Read node data
            for line in topology_data:
                data = line.split()
                if not data:  # Check for empty lines
                    break
                try:
                    node_id = data[2]
                    self.nodes.append(node_id)
                except IndexError:
                    break  # Stop reading if unexpected data format is encountered

            # Skip the next three header lines
            for _ in range(3):
                next(topology_data)

            # Read link data
            for line in topology_data:
                data = line.split()
                if not data:  # Check for empty lines
                    break
                try:
                    origin = data[2]
                    destination = data[3]
                    length = data[4]
                    elem_link = [origin, destination, length]
                    self.links.append(elem_link)
                except IndexError:
                    break  # Stop reading if unexpected data format is encountered
