import os
from sqlalchemy import create_engine, Column, String, Integer
from sqlalchemy.ext.declarative import declarative_base
from sqlalchemy.orm import sessionmaker
from sqlalchemy_utils import database_exists, create_database

# Define the base class for the ORM
Base = declarative_base()

class DB:
    """Class to manage the database connection and operations."""

    def __init__(self):
        """Initialize the database connection and session."""
        # Connect to the database using the DATABASE_URL environment variable
        self.engine = create_engine(os.getenv("DATABASE_URL"))
        # Create a session class
        Session = sessionmaker(bind=self.engine)
        # Create a session instance
        self.session = Session()

    def add(self, user):
        """Add or update a user in the database."""
        # Check if the user already exists in the database
        existing_user = self.session.query(User).filter(User.email == user.email).first()
        if existing_user:
            # Update the existing user's name
            existing_user.name = user.name
            # Add the existing user to the session
            self.session.add(existing_user)
        else:
            # Add the new user to the session
            self.session.add(user)
        # Commit the changes to the database
        self.session.commit()

class User(Base):
    """Class representing the 'users' table in the database."""

    __tablename__ = "users"
    id = Column(Integer, primary_key=True)
    name = Column(String(50))
    email = Column(String(50), unique=True)

if __name__ == "__main__":
    # Create an instance of the DB class
    db = DB()
    # Check if the database exists, if not, create it
    if not database_exists(db.engine.url):
        create_database(db.engine.url)
    # Create the tables defined in the Base class
    Base.metadata.create_all(db.engine)
