import json
import logging
import base64
import datetime
import os
from typing import List, Dict

# Mocking boto3 client and types for demonstration
class Client:
    def invoke(self, FunctionName, Payload):
        return {'FunctionError': None, 'Payload': b'Success'}

# Mocking boto3.set_stream_logger
def set_stream_logger(_, level):
    logging.basicConfig(level=level)

# Initialize AWS X-Ray
from aws_xray_sdk.core import patch_all, xray_recorder
patch_all()
set_stream_logger('', logging.INFO)

@xray_recorder.capture('invoke')
def handler(event, context):
    # Initialize Lambda client
    lambda_client = Client()

    # Extract relevant information from the event
    records: List[Dict] = event.get('Records')
    sqs_arn: str = records[0].get('eventSourceARN')
    source_queue = sqs_arn.split(':')[5]
    source_environment = source_queue.rsplit('-', 1)[1]

    # Generate timestamp for the attachment name
    current_time = f"{datetime.datetime.utcnow().isoformat(timespec='minutes')}Z"

    # Prepare payload for the notification Lambda function
    payload = {
        'message_type': 'email',
        'to': os.getenv('TO_EMAIL'),
        'template_id': os.getenv('TEMPLATE_ID'),
        'template_vars': {
            'source_queue': source_queue,
            'source_environment': source_environment
        },
        'attachment': base64.b64encode(json.dumps(records, indent=2).encode('utf-8', errors='strict')).decode("utf-8", "strict"),
        'attachment_name': f'{source_queue}_dead_messages_{current_time}.json'
    }

    # Invoke the notification Lambda function
    lambda_name = os.getenv('NOTIFY_LAMBDA_NAME')
    resp = lambda_client.invoke(FunctionName=lambda_name, Payload=json.dumps(payload))

    # Check if the invocation was successful
    if resp.get('FunctionError') is not None:
        raise RuntimeError(f'{lambda_name} failed to notify with {payload}')
    else:
        return resp['Payload'].read()

if __name__ == '__main__':
    import argparse
    from pathlib import Path

    # Parse command line arguments
    parser = argparse.ArgumentParser()
    parser.add_argument('--event', help='Path to event json file', type=Path)
    args = parser.parse_args()

    # Load event data from file
    with open(args.event) as event_file:
        evt = json.load(event_file)

    # Execute the Lambda handler function with the loaded event data
    handler(evt, {})
