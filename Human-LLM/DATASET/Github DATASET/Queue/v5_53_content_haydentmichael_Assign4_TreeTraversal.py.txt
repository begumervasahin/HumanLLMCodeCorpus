class TreeNode:
    def __init__(self, x):
        self.val = x
        self.left = None
        self.right = None

def inorderTraversal(root):
    """
    Perform an inorder traversal of the binary tree starting from the root node.

    Args:
    root (TreeNode): The root node of the binary tree.

    Returns:
    list: List containing the values in the inorder traversal order.
    """
    if root is None:
        return []
    return inorderTraversal(root.left) + [root.val] + inorderTraversal(root.right)

def preorderTraversal(root):
    """
    Perform a preorder traversal of the binary tree starting from the root node.

    Args:
    root (TreeNode): The root node of the binary tree.

    Returns:
    list: List containing the values in the preorder traversal order.
    """
    if root is None:
        return []
    return [root.val] + preorderTraversal(root.left) + preorderTraversal(root.right)

def populateTree(arr, i, n):
    """
    Populate a binary tree from the given array representation.

    Args:
    arr (list): Array representing the binary tree.
    i (int): Index of the current node.
    n (int): Length of the array.

    Returns:
    TreeNode: The root node of the populated binary tree.
    """
    root = None
    if i < n:
        if arr[i] is not None:
            root = TreeNode(arr[i])
            root.left = populateTree(arr, 2 * i + 1, n)
            root.right = populateTree(arr, 2 * i + 2, n)
    return root

def print_traversal_results(traversal_type, root):
    """
    Print the traversal results of a binary tree.

    Args:
    traversal_type (str): Type of traversal (e.g., "In-order", "Pre-order").
    root (TreeNode): The root node of the binary tree.
    """
    print(f"{traversal_type} Traversal")
    for x in traversal_type.lower() + "Traversal"(root):
        print(x)
    print()

# Test cases
arr1 = [1, None, 2, None, None, 3]
n = len(arr1)
root2 = populateTree(arr1, 0, n)
print_traversal_results("In-order", root2)
