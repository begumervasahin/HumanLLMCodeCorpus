import logging
from collections import deque
from queue import Queue

import slack
from client_helpers import fetch_image_url_for_user_id

# Set logging level to INFO
logging.getLogger().setLevel(logging.INFO)


class QueueBot:
    # Define a constant for the line block structure
    LINE_BLOCK = {
        "type": "section",
        "text": {
            "type": "mrkdwn",
            "text": (
                "Welcome to queueBot! :wave: \n\n"
                "*Here are the people in the line:*"
            ),
        },
        "accessory": {
            "type": "button",
            "text": {
                "type": "plain_text",
                "text": "Line Up"
            },
            "action_id": "action_id",
            "value": "view_alternate_2"
        }
    }

    def __init__(self, channel, name):
        # Initialize QueueBot with channel and name
        self.channel = channel
        self.name = name
        self.username = "queueBot"
        self.icon_emoji = ":robot_face:"
        self.timestamp = ""
        self.queue = deque()  # Initialize a deque to store the queue

    def get_message_payload(self):
        # Return the message payload to be sent to Slack
        return {
            "ts": self.timestamp,
            "channel": self.channel,
            "username": self.username,
            "icon_emoji": self.icon_emoji,
            "blocks": [
                self.LINE_BLOCK,
                self._get_queue_block()  # Get the block representing the current queue
            ],
        }

    def _get_queue_block(self):
        # Get the block representing the current queue
        elements = [user_to_block_kit_element(user) for user in self.queue]
        if not elements:
            # If queue is empty, display a message
            return {
                "type": "section",
                "text": {
                    "type": "mrkdwn",
                    "text": "There is no one in this queue yet! Click the \"Line Up\" button to be the first in the line!"
                }
            }
        # Insert numbers before each user
        elements = [{"type": "mrkdwn", "text": f":{i + 1}: : "} for i in range(len(elements))] + elements
        return {
            "type": "context",
            "elements": elements
        }

    def insert_queue(self, user):
        # Insert a user into the queue
        if user in self.queue:
            logging.info('Error: User is already in the queue.')
            return
        self.queue.append(user)


def user_to_block_kit_element(user):
    # Convert user data to Block Kit element
    image_url = fetch_image_url_for_user_id(user["id"])
    return {
        "type": "image",
        "image_url": image_url,
        "alt_text": user["name"]
    }
