class Node:
    def __init__(self, value):
        self.value = value
        self.next = None

class LinkedList:
    def __init__(self):
        self.head = None

    def append(self, value):
        if self.head is None:
            self.head = Node(value)
            return
        current_node = self.head
        while current_node.next:
            current_node = current_node.next
        current_node.next = Node(value)

    def __iter__(self):
        current_node = self.head
        while current_node:
            yield current_node.value
            current_node = current_node.next

    def __repr__(self):
        values = [value for value in self]
        return str(values)

def reverse(linked_list):
    """
    Reverse the input linked list
    Args:
       linked_list(obj): Linked List to be reversed
    Returns:
       obj: Reversed Linked List
    """
    reversed_llist = LinkedList()
    current_node = None
    for value in linked_list:
        if current_node is None:
            current_node = Node(value)
        else:
            new_node = Node(value)
            new_node.next = current_node
            current_node = new_node
    reversed_llist.head = current_node
    return reversed_llist

# Test the reverse function
llist = LinkedList()
for value in [4, 2, 5, 1, -3, 0]:
    llist.append(value)

flipped = reverse(llist)
is_correct = list(flipped) == [0, -3, 1, 5, 2, 4] and list(llist) == list(reverse(flipped))
print("Pass" if is_correct else "Fail")
