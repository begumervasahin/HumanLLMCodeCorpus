from __future__ import print_function
import pickle
import os.path
from googleapiclient.discovery import build
from google_auth_oauthlib.flow import InstalledAppFlow
from google.auth.transport.requests import Request
from printRequest import PrintRequest

# Define lists to hold requests in different states
in_progress = []
todo = []
finished = []
failed = []
requests = [todo, in_progress, finished, failed]
options = ['TODO', 'IN PROGRESS', 'FINISHED', 'FAILED']  # Corresponding options for request states
SCOPES = ['https://www.googleapis.com/auth/spreadsheets.readonly']

def main():
    """Main function to fetch data from Google Sheets and process requests."""
    # Load credentials from token file or authenticate user
    creds = load_or_authenticate_credentials()

    # Connect to Google Sheets API
    service = build('sheets', 'v4', credentials=creds)
    sheet = service.spreadsheets()

    # Get values from spreadsheet
    values = get_spreadsheet_values(sheet)

    # Process values and categorize requests
    if values:
        make_request_sets(values)
    else:
        print('No data found in the spreadsheet.')

def load_or_authenticate_credentials():
    """Load existing credentials or authenticate user to obtain new credentials."""
    creds = None
    if os.path.exists('token.pickle'):
        with open('token.pickle', 'rb') as token:
            creds = pickle.load(token)
    if not creds or not creds.valid:
        creds = authenticate_user()
        save_credentials(creds)
    return creds

def authenticate_user():
    """Authenticate user and obtain credentials."""
    flow = InstalledAppFlow.from_client_secrets_file('credentials.json', SCOPES)
    return flow.run_local_server()

def save_credentials(creds):
    """Save obtained credentials to a token file."""
    with open('token.pickle', 'wb') as token:
        pickle.dump(creds, token)

def get_spreadsheet_values(sheet):
    """Retrieve values from the spreadsheet."""
    result = sheet.values().get(spreadsheetId='1EEsBJlw6jhqebXVfwkq1No_FUnWRGejBe7l7Dnier24',
                                range='Form Responses 1').execute()
    return result.get('values', [])

def make_request_sets(values):
    """Categorize requests into different sets based on their status."""
    for row_num, row in enumerate(values, start=1):
        new_request = PrintRequest(row, row_num)
        if new_request.status == '0':
            todo.append(new_request)
        elif new_request.status == '1':
            in_progress.append(new_request)
        elif new_request.status == '2':
            finished.append(new_request)
        elif new_request.status == '3':
            failed.append(new_request)
    show_request_lists()

def show_request_lists():
    """Display options and request lists."""
    while True:
        print_options()
        choice = get_user_choice()
        if choice == 'q':
            quit_program()
        elif choice == 'm':
            continue
        print_list(requests[choice])

def print_options():
    """Display request state options."""
    print('Options:')
    for idx, option in enumerate(options):
        print(f'{option}[{idx}]')

def get_user_choice():
    """Get user choice for request state."""
    print("Input selection: (q) to quit (m) to return to menu")
    user_input = input()
    if user_input.lower() == 'q':
        return 'q'
    elif user_input.lower() == 'm':
        return 'm'
    else:
        return int(user_input)
