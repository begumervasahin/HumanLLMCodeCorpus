import random
import time
import threading

# Test sites and crawl statuses
test_sites = ['https://api.dev.proxycrawler.com', 'http://gatherproxy.com', 'http://foo.com', 'http://bar.com', 'http://baz.com', 'http://google.com', 'http://bing.com']
crawl_statuses = [True, False]

# Dictionary to track successful and failed crawls
successful = {site: 0 for site in test_sites}
failures = {site: 0 for site in test_sites}

# Function to display scoreboard of successes and failures
def scoreboard():
    print("--------------------------------------")
    print("Successes:")
    print(successful)
    print("--------------------------------------")
    print("Failures:")
    print(failures)
    print("--------------------------------------")

# Function to get the number of running threads
def get_running_threads():
    running = sum(1 for t in threading.enumerate() if t.is_alive())
    return running

# Worker function to perform crawling
def worker():
    print(threading.currentThread().getName(), "starting")
    time.sleep(random.randint(1, 10))
    for _ in range(5):
        url = random.choice(test_sites)
        print(threading.currentThread().getName(), "crawling %s" % url)
        success = random.choice(crawl_statuses)
        print(threading.currentThread().getName(), "crawl success=%s" % success)
        if success:
            successful[url] += 1
        else:
            failures[url] += 1
        time.sleep(random.randint(1, 12))
    print(threading.currentThread().getName(), "stopping")

# Function to create worker threads
def make_workers():
    workers = []
    for i in range(5):
        worker_name = "worker_%s" % i
        wkr = threading.Thread(name=worker_name, target=worker)
        workers.append(wkr)
    return workers

# Create worker threads
workers = make_workers()

# Daemon function to manage workers
def daemon():
    print(threading.currentThread().getName(), 'Starting daemon.')
    for w in workers:
        w.start()
    time.sleep(15)
    while True:
        scoreboard()
        if get_running_threads() == 1:
            break
        time.sleep(5)
    print(threading.currentThread().getName(), 'Stopping daemon.')

# Create and start daemon thread
dmn = threading.Thread(name='daemon', target=daemon)
dmn.daemon = True
dmn.start()
