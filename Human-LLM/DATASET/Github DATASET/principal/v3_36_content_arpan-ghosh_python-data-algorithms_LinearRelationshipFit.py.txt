import pandas as pd
import matplotlib.pyplot as plt
import numpy as np
from scipy.stats import linregress

def load_datasets(file_paths):
    """Load datasets from file paths"""
    datasets = []
    for file_path in file_paths:
        dataset = pd.read_csv(file_path, names=['x', 'y'], header=None)
        datasets.append(dataset)
    return datasets

def plot_analysis(dataset, filename):
    """Plot and analyze dataset"""
    # Create a figure for plotting
    fig, axes = plt.subplots(1, 2, figsize=(12, 4))
    
    # Scatter plot
    axes[0].scatter(dataset['x'], dataset['y'])
    axes[0].set_xlabel('x')
    axes[0].set_ylabel('y')
    axes[0].set_title(f'Scatter Plot of {filename}')
    
    # Polynomial fit
    axes[1].scatter(dataset['x'], dataset['y'])
    for degree, style in zip([1, 2, 3], ['r-', 'b--', 'm:']):
        coeffs = np.polyfit(dataset['x'], dataset['y'], degree)
        fit_values = np.polyval(coeffs, dataset['x'])
        axes[1].plot(dataset['x'], fit_values, style, label=f'Degree {degree} Fit')
    axes[1].set_xlabel('x')
    axes[1].set_ylabel('y')
    axes[1].set_title(f'Polynomial Fits of {filename}')
    axes[1].legend()
    
    # Compute R-squared using numpy
    residuals = dataset['y'] - np.polyval(coeffs, dataset['x'])
    SS_resid = sum(residuals ** 2)
    SS_total = len(dataset['y']) * np.var(dataset['y'])
    r_squared = 1 - SS_resid / SS_total
    print(f'R-squared (computed with numpy) for {filename}: {r_squared}')
    
    # Compute R-squared using scipy
    slope, intercept, r_value, p_value, std_err = linregress(dataset['x'], dataset['y'])
    print(f'R-squared (computed with scipy) for {filename}: {r_value ** 2}')
    print(f'p-value for {filename}: {p_value}')
    
    plt.show()

# Define file paths for datasets
data_files = [
    '/home/idies/workspace/AS.171.205/data/bfit.csv',
    '/home/idies/workspace/AS.171.205/data/cfit.csv',
    '/home/idies/workspace/AS.171.205/data/dfit.csv',

]

# Load datasets
datasets = load_datasets(data_files)

# Plot and analyze each dataset
for dataset, filename in zip(datasets, data_files):
    plot_analysis(dataset, filename)
