import numpy as np
import pandas as pd
from sklearn.decomposition import PCA
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from openpyxl import load_workbook

# Load and preprocess the data
def load_and_preprocess_data(file_path, sheet_name='Itraxx'):
    df = pd.read_excel(file_path, sheet_name=sheet_name)
    df['Date'] = pd.to_datetime(df['Date']).dt.strftime('%Y-%m-%d')
    return df

def reorder_columns(df):
    column_order = [
        'S26_3Y', 'S24_5Y', 'S28_3Y', 'S30_3Y', 'S26_5Y', 'S31_3Y', 'S24_7Y', 'S28_5Y', 'S26_7Y',
        'S30_5Y', 'S31_5Y', 'S28_7Y', 'S24_10Y', 'S30_7Y', 'S31_7Y', 'S26_10Y', 'S28_10Y', 'S30_10Y', 'S31_10Y'
    ]
    df = df[column_order]
    return df

def calculate_differences(df):
    differences = df.diff().dropna()
    return differences

def calculate_multiplier(differences):
    new_date = differences.index
    multiplier = np.zeros(len(new_date) - 1)
    for i in range(len(new_date) - 1):
        delta = (new_date[i] - new_date[i + 1]).days
        multiplier[i] = 1 / (np.sqrt(((delta) / 365) * 260)))
    return differences.mul(multiplier, axis=0)

# Rolling PCA
def rolling_pca(differences, window_size=20):
    residuals = pd.DataFrame()
    for i in range(len(differences) - window_size):
        window_data = differences.iloc[i:i + window_size]
        window_scaled = StandardScaler(with_std=False).fit_transform(window_data)
        pca = PCA(n_components=3)
        pca_scores = pca.fit_transform(window_scaled)
        loadings = pca.components_
        projected_data = pca.inverse_transform(pca_scores) + window_data.mean().values
        residual = (window_data - projected_data).iloc[0]
        residuals = residuals.append(residual, ignore_index=True)
    return residuals

# Export results to Excel
def export_to_excel(df, diff_df, residuals, output_file='PCA_ItraxxMain.xlsx'):
    with pd.ExcelWriter(output_file, engine='openpyxl') as writer:
        df.to_excel(writer, sheet_name='Original_Data', index=True, header=True)
        diff_df.to_excel(writer, sheet_name='DailyChange', index=True, header=True)
        residuals.to_excel(writer, sheet_name='Residuals', index=True, header=True)

# Plotting
def plot_residuals(residuals):
    for column in residuals.columns:
        plt.plot(residuals.index, residuals[column], label=column)
    plt.legend(loc='upper left', bbox_to_anchor=(1, 0.6))
    plt.xticks(np.arange(0, len(residuals), step=3))
    plt.xlabel('Date')
    plt.ylabel('Residual')
    plt.gcf().autofmt_xdate()
    plt.show()

# Main function
def main():
    file_path = 'Itraxxdata_22Mar19_04Jun19.xlsx'
    df = load_and_preprocess_data(file_path)
    df = reorder_columns(df)
    differences = calculate_differences(df)
    differences = calculate_multiplier(differences)
    residuals = rolling_pca(differences)
    export_to_excel(df, differences, residuals)
    plot_residuals(residuals)

if __name__ == "__main__":
    main()
