import numpy as np
import scipy as sp

def linear_pca(data, cv=None, normalize=False):
    """
    Perform Principal Component Analysis using Linear PCA.

    Parameters:
        data (array-like): Input data matrix.
        cv (float, optional): Target cumulative variance to retain.
        normalize (bool, optional): Flag to indicate whether to normalize data.

    Returns:
        array-like: Principal components.
    """
    if normalize:
        mean = np.mean(data, axis=0)
        variance = np.var(data, axis=0, ddof=1)
        data_normalized = (data - np.ones((data.shape[0], 1)) * mean) / variance
    else:
        data_normalized = data

    _, _, Vt = np.linalg.svd(data_normalized)
    cumulative_variance_explained = np.cumsum(100 * np.square(Vt) / np.sum(np.square(Vt)))

    if cv is None:
        cv = 95

    num_components = np.argmax(cumulative_variance_explained >= cv)
    return Vt[:num_components, :]

def sparse_pca(data, cv=None, normalize=False):
    """
    Perform Principal Component Analysis using Sparse PCA.

    Parameters:
        data (array-like): Input data matrix.
        cv (float, optional): Target cumulative variance to retain.
        normalize (bool, optional): Flag to indicate whether to normalize data.

    Returns:
        array-like: Principal components.
    """
    if normalize:
        normalize = False

    # Construct similarity matrix using exponential of the correlation matrix
    similarity_matrix = sp.linalg.expm(np.corrcoef(data, rowvar=False) / 0.1)
    diagonal_matrix = np.diag(np.sum(similarity_matrix, axis=0))
    scaled_data = np.matmul(np.linalg.inv(sp.linalg.sqrtm(diagonal_matrix)),
                            np.matmul(similarity_matrix, np.linalg.inv(sp.linalg.sqrtm(diagonal_matrix))))

    _, _, Vt = np.linalg.svd(scaled_data)
    cumulative_variance_explained = np.cumsum(100 * np.square(Vt) / np.sum(np.square(Vt)))

    if cv is None:
        cv = 95

    num_components = np.argmax(cumulative_variance_explained >= cv)
    return np.matmul(np.linalg.inv(sp.linalg.sqrtm(diagonal_matrix)), Vt.T)[:num_components, :].T
