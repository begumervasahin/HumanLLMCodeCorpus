import sys
import math
from optparse import OptionParser

# Introduction
print("Maldi2PCA - Prepares Maldi data for PCA: reduce & normalize TAB-delimited data files")
print("  Author:\tWim Fremout / Royal Institute for Cultural Heritage, Brussels, Belgium (4 Oct 2016)")
print("  Licence:\tGNU GPL version 3.0\n")

# Usage information
usage = "usage: %prog [options] INFILES"
parser = OptionParser(usage=usage, version="%prog 0.2")

# Define command-line options
parser.add_option("--nolimits", help="do not set X range", action="store_false", dest="FILT", default=True)
parser.add_option("--low", help="lower limit (default: 900)", action="store", type="int", dest="LOW", default=900)
parser.add_option("--high", help="higher limit (default: 2000)", action="store", type="int", dest="HIGH", default=2000)
parser.add_option("-n", help="normalize Y data, no normalization when zero (default: 999)", action="store", type="int", dest="NORM", default=999)
parser.add_option("-c", help="display columns (default: 15, 13 in case of -n0) \n 1-Xround\n 2-Xpeak\n 3-Ypeak\n 4-Ysum\n 5-Ypeak*\n 6-Ysum*\n 7-iterations", action="store", type="int", dest="COL", default="15")
parser.add_option("--headerless", help="do not display the table header", action="store_false", dest="HEADER", default=True)
parser.add_option("--comma", help="comma as digital separator", action="store_true", dest="COMMA", default=False)
parser.add_option("-o", help="output file", action="store", type="string", dest="OUTFILE")
parser.add_option("-v", "--verbose", help="be very verbose", action="store_true", dest="VERBOSE", default=False)

# Parse command-line arguments
(options, args) = parser.parse_args()

# Check for correct number of arguments
if len(args) == 0:
    parser.error("incorrect number of arguments")

# Adjust options based on input
if not options.FILT:
    options.LOW = 0
    options.HIGH = float('inf')
else:
    options.LOW -= 1

options.COL = str(options.COL).replace("5", "3").replace("6", "4")

digsep = "," if options.COMMA else "."

# Print parameters if verbose
if options.VERBOSE:
    print("PARAMETERS")
    print(f"  mass range: {options.FILT}")
    print(f"    lower limit: {options.LOW}")
    print(f"    higher limit: {options.HIGH}")
    print(f"  normalization: {options.NORM}")
    print(f"  columns: {options.COL}")
    print(f"  header: {options.HEADER}")
    print(f"  digital separator: {digsep}")
    print(f"  OUTFILE: {options.OUTFILE}")
    print(f"  INFILES: {args}\n")

# Initialize lists to store data
datapointindex = [0]
xround = [0]
xpeak = [0]
ysum = [0]
ypeak = [0]
iterations = [0]
n = 0

# Reading and reducing the dataset
if options.VERBOSE:
    print("READING AND REDUCING THE DATASET...")

for z in args:
    with open(z, 'r') as sourcefile:
        if options.VERBOSE:
            print(f"  opening file: {z}")

        for line in sourcefile:
            line = line.strip()
            temp = line.split()
            xroundtemp = math.trunc(round(float(temp[0]), 0))

            if options.LOW < xroundtemp <= options.HIGH:
                if xroundtemp == xround[n]:
                    if "7" in options.COL:
                        iterations[n] += 1
                    if any(x in options.COL for x in ("4", "6")):
                        ysum[n] += round(float(temp[1]), 0)
                    if any(x in options.COL for x in ("2", "3", "5")):
                        ypeaktemp = round(float(temp[1]), 0)
                        if ypeaktemp > ypeak[n]:
                            ypeak[n] = ypeaktemp
                            if "2" in options.COL:
                                xpeak[n] = round(float(temp[0]), 4)
                    if options.VERBOSE:
                        print(f"    read line from {z}: {line} --> ADD")
                else:
                    xround.append(xroundtemp)
                    if "2" in options.COL:
                        xpeak.append(round(float(temp[0]), 4))
                    if any(x in options.COL for x in ("4", "6")):
                        ysum.append(round(float(temp[1]), 0))
                    if any(x in options.COL for x in ("2", "3", "5")):
                        ypeak.append(round(float(temp[1]), 0))
                    if "7" in options.COL:
                        iterations.append(1)
                    n += 1
                    if options.VERBOSE:
                        print(f"    read line from {z}: {line} --> NEW")
        datapointindex.append(n)

datapointindex.pop(0)
xround.pop(0)
xpeak.pop(0)
ysum.pop(0)
ypeak.pop(0)
iterations.pop(0)

# Normalize data if needed
if options.NORM > 0:
    if options.VERBOSE:
        print("\n\nNORMALIZING...")
    
    ypeaknorm = []
    ysumnorm = []

    for z in range(len(datapointindex) - 1):
        if options.VERBOSE:
            print(f"  normalizing file {args[z]}")
        
        if "5" in options.COL:
            vMax = max(ypeak[datapointindex[z]:datapointindex[z + 1]])
            ypeaknorm.extend([round(x / (vMax * 1.0) * options.NORM, 0) for x in ypeak[datapointindex[z]:datapointindex[z + 1]]])
        
        if "6" in options.COL:
            vMax = max(ysum[datapointindex[z]:datapointindex[z + 1]])
            ysumnorm.extend([round(x / (vMax * 1.0) * options.NORM, 0) for x in ysum[datapointindex[z]:datapointindex[z + 1]]])

# Creating output
if options.VERBOSE:
    print("\n\nCREATING OUTPUT...\n")

if options.OUTFILE:
    with open(options.OUTFILE, 'w') as outfile:
        if options.COL.startswith("1"):
            options.COL = options.COL[1:]

        header_lines = ""
        for z in args:
            for col in options.COL:
                if col == "1":
                    header_lines += f"{z}\tX\t"
                elif col == "2":
                    header_lines += f"{z}\tXpeak\t"
                elif col == "3":
                    header_lines += f"{z}\tYpeak\t"
                elif col == "4":
                    header_lines += f"{z}\tYsum\t"
                elif col == "5":
                    header_lines += f"{z}\tYpeak*\t"
                elif col == "6":
                    header_lines += f"{z}\tYsum*\t"
                elif col == "7":
                    header_lines += f"{z}\titerations\t"
        outfile.write(header_lines.strip() + "\n")

else:
    header_lines = ""
    for z in args:
        for col in options.COL:
            if col == "1":
                header_lines += f"{z}\tX\t"
            elif col == "2":
                header_lines += f"{z}\tXpeak\t"
            elif col == "3":
                header_lines += f"{z}\tYpeak\t"
            elif col == "4":
                header_lines += f"{z}\tYsum\t"
            elif col == "5":
                header_lines += f"{z}\tYpeak*\t"
            elif col == "6":
                header_lines += f"{z}\tYsum*\t"
            elif col == "7":
                header_lines += f"{z}\titerations\t"
    print(header_lines.strip())

for x in range(min(xround), max(xround) + 1):
    outputline = str(x) + "\t" if "1" in options.COL else ""
    for z in range(len(datapointindex) - 1):
        if xround[datapointindex[z]] == x:
            for col in options.COL:
                if col == "1":
                    outputline += str(xround[datapointindex[z]]).replace(".", digsep) + "\t"
                elif col == "2":
                    outputline += str(xpeak[datapointindex[z]]).replace(".", digsep) + "\t"
                elif col == "3":
                    outputline += str(ypeak[datapointindex[z]]).replace(".", digsep) + "\t"
                elif col == "4":
                    outputline += str(ysum[datapointindex[z]]).replace(".", digsep) + "\t"
                elif col == "5":
                    outputline += str(ypeaknorm[datapointindex[z]]).replace(".", digsep) + "\t"
                elif col == "6":
                    outputline += str(ysumnorm[datapointindex[z]]).replace(".", digsep) + "\t"
                elif col == "7":
                    outputline += str(iterations[datapointindex[z]]) + "\t"
            xround.pop(datapointindex[z])
            if "2" in options.COL:
                xpeak.pop(datapointindex[z])
            if "3" in options.COL:
                ypeak.pop(datapointindex[z])
            if "4" in options.COL:
                ysum.pop(datapointindex[z])
            if "5" in options.COL:
                ypeaknorm.pop(datapointindex[z])
            if "6" in options.COL:
                ysumnorm.pop(datapointindex[z])
            if "7" in options.COL:
                iterations.pop(datapointindex[z])
            for y in range(z + 1, len(datapointindex) - 1):
                datapointindex[y] -= 1
        else:
            outputline += "\t" * len(options.COL)

    if options.OUTFILE:
        with open(options.OUTFILE, 'a') as outfile:
            outfile.write(outputline.strip() + "\n")
    else:
        print(outputline.strip())

print("All done.")
