# Import necessary libraries and modules
import os
from subprocess import run
from sys import executable

# Change working directory
os.chdir('N:/PATH/TO/WORKING/DIR')

# Set the number of jobs
n_jobs = 8

# Import specific functions and objects from libraries
from numpy import log2, array, load, save
from sklearn.decomposition import PCA
from sklearn.externals.joblib import Parallel, delayed
from open3d import read_point_cloud

# Define a function to compute logarithm of variances using PCA
def log_var(pts):
    pca = PCA()  # Initialize PCA object
    pca.fit(pts)  # Fit PCA on input points
    pca.explained_variance_[:-1] -= pca.explained_variance_[1:]  # Compute differences of consecutive variances
    return log2(pca.explained_variance_)  # Return logarithm of variances

# Define a function to export PCA results
def export_pca(ply_vox, npy_knn, temp_out):
    vox = array(read_point_cloud(ply_vox).points)  # Read point cloud data from .ply file
    knn = load(npy_knn)  # Load precomputed nearest neighbor indices
    # Compute PCA asynchronously in parallel for each set of neighbors
    save(temp_out, Parallel(n_jobs=int(n_jobs), verbose=1)(delayed(log_var)(vox[n]) for n in knn))
    return None

# Perform PCA computation and export for different input files
export_pca('vox1.ply', '128_1.npy', 'pca_1')  # Process vox1.ply with corresponding neighbors
export_pca('vox2.ply', '128_2.npy', 'pca_2')  # Process vox2.ply with corresponding neighbors
export_pca('vox4.ply', '128_4.npy', 'pca_4')  # Process vox4.ply with corresponding neighbors

# Execute various Python scripts
run((executable, 'concat.py'))  # Combine results from different PCA computations
run((executable, 'propagate.py'))  # Perform propagation tasks
run((executable, 'test.py'))  # Run tests
run((executable, 'vis.py'))  # Visualize results
run((executable, 'treetop.py'))  # Perform treetop operations
