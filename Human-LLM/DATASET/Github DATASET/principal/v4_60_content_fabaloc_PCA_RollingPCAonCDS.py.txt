# Importing necessary libraries
import numpy as np
import pandas as pd
from sklearn.decomposition import PCA
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from openpyxl import load_workbook

# Reading the data from an Excel file
data_path = '~/Itraxxdata_22Mar19_04Jun19.xlsx'
df = pd.read_excel(data_path, sheet_name='Itraxx')

# Preprocessing the data
df['Date'] = pd.to_datetime(df['Date'])
df['Date'] = df['Date'].dt.strftime('%Y-%m-%d')
date = pd.to_datetime(df['Date'])
date = date.dt.strftime('%Y-%m-%d')
df = df.iloc[:, 1:].set_index(date)

# Renaming columns
col_names = ['S31_3Y', 'S31_5Y', 'S31_7Y', 'S31_10Y', 'S30_3Y', 'S30_5Y', 'S30_7Y', 'S30_10Y',
             'S28_3Y', 'S28_5Y', 'S28_7Y', 'S28_10Y', 'S26_3Y', 'S26_5Y', 'S26_7Y', 'S26_10Y',
             'S24_5Y', 'S24_7Y', 'S24_10Y']
df.columns = col_names

# Reordering columns
new_order = ['S26_3Y', 'S24_5Y', 'S28_3Y', 'S30_3Y', 'S26_5Y',
             'S31_3Y', 'S24_7Y', 'S28_5Y', 'S26_7Y',
             'S30_5Y', 'S31_5Y', 'S28_7Y', 'S24_10Y',
             'S30_7Y', 'S31_7Y', 'S26_10Y', 'S28_10Y',
             'S30_10Y', 'S31_10Y']
df = df.reindex(columns=new_order)

# Plotting the original data
plt.figure(figsize=(8, 4))
plt.plot(df.T)
plt.xticks(rotation=45)
plt.show()

# Calculating differences and scaling
diff_df = df.diff().dropna()
multiplier = np.zeros(len(diff_df) - 1)

# Applying Rolling PCA
for i in range(len(diff_df) - 1):
    delta = (diff_df.index[i] - diff_df.index[i + 1]).days
    multiplier[i] = 1 / (np.sqrt(((delta) / 365) * 260))
d = diff_df.mul(multiplier, axis=0)

# Initializing variables for PCA
n, m = d.shape
t = 20
dates = date[0:n - t]

# Looping over rolling windows
for i in range(n - t):
    d_roll = d.iloc[i:i + t, :]
    dstd_roll = StandardScaler(with_std=False).fit_transform(d_roll)
    pca = PCA(n_components=3)
    scores = pca.fit_transform(dstd_roll)
    loadings = pca.components_
    eigvalues = pca.explained_variance_
    expvarratio = pca.explained_variance_ratio_ * 100

# Saving results to Excel
output_path = r'~/PCA_ItraxxMain.xlsx'
with pd.ExcelWriter(output_path, engine='openpyxl') as writer:
    df.to_excel(writer, sheet_name='Original_Data')
    d.to_excel(writer, sheet_name='DailyChange')

# Plotting results
plt.plot(loadings[0, :], label='PC1')
plt.plot(loadings[1, :], label='PC2')
plt.plot(loadings[2, :], label='PC3')
plt.legend()

# Plotting specific columns
plt.plot(loadings[0, :4], label='PC1')
plt.plot(loadings[1, :4], label='PC2')
plt.plot(loadings[2, :4], label='PC3')
plt.legend()

# Plotting residuals
plt.plot(diff_df.iloc[:, :4], label=['res31_3Y', 'res31_5Y', 'res31_7Y', 'res31_10Y'])
plt.legend(loc='upper left', bbox_to_anchor=(1, 0.6))
plt.xticks(np.arange(0, len(diff_df), step=3))
plt.xlabel('Date')
plt.ylabel('Residual')
plt.gcf().autofmt_xdate()
plt.show()
