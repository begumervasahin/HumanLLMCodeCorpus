import os
import cv2
import numpy as np

# Function to reshape a vector into a matrix
def unFlatten(vector, rows, cols):
    img = []
    cutter = 0
    while cutter + cols <= rows * cols:
        try:
            img.append(vector[cutter:cutter + cols])
        except:
            img = vector[cutter:cutter + cols]
        cutter += cols
    img = np.array(img)
    return img

# Dimensions for resizing images
w = 20
h = 20

# List of vowels
vowels = ['A', 'E', 'I', 'O', 'U']

# Main function
def main():
    # Loop through each vowel
    for v in vowels:
        in_matrix = None  # Initialize input matrix
        imgcnt = 0  # Counter for images processed
        print('Reading from: ' + v + ' Directory')

        # Iterate through files in the directory for the current vowel
        for f in os.listdir(os.path.join('training/', v)):
            imgcnt += 1
            print(f)  # Print current file being processed
            # Read image in grayscale
            img = cv2.imread(os.path.join('training/', v, f), cv2.IMREAD_GRAYSCALE)
            # Resize image to specified dimensions
            img_resized = cv2.resize(img, (w, h))
            # Flatten the image into a vector
            vec = img_resized.reshape(w * h)

            # Append the vector to the input matrix
            try:
                in_matrix = np.vstack((in_matrix, vec))
            except:
                in_matrix = vec

        # Perform PCA if input matrix is not empty
        if in_matrix is not None:
            mean, eigenvectors = cv2.PCACompute(in_matrix, np.mean(in_matrix, axis=0).reshape(1, -1))
            # Reshape mean vector to obtain the average image
            img = unFlatten(mean.transpose(), w, h)
            # Save the average image
            cv2.imwrite('trained/pca_vowels_' + v + '.png', img)

    cv2.waitKey(0)
    cv2.destroyAllWindows()

# Entry point of the program
if __name__ == "__main__":
    main()
