import cv2
import numpy as np
from sklearn.decomposition import PCA

def rescale_image(img):
    """
    Rescale the given image to fit within a width of 900 pixels while maintaining aspect ratio.

    Parameters:
    img (numpy.ndarray): Input image.

    Returns:
    tuple: Rescaled dimensions (width, height).
    """
    height, width, _ = img.shape
    target_width = 900
    scaling_factor = target_width / width
    new_width, new_height = int(width * scaling_factor), int(height * scaling_factor)
    return new_width, new_height

def compress_image_with_pca(image):
    """
    Compresses the given image using PCA.

    Parameters:
    image (numpy.ndarray): Input image.

    Returns:
    numpy.ndarray: Compressed image.
    """
    height, width, channels = image.shape
    flattened_img = np.reshape(image, (height, width * channels))

    pca = PCA(n_components=350)
    pca_fit = pca.fit(flattened_img)
    transformed_img = pca_fit.transform(flattened_img)

    reconstructed_img = pca_fit.inverse_transform(transformed_img)
    compressed_img = np.uint8(np.reshape(reconstructed_img, (height, width, channels)))
    return compressed_img

def main():
    """
    Main function to process and compress images using PCA.
    """
    for num in range(1, 3):
        print(f"Processing image {num}")
        
        # Read the image
        image = cv2.imread(f"{num}.jpg")
        image_copy = image.copy()

        # Compress image using PCA
        compressed_img = compress_image_with_pca(image)

        # Save compressed image
        cv2.imwrite(f'quantizedImages/op-{num}.jpg', compressed_img)

        # Resize and display original and compressed images
        resized_original = cv2.resize(image_copy, rescale_image(image))
        cv2.imshow("Original", resized_original)
        
        resized_compressed = cv2.resize(compressed_img, rescale_image(image))
        cv2.imshow("Compressed", resized_compressed)

        # Wait for user input to proceed to the next image
        cv2.waitKey()

if __name__ == "__main__":
    main()
