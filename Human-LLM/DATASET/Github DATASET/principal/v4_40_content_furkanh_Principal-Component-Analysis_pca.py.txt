import numpy as np

class PCA:
    def __init__(self, k=None):
        self.k = k
        
    def fit(self, X):
        if self.k is None or self.k > X.shape[0]:
            self.k = X.shape[0]
            
        X_mean = np.sum(X, axis=0) / X.shape[0]
        self.X_mean = X_mean
        X_centered = X - X_mean
        
        covariance_matrix = np.matmul(X_centered.transpose(), X_centered) / (X.shape[0] - 1)
        eig_values, eig_vectors = np.linalg.eig(covariance_matrix)
        
        sorted_indices = eig_values.argsort()[::-1]
        eig_vectors = eig_vectors[:, sorted_indices]
        eig_values = eig_values[sorted_indices][::-1]
        
        self.eig_values = np.real(eig_values)
        self.eig_vectors = np.real(eig_vectors)
        
        total_eig_sum = np.sum(self.eig_values)
        cumulative_variance_explained = 0
        self.pve_list = []
        
        for i in range(self.eig_values.shape[0]):
            cumulative_variance_explained += self.eig_values[i]
            self.pve_list.append(cumulative_variance_explained / total_eig_sum)
            
    def transform(self, X):
        X_centered = X - self.X_mean
        transformed_data = np.matmul(X_centered, self.eig_vectors)[:, :self.k]
        return transformed_data
