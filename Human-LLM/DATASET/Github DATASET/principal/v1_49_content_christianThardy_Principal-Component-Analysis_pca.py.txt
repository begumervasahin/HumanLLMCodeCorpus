import numpy as np
import timeit
import matplotlib.pyplot as plt
from load_data import load_mnist
from sklearn.decomposition import PCA as SKPCA
import pandas as pd

# Load MNIST data
MNIST = load_mnist()
images, labels = MNIST['data'], MNIST['target']

# Function to normalize data
def normalize(X):
    mu = np.mean(X, axis=0)
    std = np.std(X, axis=0)
    std_filled = std.copy()
    std_filled[std == 0] = 1.
    Xbar = (X - mu) / std_filled
    return Xbar, mu, std

# Function to perform PCA
def PCA(X, num_components):
    X, mean, std = normalize(X)
    S = np.cov(X, rowvar=False, bias=True)
    eig_vals, eig_vecs = np.linalg.eig(S)
    idx = eig_vals.argsort()[::-1]
    eig_vals = eig_vals[idx]
    eig_vecs = eig_vecs[:, idx]
    P = eig_vecs[:, :num_components]
    X_reconstruct = np.dot(P.T, X.T).T
    return X_reconstruct

# Function to compute mean squared error
def mse(predict, actual):
    return np.square(predict - actual).sum(axis=1).mean()

# Function to compute running time
def time(f, repeat=10):
    times = []
    for _ in range(repeat):
        start = timeit.default_timer()
        f()
        stop = timeit.default_timer()
        times.append(stop - start)
    return np.mean(times), np.std(times)

# Function to perform PCA for high-dimensional datasets
def PCA_high_dim(X, n_components):
    N, D = X.shape
    M = (X @ X.T) / N
    eig_vals, eig_vecs = np.linalg.eig(M)
    idx = eig_vals.argsort()[::-1]
    eig_vals = eig_vals[idx]
    eig_vecs = eig_vecs[:, idx]
    U = eig_vecs[:, :n_components]
    P = U @ np.linalg.inv(U.T @ U) @ U.T
    X_reconstruct = P @ X.T
    return X_reconstruct.T

# Load data and normalize
NUM_DATAPOINTS = 1000
X = (images.reshape(-1, 28 * 28)[:NUM_DATAPOINTS]) / 255.
Xbar, mu, std = normalize(X)

# Test PCA and PCA_high_dim
np.testing.assert_almost_equal(PCA(Xbar, 2), PCA_high_dim(Xbar, 2))

# Compute MSE for different number of principal components
loss = []
for num_component in range(1, 100):
    reconst = PCA(Xbar, num_component)
    error = mse(reconst, Xbar)
    loss.append((num_component, error))
loss = np.asarray(loss)

# Plot MSE vs number of principal components
plt.plot(loss[:, 0], loss[:, 1])
plt.axhline(100, linestyle='--', color='r', linewidth=2)
plt.xlabel('num_components')
plt.ylabel('MSE')
plt.title('MSE vs number of principal components')
plt.show()

# Measure running time for different dataset sizes
times_mm0 = []
times_mm1 = []
for datasetsize in np.arange(4, 784, step=50):
    XX = Xbar[:datasetsize]
    mu, sigma = time(lambda: XX.T @ XX)
    times_mm0.append((datasetsize, mu, sigma))
    mu, sigma = time(lambda: XX @ XX.T)
    times_mm1.append((datasetsize, mu, sigma))
times_mm0 = np.asarray(times_mm0)
times_mm1 = np.asarray(times_mm1)

# Plot running time vs dataset size
plt.errorbar(times_mm0[:, 0], times_mm0[:, 1], times_mm0[:, 2], label='$X^T X$ (PCA)', linewidth=2)
plt.errorbar(times_mm1[:, 0], times_mm1[:, 1], times_mm1[:, 2], label='$X X^T$ (PCA_high_dim)', linewidth=2)
plt.xlabel('size of dataset')
plt.ylabel('running time')
plt.legend()
plt.show()

# Measure running time for different number of datapoints
times0 = []
times1 = []
for datasetsize in np.arange(4, 784, step=100):
    XX = Xbar[:datasetsize]
    npc = 2
    mu, sigma = time(lambda: PCA(XX, npc), repeat=10)
    times0.append((datasetsize, mu, sigma))
    mu, sigma = time(lambda: PCA_high_dim(XX, npc), repeat=10)
    times1.append((datasetsize, mu, sigma))
times0 = np.asarray(times0)
times1 = np.asarray(times1)

# Plot running time vs number of datapoints
plt.errorbar(times0[:, 0], times0[:, 1], times0[:, 2], label='PCA', linewidth=2)
plt.errorbar(times1[:, 0], times1[:, 1], times1[:, 2], label='PCA_high_dim', linewidth=2)
plt.xlabel('number of datapoints')
plt.ylabel('run time')
plt.legend()
plt.show()
