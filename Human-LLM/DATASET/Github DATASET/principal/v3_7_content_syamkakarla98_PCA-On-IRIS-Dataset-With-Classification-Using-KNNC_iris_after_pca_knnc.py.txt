# Import necessary libraries
import numpy as np
import pandas as pd
from sklearn.model_selection import train_test_split
from sklearn.neighbors import KNeighborsClassifier
from sklearn import metrics

# Load the preprocessed Iris dataset after PCA
dataset_path = "D:\Python_programs\ML\iris_after_pca.csv"
df = pd.read_csv(dataset_path)

# Define features and target variable
features = ['PC-1', 'PC-2']
X = df[features].values
Y = df['target'].values

# Split the dataset into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(X, Y, test_size=0.3, random_state=100)

# Initialize KNN model
knn_model = KNeighborsClassifier()

# Fit the model
knn_model.fit(X_train, y_train)

# Predict on test set
y_pred_test = knn_model.predict(X_test)

# Print header for accuracy output
print('*'*11, 'Accuracy of IRIS Dataset after PCA', '*'*11, '\n')

# Loop over different values of K to evaluate accuracy
for K in range(1, 26):
    # Initialize KNN classifier with current K value
    knn_classifier = KNeighborsClassifier(n_neighbors=K, weights='uniform', algorithm='auto')
    knn_classifier.fit(X_train, y_train)  # Fit the classifier
    y_pred = knn_classifier.predict(X_test)  # Predict using test data
    accuracy = metrics.accuracy_score(y_test, y_pred)  # Calculate accuracy
    # Print accuracy for current K value
    print(f"Accuracy is {accuracy*100:.2f}% for K-Value: {K}")
