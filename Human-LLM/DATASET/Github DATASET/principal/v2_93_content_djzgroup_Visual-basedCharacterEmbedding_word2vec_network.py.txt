import numpy as np
from gensim.models import Word2Vec
from keras.layers import LSTM, Embedding, Dense, Dropout
from keras.models import Sequential
from keras.preprocessing.sequence import pad_sequences
from keras.metrics import categorical_accuracy
from matplotlib import pyplot as plt

def one_hot(labels, num_classes=12):
    return np.eye(num_classes)[labels]

def two_layer_LSTM(model, dims):
    model.add(LSTM(256, input_shape=(None, dims), activation='relu', return_sequences=True))
    model.add(LSTM(128, return_sequences=True))
    model.add(LSTM(64))

def one_layer_GRU(model, dims):
    model.add(LSTM(64, input_shape=(None, dims), activation='relu'))

def two_layer_GRU(model, dims):
    model.add(LSTM(128, input_shape=(None, dims), return_sequences=True))
    model.add(LSTM(64))

def one_layer_LSTM(model, dims):
    model.add(LSTM(64, input_shape=(None, dims)))

def run(dim, layer, train_size, padding_len):
    bin_fname = 'word2vec/word2vec_wx'
    model_word2vec = Word2Vec.load(bin_fname)

    EMBEDDING_DIMS = 256
    d = {}

    for split in ['train', 'test', 'val']:
        x = []
        y = []
        src_name = 'data/zh_simplified_{}.txt'.format(split)
        with open(src_name, 'r', encoding='utf-8') as f:
            sequence = f.readline()
            while sequence:
                New_seq = []
                sequence = sequence.split('\t')
                sequence[1] = sequence[1].replace('\n', '')
                for char in sequence[1]:
                    if char not in d:
                        d[char] = len(d)
                x.append(New_seq)
                y.append(sequence[0])
                sequence = f.readline()

        np.save('zh_simplified_word2vec_{}_data.npy'.format(split), x)
        np.save('zh_simplified_word2vec_{}_label.npy'.format(split), y)

    embedding_matrix = np.zeros((len(d) + 1, EMBEDDING_DIMS))
    model_word2vec = model_word2vec.wv

    for word, i in d.items():
        if word in model_word2vec:
            embedding_matrix[i] = model_word2vec[word]

    del d, model_word2vec

    x_train = np.load('zh_simplified_word2vec_train_data.npy')
    y_train = np.load('zh_simplified_word2vec_train_label.npy')
    x_val = np.load('zh_simplified_word2vec_val_data.npy')
    y_val = np.load('zh_simplified_word2vec_val_label.npy')
    x_test = np.load('zh_simplified_word2vec_test_data.npy')
    y_test = np.load('zh_simplified_word2vec_test_label.npy')

    embedding_layer = Embedding(
        len(embedding_matrix),
        EMBEDDING_DIMS,
        weights=[embedding_matrix],
        input_length=padding_len,
        trainable=True
    )

    model = Sequential()
    model.add(embedding_layer)

    if layer == 1:
        one_layer_GRU(model, dim)
    elif layer == 2:
        two_layer_GRU(model, dim)
    elif layer == 3:
        one_layer_LSTM(model, dim)
    elif layer == 4:
        two_layer_LSTM(model, dim)

    model.add(Dropout(0.5))
    model.add(Dense(12, activation='softmax'))

    model.compile(loss='categorical_crossentropy', optimizer='adam', metrics=['accuracy', categorical_accuracy])

    print(model.summary())

    history = model.fit(x_train, one_hot(y_train), batch_size=128, epochs=50, validation_split=1 - train_size, shuffle=True)

    plt.plot(history.history['loss'])
    plt.plot(history.history['val_loss'])
    plt.title('Model Train vs Validation Loss')
    plt.ylabel('Loss')
    plt.xlabel('Epoch')
    plt.legend(['Train', 'Validation'], loc='upper right')
    plt.show()

    score = model.evaluate(x_test, one_hot(y_test), batch_size=128)
    print("\nTest Score: {} Test Accuracy: {}".format(score[0], score[1]))

    np.save('result/word2vec_{}_{}_len{}.npy'.format(dim, layer, padding_len), score)

    with open('result/word2vec_{}_{}_len{}.txt'.format(dim, layer, padding_len), 'w') as f:
        f.writelines('Test Score: {} Test Accuracy: {}'.format(score[0], score[1]))


if __name__ == '__main__':
