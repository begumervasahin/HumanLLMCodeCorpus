import numpy as np
from scipy.signal import welch, hilbert
import matplotlib.pyplot as plt

class PCA:
    def __init__(self, sample, No_modes):
        self.sample = sample
        self.No_modes = No_modes
        self.func_dict = {'id': lambda x: x}
        self.PCA_modes = []
        self.PCA_eigen = []

    def pre_process(self, signal):
        """Remove the mean component from the sample and return the fluctuation component"""
        signal_mean = np.mean(signal, axis=0)
        signal_proc = signal - np.outer(np.ones(signal.shape[0]), signal_mean)
        return signal_proc

    def cov_(self, signal):
        """Return the covariance of preprocessed sample"""
        signal_proc = self.pre_process(signal)
        L = len(signal_proc)
        signal_cov = np.dot(signal_proc.T.conj(), signal_proc) / L
        return signal_cov

    def svd(self, signal):
        """Return PCA modes using singular value decomposition"""
        signal_cov = self.cov_(signal)
        u, s, vh = np.linalg.svd(signal_cov, full_matrices=True)
        self.PCA_modes = u[:, :self.No_modes]
        self.PCA_eigen = s[:self.No_modes]

    def principal_comp(self):
        """Compute the time coefficient (principal components) of each mode"""
        signal_proc = self.pre_process(self.sample)
        principal_components = np.dot(self.PCA_modes.T, signal_proc.T)
        return principal_components

    def PSD_PC(self):
        """Plot the frequency transform of time coefficient of modes"""
        PC = self.principal_comp()
        sampling_frequency = 700
        Nfft = 2 ** 13
        fig, ax = plt.subplots()
        for i in range(min(5, self.No_modes)):
            freq, spec = welch(PC[i], nperseg=Nfft, nfft=Nfft, noverlap=Nfft // 2, fs=sampling_frequency)
            ax.plot(freq, spec, label=f'Mode {i + 1}')
        ax.set_xlim(0, 10)
        ax.set_xlabel('Frequency (Hz)')
        ax.set_ylabel('Spectral Density (1/Hz)')
        ax.legend(loc='best')
        plt.show()

    def execute(self):
        """Execute PCA and return PCA modes and their corresponding eigenvalues"""
        self.svd(self.sample)
        return self.PCA_modes, self.PCA_eigen

    def visualize(self):
        """Visualize PCA modes"""
        self.func_dict['id'](self.PCA_modes[:, :min(self.No_modes, 5)], ['red', 'white', 'blue'], txt=True, num=np.arange(1, min(self.No_modes, 5) + 1))
        self.PSD_PC()

class Dynamic_PCA(PCA):
    def __init__(self, sample, No_modes, f1, f2, sampling_freq):
        super().__init__(sample, No_modes)
        self.f1 = f1
        self.f2 = f2
        self.fs = sampling_freq
        self.DPCA_modes = []
        self.DPCA_eigen = []

    def bandpass_filter(self, signal):
        """Remove frequencies outside the band pass range"""
        signal_proc = self.pre_process(signal)
        N = len(signal_proc)
        dF = self.fs / N
        f = np.linspace(start=-self.fs / 2, stop=self.fs / 2 - dF, num=self.sample.shape[0])
        Boolean = (self.f1 < np.abs(f)) & (np.abs(f) < self.f2)
        FFT = np.fft.fft(signal_proc, axis=0)
        FFT_shift = np.fft.fftshift(FFT) / N
        sp = Boolean.astype(np.int) * FFT_shift
        FFT_shift_inverse = np.fft.ifftshift(sp)
        FFT_inverse = np.fft.ifft(FFT_shift_inverse, axis=0)
        sample_filtered = np.real(FFT_inverse)
        return sample_filtered

    def hilbert_trans(self, signal):
        """Create analytic signal using Hilbert transform"""
        Signal_filtered = self.bandpass_filter(signal)
        Analytic_signal = hilbert(Signal_filtered, axis=0)
        return Analytic_signal

    def get_modes(self, signal):
        """Return Dynamic PCA modes"""
        Analytic_signal = self.hilbert_trans(signal)
        signal_cov = self.cov_(Analytic_signal)
        u, s, vh = np.linalg.svd(signal_cov)
        self.DPCA_modes = u[:, :self.No_modes]
        self.DPCA_eigen = s[:self.No_modes]

    def visualize(self, mode_no):
        """Visualize the animated movie of PCA mode"""
        D_PCA = self.DPCA_modes[:, mode_no]
        Z = D_PCA.reshape(-1, 1)
        phase_shift = np.exp(1j * np.linspace(0, 2 * np.pi, 100)).reshape(1, -1)
        ZZ = np.real(Z * phase_shift)
        plt.ion()
        for i in range(100):
            press = ZZ[:, i]
            plt.plot(press)
            plt.pause(0.1)
            plt.clf()

# Example usage:

# Generating sample data
sample_data = np.random.randn(1000, 10)

# Using PCA
pca = PCA(sample_data, No_modes=3)
pca.execute()
pca.visualize()

# Using Dynamic PCA
dynamic_pca = Dynamic_PCA(sample_data, No_modes=3, f1=1, f2=10, sampling_freq=100)
dynamic_pca.execute()
dynamic_pca.visualize(mode_no=0)
