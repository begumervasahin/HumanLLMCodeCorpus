# Importing SizerPy module
import SizerPy as sp

def main():
    # Parsing arguments for raster processing
    args = sp.RasterParser()
    
    # Creating an object for input raster
    inRasterObj = sp.SizeRaster(args.inputRaster)
    
    # Resizing the input raster and saving the result to an output raster
    inRasterObj.SizeBands(args.outRaster, factor=args.factor, d1Interpolate=args.d1Interpolate)
    
    # Indicating completion of the resizing process
    print('Resizing completed.')

if __name__ == '__main__':
    main()  # Calling the main function
