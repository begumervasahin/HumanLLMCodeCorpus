# Importing necessary libraries
import numpy as np
import pandas as pd
import os
from scipy.stats import norm
from scipy.special import erfinv

# Defining file paths
base_path = os.path.split(__file__)[0]  # Assuming __file__ is available in this context
data_file = os.path.join(base_path, 'IndexData.csv')
corr_file = os.path.join(base_path, 'IndexCorrelations.csv')

# Reading data from CSV file and dropping any rows with missing values
data = pd.read_csv(data_file).dropna()

# Extracting the relevant columns for analysis
X = data[['Economic', 'Personal', 'Social']].copy()

# Standardizing the data by subtracting the mean
X -= X.mean(0)

# Getting the dimensions of the data
n = X.shape[0]

# Performing Singular Value Decomposition (SVD)
(U, S, VT) = np.linalg.svd(X.T / np.sqrt(n - 1))

# Projecting the data onto the principal components
projection = np.dot(U.T, X.T).T

# Creating a combined index by taking the negative of the first principal component and centering it around 5
data['Combined'] = -projection[:, 0]
data['Combined'] += (5 - data['Combined'].mean())

# Sorting the data by the combined index in descending order
sorted_data = data.set_index('Countries (2014)').sort_values('Combined', ascending=False)

# Saving the sorted data to a new CSV file
sorted_data.to_csv(data_file)

# Calculating and saving the correlation matrix of the data
correlation_matrix = data.corr()
correlation_matrix.to_csv(corr_file)
