import numpy as np
import pandas as pd
import matplotlib.pyplot as plt

# Assuming you have your data in an Excel file with a sheet named 'sheet'
path = '~/yourdata'

# Read data from Excel file
d = pd.read_excel(path, sheet_name='sheet')

# Convert data to DataFrame
d = pd.DataFrame(d)

# Calculate mean vector
mean_vec = np.mean(d, axis=0)

# Calculate variance vector
var_vec = np.var(d, axis=0)

# Get dimensions of the data
n, m = d.shape

# Standardize the data
d_std = d - mean_vec

# Calculate covariance matrix
cov_mat = (d_std.T.dot(d_std)) / (n - 1)

# Calculate eigenvalues and eigenvectors of the covariance matrix
eigval, eigvec = np.linalg.eig(cov_mat)

# Check if eigenvectors are unit vectors
for i in eigvec:
    np.testing.assert_array_almost_equal(1.0, np.linalg.norm(i))
    print('Everything ok!')

# Sort eigenvalues and eigenvectors
idx = eigval.argsort()[::-1]
eigval = eigval[idx]
eigvec = eigvec[:, idx]

# Calculate explained variance and cumulative explained variance
tot = sum(eigval)
var_exp = [i / tot * 100 for i in sorted(eigval, reverse=True)]
cum_var_exp = np.cumsum(var_exp)

# Principal Component labels
PC = ['PC%s' % s for s in range(1, len(eigval) + 1)]

# Plot explained variance
plt.scatter(var_exp, PC, alpha=0.5)
plt.title('Explained variance')
plt.xlabel('Principal Components')
plt.ylabel('Explained Variance (%)')
plt.show()

# Select number of components
n_components = 3

# Calculate loadings
loadings = eigvec[:, :n_components]

# Calculate scores
scores = d_std.dot(loadings)

# Reconstruct the data
d_hat = scores.dot(loadings.T)
d_hat_raw = d_hat + mean_vec.values

# Calculate residuals
res = d - d_hat_raw
