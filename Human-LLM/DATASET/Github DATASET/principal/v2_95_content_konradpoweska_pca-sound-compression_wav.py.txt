import numpy as np
import wave
import struct

# Dictionary mapping sample width to struct format
widthStruct = {
    1: 'b',  # 8-bit
    2: 'h'   # 16-bit
}

def open_wave_file(filename):
    # Open the WAV file in read mode
    file = wave.open(filename, 'r')
    
    # Extract information about the WAV file
    nchannels = file.getnchannels()    # Number of channels
    sampwidth = file.getsampwidth()    # Sample width (in bytes)
    framerate = file.getframerate()    # Frame rate (frames per second)
    nframes = file.getnframes()        # Total number of frames
    
    # Store the extracted information as headers
    headers = nchannels, sampwidth, framerate, nframes
    
    # Determine the struct format for unpacking the frames
    structFormat = "<{0}{1}".format(nframes, widthStruct[sampwidth])
    
    # Read and unpack the frames into samples using numpy
    frames = file.readframes(nframes)
    samples = np.array(struct.unpack(structFormat, frames))
    
    # Close the WAV file
    file.close()
    
    return headers, samples

def save_wave_file(filename, headers, samples):
    # Open a new WAV file in write mode
    file = wave.open(filename, 'w')
    
    # Unpack headers
    nchannels, sampwidth, framerate, nframes = headers
    
    # Set parameters for the new WAV file
    file.setnchannels(nchannels)
    file.setsampwidth(sampwidth)
    file.setframerate(framerate)
    file.setnframes(nframes)
    
    # Clip samples to fit within the specified sample width range
    samples = np.clip(samples[:nframes], -2**(sampwidth*8-1), 2**(sampwidth*8-1)-1)
    
    # Determine the struct format for packing the samples
    structFormat = "<{0}{1}".format(nframes, widthStruct[sampwidth])
    
    # Pack the samples and write them to the new WAV file
    frames = struct.pack(structFormat, *samples)
    file.writeframes(frames)
    
    # Close the WAV file
    file.close()
