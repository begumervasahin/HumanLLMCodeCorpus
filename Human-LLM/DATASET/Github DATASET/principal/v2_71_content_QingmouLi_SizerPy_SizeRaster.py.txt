# Importing necessary modules
import SizerPy as sp

def main():
    # This function demonstrates resizing of raster data using SizerPy
    args = sp.RasterParser()  # Parsing command line arguments related to raster data
    input_raster = args.inputRaster  # Getting input raster file from parsed arguments
    out_raster = args.outRaster  # Getting output raster file from parsed arguments
    factor = args.factor  # Getting resizing factor from parsed arguments
    d1_interpolate = args.d1Interpolate  # Getting interpolation method from parsed arguments
    
    # Creating an instance of SizeRaster class from SizerPy module
    raster_object = sp.SizeRaster(input_raster)
    
    # Resizing bands of the raster and saving it to the output raster file
    raster_object.SizeBands(out_raster, factor=factor, d1Interpolate=d1_interpolate)
    
    # Printing a message indicating successful completion
    print('Resizing of raster bands completed.')

if __name__ == '__main__':
    main()
