import os
import numpy as np
import matplotlib.pyplot as plt

# Define the main function
def main():
    # Define parameters
    D_range = range(1, 5)
    Dest, Pval, save_data = 1, 1, 0
    gen_seed_list = range(25)
    n_s = float(len(gen_seed_list))
    alpha_range = np.arange(0, 1.01, 0.1)
    
    # Close any existing plots
    plt.close('all')
    
    # Plot Dest and Pval
    plot_results(13, 16, 3, 0.1, svfg=1, v_='', optn='(a)', lgnd=1, data_type='Dest')
    plot_results(13, 16, 3, 0.1, svfg=1, v_='', optn='(b)', lgnd=1, data_type='Pval')

# Function to plot Dest and Pval
def plot_results(p1, p2, p3, p4, svfg_=0, v_='', optn=0, lgnd=0, data_type='Dest'):
    param_list = [1, 2, 3, 4]
    for param in param_list:
        grp_N, M = p1, p2
        K = p3
        D_t = param
        sigma_ = p4
    
    # Load data
    vals_m = np.zeros((len(param_list), len(alpha_range), 6))
    for g_seed in gen_seed_list:
        vals_m += np.load('%s%s(%dx%d)x%d_D%ds%d_%d.npy' % (
            data_type, v_, grp_N, M, K, D_t, sigma_*100, g_seed))
    vals = vals_m / n_s
    
    # Calculate standard deviation
    vals_sd = np.zeros((len(param_list), len(alpha_range), 6))
    for g_seed in gen_seed_list:
        load_ = np.load('%s%s(%dx%d)x%d_D%ds%d_%d.npy' % (
            data_type, v_, grp_N, M, K, D_t, sigma_*100, g_seed))
        vals_sd += (load_ - vals) ** 2
    vals2 = vals_sd / n_s
    
    # Print mean values
    for i in range(6):
        print('\t'.join(list(np.around(np.mean(vals[:, :, i], axis=0), 2).astype(str))))
    
    # Plot the data
    plt.figure()
    plt.subplot(1, 1, 1)
    for i, color in enumerate(['c', 'b', 'r', 'g', 'k']):
        plt.plot(alpha_range, np.mean(vals[:, :, i], axis=0), color + '*', ms=12)
        plt.errorbar(alpha_range, np.mean(vals[:, :, i], axis=0),
                     yerr=np.sqrt(np.mean(vals2[:, :, i], axis=0)) / np.sqrt(n_s), color=color, ms=12)
        plt.plot(alpha_range, np.mean(vals[:, :, i], axis=0), color + '--', ms=12)
    plt.xlabel(r'$\alpha$', fontsize=16)
    plt.ylim(-0.1, 1.1)
    if lgnd:
        legend_labels = ['gPCA', 'JIVE', 'LP', 'BIC', 'KG', 'KN'] if data_type == 'Dest' else [r'$\alpha$', r'$p_{\alpha 1}$',  r'$p_{\alpha 2}$']
        plt.legend(legend_labels, loc='upper left', fontsize=16)
    plt.title('%s: Proportion of correct rank selections' % optn, fontsize=16)
    if svfg_:
        plt.savefig('%s%s(%dx%d)x%d_D%ds%d_all%d.png' % (data_type, v_, grp_N, M, 
            K, D_t, sigma_*100, gen_seed_list[-1]), format='png', bbox_inches='tight')
    plt.show()

# Entry point of the script
if __name__ == "__main__":
    main()
