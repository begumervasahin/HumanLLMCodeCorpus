import numpy as np
from numpy.linalg import eig
import matplotlib.pyplot as plt
from scipy.stats import multivariate_normal

# Load the dataset
with open('data_online.txt') as f:
    train_data = np.array([list(map(float, line.split())) for line in f])

# Perform Principal Component Analysis (PCA)
cov_matrix = np.cov(train_data.T)
eigenvalues, eigenvectors = eig(cov_matrix)
eigenvectors = eigenvectors.T
pca1_idx, pca2_idx = np.argsort(eigenvalues)[-2:]
pc1, pc2 = eigenvectors[pca1_idx], eigenvectors[pca2_idx]
transformation_matrix = np.array([pc1, pc2])
trans_data = np.dot(train_data, transformation_matrix.T)

# Function to draw the main scatter plot
def draw_main_scatter():
    plt.scatter(trans_data[:, 0], trans_data[:, 1], alpha=0.2)
    plt.title('Principal Component Analysis')
    plt.xlabel('Principal Component 1')
    plt.ylabel('Principal Component 2')
    plt.show()

# Initializations for Gaussian Mixture Model (GMM) clustering
num_clusters = 3
probabilities = np.random.rand(len(trans_data), num_clusters)
probabilities /= np.sum(probabilities, axis=1)[:, None]

means = np.array([trans_data[np.random.randint(0, len(trans_data))] for _ in range(num_clusters)])
covariances = np.array([[[1, 0.1], [0.1, 1]]] * num_clusters)
weights = np.ones(num_clusters) / num_clusters

# Function to draw the Gaussian mixture model
def draw_gmm():
    plt.figure(figsize=(8, 6))
    for k in range(num_clusters):
        x, y = np.mgrid[-4:10:.5, -6:6:.5]
        pos = np.empty(x.shape + (2,))
        pos[:, :, 0] = x
        pos[:, :, 1] = y
        rv = multivariate_normal(means[k], covariances[k])
        plt.contour(x, y, rv.pdf(pos), alpha=0.5)
        plt.scatter(means[k][0], means[k][1], s=300, c='black', marker="d", alpha=0.5)
    plt.show()

# Function to check convergence of the GMM
def has_converged(prev_likelihood):
    curr_likelihood = 0.0
    for i in range(len(trans_data)):
        likelihood = 0.0
        for k in range(num_clusters):
            likelihood += weights[k] * multivariate_normal.pdf(trans_data[i], means[k], covariances[k])
        curr_likelihood += np.log(likelihood)
    return abs(curr_likelihood - prev_likelihood) <= 0.001

# Training loop for GMM
max_epochs = 2001
prev_likelihood = 0

for epoch in range(max_epochs):
    if has_converged(prev_likelihood):
        break
    
    for i in range(len(trans_data)):
        for k in range(num_clusters):
            probabilities[i][k] = weights[k] * multivariate_normal.pdf(trans_data[i], means[k], covariances[k])
        probabilities[i] /= np.sum(probabilities[i])

    for k in range(num_clusters):
        means[k] = np.dot(probabilities[:, k], trans_data) / np.sum(probabilities[:, k])
        covariances[k] = np.dot((probabilities[:, k] * (trans_data - means[k])).T, (trans_data - means[k])) / np.sum(probabilities[:, k])
        weights[k] = np.sum(probabilities[:, k]) / len(trans_data)

    if epoch % 100 == 0:
        draw_gmm()
        print("Epoch:", epoch)

# Final drawing of the GMM
draw_gmm()

# Print the probability sum for each cluster
for k in range(num_clusters):
    print("Cluster", k, "probability sum =", np.sum(probabilities[:, k]))
