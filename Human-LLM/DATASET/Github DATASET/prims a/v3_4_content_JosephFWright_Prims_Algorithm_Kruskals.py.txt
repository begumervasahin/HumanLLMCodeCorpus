def kruskals(edge_file_name, start_vertex=0):
    """
    Computes the minimum spanning tree of a graph using Kruskal's algorithm.
    
    Args:
    - edge_file_name: Name of the file containing graph edges
    - start_vertex: The starting vertex for the algorithm (default is 0)
    
    Returns:
    - minimum_spanning_tree: List of edges in the minimum spanning tree
    """
    
    # Helper function to find the parent of a vertex
    def find(parent, i):
        if parent[i] == i:
            return i
        return find(parent, parent[i])

    # Helper function to perform union operation
    def union(parent, rank, x, y):
        x_root = find(parent, x)
        y_root = find(parent, y)

        if rank[x_root] < rank[y_root]:
            parent[x_root] = y_root
        elif rank[x_root] > rank[y_root]:
            parent[y_root] = x_root
        else:
            parent[y_root] = x_root
            rank[x_root] += 1

    # Create a weighted graph instance
    graph = Weighted_Graph(edge_file_name)

    # Sort edges based on weight
    sorted_edges = sorted(graph.edges(), key=lambda x: x[2])

    # Initialize parent and rank arrays
    num_vertices = len(graph.vertex_set())
    parent = [i for i in range(num_vertices)]
    rank = [0] * num_vertices

    # Initialize minimum spanning tree
    minimum_spanning_tree = []

    # Iterate through sorted edges
    for edge in sorted_edges:
        u, v, weight = edge
        x = find(parent, u)
        y = find(parent, v)

        # Check if adding the edge creates a cycle
        if x != y:
            minimum_spanning_tree.append(edge)
            union(parent, rank, x, y)

    # Return the minimum spanning tree
    return minimum_spanning_tree
