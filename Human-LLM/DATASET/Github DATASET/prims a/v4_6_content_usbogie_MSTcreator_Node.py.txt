class Node:
    def __init__(self, xloc, yloc, kval, adjList):
        # Initialize the attributes of the Node class
        self.xloc = xloc  # x-coordinate location
        self.yloc = yloc  # y-coordinate location
        self.kval = kval  # k-value
        self.adjList = adjList  # List of adjacent nodes

    def __eq__(self, rhs):
        # Define equality between nodes based on their x, y, and k values
        return self.xloc == rhs.xloc and self.yloc == rhs.yloc and self.kval == rhs.kval
