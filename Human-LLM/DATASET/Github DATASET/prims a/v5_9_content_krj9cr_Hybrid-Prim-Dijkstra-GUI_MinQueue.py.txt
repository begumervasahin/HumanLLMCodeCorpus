import sys
from Node import Node

class MinQueue(object):
    def __init__(self, nodes):
        # Initialize MinQueue with a list of nodes
        self.nodes = nodes[:]  # Creating a copy of the input list

    def __len__(self):
        # Return the length of the MinQueue
        return len(self.nodes)

    def contains(self, key):
        # Check if a node with a given key exists in the MinQueue
        return any(node.key == key for node in self.nodes)

    def extractMin(self):
        # Extract the node with the minimum key from the MinQueue
        min_key = sys.maxint  # Setting initial minimum key to maximum integer value
        min_node = None  # Placeholder for the node with minimum key
        for node in self.nodes:
            if node.key < min_key:
                min_key = node.key
                min_node = node
        self.nodes.remove(min_node)  # Removing the node with minimum key from the list
        return min_node.idx  # Returning the index of the node with minimum key
