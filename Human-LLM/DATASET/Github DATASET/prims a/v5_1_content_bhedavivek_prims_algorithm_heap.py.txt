class Heap:
    def __init__(self):
        # Use the first element to store the size of the heap
        self._size = 0
        self._data = [0]

    def insert(self, key_value):
        """
        Inserts a new value into the heap, maintaining the heap property.
        """
        self._data.append(key_value)
        self._size += 1
        self._heapify_up(self._size)

    def _heapify_up(self, index):
        """
        Moves the element at the specified index up to maintain the heap property.
        """
        while index > 1:
            parent_index = index // 2
            if self._data[parent_index].distance > self._data[index].distance:
                # Swap if the parent's distance is greater than the child's
                self._data[parent_index], self._data[index] = self._data[index], self._data[parent_index]
                index = parent_index
            else:
                break

    def extract_min(self):
        """
        Removes and returns the element with the smallest distance.
        """
        if self._size < 1:
            return None  # Heap is empty

        min_element = self._data[1]
        self._data[1] = self._data[self._size]
        self._size -= 1
        if self._size > 1:
            self._heapify_down(1)
        self._data.pop()
        return min_element

    def _heapify_down(self, index):
        """
        Moves the element at the specified index down to maintain the heap property.
        """
        while 2 * index <= self._size:
            left_child_index = 2 * index
            right_child_index = left_child_index + 1

            # Determine which child is smaller
            smaller_child_index = left_child_index
            if (right_child_index <= self._size and
                    self._data[right_child_index].distance < self._data[left_child_index].distance):
                smaller_child_index = right_child_index

            # If the current element is greater than the smaller child, swap them
            if self._data[index].distance > self._data[smaller_child_index].distance:
                self._data[index], self._data[smaller_child_index] = self._data[smaller_child_index], self._data[index]
                index = smaller_child_index
            else:
                break
