class DisjointSet:
    def __init__(self, size):
        # Check if size is non-negative
        if size < 0:
            raise ValueError("size must be non-negative")
        
        # Initialize the size of the disjoint set
        self.size = size
        
        # Initialize parent pointers for each element
        # Initially, each element is its own parent
        self.parent = [i for i in range(size + 1)]
        
        # Initialize ranks for each element
        # Ranks are used to keep track of the depth of trees
        self.rank = [0] * (size + 1)

    def find(self, x):
        """
        Find operation: returns the representative element of the set containing x.
        Utilizes path compression to flatten the tree structure for faster future finds.
        """
        # If x is not the parent of itself, recursively find the parent
        if self.parent[x] != x:
            # Path compression: set the parent of x to its root
            self.parent[x] = self.find(self.parent[x])
        
        # Return the root of the set containing x
        return self.parent[x]

    def union(self, x, y):
        """
        Union operation: merges the sets containing elements x and y.
        Utilizes union by rank to optimize the merging process.
        """
        # Find the roots of the sets containing x and y
        rootX = self.find(x)
        rootY = self.find(y)

        # If x and y belong to different sets, merge them
        if rootX != rootY:
            # Compare ranks to determine which tree to attach to the other
            if self.rank[rootX] > self.rank[rootY]:
                # Attach the root of the smaller tree (rootY) to the root of the larger tree (rootX)
                self.parent[rootY] = rootX
            elif self.rank[rootX] < self.rank[rootY]:
                # Attach the root of the smaller tree (rootX) to the root of the larger tree (rootY)
                self.parent[rootX] = rootY
            else:
                # If ranks are equal, arbitrarily choose one root as the parent and increment its rank
                self.parent[rootY] = rootX
                self.rank[rootX] += 1
