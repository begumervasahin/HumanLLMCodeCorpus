import networkx as nx
from pqdict import PQDict

def prim_minimum_spanning_tree(graph, start_node):
    """
    Find the Minimum Spanning Tree (MST) of a graph using Prim's algorithm.

    Args:
        graph: NetworkX graph object.
        start_node: Starting node for the algorithm.

    Returns:
        mst_edges: List of edges forming the Minimum Spanning Tree.
    """
    num_nodes = graph.number_of_nodes()
    num_edges_to_visit = num_nodes - 1
    visited_nodes = set()
    edge_priority_queue = PQDict()
    mst_edges = []

    current_node = start_node

    while len(mst_edges) < num_edges_to_visit:
        # Visit neighbors of the current node and add edges to the priority queue
        for neighbor_node in graph.neighbors(current_node):
            if neighbor_node not in visited_nodes and current_node not in visited_nodes:
                # Add the edge to the priority queue if it's not already added
                if (current_node, neighbor_node) not in edge_priority_queue and \
                   (neighbor_node, current_node) not in edge_priority_queue:
                    weight = graph.edges[current_node, neighbor_node]['weight']
                    edge_priority_queue.additem((current_node, neighbor_node), weight)

        # Mark the current node as visited
        visited_nodes.add(current_node)

        # Get the next edge with the minimum weight from the priority queue
        while True:
            edge, weight = edge_priority_queue.popitem()
            if edge[1] not in visited_nodes:
                break

        # Add the edge to the MST
        mst_edges.append(edge)

        # Update the current node for the next iteration
        current_node = edge[1]

    return mst_edges

# Example usage:
# Create a graph
graph = nx.Graph()
graph.add_weighted_edges_from([(0, 1, 4), (0, 2, 8), (1, 2, 2), (1, 3, 6), (2, 3, 3)])

# Call the prim_minimum_spanning_tree function to find the Minimum Spanning Tree (MST)
mst = prim_minimum_spanning_tree(graph, 0)
print("Minimum Spanning Tree:", mst)
