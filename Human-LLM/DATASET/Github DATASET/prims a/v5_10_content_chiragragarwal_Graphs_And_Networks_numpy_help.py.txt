import numpy as np

# Define a string representing a matrix
matrix_string = '1 2 3 4; 5 6 7 8; 0 1 2 3'

# Convert the string into a NumPy matrix
matrix = np.matrix(matrix_string)

# Check the shape of the matrix
shape = matrix.shape
print("Shape of the matrix:", shape)

# Access elements of the matrix
element_0_1 = matrix[0, 1]
element_2_3 = matrix[2, 3]
print("Element at [0, 1]:", element_0_1)
print("Element at [2, 3]:", element_2_3)

# Modify an element of the matrix
matrix[0, 1] = -8

# Access a column of the matrix
column_1 = matrix[:, 1]
print("Column 1 of the matrix:")
print(column_1)
