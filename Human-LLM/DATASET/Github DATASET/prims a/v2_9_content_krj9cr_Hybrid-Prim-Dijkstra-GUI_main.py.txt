import sys

# Check Python version and import the appropriate Tkinter module
if sys.version_info[0] < 3:
    import Tkinter as Tk
else:
    import tkinter as Tk

# Import the App class from the App module
from App import App

# Define the main function
def main():
    # Create a Tkinter root window
    root = Tk.Tk()
    # Set the title of the window
    root.wm_title("Prim & Dijkstra's Algorithms")
    # Create an instance of the App class and pass the root window
    app = App(root)
    # Start the Tkinter event loop
    Tk.mainloop()

# If this script is executed directly, call the main function
if __name__ == "__main__":
    main()
