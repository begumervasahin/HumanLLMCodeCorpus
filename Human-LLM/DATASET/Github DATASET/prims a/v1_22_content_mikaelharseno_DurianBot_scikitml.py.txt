import pandas as pd
import pickle
import os
from sklearn import model_selection
from sklearn.linear_model import LogisticRegression

# Assuming 'results.txt' contains data in the format mentioned
basePath = os.path.dirname(os.path.abspath(__file__))
path = os.path.join(basePath, 'results.txt')

# Reading data from 'results.txt'
df = pd.read_csv(path, sep=" ", header=None)
df.columns = ['worstcaseprob', 'scout_report', 'actualvalue']

# Splitting data into features (X) and target (Y)
X, Y = df.iloc[:, :2], df.iloc[:, 2]

# Splitting data into train and test sets
test_size = 0.2
X_train, X_test, Y_train, Y_test = model_selection.train_test_split(X, Y, test_size=test_size)

# Creating and training a Logistic Regression model
model = LogisticRegression()
model.fit(X_train, Y_train)

# Printing predicted probabilities for test set
print(list(model.predict_proba(X_test)))

# Calculating and printing the accuracy of the model on the test set
accuracy = model.score(X_test, Y_test)
print("Accuracy:", accuracy)

# Saving the trained model to a file
filename = 'finalized_model.sav'
pickle.dump(model, open(filename, 'wb'))

# Loading the saved model from file
loaded_model = pickle.load(open(filename, 'rb'))

# Evaluating the loaded model on the test set
loaded_model_accuracy = loaded_model.score(X_test, Y_test)
print("Loaded Model Accuracy:", loaded_model_accuracy)
