class DisjointSet:
    def __init__(self, init_arr):
        self._disjoint_set = []
        if init_arr:
            self._initialize_disjoint_set(init_arr)

    def _initialize_disjoint_set(self, arr):
        unique_elements = set(arr)
        self._disjoint_set = [[item] for item in unique_elements]

    def _find_set_index(self, elem):
        for index, item in enumerate(self._disjoint_set):
            if elem in item:
                return index
        return None

    def find(self, elem):
        for item in self._disjoint_set:
            if elem in item:
                return item
        return None

    def union(self, elem1, elem2):
        index_elem1 = self._find_set_index(elem1)
        index_elem2 = self._find_set_index(elem2)
        
        if index_elem1 is not None and index_elem2 is not None and index_elem1 != index_elem2:
            self._disjoint_set[index_elem2] += self._disjoint_set[index_elem1]
            del self._disjoint_set[index_elem1]
        
        return self._disjoint_set

    def get(self):
        return self._disjoint_set
