import heapq

class GlobalVars:
    EDGES = []        # List to store edges
    VERTICES = set()  # Set to store vertices
    PARENT = {}       # Dictionary to store parent of each vertex
    RANK = {}         # Dictionary to store rank of each vertex
    MST = []          # List to store edges in the Minimum Spanning Tree (MST)

glb = GlobalVars()

# Function to read the graph from a file
def read_graph(file, delimiter):
    with open(file) as file_path:
        for line in file_path:
            # Split the line based on the delimiter
            v1, v2, weight = map(str.strip, line.split(delimiter))
            # Check if the edge (v2, v1, weight) already exists, if so, skip
            if (int(weight), v2, v1) in glb.EDGES:
                continue
            # Add the edge to the list of edges
            heapq.heappush(glb.EDGES, (int(weight), v1, v2))
            # Add vertices to the set of vertices
            glb.VERTICES.add(v1)
            glb.VERTICES.add(v2)
    return glb.EDGES

# Function to create a set for each vertex
def make_set(v):
    glb.PARENT[v] = v  # Set the parent of the vertex to itself
    glb.RANK[v] = 0     # Initialize the rank of the vertex to 0

# Function to find the root of the set containing a vertex (with path compression)
def find(v):
    if glb.PARENT[v] != v:
        glb.PARENT[v] = find(glb.PARENT[v])  # Path compression
    return glb.PARENT[v]

# Function to merge two sets (with union by rank)
def union(v1, v2):
    root1 = find(v1)
    root2 = find(v2)
    if root1 != root2:
        if glb.RANK[root1] > glb.RANK[root2]:
            glb.PARENT[root2] = root1
        else:
            glb.PARENT[root1] = root2
            if glb.RANK[root1] == glb.RANK[root2]:
                glb.RANK[root2] += 1

# Kruskal's algorithm to find Minimum Spanning Tree (MST)
def kruskals():
    total_weight = 0  # Variable to store the total weight of the MST
    for v in glb.VERTICES:
        make_set(v)  # Create a set for each vertex
    while glb.EDGES:
        # Pop the edge with the smallest weight
        weight, v1, v2 = heapq.heappop(glb.EDGES)
        if find(v1) != find(v2):  # Check if including this edge forms a cycle
            union(v1, v2)  # Merge the sets containing v1 and v2
            total_weight += weight  # Update the total weight of the MST
            glb.MST.append((v1, v2, str(weight), str(total_weight)))  # Add edge to MST
    return (glb.MST, total_weight)

# Example usage:
graph_file = "graph.txt"
delimiter = ","
read_graph(graph_file, delimiter)
mst, total_weight = kruskals()
print("Minimum Spanning Tree:")
for edge in mst:
    print(edge)
print("Total Weight of MST:", total_weight)
