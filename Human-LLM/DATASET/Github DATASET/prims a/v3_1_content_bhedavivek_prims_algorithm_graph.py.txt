class Vertex:
    def __init__(self, id, parent_id, distance, position=0):
        """
        Represents a vertex in the graph.

        Args:
        - id: Unique identifier of the vertex.
        - parent_id: Parent vertex id (if any).
        - distance: Distance from the parent vertex.
        - position: Position attribute (default is 0).
        """
        self.id = id
        self.parent_id = parent_id
        self.distance = distance
        self.position = position

class Graph:
    def __init__(self, num_vertices):
        """
        Represents a graph structure.

        Args:
        - num_vertices: Number of vertices in the graph.
        """
        # Initialize the adjacency list with empty lists for each vertex
        self.adjacency_list = [[] for _ in range(num_vertices)]

    def insert_edge(self, source_vertex, target_vertex, weight):
        """
        Inserts an edge between vertices with a given weight.

        Args:
        - source_vertex: Source vertex id.
        - target_vertex: Target vertex id.
        - weight: Weight of the edge.
        """
        # Create a new vertex object for the target vertex with the source vertex as parent and given weight
        new_vertex = Vertex(target_vertex, source_vertex, weight)
        # Add the new vertex to the adjacency list of the source vertex
        self.adjacency_list[source_vertex - 1].append(new_vertex)

# Example usage:
if __name__ == "__main__":
    # Create a new graph with 5 vertices
    graph = Graph(5)
    # Insert edges between vertices with corresponding weights
    graph.insert_edge(1, 2, 10)
    graph.insert_edge(1, 3, 15)
    graph.insert_edge(2, 4, 20)
    graph.insert_edge(3, 5, 25)
    # Print the adjacency list of the graph
    print(graph.adjacency_list)
