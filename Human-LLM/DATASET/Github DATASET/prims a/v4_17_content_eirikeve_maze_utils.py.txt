import argparse

# Define a function to check if a value is a positive integer
def check_positive(value):
    try:
        # Attempt to convert the value to an integer
        value = int(value)
    except ValueError:
        # If the conversion fails, raise an error indicating that the value is not an integer
        raise argparse.ArgumentTypeError(f"{value} is not an integer value")
    else:
        # If the conversion succeeds, check if the value is positive
        if value < 0:
            # If the value is negative, raise an error indicating that it's not positive
            raise argparse.ArgumentTypeError(f"{value} is not a positive integer value")
    # If the value is a positive integer, return it
    return value
