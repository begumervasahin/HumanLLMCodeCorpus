# Define a class to represent a Node
class Node:
    # Constructor to initialize the Node object
    def __init__(self, x_location, y_location, k_value, adjacency_list):
        self.x_location = x_location  # X coordinate of the node
        self.y_location = y_location  # Y coordinate of the node
        self.k_value = k_value  # Some value associated with the node
        self.adjacency_list = adjacency_list  # List of adjacent nodes
    
    # Method to check equality between two nodes
    def __eq__(self, other_node):
        # Two nodes are equal if their x, y locations, and k values are equal
        return (self.x_location == other_node.x_location and 
                self.y_location == other_node.y_location and 
                self.k_value == other_node.k_value)

# Creating instances of Node class for demonstration
node1 = Node(1, 2, 3, [4, 5, 6])  # A node at (1, 2) with k value 3 and adjacency list [4, 5, 6]
node2 = Node(1, 2, 3, [4, 5, 6])  # Another node with the same properties as node1
node3 = Node(4, 5, 6, [7, 8, 9])   # A different node with different properties

# Testing equality between nodes
print("Is node1 equal to node2?", node1 == node2)  # This should print True
print("Is node1 equal to node3?", node1 == node3)  # This should print False
