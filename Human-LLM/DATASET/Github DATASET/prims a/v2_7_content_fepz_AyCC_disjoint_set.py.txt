class DisjointSets:
    def __init__(self, elements):
        # Initialize the disjoint sets with elements and set labels
        self.elements = list(elements)
        self.sets = list(range(len(elements)))  # Each element starts in its own set
        self.ranks = [1] * len(elements)  # Keep track of the rank of each set

    def find(self, x):
        """Find the label of the set containing object x."""
        # Initialize indices
        i = r = self.elements.index(x)
        
        # Find the root of the set
        while self.sets[r] != r:
            r = self.sets[r]
        
        # Path compression: update the parent of each element on the path to the root
        while i != r:
            j = self.sets[i]
            self.sets[i] = r
            i = j
        
        # Return the label of the root set
        return r

    def merge(self, a, b):
        """Merge the sets labeled a and b; we assume a != b."""
        # Check the ranks to optimize the merge operation
        if self.ranks[a] == self.ranks[b]:
            self.ranks[a] += 1  # Increment the rank of the root set
            self.sets[b] = a  # Make a the parent of b
        else:
            # Merge the smaller set into the larger set
            if self.ranks[a] > self.ranks[b]:
                self.sets[b] = a
            else:
                self.sets[a] = b


# Example usage:
elements = ['A', 'B', 'C', 'D', 'E']
disjoint_sets = DisjointSets(elements)

print("Initial sets:", disjoint_sets.sets)
print("Find 'A':", disjoint_sets.find('A'))

disjoint_sets.merge(0, 1)
print("Sets after merging 'A' and 'B':", disjoint_sets.sets)
print("Find 'B':", disjoint_sets.find('B'))

disjoint_sets.merge(2, 3)
print("Sets after merging 'C' and 'D':", disjoint_sets.sets)
print("Find 'D':", disjoint_sets.find('D'))
