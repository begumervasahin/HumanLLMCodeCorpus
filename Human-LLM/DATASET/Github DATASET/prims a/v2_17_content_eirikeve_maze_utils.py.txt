import argparse

# Function to check if a value is a positive integer
def check_positive(value):
    try:
        # Try converting the value to an integer
        value = int(value)
    except ValueError:
        # If conversion fails, raise an error indicating that the value is not an integer
        raise argparse.ArgumentTypeError(f"{value} is not an integer")
    else:
        # If conversion succeeds, check if the value is positive
        if value < 0:
            # If it's not positive, raise an error indicating that the value is not positive
            raise argparse.ArgumentTypeError(f"{value} is not a positive integer")
    return value

# Main function to parse command-line arguments and print the result
def main():
    # Create an ArgumentParser object with a description
    parser = argparse.ArgumentParser(description="Example Argument Parser")
    
    # Add an argument for a positive integer value, using the check_positive function for validation
    parser.add_argument("integer", type=check_positive, help="A positive integer value")
    
    # Parse the command-line arguments
    args = parser.parse_args()
    
    # Print the provided positive integer value
    print("The provided positive integer value is:", args.integer)

# Check if the script is being run as the main program
if __name__ == "__main__":
    # If it is, call the main function
    main()
