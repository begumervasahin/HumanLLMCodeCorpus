# Import necessary classes and functions
from Weighted_Graph import Weighted_Graph
from functions import update, total_sum

# Define the function to find Minimum Spanning Tree using Prim's algorithm
def find_minimum_spanning_tree(edge_file_name, start_vertex=0, draw=False):
    # Initialize the Minimum Spanning Tree
    minimum_spanning_tree = ({start_vertex}, [])

    # Create a weighted graph object from the provided edge file
    graph = Weighted_Graph(edge_file_name)

    # Optionally draw the original graph
    if draw:
        graph.draw_graph()

    # Continue expanding the Minimum Spanning Tree until it covers all vertices
    while minimum_spanning_tree[0] != graph.vertex_set():
        # Update the Minimum Spanning Tree using Prim's algorithm
        minimum_spanning_tree = update(minimum_spanning_tree, graph)

        # Optionally draw the current subgraph of the Minimum Spanning Tree
        if draw:
            graph.draw_subgraph(minimum_spanning_tree)

    # Calculate the total cost of the Minimum Spanning Tree
    total_cost = total_sum(minimum_spanning_tree, graph)

    # Print the total cost of the Minimum Spanning Tree
    print("The minimum spanning tree costs: " + str(total_cost))

    # Return the Minimum Spanning Tree
    return minimum_spanning_tree

# Example usage when running the script
if __name__ == "__main__":
    # Replace "your_edge_file_name.txt" with the actual edge file name
    edge_file_name = "your_edge_file_name.txt"

    # Find and display the Minimum Spanning Tree
    minimum_spanning_tree = find_minimum_spanning_tree(edge_file_name, start_vertex=0, draw=True)
