import pygame
from time import sleep

# Define colors
RED = (255, 0, 0)
BLUE = (0, 0, 255)
GREEN = (0, 255, 0)
BLACK = (0, 0, 0)
WHITE = (255, 255, 255)

# Define Position class to represent a position on the board
class Position:
    def __init__(self, x, y):
        self.x = x
        self.y = y

# Define BoardDisplay class to manage the display of the board
class BoardDisplay:
    def __init__(self, num_rows, num_columns, width):
        # Initialize pygame
        pygame.init()
        # Set display dimensions based on the number of rows, columns, and width of each cell
        self.display = pygame.display.set_mode((num_columns * width, num_rows * width))
        # Store number of rows, columns, and cell width
        self.num_rows = num_rows
        self.num_columns = num_columns
        self.width = width
        # Fill the display with a black background
        self.display.fill(BLACK)

    # Method to draw a cell on the board at specified row and column with given color
    def draw_position(self, row, col, color):
        # Ensure row and column inputs are integers
        assert(type(row) is int)
        assert(type(col) is int)
        # Draw a rectangle representing the cell at the specified position with the given color
        pygame.draw.rect(self.display, color, pygame.Rect(col * self.width, row * self.width, self.width, self.width))

    # Method to start displaying a trace of positions on the board
    def start(self, trace):
        # Flag to control the main loop
        running = True
        # Iterate over each position in the trace
        for p in trace:
            # Check for any events (e.g., window close event)
            for event in pygame.event.get():
                # If the event is a window close event, set running flag to False to exit the loop
                if event.type == pygame.QUIT:
                    running = False
            # If running flag is False, break out of the loop
            if not running:
                break

            # Draw the current position in green
            self.draw_position(p.x, p.y, GREEN)
            # Update the display to show the changes
            pygame.display.flip()
            # Pause briefly to visualize the movement
            sleep(0.1)
            # Erase the current position by drawing it in white
            self.draw_position(p.x, p.y, WHITE)
            # Update the display again
            pygame.display.flip()

        # Quit pygame when the loop is finished
        pygame.quit()

# Example usage
if __name__ == "__main__":
    # Define board dimensions and cell width
    num_rows = 10
    num_columns = 10
    width = 30
    # Initialize BoardDisplay object
    bd = BoardDisplay(num_rows, num_columns, width)
    # Define a trace of positions
    trace = [Position(2, 3), Position(4, 5), Position(6, 7)]  # Example trace
    # Start displaying the trace on the board
    bd.start(trace)
