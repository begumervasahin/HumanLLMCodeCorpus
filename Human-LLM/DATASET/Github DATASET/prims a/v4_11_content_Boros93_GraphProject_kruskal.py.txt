# Import necessary libraries
import networkx as nx
import operator
import matplotlib.pyplot as plt
from disjoint_set import DisjointSet

# Create an empty graph and disjoint set
G = nx.Graph()
disjoint_set = DisjointSet(['a', 'b', 'c', 'd'])

# Add nodes to the graph with positions and keys
G.add_node('a', pos=(0, 1), key=0)
G.add_node('b', pos=(2, 1), key=0)
G.add_node('c', pos=(1, 0), key=0)
G.add_node('d', pos=(1, 2), key=0)

# Add edges to the graph with weights
G.add_edge('a', 'b', weight=2)
G.add_edge('a', 'c', weight=10)
G.add_edge('b', 'c', weight=90)
G.add_edge('b', 'd', weight=1)
G.add_edge('d', 'a', weight=3)

# Get edge and node attributes
edges = nx.get_edge_attributes(G, 'weight')
pos = nx.get_node_attributes(G, 'pos')

# Initialize an empty graph for Kruskal's algorithm
KruskalG = nx.Graph()

# Sort edges by weight
sorted_edges = sorted(edges.items(), key=operator.itemgetter(1))

# Iterate through sorted edges
for uv in sorted_edges:
    u = uv[0][0]
    v = uv[0][1]
    
    # Check if adding edge (u, v) forms a cycle
    if disjoint_set.find(u) != disjoint_set.find(v):
        disjoint_set.union(u, v)
        KruskalG.add_edge(u, v, weight=uv[1], color='b')  # Add edge to Kruskal's graph with blue color
    else:
        KruskalG.add_edge(u, v, weight=uv[1], color='r')  # Add edge to Kruskal's graph with red color

# Get edges and corresponding colors and weights for visualization
edgesK = KruskalG.edges()
colors = [KruskalG[u][v]['color'] for u, v in edgesK]
weights = [KruskalG[u][v]['weight'] for u, v in edgesK]

# Plot original graph and Kruskal's MST side by side
plt.subplot(121)
nx.draw(G, pos, with_labels=True)
nx.draw_networkx_edge_labels(G, pos, edge_labels=edges)
plt.subplot(122)
nx.draw(G, pos, with_labels=True)
nx.draw_networkx_edge_labels(G, pos, edge_labels=edges)
nx.draw(KruskalG, pos, edges=edgesK, edge_color=colors)
plt.show()
