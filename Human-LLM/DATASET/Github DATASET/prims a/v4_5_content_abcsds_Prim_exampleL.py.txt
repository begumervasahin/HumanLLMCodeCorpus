# Import necessary libraries
try:
    import matplotlib.pyplot as plt
    import networkx as nx
except ImportError as e:
    raise ImportError("Required libraries not found:", e)

# Import Prim's algorithm function
from prim import prim

# Read the graph from an edgelist file
try:
    G = nx.read_edgelist("flights.edgelist")
except FileNotFoundError:
    raise FileNotFoundError("Edgelist file not found")

# Find the minimal spanning tree using Prim's algorithm starting from 'Madrid'
mst = prim(G, 'Madrid')

# Find other edges not in the minimal spanning tree
othr = [n for n in G.edges() if n not in mst]

# Print minimal spanning tree
print("Minimal Spanning Tree:", mst)

# Generate layout for visualization
pos = nx.spring_layout(G)

# Draw nodes
nx.draw_networkx_nodes(G, pos, node_size=500)

# Draw edges of the minimal spanning tree
nx.draw_networkx_edges(G, pos, edgelist=mst, width=6)

# Draw other edges with different style
nx.draw_networkx_edges(G, pos, edgelist=othr, width=6, alpha=0.5, edge_color='b', style='dashed')

# Draw node labels
nx.draw_networkx_labels(G, pos, font_size=20, font_family='sans-serif')

# Turn off axis
plt.axis('off')

# Display the graph
plt.show()
