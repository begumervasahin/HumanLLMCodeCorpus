import sys

class Node(object):
    def __init__(self, x, y, i):
        self.x = x
        self.y = y
        self.idx = i
        self.key = sys.maxsize  # In Python 3, sys.maxsize is used instead of sys.maxint
        self.parent = None

    def __str__(self):
        return str(self.idx) + ': (' + str(self.x) + ',' + str(self.y) + ')'

# Example usage:
node1 = Node(1, 2, 0)
print(node1)  # Output: 0: (1,2)
print("Key:", node1.key)  # Output: Key: 9223372036854775807 (maximum size of an integer)
