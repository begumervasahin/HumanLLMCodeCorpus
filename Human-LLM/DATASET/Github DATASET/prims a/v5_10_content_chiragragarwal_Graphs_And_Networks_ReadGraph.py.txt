import numpy
from path import Path

class GraphReader:
    '''
    This class reads a graph matrix from a text file and stores it in a numpy matrix.
    It then returns the same matrix.
    '''

    def __init__(self, file_path):
        '''
        Initialize the GraphReader object with the input file path.
        '''
        self.file_path = file_path

    def read_graph_matrix(self):
        '''
        Read the contents of the file and convert it into a numpy matrix.
        '''
        # Read the contents of the file
        graph_content = self._read_file_contents()

        # Convert the string representation of the graph into a numpy matrix
        graph_matrix = numpy.matrix(graph_content)

        return graph_matrix

    def _read_file_contents(self):
        '''
        Helper method to read the contents of the file.
        '''
        with open(self.file_path, 'r') as file:
            graph_content = file.read()

        return graph_content
