from Prims import Prims

def display_menu():
    print("MST Theory:")
    print("Choose 1 for Prims")
    print("Choose 2 for Kruskals*")
    print("\nChoose 9 to exit.")

def get_choice():
    return input("Please make a choice: ")

def handle_prims():
    file_name = input("Please enter the filename of your graph file: ")
    start_vertex = input("Choose a starting vertex: ")
    draw_graph = input("Would you like the graph to be drawn for you? (Y/N) ").lower()
    
    if draw_graph == "y":
        Prims(file_name, int(start_vertex), True)
    elif draw_graph == "n":
        Prims(file_name, int(start_vertex), False)
    else:
        print("You didn't choose yes or no. Going back to the main menu...")

def main():
    while True:
        display_menu()
        choice = get_choice()
        
        if choice == "1":
            handle_prims()
        elif choice == "2":
            print("*Kruskals is not available right now.")
        elif choice == "9":
            print("Thank you. Goodbye!")
            break
        else:
            print("Not a valid choice!")

if __name__ == "__main__":
    main()
