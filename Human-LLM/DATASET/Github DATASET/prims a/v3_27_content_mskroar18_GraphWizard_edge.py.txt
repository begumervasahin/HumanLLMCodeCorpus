class Edge:
    def __init__(self, vertex1, vertex2, weight, extra=0, selected=False):
        self.vertex1 = vertex1
        self.vertex2 = vertex2
        self.weight = weight
        self.extra = extra
        self.selected = selected

    def __repr__(self):
        return f"Edge(Vertex1: {self.vertex1}, Vertex2: {self.vertex2}, Weight: {self.weight}, Extra: {self.extra}, Selected: {self.selected})"

# Example usage
if __name__ == "__main__":
    # Create some sample edges
    edge1 = Edge(vertex1=1, vertex2=2, weight=5)
    edge2 = Edge(vertex1=2, vertex2=3, weight=7, extra=2, selected=True)
    
    # Print the edges
    print("Edge 1:", edge1)
    print("Edge 2:", edge2)
