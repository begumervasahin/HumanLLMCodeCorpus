# This program is designed to load a graph from a text file and provide various functionalities related to the graph.

# Importing the Graph class from the Graph module
from Graph import Graph

class Repository:
    # Constructor method to initialize the repository with a graph loaded from a file
    def __init__(self, filename="graph1.txt"):
        self._loadFromFile(filename)

    # Method to load data from a file into the graph
    def _loadFromFile(self, filename):
        # Opening the file in read mode
        file = open(filename, 'r')
        # Reading the first line which contains the number of vertices in the graph
        text = file.readline()
        text = text.split()
        # Creating a new graph object with the specified number of vertices
        self.graph = Graph(int(text[0]))
        
        # Reading subsequent lines which contain edges information until the end of the file is reached
        text = file.readline()
        while text != '':
            text = text.split()
            # Adding edges to the graph (undirected graph, so adding edges in both directions)
            self.graph.addEdge(int(text[0]), int(text[1]))
            self.graph.addEdge(int(text[1]), int(text[0]))
            text = file.readline()
        
        # Closing the file
        file.close()

    # Method to get the number of vertices in the graph
    def getNrVertices(self):
        return self.graph.getNrVertices()

    # Method to get the number of edges in the graph
    def getNrEdges(self, g):
        return g.getNrEdges()

    # Method to check if there is an edge between two vertices
    def isEdge(self, v1, v2):
        return self.graph.isEdge(v1, v2)

    # Method to get the degree of a vertex
    def getDegree(self, v):
        return self.graph.getDegree(v)

    # Method to get the neighbors of a vertex
    def getNeighbours(self, v):
        return self.graph.parseNeighbours(v)

    # Method to show all vertices and their respective neighbors
    def showAll(self):
        return self.graph.parseAll()

    # Method to add an edge with a certain weight between two vertices
    def addEdge(self, v1, v2, c):
        return self.graph.addEdge(v1, v2, c)
