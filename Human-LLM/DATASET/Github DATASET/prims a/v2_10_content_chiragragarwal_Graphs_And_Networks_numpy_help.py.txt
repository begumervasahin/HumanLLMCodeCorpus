import numpy as np

# Define the matrix as a string
matrix_string = '1 2 3 4 ; 5 6 7 8 ; 0 1 2 3'

# Create a numpy matrix from the string
matrix = np.matrix(matrix_string)

# Print the shape of the matrix
print("The matrix has a shape of:", matrix.shape)

# Access and print an element in the matrix
print("The element at row 0, column 1 is:", matrix[0, 1])

# Access and print another element in the matrix
print("The element at row 2, column 3 is:", matrix[2, 3])

# Modify an element in the matrix
matrix[0, 1] = -8
print("Modified matrix:")
print(matrix)

# Print the second column of the matrix
print("The second column of the matrix is:")
print(matrix[:, 1])
