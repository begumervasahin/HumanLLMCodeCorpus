import sys
from Node import Node

class MinQueue(object):
    def __init__(self, n):
        self.list = n[:]  # Creating a copy of the input list

    def __len__(self):
        return len(self.list)

    def contains(self, x):
        if x in self.list:
            return True
        return False

    def extractMin(self):
        min_val = sys.maxint  # Setting initial minimum value to maximum integer value
        min_node = None  # Placeholder for the node with minimum key
        for node in self.list:
            if node.key < min_val:
                min_val = node.key
                min_node = node
        self.list.remove(min_node)  # Removing the node with minimum key from the list
        return min_node.idx  # Returning the index of the node with minimum key
