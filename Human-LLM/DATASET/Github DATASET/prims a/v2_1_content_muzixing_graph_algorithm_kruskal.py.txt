def kruskal(graph):
    # Check if graph is a dictionary
    assert isinstance(graph, dict)
    
    # Get a list of nodes in the graph
    nodes = list(graph.keys())
    
    # Initialize an empty set to keep track of visited nodes
    visited = set()
    
    # Initialize an empty list to store the final path
    path = []
    
    # Initialize a variable to store the next node to visit
    next_node = None
    
    # Continue until all nodes are visited
    while len(visited) < len(nodes):
        # Set the initial distance to infinity
        distance = float('inf')
        
        # Iterate over all nodes in the graph
        for s in nodes:
            for d in nodes:
                # Skip if both nodes are already visited or if it's the same node
                if s in visited and d in visited or s == d:
                    continue
                
                # Update the distance and next node if a shorter path is found
                if graph[s][d] < distance:
                    distance = graph[s][d]
                    pre = s
                    next_node = d
        
        # Add the edge to the path and mark both nodes as visited
        path.append((pre, next_node))
        visited.add(pre)
        visited.add(next_node)
    
    # Return the final path
    return path
