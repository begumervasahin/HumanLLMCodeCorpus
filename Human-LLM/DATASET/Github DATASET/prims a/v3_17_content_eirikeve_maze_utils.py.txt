import argparse

def check_positive_integer(value):
    try:
        # Attempt to convert the value to an integer
        int_value = int(value)
    except ValueError:
        # If conversion fails, raise an error indicating that the value is not an integer
        raise argparse.ArgumentTypeError(f"Invalid integer: {value}")
    else:
        # If conversion succeeds, check if the value is positive
        if int_value < 0:
            # If it's not positive, raise an error indicating that the value is not positive
            raise argparse.ArgumentTypeError(f"Negative integer not allowed: {int_value}")
    return int_value

def parse_arguments():
    # Create an ArgumentParser object with a description
    parser = argparse.ArgumentParser(description="Example Argument Parser")

    # Add an argument for a positive integer value, using the check_positive_integer function for validation
    parser.add_argument("integer", type=check_positive_integer, help="A positive integer value")

    # Parse the command-line arguments
    return parser.parse_args()

def main():
    # Parse command-line arguments
    args = parse_arguments()

    # Print the provided positive integer value
    print("The provided positive integer value is:", args.integer)

if __name__ == "__main__":
    main()
