import sys
import kruskals
import prims
import glb

def usage():
    """
    Prints the usage instructions for the script.
    """
    print("USAGE: main.py [\"file.txt\"] [algo] where p = prims , k = kruskals")
    sys.exit()

def main():
    """
    Main function of the script.
    """
    if len(sys.argv) < 3:
        usage()

    file = sys.argv[1]
    delimiter = ' '
    algo = sys.argv[2]

    if algo == 'p':
        # Execute Prim's algorithm
        prims.read_graph(file, delimiter)
        graph, tdist = prims.prims()
        glb.print_pretty()
        print(f"Total Distance : {tdist}")
    elif algo == 'k':
        # Execute Kruskal's algorithm
        kruskals.read_graph(file, delimiter)
        graph, tdist = kruskals.kruskals()
        glb.print_pretty()
        print(f"Total Distance : {tdist}")
    else:
        # Invalid algorithm specified, display usage
        usage()

if __name__ == "__main__":
    main()
