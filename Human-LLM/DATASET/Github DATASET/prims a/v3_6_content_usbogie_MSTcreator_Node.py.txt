class Node:
    def __init__(self, x, y, k, adjacency_list):
        """
        Initialize a Node object.

        Parameters:
            x (int): X coordinate of the node.
            y (int): Y coordinate of the node.
            k (int): Some value associated with the node.
            adjacency_list (list): List of adjacent nodes.
        """
        self.x = x
        self.y = y
        self.k = k
        self.adjacency_list = adjacency_list

    def __eq__(self, other):
        """
        Check if two nodes are equal.

        Two nodes are considered equal if their x, y coordinates, and k value are the same.

        Parameters:
            other (Node): Another Node object for comparison.

        Returns:
            bool: True if the nodes are equal, False otherwise.
        """
        return (self.x == other.x and
                self.y == other.y and
                self.k == other.k)

# Demonstration
node1 = Node(1, 2, 3, [4, 5, 6])
node2 = Node(1, 2, 3, [4, 5, 6])
node3 = Node(4, 5, 6, [7, 8, 9])

print("Is node1 equal to node2?", node1 == node2)  # This should print True
print("Is node1 equal to node3?", node1 == node3)  # This should print False
