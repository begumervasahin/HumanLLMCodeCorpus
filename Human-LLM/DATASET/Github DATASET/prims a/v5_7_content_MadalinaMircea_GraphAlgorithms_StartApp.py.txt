# This script initializes an application for managing data through a user interface.

# Importing necessary modules
from Repository import Repository
from Controller import Controller
from UI import UI

def main():
    # Creating a repository instance
    data_repository = Repository()

    # Creating a controller instance with the repository
    data_controller = Controller(data_repository)

    # Creating a user interface instance with the controller
    user_interface = UI(data_controller)

    # Starting the user interface
    user_interface.start()

if __name__ == "__main__":
    main()
