import numpy as np

# Define the matrix as a string
mat = '1 2 3 4 ; 5 6 7 8 ; 0 1 2 3'

# Create a numpy matrix from the string
graphMat = np.matrix(mat)

# Print the shape of the matrix
print("Shape of the matrix:", graphMat.shape)

# Access and print an element in the matrix
print("Element at (0,1):", graphMat[0, 1])

# Access and print another element in the matrix
print("Element at (2,3):", graphMat[2, 3])

# Modify an element in the matrix
graphMat[0, 1] = -8
print("Modified matrix:")
print(graphMat)

# Print the second column of the matrix
print("Second column:")
print(graphMat[:, 1])
