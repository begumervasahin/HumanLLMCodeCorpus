import numpy
from pathlib import Path

class ReadGraph:
    '''
    This class reads a graph matrix from a text file and stores it in a numpy matrix.
    It returns the same matrix.
    '''

    def __init__(self, input_file):
        # Initialize the ReadGraph object with the provided input file path
        self.input_file = input_file

    def read(self):
        '''
        Reads the contents of the input file and converts it into a matrix.
        '''

        # Open the input file and read its contents
        with open(self.input_file, 'r') as file:
            graph_string = file.read()

        # Convert the string representation of the graph into a numpy matrix
        graph_matrix = numpy.matrix(graph_string)

        # Return the resulting matrix
        return graph_matrix

# Example usage:
if __name__ == "__main__":
    # Specify the path to the input file containing the graph matrix
    input_file_path = Path("your_graph_file.txt")

    # Create an instance of the ReadGraph class with the specified input file path
    graph_reader = ReadGraph(input_file_path)

    # Read the graph matrix from the file
    graph_matrix = graph_reader.read()

    # Print the resulting graph matrix
    print(graph_matrix)
