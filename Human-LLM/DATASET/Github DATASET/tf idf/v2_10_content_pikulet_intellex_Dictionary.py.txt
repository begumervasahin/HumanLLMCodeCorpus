# Function to store data in a file
def store_data(file_path, data):
    # Open the file in write mode
    with open(file_path, 'w') as file:
        # Write each key-value pair to the file
        for key, value in data.items():
            file.write(f"{key}: {value}\n")

# Class definition for the dictionary
class Dictionary:
    # Class constants for indices
    DF = 0
    IDF = 0
    TERMID = 1
    TERM_OFFSET = 1

    # Constructor to initialize the dictionary
    def __init__(self, file):
        # Initialize an empty dictionary to store terms and their properties
        self.terms = {}
        # Store the file path where the data will be saved
        self.file = file
        # Initialize a variable to keep track of the total number of documents
        self.total_num_documents = 0

    # Method to check if a term is present in the dictionary
    def has_term(self, t):
        return t in self.terms

    # Method to get the term ID for a given term
    def get_termID(self, t):
        return self.terms[t][Dictionary.TERMID]

    # Method to add a new term to the dictionary
    def add_term(self, t, termID):
        # Initialize the term with document frequency 1 and the provided term ID
        self.terms[t] = [1, termID]

    # Method to retrieve all terms in the dictionary
    def get_terms(self):
        return self.terms

    # Method to increment the document frequency of a term
    def add_df(self, t):
        # If the term already exists, increment its document frequency, else initialize it with frequency 1
        if t in self.terms:
            self.terms[t][Dictionary.DF] += 1
        else:
            self.terms[t] = [1, None]

    # Method to set the IDF (Inverse Document Frequency) for a term
    def set_idf(self, t, idf):
        # Set the IDF value for the term
        self.terms[t][Dictionary.IDF] = idf

    # Method to set the offset for a term
    def set_offset(self, t, offset):
        # Set the offset value for the term
        self.terms[t][Dictionary.TERM_OFFSET] = offset

    # Method to save the dictionary data to disk
    def save_to_disk(self):
        # Call the store_data function to save the data to the specified file
        store_data(self.file, self.terms)

# Example usage of the Dictionary class
if __name__ == "__main__":
    # Initialize a Dictionary object with the specified file path
    dictionary = Dictionary("terms_data.txt")

    # Add terms and their corresponding term IDs to the dictionary
    terms = ["apple", "banana", "orange"]
    for idx, term in enumerate(terms):
        dictionary.add_term(term, idx)

    # Increment document frequencies for each term
    for term in terms:
        dictionary.add_df(term)

    # Set IDF for a specific term
    dictionary.set_idf("apple", 0.5)

    # Set offset for a specific term
    dictionary.set_offset("apple", 100)

    # Save the dictionary data to disk
    dictionary.save_to_disk()
