import pandas as pd
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.metrics.pairwise import linear_kernel

# Load the movie dataset
cinema = pd.read_csv("tmdb_5000_movies.csv", encoding="latin-1")

# Display the first few rows of the dataset
cinema.head(3)

# Get the total number of movies in the dataset
total_movies = len(cinema)

# Group movies by their original title and count their popularity
movie_grouped = cinema.groupby(['original_title']).agg({'popularity':'count'}).reset_index()

# Count the number of unique movie titles
unique_movie_count = len(cinema['original_title'].unique())

# Get the unique languages present in the dataset
unique_languages = cinema['original_language'].unique()

# Initialize the TF-IDF Vectorizer with English stop words
tfidf = TfidfVectorizer(stop_words='english')

# Fill missing values in the 'overview' column with an empty string
cinema['overview'] = cinema['overview'].fillna('')

# Compute the TF-IDF matrix
tfidf_matrix = tfidf.fit_transform(cinema['overview'])

# Get the shape of the TF-IDF matrix
matrix_shape = tfidf_matrix.shape

# Compute the cosine similarity matrix
cosine_sim = linear_kernel(tfidf_matrix, tfidf_matrix)

# Create a Series with movie titles as index and their corresponding indices as values
indices = pd.Series(cinema.index, index=cinema['original_title']).drop_duplicates()

# Define a function to recommend similar movies based on cosine similarity
def get_recommendations(title, cosine_sim=cosine_sim):
    # Get the index of the input movie title
    idx = indices[title]
    # Calculate similarity scores between the input movie and all other movies
    sim_scores = list(enumerate(cosine_sim[idx]))
    # Sort the movies based on similarity scores in descending order
    sim_scores = sorted(sim_scores, key=lambda x: x[1], reverse=True)
    # Exclude the input movie itself
    sim_scores = sim_scores[1:20]
    # Get the indices of the most similar movies
    movie_indices = [i[0] for i in sim_scores]
    # Return the titles of the most similar movies
    return cinema['original_title'].iloc[movie_indices]

# Example usage: Get recommendations for the movie "No Reservations"
recommended_movies = get_recommendations("No Reservations")
