import numpy as np
import distance

class Agnes:
    def fit(self, X, n_clusters):
        '''
        Parameters
        ----------
        X : numpy.ndarray
            Training data with shape (n_samples, n_features)
        n_clusters : int
            The number of clusters

        Returns
        -------
        y : numpy.ndarray
            Predicted cluster label per sample with shape (n_samples,)
        '''
        n_samples = X.shape[0]  # Get the number of samples
        clusters = [[i] for i in range(n_samples)]  # Initialize each sample as its own cluster
        
        # Merge clusters until the desired number of clusters is reached
        for j in reversed(range(n_clusters, n_samples)):
            centers = np.array([np.mean(X[cluster], axis=0).ravel() for cluster in clusters])  # Calculate cluster centers
            distances = np.apply_along_axis(distance.euclidean_distance, 1, centers, centers)  # Calculate distances between cluster centers
            near_indexes = np.unravel_index(np.argmin(distances + np.diag(np.full(j + 1, np.inf))), distances.shape)  # Find the two nearest clusters
            clusters[near_indexes[0]].extend(clusters[near_indexes[1]])  # Merge the nearest clusters
            del clusters[near_indexes[1]]

        # Assign cluster labels to each sample
        y = np.zeros(n_samples)
        for i, cluster in enumerate(clusters):
            y[cluster] = i

        return y
