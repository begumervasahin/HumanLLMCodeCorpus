from scraper import linkify, getLinks, getWords
from algorithms import tf, getDocLinkStrength, idf, tfidf

def get_user_input():
    return input("Enter the Name of Personality\n")

def retrieve_word_list(page_link):
    return getWords(page_link)

def retrieve_link_list(page_link):
    return getLinks(page_link)

def calculate_term_frequency(word_list):
    return tf(word_list)

def calculate_link_strength(link_list, term_frequency_dict):
    return getDocLinkStrength(link_list, term_frequency_dict)

def retrieve_unique_links(doc_links):
    return list(set(link for link, _ in doc_links))

def retrieve_top_documents(word_links, top_n=10):
    top_documents = []
    for link in word_links[:top_n]:
        temp_link = linkify(link)
        top_documents.append(getWords(temp_link))
    return top_documents

def calculate_inverse_document_frequency(term_frequency_dict, documents):
    return idf(term_frequency_dict, documents)

def calculate_tfidf(term_frequency_dict, inverse_document_frequency_dict):
    return tfidf(term_frequency_dict, inverse_document_frequency_dict)

def sort_tfidf_by_value(tfidf_dict):
    return sorted(tfidf_dict.items(), key=lambda x: x[1], reverse=True)

def extract_keywords(link_list, tfidf_dict):
    return [link for link, _ in getDocLinkStrength(link_list, tfidf_dict)]

def print_keywords(keywords):
    for keyword in keywords:
        print(keyword)

def main():
    name = get_user_input()
    page_link = linkify(name)
    word_list = retrieve_word_list(page_link)
    link_list = retrieve_link_list(page_link)
    term_frequency_dict = calculate_term_frequency(word_list)
    doc_links = calculate_link_strength(link_list, term_frequency_dict)
    unique_links = retrieve_unique_links(doc_links)
    top_documents = retrieve_top_documents(unique_links)
    inverse_document_frequency_dict = calculate_inverse_document_frequency(term_frequency_dict, top_documents)
    tfidf_dict = calculate_tfidf(term_frequency_dict, inverse_document_frequency_dict)
    sorted_tfidf = sort_tfidf_by_value(tfidf_dict)
    keywords = extract_keywords(link_list, tfidf_dict)
    print_keywords(keywords)

if __name__ == "__main__":
    main()
