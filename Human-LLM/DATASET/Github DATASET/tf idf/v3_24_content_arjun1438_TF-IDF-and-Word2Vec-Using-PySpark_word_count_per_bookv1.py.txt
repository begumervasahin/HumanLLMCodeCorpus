from pyspark import SparkConf, SparkContext
from operator import add
import re
import sys

class WordCountApp:
    def __init__(self, sc):
        self.sc = sc

    def process_document(self, document_path_and_contents):
        """Process a document to count words."""
        document_path, contents = document_path_and_contents
        document_name = document_path.split('/')[-1]
        words = re.findall(r'\w+', contents.lower())
        return [((document_name, word), 1) for word in words]

    def count_words(self, input_path, output_path):
        """Count words in input documents and save the results."""
        input_rdd = self.sc.wholeTextFiles(input_path).cache()
        word_count = input_rdd.flatMap(self.process_document).reduceByKey(add, numPartitions=1).sortByKey()
        word_count.saveAsTextFile(output_path)

def main():
    # Check if the correct number of command-line arguments is provided
    if len(sys.argv) != 3:
        print("Usage: python script.py <input_file> <output_directory>")
        sys.exit(1)

    # Initialize Spark context
    conf = SparkConf().setAppName('Word Count Per Document').set('spark.executor.memory', '2g').set('spark.ui.port', 6200)
    sc = SparkContext(conf=conf)

    # Get input and output file paths from command-line arguments
    input_path = sys.argv[1]
    output_path = sys.argv[2]

    # Instantiate WordCountApp and count words
    word_counter = WordCountApp(sc)
    word_counter.count_words(input_path, output_path)

    # Stop Spark context
    sc.stop()

if __name__ == '__main__':
    main()
