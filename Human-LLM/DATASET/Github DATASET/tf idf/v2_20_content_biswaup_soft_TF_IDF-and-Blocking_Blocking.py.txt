# Define a custom dictionary class
class MyDictionary:
    def __init__(self):
        # Initialize an empty dictionary
        self.dictionary = {}

    # Method to add a key-value pair to the dictionary
    def add(self, key, value):
        self.dictionary[key] = value

    # Method to update the value of an existing key in the dictionary
    def update(self, key, value):
        self.dictionary[key] = value

    # Method to return the dictionary
    def my_dictionary(self):
        return self.dictionary


# Class for blocking functionality
class Blocking:
    # Method to create a hash table from given data
    def CreateHashTable(self, data):
        # Initialize an empty dictionary object
        hash_table = MyDictionary()
        # Initialize a counter for assigning unique identifiers to data
        count = -1
        # Iterate through each data item
        for d in data:
            # Generate a hash key for the data item
            key = self.CreateHashKey(d)
            # Increment the counter
            count = count + 1
            # Append a unique identifier to the data item
            d = d + ',' + str(count)
            # Check if the key already exists in the hash table
            if key in hash_table.my_dictionary():
                # If the key exists, update the value with the new data item
                hash_table.update(key, d)
            else:
                # If the key doesn't exist, add the key-value pair to the hash table
                hash_table.add(key, d)
        # Return the populated hash table
        return hash_table

    # Method to create a hash key from a given name
    def CreateHashKey(self, name):
        # Initialize an empty string to store the key
        key = ''
        # Iterate through each word in the name
        for w in name.split():
            # Append the first character of each word to the key
            key = key + w[0]
        # Sort the characters in the key alphabetically
        key = ''.join(sorted(key))
        # Return the generated key
        return key

    # Method to find keys based on a given key
    def FindKeys(self, key):
        # Calculate the size of the search space
        size = 1 << len(key)
        # Get the length of the key
        num = len(key)
        # Initialize an empty list to store found keys
        lstKeys = []
        # Iterate through the search space
        for i in range(1, size - 1, 1):
            # Convert the index to binary and pad with zeros
            val = bin(i)[2:].zfill(num)
            # Initialize an empty string to store the found key
            tot = ''
            # Iterate through each character in the binary representation
            for j in range(0, len(val)):
                # Check if the corresponding character in the key is '1'
                if j < num and val[j] == '1':
                    # If so, append the corresponding character from the original key
                    if not key[j].isdigit():
                        tot = tot + key[j]
            # Append the found key to the list of keys
            lstKeys.append(tot)
        # Return a list of unique keys
        return list(set(lstKeys))


# Example usage:
if __name__ == "__main__":
    # Sample data
    data = ["John Doe", "Jane Smith", "Alice Johnson"]
    # Initialize the Blocking object
    blocking = Blocking()
    # Create a hash table from the sample data
    hash_table = blocking.CreateHashTable(data)
    # Print the generated hash table
    print("Hash Table:")
    print(hash_table.my_dictionary())

    # Key for blocking
    key = "JhonsDoe"
    # Find keys after blocking
    keys = blocking.FindKeys(key)
    # Print the found keys
    print("\nKeys for '{}' after blocking:".format(key))
    print(keys)
