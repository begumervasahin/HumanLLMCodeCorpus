from operator import itemgetter
import sys
from math import log10

# Initialize variables
prev_word = None
count = 1
doc_word_count = {}  # Dictionary to store word counts per document
temp_arr = []  # List to store word-document pairs
corpus_size = 10.0  # Total number of documents in the corpus

# Process input lines
for line in sys.stdin:
    line = line.strip()
    word, details = line.split('\t', 1)
    filename, word_count, doc_word_total, word_freq = details.split(' ', 3)

    # Check if the word is the same as the previous one
    if prev_word == word:
        count += int(word_freq)
    else:
        # If it's a new word, store the previous word's information
        if prev_word is not None:
            counts = word_count + ' ' + doc_word_total + ' ' + str(count)      
            doc_word_count[prev_word] = counts
            pair = prev_word + ' ' + filename
            temp_arr.append(pair)

        # Reset count for the new word
        count = 1
        prev_word = word

# Store information for the last word encountered
counts = word_count + ' ' + doc_word_total + ' ' + str(count)
doc_word_count[prev_word] = counts
pair = prev_word + ' ' + filename
temp_arr.append(pair)

# Calculate TF-IDF for each word-document pair
for pair in temp_arr:
    word, filename = pair.split(' ', 1)
    for w in doc_word_count:
        if word == w:
            n, N, m = doc_word_count[w].split(' ', 2)
            n = float(n)
            N = float(N)
            m = float(m)
            tfidf = (n / N) * log10(corpus_size / m)
            print('%s\t%s' % (pair, tfidf))
