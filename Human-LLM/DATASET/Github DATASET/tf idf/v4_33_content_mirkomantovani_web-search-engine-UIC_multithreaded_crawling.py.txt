import threading
from queue import Queue
from crawler import Crawler
from domain_utils import get_domain_name
from crawling_utils import get_set_from_file

# Constants
FOLDER = 'uic'
HOMEPAGE = 'https://www.cs.uic.edu/'
DOMAIN_NAME = get_domain_name(HOMEPAGE)
QUEUE_PATH = FOLDER + '/queue.txt'
CRAWLED_PATH = FOLDER + '/crawled.txt'
THREAD_NUMBER = 20

# Queue initialization
queue = Queue()

# Initialize counter
counter = 100

# Instantiate a crawler object
crawler = Crawler(FOLDER, HOMEPAGE, DOMAIN_NAME)

# Function to start crawling
def start_crawling():
    create_workers()
    crawl()

# Function to create worker threads
def create_workers():
    for _ in range(THREAD_NUMBER):
        t = threading.Thread(target=work)
        t.daemon = True  # Daemon threads exit when the main program finishes
        t.start()

# Worker function to process URLs
def work():
    while True:
        url = queue.get()
        # Call the crawl_page method of the crawler object
        crawler.crawl_page(threading.current_thread().name, url)
        queue.task_done()

# Function to add jobs to the queue
def create_jobs():
    # Get links from the queue file and add them to the queue
    for link in get_set_from_file(QUEUE_PATH):
        queue.put(link)
    # Wait for all tasks in the queue to be processed
    queue.join()
    # Call crawl function again to process any new links added during crawling
    crawl()

# Function to initiate crawling
def crawl():
    # Get queued links from the queue file
    queued_links = get_set_from_file(QUEUE_PATH)
    # If there are links in the queue, proceed with crawling
    if len(queued_links) > 0:
        print(str(len(queued_links)) + ' links in the queue')
        create_jobs()

# Start crawling
start_crawling()
