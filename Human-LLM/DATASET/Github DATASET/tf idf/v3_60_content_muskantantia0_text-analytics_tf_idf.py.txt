import re
import string
import pandas as pd
from math import log

# Read the CSV file containing TF-IDF data
data = pd.read_csv('tf_idf.csv')

# Extract reviewer IDs and document titles
reviewer_ids = data['id'].tolist()
reviewer_ids.insert(0, "reviewerID")  # Add a header for reviewer IDs
document_titles = data['title'].tolist()

# Define a regex pattern to remove punctuation
punctuation_pattern = re.compile('[%s]' % re.escape(string.punctuation))

# Function to flatten nested lists
def flatten_lists(nested_lists):
    return [item for sublist in nested_lists for item in sublist]

# Preprocess document titles: lowercase, remove punctuation, and split into terms
def preprocess_title(title):
    title = str(title).lower()  # Convert to lowercase
    title = punctuation_pattern.sub('', title)  # Remove punctuation
    return title.split()  # Split into terms

# Flatten the list of terms from all document titles
all_terms = flatten_lists([preprocess_title(title) for title in document_titles])

# Function to calculate word frequencies and TF-IDF values
def calculate_tfidf(words):
    word_freq = {word: words.count(word) for word in words}
    sorted_freq = sorted(word_freq.items(), key=lambda x: x[1], reverse=True)
    total_words = len(words)
    tfidf_results = [(word, freq, round(log(total_words / freq, 10), 4), freq * log(total_words / freq, 10)) for word, freq in sorted_freq]
    return tfidf_results

# Calculate word frequencies along with TF-IDF values
word_freq_tfidf = calculate_tfidf(all_terms)

# Print the results
print(word_freq_tfidf)
