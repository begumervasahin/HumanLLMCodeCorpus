import pandas as pd
from format import clean_str, shuffle_both

def read_train_data(file_name, x_col, y_col, separator='\t', with_cleaning=True, with_header=False, with_shuffle=True):
    # Open the file
    with open(file_name) as file:
        # Check if header is present
        if not with_header:
            # Read the CSV file skipping the header
            data = pd.read_csv(file, sep=separator, skiprows=[0], header=None)
        else:
            # Read the CSV file with header
            data = pd.read_csv(file, sep=separator, header=None)

    # Extract input text and labels
    x_text = _read_unicode_lines(data[x_col].tolist(), with_cleaning)
    y_text = data[y_col].tolist()

    # Shuffle the data if required
    if with_shuffle:
        return shuffle_both(x_text, y_text)
    else:
        return x_text, y_text

def read_test_data(file_name, x_col, with_cleaning=True):
    # Read the CSV file skipping the header
    data = pd.read_csv(file_name, skiprows=[0], header=None)
    # Extract input text
    x_text = _read_unicode_lines(data[x_col], with_cleaning)
    return x_text

def _read_unicode_lines(lines, with_cleaning=True):
    if with_cleaning:
        # Clean each line of text
        return [clean_str(sentence.decode('utf-8')) for sentence in lines]
    else:
        # Return lines as they are
        return [sentence.decode('utf-8') for sentence in lines]
