import os
import json
import jsonlines
import nltk
import codecs
import utilities

def get_relevant_sentences(relevant_docs, entities, wiki_split_docs_dir):        
    relevant_sentences = []
    
    for relevant_doc in relevant_docs:
        with codecs.open(os.path.join(wiki_split_docs_dir, relevant_doc + ".json"), "r", "utf-8") as file:
            data = json.load(file)
            full_lines = data["lines"]
            lines = [line['content'] for line in full_lines if line['content'].strip()]
            
            for i, line in enumerate(lines):
                line = line.strip()
                line = line.replace("-LRB-", " ( ")
                line = line.replace("-RRB-", " ) ")
                
                temp = {
                    'id': relevant_doc,
                    'line_num': i,
                    'sentence': line
                }
                
                relevant_sentences.append(temp)
                
    return relevant_sentences

def get_sentence(wiki_doc_dir, doc_filename, sentence_id):
    with codecs.open(os.path.join(wiki_doc_dir, doc_filename + ".txt"), "r", "utf-8") as doc:
        doc_splitted_lines = doc.readlines()
        return doc_splitted_lines[sentence_id].split("\t")[1]
