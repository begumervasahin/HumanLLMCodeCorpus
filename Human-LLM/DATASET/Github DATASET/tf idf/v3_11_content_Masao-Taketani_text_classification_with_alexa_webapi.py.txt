import json
from flask import Flask, request
import predict_using_mlp

# Configuration
PORT_NO = 8080
DEBUG_MODE = False
HOST = '0.0.0.0'

# Flask application instance
app = Flask(__name__)

# Route for handling GET requests to the root URL
@app.route('/', methods=['GET'])
def index():
    # Read the contents of the index.html file
    with open("index.html", "rb") as f:
        return f.read()

# Route for handling GET requests to the '/api' endpoint
@app.route('/api', methods=['GET'])
def api():
    # Retrieve the value of the 'q' query parameter from the request URL
    q = request.args.get('q', '')

    # If the query parameter is empty, return a JSON response indicating an empty text
    if not q:
        return json.dumps({"label": "テキストが空です", "percent": 0})

    # Print the received query for debugging purposes
    print("Received query:", q)

    # Predict genre label and confidence percentage using MLP model
    label, percent, _ = predict_using_mlp.predict_genre(q)
    
    # Round the percentage to two decimal places
    percent = round(percent * 100, 2)

    # Return a JSON response containing predicted label and confidence percentage
    return json.dumps({"label": label, "percent": percent})

# Route for handling POST requests to the root URL
@app.route('/', methods=['POST'])
def api_2():
    # Check if the Content-Type of the request is 'application/json'
    if request.headers['Content-Type'] != 'application/json':
        print('Invalid Content-Type:', request.headers['Content-Type'])
        # Return a JSON response indicating an error if Content-Type is not 'application/json'
        return json.dumps({"res": "error"}), 400
    else:
        # Parse JSON data from the request body
        data = request.get_json()
        print("Received text:", data['text'])  # Print the received text for debugging
        
        # Predict genre label using MLP model
        label, _, _ = predict_using_mlp.predict_genre(data['text'])
        
        # Return a JSON response containing the predicted label
        return json.dumps({"label": label})

# Run the Flask application
if __name__ == '__main__':
    app.run(debug=DEBUG_MODE, host=HOST, port=PORT_NO, threaded=False)
