import json
from flask import Flask, request
import predict_using_mlp  # Importing your module for MLP prediction

PORT_NO = 8080  # Port number for running the Flask server
app = Flask(__name__)  # Creating a Flask application instance

# Route for handling GET requests to the root URL
@app.route('/', methods=['GET'])
def index():
    # Open and read the contents of the index.html file
    with open("index.html", "rb") as f:
        return f.read()

# Route for handling GET requests to the '/api' endpoint
@app.route('/api', methods=['GET'])
def api():
    # Getting the value of the 'q' query parameter from the request URL
    q = request.args.get('q', '')

    # If the query parameter is empty, return a JSON response indicating an empty text
    if q == '':
        return '{"label": "テキストが空です", "percent":0}'

    # Printing the received query for debugging purposes
    print("q=", q)

    # Predicting genre label and confidence percentage using MLP model
    label, percent, _ = predict_using_mlp.predict_genre(q)
    
    # Rounding the percentage to two decimal places
    percent = round(percent * 100, 2)

    # Returning a JSON response containing predicted label and confidence percentage
    return json.dumps({
        "label": label,
        "percent": percent
    })

# Route for handling POST requests to the root URL
@app.route('/', methods=['POST'])
def api_2():
    # Checking if the Content-Type of the request is 'application/json'
    if request.headers['Content-Type'] != 'application/json':
        print('Content-Type:', request.headers['Content-Type'])
        # Returning a JSON response indicating an error if Content-Type is not 'application/json'
        return json.dumps({"res": "error"}), 400
    else:
        # Parsing JSON data from the request body
        data = request.get_json()
        print(data['text'])  # Printing the received text for debugging
        
        # Predicting genre label using MLP model
        label, _, _ = predict_using_mlp.predict_genre(data['text'])
        
        # Returning a JSON response containing the predicted label
        return json.dumps({"label": label})

# Running the Flask application
if __name__ == '__main__':
    app.run(debug=False, host='0.0.0.0', port=PORT_NO, threaded=False)
