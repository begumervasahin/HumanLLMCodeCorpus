import numpy as np
import distance

class FCM:
    def fit(self, X, n_clusters, m, epochs):
        '''
        Parameters
        ----------
        X : shape (n_samples, n_features)
            Training data
        n_clusters : int
            The number of clusters
        m : float
            Weighted index number
        epochs : int
            The number of epochs
        
        Returns
        -------
        y : shape (n_samples,)
            Predicted cluster label per sample.
        '''
        # Getting the shape of the input data
        n_samples, n_features = X.shape
        
        # Initializing the matrix to store weights
        weights = np.zeros((n_samples, n_clusters))   
        
        # Initializing cluster centers randomly
        centers = X[np.random.choice(n_samples, n_clusters)]
        
        # Running the optimization loop for the specified number of epochs
        for _ in range(epochs):
            # Calculating distances between data points and cluster centers
            distances = np.apply_along_axis(distance.euclidean_distance, 1, centers, X).T + 1e-8
            
            # Calculating membership weights for each data point
            for i in range(n_clusters):
                weights[:, i] = 1 / np.sum((distances[:, i].reshape((-1, 1)) / distances) ** (2 / (m - 1)), axis=1)
            
            # Updating cluster centers based on membership weights
            for i in range(n_clusters):
                centers[i] = np.sum(weights[:, i].reshape(-1, 1) ** m * X, axis=0) / np.sum(weights[:, i] ** m)
        
        # Returning the index of the cluster with maximum weight for each data point
        return np.argmax(weights, axis=1)
