import os
import pandas as pd
import csv
import re

# Define constants
DATA_DIR = "data"
MBTI_CLEAN_CSV_PATH = os.path.join(DATA_DIR, "mbti_clean.csv")
DIMENSIONS = ("IE", "NS", "TF", "PJ")

# Read the MBTI data CSV file into a DataFrame
df = pd.read_csv(MBTI_CLEAN_CSV_PATH)

# Function to filter posts and write to CSV
def write_posts_to_csv(posts, letter):
    csv_path = os.path.join(DATA_DIR, f"test_{letter}.csv")
    with open(csv_path, "w", newline="", encoding="utf-8") as csv_file:
        writer = csv.writer(csv_file)
        for post in posts:
            writer.writerow([post])

# Iterate over each MBTI dimension
for dimension in DIMENSIONS:
    # Split the dimension into two letters
    letter_1, letter_2 = dimension
    # Iterate over each letter in the dimension
    for letter in [letter_1, letter_2]:
        # List to store posts for the current letter
        posts = []
        # Iterate over each row (profile) in the MBTI data
        for _, row in df.iterrows():
            # Check if the current profile contains the letter
            if letter in row["type"]:
                # Split the posts into individual posts
                individual_posts = row["posts"].split("|||")
                # Filter and append valid posts
                for post in individual_posts:
                    if ("http" not in post) and post and re.search("[a-zA-Z]", post):
                        posts.append(post)
        # Write filtered posts to a CSV file for the current letter
        write_posts_to_csv(posts, letter)

print("CSV files generated successfully.")
