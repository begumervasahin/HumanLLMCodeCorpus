import os
import pandas as pd

# Function to get a list of files in a directory
def getFilelist(directory):
    path = directory
    filelist = []
    files = os.listdir(path)
    for file in files:
        filelist.append(file)
    return filelist

# Getting file lists from two different directories
allfile = getFilelist('/TF-IDF')
allfile1 = getFilelist('/TF-IDF1')

# Changing directory
os.chdir('/TF-IDF')
path = '/TF-IDF1'

# Initializing a counter
j = 0

# Iterating through each file in the first directory
for filename in allfile:
    # Checking if the file is not present in the second directory
    if filename not in allfile1:
        try:
            # Reading the CSV file with specific settings
            data = pd.read_csv(filename, sep='    ', header=None, engine='python')
            # Filtering rows where the second column is greater than 0
            filtered_data = data[data[1] > 0]
            # Sorting the filtered data by the second column in descending order
            sorted_data = filtered_data.sort_values(by=[1], axis=0, ascending=False)
            
            # Checking the length of sorted data
            if len(sorted_data) <= 50:
                # Writing sorted data to a new file
                new_filename = filename.replace('.txt', '')
                new_path = path + '/' + new_filename + '.txt'
                sorted_data.to_csv(new_path, sep='\t', header=None, index=False)
            else:
                # Selecting the top 50 rows if the length exceeds 50
                top_50 = sorted_data.iloc[0:50, :]
                new_filename = filename.replace('.txt', '')
                new_path = path + '/' + new_filename + '.txt'
                top_50.to_csv(new_path, sep='\t', header=None, index=False)
        except Exception as e:
            j += 1

# Printing a message after execution
print('Execution complete!')
