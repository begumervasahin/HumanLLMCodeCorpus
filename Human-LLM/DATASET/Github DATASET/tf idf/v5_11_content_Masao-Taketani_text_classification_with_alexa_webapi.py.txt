import json
from flask import Flask, request
import predict_using_mlp

# Define the port number for the Flask application
PORT_NO = 8080

# Initialize the Flask app
app = Flask(__name__)

# Route for the home page
@app.route('/', methods=['GET'])
def home():
    # Open and read the index.html file
    with open("index.html", "rb") as f:
        return f.read()

# Route for the API endpoint
@app.route('/api', methods=['GET'])
def predict_genre_api():
    # Get the value of the 'q' parameter from the query string
    q = request.args.get('q', '')
    
    # If 'q' parameter is empty, return a JSON response indicating empty text
    if not q:
        return json.dumps({"label": "テキストが空です", "percent": 0})
    
    # Print the received text
    print("Received text:", q)
    
    # Predict the genre label and percentage using the MLP model
    label, percent, _ = predict_using_mlp.predict_genre(q)
    
    # Round the percentage to two decimal places
    percent = round(percent * 100, 2)
    
    # Return a JSON response with the predicted label and percentage
    return json.dumps({"label": label, "percent": percent})

# Route for handling POST requests
@app.route('/', methods=['POST'])
def predict_genre_post():
    # Check if the content type is JSON
    if request.headers['Content-Type'] != 'application/json':
        # If not JSON, return an error response
        print('Error: Content-Type is not application/json')
        return json.dumps({"error": "Content-Type must be application/json"}), 400
    
    # Get the JSON data from the request
    data = request.get_json()
    
    # Print the received text
    print("Received text:", data.get('text', ''))
    
    # Predict the genre label using the MLP model
    label, _, _ = predict_using_mlp.predict_genre(data.get('text', ''))
    
    # Return a JSON response with the predicted label
    return json.dumps({"label": label})

# Start the Flask application
if __name__ == '__main__':
    app.run(debug=False, host='0.0.0.0', port=PORT_NO, threaded=False)
