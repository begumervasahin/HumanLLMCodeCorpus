import re
import string
import pandas as pd
from math import log

# Read the CSV file containing TF-IDF data
data = pd.read_csv('tf_idf.csv')

# Get reviewer IDs and document titles
reviewer_ids = data['id'].tolist()
reviewer_ids.insert(0, "reviewerID")  # Adding a header for reviewer IDs
document_titles = data['title'].tolist()

# Initialize regex pattern to remove punctuation
punctuation_pattern = re.compile('[%s]' % re.escape(string.punctuation))

# Function to flatten nested lists
def flatten_lists(nested_lists):
    flattened_list = []
    for sublist in nested_lists:
        for item in sublist:
            flattened_list.append(item)
    return flattened_list

# Lowercase, remove punctuation, and split documents into terms
document_terms = []
for title in document_titles:
    title = str(title).lower()  # Convert to lowercase
    title = punctuation_pattern.sub('', title)  # Remove punctuation
    terms = title.split(' ')  # Split into terms
    document_terms.append(terms)

# Flatten the list of terms
all_terms = flatten_lists(document_terms)

# Function to convert list of words to frequency dictionary
def word_list_to_frequency_dict(words):
    word_freq = [words.count(word) for word in words]
    freq_dict = dict(zip(words, word_freq))
    sorted_freq = [(freq, word) for word, freq in freq_dict.items()]
    sorted_freq.sort(reverse=True)
    final_result = [(word, freq, round(log(round((len(words) / freq), 4), 10), 4), freq * log(len(words) / freq, 10)) for freq, word in sorted_freq]
    return final_result

# Calculate word frequencies along with TF-IDF values
word_freq_tfidf = word_list_to_frequency_dict(all_terms)

# Print the results
print(word_freq_tfidf)
