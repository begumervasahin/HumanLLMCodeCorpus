import re
import json
import time
from string import punctuation
from joblib import load

def clean_text(text):
    # Remove unwanted characters and normalize spaces
    cleaned_text = re.sub(r"[-()\"]", "", text)
    cleaned_text = re.sub(r"\s+", " ", cleaned_text)
    return cleaned_text.strip()

def filter_punctuation(text):
    # Filter out specific punctuation marks
    punc = list(punctuation)
    return filter(lambda x: x not in punc + [',', "’", '!', ':', "–"], text)

# Measure script execution time
start_time = time.process_time()

# Initialize an empty list to store cleaned text data
x_test = []

# Open the file containing JSON objects
with open('test_X_languages_homework.json.txt', encoding='utf-8') as file:
    # Iterate over each JSON object in the file
    for line in file:
        # Load the JSON object
        feature = json.loads(line)
        # Extract the 'text' field from the JSON object and clean it
        cleaned_feature = clean_text(feature['text'])
        # Filter out specific punctuation marks and convert to lowercase
        cleaned_feature = "".join(filter_punctuation(cleaned_feature)).lower()
        # Append the cleaned text to the list
        x_test.append(cleaned_feature)

# Load the trained model
pipe = load('model.bin')

# Use the model to predict labels for the test data
y_predicted = pipe.predict(x_test)

# Write the predicted labels to a file
with open('predictions.txt', 'w+', encoding='utf-8') as output_file:
    # Iterate over each predicted label and write to the file
    for label in y_predicted:
        json.dump({'classification': label}, output_file)
        output_file.write('\n')

# Print the time taken for the script to execute
print("Execution time:", time.process_time() - start_time, 'seconds')
