import socketserver
import http.server
import sys
import logging
import extractor

# Initialize the keyword extractor
key_extractor = extractor.Extractor()

# Configure logging
logging.basicConfig(level=logging.DEBUG)

# Define a custom HTTP request handler
class MyHTTPServer(http.server.BaseHTTPRequestHandler):
    def do_POST(self):
        try:
            # Read the content length and data from the request
            content_length = int(self.headers['Content-Length'])
            post_data = self.rfile.read(content_length)
            logging.debug('Received: {}...'.format(post_data))

            # Send a 200 OK response and set response headers
            self.send_response(200)
            self.send_header('Content-type', 'text/plain')
            self.end_headers()

            # Extract keywords and send response
            response = key_extractor.extract(post_data, top=15)
            self.wfile.write('\n'.join(response).encode('utf-8'))
        except Exception as e:
            # Log and handle errors
            logging.error('Failed to extract keyword: {}'.format(e))
            self.send_response(500)
            self.end_headers()

if __name__ == '__main__':
    # Set the default port number
    PORT = 27896

    # Check if a custom port is provided via command-line arguments
    if len(sys.argv) > 1:
        PORT = int(sys.argv[1])

    # Log the server start message
    logging.info('Server is running at localhost:{}'.format(PORT))

    # Create an HTTP server object
    server_address = ('127.0.0.1', PORT)
    httpd = socketserver.TCPServer(server_address, MyHTTPServer)

    # Log connection information
    sa = httpd.socket.getsockname()
    logging.info('Connected to {}'.format(sa))

    # Start the server and handle KeyboardInterrupt to gracefully shutdown
    try:
        logging.info('Server is running. Press Ctrl+C to shutdown.')
        httpd.serve_forever()
    except KeyboardInterrupt:
        logging.info('Server is shutting down...')
        httpd.shutdown()
