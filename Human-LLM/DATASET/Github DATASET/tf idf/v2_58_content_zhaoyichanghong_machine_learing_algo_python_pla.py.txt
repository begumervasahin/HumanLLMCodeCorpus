import numpy as np
from sklearn.metrics import accuracy_score

class PerceptronLearningAlgorithm:
    def fit(self, X_train, y_train):
        '''
        Fit the Perceptron Learning Algorithm to the training data.

        Parameters
        ----------
        X_train : numpy.ndarray
            Training data of shape (n_samples, n_features).
        y_train : numpy.ndarray
            Target values of shape (n_samples,).
        '''
        n_samples, n_features = X_train.shape
        # Initialize weights and bias
        self.weights = np.zeros(n_features)
        self.bias = 0

        # Iterate until convergence
        while True:
            # Iterate through each sample
            for i in range(n_samples):
                # Predict the label for the current sample
                prediction = self.predict(X_train[i])
                # Update weights and bias if prediction is incorrect
                if y_train[i] * prediction <= 0:
                    self.weights += (y_train[i] * X_train[i]).reshape(self.weights.shape)
                    self.bias += y_train[i]

            # Compute predictions on the entire training set
            predictions = self.predict(X_train)
            # Check for convergence
            if accuracy_score(y_train, predictions) == 1:
                break

    def predict(self, X):
        '''
        Predict class labels for input data.

        Parameters
        ----------
        X : numpy.ndarray
            Input data of shape (n_samples, n_features).

        Returns
        -------
        numpy.ndarray
            Predicted class labels of shape (n_samples,).
        '''
        # Compute the dot product of input data and weights, add bias, and apply sign function
        return np.sign(X.dot(self.weights) + self.bias)

# Example usage
if __name__ == "__main__":
    # Example training data
    X_train = np.array([[1, 2], [2, 3], [3, 4], [4, 5]])
    y_train = np.array([1, 1, -1, -1])

    # Create Perceptron Learning Algorithm instance
    pla = PerceptronLearningAlgorithm()

    # Train the model
    pla.fit(X_train, y_train)

    # Example testing data
    X_test = np.array([[1, 1], [2, 2], [3, 3], [4, 4]])

    # Make predictions
    predictions = pla.predict(X_test)
    print("Predictions:", predictions)
