import SocketServer
import BaseHTTPServer
import SimpleHTTPServer
import sys
import extractor
import logging

# Create an instance of the Extractor class from the extractor module
key_extractor = extractor.Extractor()

# Define a custom HTTP request handler class
class MyHTTPServer(BaseHTTPServer.BaseHTTPRequestHandler):
    
    # Define the behavior for POST requests
    def do_POST(self, *args):
        try:
            # Extract the length of the incoming request's content
            length = int(self.headers.getheader('content-length'))
            # Read the content of the request
            query = self.rfile.read(length)
            # Log the received query
            logging.debug('Received: {}...'.format(query))
            # Send a success response (HTTP 200)
            self.send_response(200)
            self.send_header('Content-type', 'text/plain')
            self.end_headers()
            # Extract keywords from the query using the key_extractor instance
            response = key_extractor.extract(query, top=15)
            # Write the extracted keywords to the response
            self.wfile.write('\n'.join(response))     
        except:
            # If an error occurs during keyword extraction, send a server error response (HTTP 500)
            self.send_response(500)
            # Log the error
            logging.error('Failed to extract keyword.')

# Set the default port for the server
PORT = 27896

# Check if a custom port is provided as a command-line argument
if len(sys.argv) > 1:
    PORT = int(sys.argv[1])

# Log the server's address
logging.info('Running at localhost:{}'.format(PORT))  

# Set up the server address
server_address = ('127.0.0.1', PORT)

# Create an instance of the TCP server using the custom HTTP request handler
httpd = SocketServer.TCPServer(("", PORT), MyHTTPServer)

# Get the server's socket address
sa = httpd.socket.getsockname()

# Log the connection from the server's socket address
logging.info('Connection from {}'.format(sa))

# Start serving HTTP requests indefinitely
httpd.serve_forever()
