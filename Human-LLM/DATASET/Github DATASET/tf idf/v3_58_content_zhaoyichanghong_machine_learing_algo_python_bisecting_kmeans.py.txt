class BisectingKMeans:
    def fit(self, X, n_clusters):
        '''
        Fit Bisecting K-Means clustering algorithm.

        Parameters
        ----------
        X : numpy.ndarray
            Training data of shape (n_samples, n_features).
        n_clusters : int
            The desired number of clusters.

        Returns
        -------
        y : numpy.ndarray
            Predicted cluster label per sample of shape (n_samples,).
        '''
        n_samples = X.shape[0]
        data = X
        clusters = []

        while True:
            # Initialize KMeans model
            model = KMeans()
            # Fit the KMeans model with 2 clusters and a maximum of 100 iterations
            labels = model.fit(data, n_clusters=2, max_iter=100)
            
            # Assign samples to clusters based on labels
            cluster1_indices = np.flatnonzero(labels == 0)
            cluster2_indices = np.flatnonzero(labels == 1)
            clusters.append(cluster1_indices)
            clusters.append(cluster2_indices)

            # Check if desired number of clusters is achieved
            if len(clusters) == n_clusters:
                break

            # Calculate sum of squared errors (SSE) for each cluster
            sse = [np.var(data[cluster]) for cluster in clusters]
            # Choose the cluster with the highest SSE for further splitting
            data = data[clusters[np.argmax(sse)]]
            # Remove the cluster that was split
            del clusters[np.argmax(sse)]

        # Assign cluster labels to each sample
        y = np.zeros(n_samples)
        for i, cluster_indices in enumerate(clusters):
            y[cluster_indices] = i

        return y
