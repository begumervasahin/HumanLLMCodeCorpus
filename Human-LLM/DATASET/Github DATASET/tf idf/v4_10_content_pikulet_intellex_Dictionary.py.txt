# Importing necessary function from a helper module
from data_helper import store_data

# Dictionary class for tracking terms, document frequency, IDF, term ID, and term offset
class Dictionary:
    # Constants for dictionary indices
    DOCUMENT_FREQUENCY = 0
    IDF = 0
    TERM_ID = 1
    TERM_OFFSET = 1

    # Constructor initializing the dictionary and file
    def __init__(self, file):
        self.terms = {}  # Dictionary to store terms
        self.file = file  # File to store data
        self.total_num_documents = 0  # Total number of documents

    # Method to check if the term exists in the dictionary
    def has_term(self, term):
        return term in self.terms

    # Method to get the term ID
    def get_term_id(self, term):
        return self.terms[term][Dictionary.TERM_ID]

    # Method to add a new term to the dictionary
    def add_term(self, term, term_id):
        # Initialize term with document frequency and term ID
        self.terms[term] = [1, term_id]

    # Method to retrieve all terms in the dictionary
    def get_terms(self):
        return self.terms

    # Method to increment the document frequency of a term
    def add_document_frequency(self, term):
        self.terms[term][Dictionary.DOCUMENT_FREQUENCY] += 1

    # Method to set the IDF (Inverse Document Frequency) of a term
    def set_idf(self, term, idf):
        self.terms[term][Dictionary.IDF] = idf

    # Method to set the offset of a term
    def set_offset(self, term, offset):
        self.terms[term][Dictionary.TERM_OFFSET] = offset

    # Method to save the dictionary to disk
    def save_to_disk(self):
        # Using helper function to store data in a file
        store_data(self.file, self.terms)
