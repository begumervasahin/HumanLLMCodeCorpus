from nltk.tokenize import word_tokenize

# Function to calculate term frequency
def term_frequency(doc, text):
    # Tokenize the text
    word_tokens = word_tokenize(doc[text])
    # Count the frequency of each word
    words_freq = nltk.FreqDist(word_tokens)
    return words_freq

# Function to calculate inverse document frequency
def inverse_document_frquency(doc, word):
    # Check if the word appears in each document
    word_cnt = [word in doc[file] for file in doc]
    # Calculate IDF
    idf = math.log(len(word_cnt) / sum(word_cnt))
    return idf

# Function to calculate TF-IDF scores
def tf_idf(doc, text):
    tfidf_scores = {}
    tf = term_frequency(doc, text)
    for term in tf:
        # Consider only alphabetic terms
        if term.isalpha():
            # Calculate IDF for the term
            idf = inverse_document_frquency(doc, term)
            # Get term frequency
            tf_term = term_frequency(doc, text)[term]
            # Calculate TF-IDF score
            tf_idf = idf * tf_term
            tfidf_scores[term] = round(tf_idf, 3)
    return tfidf_scores
