import math
import string

# Function to calculate tf-idf score
def tf_idf(corpus, document, term):
    # Convert document to lowercase
    document = document.lower()
    
    # Remove punctuation from the document
    exclude = set(string.punctuation)
    document = ''.join(ch for ch in document if ch not in exclude)
    
    # Split document into terms
    doc_term_list = document.split()
    
    # Count total number of terms in the document
    doc_term_count = len(doc_term_list)
    
    # Count the number of occurrences of the term in the document
    term_count_in_document = 0
    for t in doc_term_list:
        if t == term:
            term_count_in_document += 1
    
    # Calculate term frequency (tf)
    tf = term_count_in_document / doc_term_count
    
    # Count the number of documents in the corpus
    doc_count = len(corpus)
    
    # Count the number of documents containing the term
    num_docs_with_term = 0
    for doc in corpus:
        # Convert document to lowercase
        doc = doc.lower()
        
        # Remove punctuation from the document
        doc = ''.join(ch for ch in doc if ch not in exclude)
        
        # Split document into terms
        d_words = doc.lower().split()
        
        # Check if the term is present in the document
        if term in d_words:
            num_docs_with_term += 1
    
    # Calculate inverse document frequency (idf)
    if num_docs_with_term != 0:
        temp_idf = doc_count / num_docs_with_term
        idf = math.log(temp_idf)
    else:
        idf = 0
    
    # Calculate tf-idf score
    return tf * idf
