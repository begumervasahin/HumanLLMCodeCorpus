import os
import sys
import glob

# Ensure the 'Texts' directory exists, create it if not
text_directory = "Texts"
if not os.path.exists(text_directory):
    os.makedirs(text_directory)

# Function to create index and search in folder
def create_index_and_search_in_folder():
    # Get a list of text files in the 'Texts' directory
    text_files = glob.glob("./Texts/*.txt")
    # Prepare command-line arguments for createIndex_tf.py
    args = " ".join(['"' + file + '"' for file in text_files])
    # Execute createIndex_tf.py and queryIndex_tfidf.py
    os.system("python ./createIndex_tf.py " + args)
    os.system("python ./queryIndex_tfidf.py ")

# Function to search in folder
def search_in_folder():
    # Execute queryIndex_tf.py
    os.system("python ./queryIndex_tf.py ")

# Function to convert PDF files to text
def convert_pdf_to_text():
    # Execute pdftotext.py
    os.system("python ./pdftotext.py ")

# Function to search in file using KMP algorithm
def search_in_file_by_kmp():
    # Prompt user to enter search term
    search_term = input("Enter search term: ")
    # Execute kmp.py with the provided search term
    os.system("python ./kmp.py " + search_term)

# Function to find recently modified or created files
def recently_modified_or_created_files():
    # Get a list of recently modified or created text files in the 'Texts' directory
    text_files = glob.glob("./Texts/*.txt")
    # Prepare command-line arguments for time.py
    args = " ".join(['"' + file + '"' for file in text_files])
    # Execute time.py with the provided arguments
    os.system("python ./time.py " + args)

# Main function to display menu and handle user input
def main():
    print("1) To create index and search in folder")
    print("2) To search in folder")
    print("3) To convert PDF to text")
    print("4) Search in file by KMP")
    print("5) Recently modified or created files")
    print("6) Exit program")
    
    while True:
        choice = int(input("Enter your choice: "))
        
        if choice == 1:
            create_index_and_search_in_folder()
        elif choice == 2:
            search_in_folder()
        elif choice == 3:
            convert_pdf_to_text()
        elif choice == 4:
            search_in_file_by_kmp()
        elif choice == 5:
            recently_modified_or_created_files()
        elif choice == 6:
            break
        else:
            print("Invalid choice. Please enter a number between 1 and 6.")

# Execute main function when the script is run
if __name__ == "__main__":
    main()
