import numpy as np
import scipy.special

class PCA:
    def __init__(self, n_components, whiten=False):
        self.n_components = n_components
        self.whiten = whiten

    def fit_transform(self, X):
        # Randomly initialize PCA components
        self.components_ = np.random.rand(X.shape[1], self.n_components)
        # Transform input data using PCA components
        return X.dot(self.components_)

class Optimizer:
    def optimize(self, gradients):
        # Basic optimizer that returns gradients as is
        return gradients

class Ica:
    def fit_transform(self, X, epochs, optimizer):
        '''
        Independent Component Analysis (ICA) fitting and transformation.

        Parameters
        ----------
        X : array-like of shape (n_samples, n_features)
            Input data for training.
        epochs : int
            The number of training epochs.
        optimizer : Optimizer object
            Optimization algorithm to update model parameters.

        Returns
        -------
        s : array-like of shape (n_samples, n_features)
            Predicted source per sample.
        '''
        n_samples, n_features = X.shape
        
        # Apply PCA for whitening
        pca_model = PCA(n_features, whiten=True)
        X_whiten = pca_model.fit_transform(X)
        
        # Initialize random weight matrix
        self.__W = np.random.rand(n_features, n_features)
        
        # Training loop
        for _ in range(epochs):
            # Initialize gradient of weight matrix
            g_W = np.zeros_like(self.__W)
            
            # Compute gradient for each sample
            for x in X_whiten:
                # Compute gradient using ICA update rule
                g_W += (1 - 2 * scipy.special.expit(self.__W.dot(x.T))).dot(x) + np.linalg.inv(self.__W.T)
            
            # Average gradient across samples
            g_W /= n_samples
            
            # Optimize using provided optimizer
            g_W = optimizer.optimize([g_W])[0]
            
            # Update weight matrix
            self.__W += g_W
        
        # Return transformed data
        return X_whiten.dot(self.__W)

# Example usage
if __name__ == "__main__":
    # Generate some dummy data
    np.random.seed(0)
    X = np.random.rand(100, 5)

    # Create optimizer
    optimizer = Optimizer()

    # Initialize and fit ICA model
    ica = Ica()
    transformed_data = ica.fit_transform(X, epochs=10, optimizer=optimizer)

    print("Transformed data shape:", transformed_data.shape)
