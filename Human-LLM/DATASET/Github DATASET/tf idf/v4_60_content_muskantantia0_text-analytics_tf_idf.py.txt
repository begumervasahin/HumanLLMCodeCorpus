import nltk.corpus
import re
import string
import pandas as pd

# Read the data from a CSV file
data = pd.read_csv('tf_idf.csv')

# Get the list of reviewer IDs and titles from the dataframe
reviewer = data['id'].tolist()
reviewer.insert(0, "reviewerID")
titles = data['title'].tolist()

# Prepare punctuation removal and initialize a list to hold terms
punctuation = re.compile('[%s]' % re.escape(string.punctuation))
term_vec = []

# Process each title
for title in titles:
    title = str(title)  # Ensure it's a string
    title = title.lower()  # Convert to lowercase
    title = punctuation.sub('', title)  # Remove punctuation
    title = title.split(' ')  # Split into individual words
    term_vec.append(title)  # Append to term vector

wordlist = []

# Flatten the list of lists into a single list
def flatten(lists):
    for sublist in lists:
        for word in sublist:
            wordlist.append(word)
    return wordlist

wordlist = flatten(term_vec)

# Count the frequency of each word
wordfreq = []

def wordListToFreqDict(wordlist):
    wordfreq = [wordlist.count(word) for word in wordlist]
    word_freq_dict = dict(zip(wordlist, wordfreq))
    
    # Sort the dictionary by values in descending order
    sorted_word_freq = sorted(word_freq_dict.items(), key=lambda x: x[1], reverse=True)
    
    # Calculate TF-IDF scores
    final_result = [(word, freq, round(log(len(wordlist)/freq, 10), 4), freq * log(len(wordlist)/freq, 10)) for word, freq in sorted_word_freq]
    
    return final_result

finals = wordListToFreqDict(wordlist)

print(finals)
