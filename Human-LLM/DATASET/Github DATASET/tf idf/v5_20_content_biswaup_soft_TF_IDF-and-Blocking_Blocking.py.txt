import Dictionary as dict

class Blocking:
    def __init__(self):
        self.hash_table = dict.my_dictionary()

    def create_hash_table(self, data):
        count = -1
        for d in data:
            key = self.create_hash_key(d)
            count += 1
            d = f"{d},{count}"
            if key in self.hash_table:
                self.hash_table.update(key, d)
            else:
                self.hash_table.add(key, d)
        return self.hash_table

    def create_hash_key(self, name):
        key = ''.join(w[0] for w in name.split())
        return ''.join(sorted(key))

    def find_keys(self, key):
        size = 1 << len(key)
        num = len(key)
        keys = set()
        for i in range(1, size-1):
            val = bin(i)[2:].zfill(num)
            tot = ''
            for j in range(num):
                if val[j] == '1' and not key[j].isdigit():
                    tot += key[j]
            keys.add(tot)
        return list(keys)
