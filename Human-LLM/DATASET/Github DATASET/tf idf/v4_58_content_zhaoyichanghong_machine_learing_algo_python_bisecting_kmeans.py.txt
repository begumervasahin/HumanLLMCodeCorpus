import numpy as np
from k_means import KMeans

class BisectingKMeans:
    def fit(self, X, n_clusters):
        '''
        Parameters
        ----------
        X : shape (n_samples, n_features)
            Training data
        n_clusters : The number of clusters
        Returns
        -------
        y : shape (n_samples,)
            Predicted cluster label per sample.
        '''
        n_samples = X.shape[0]
        data = X
        clusters = []
        
        while True:
            # Create a KMeans model
            model = KMeans()
            # Fit the model with data into 2 clusters, max 100 iterations
            label = model.fit(data, 2, 100)
            
            # Append the indices of data points for each cluster
            clusters.append(np.flatnonzero(label == 0))
            clusters.append(np.flatnonzero(label == 1))
            
            # If the desired number of clusters is reached, exit the loop
            if len(clusters) == n_clusters:
                break
            
            # Calculate the sum of squared errors for each cluster
            sse = [np.var(data[cluster]) for cluster in clusters]
            # Choose the cluster with the highest SSE
            data = data[clusters[np.argmax(sse)]]
            # Remove the chosen cluster from the list
            del clusters[np.argmax(sse)]
        
        # Assign cluster labels to each data point
        y = np.zeros(n_samples)
        for i in range(len(clusters)):
            y[clusters[i]] = i
        
        return y
