from flask import Flask, render_template, redirect, url_for, request
import scripts
import index

# Create a Flask web application instance
app = Flask(__name__)

# Route for the landing page of the search engine web application
@app.route('/', methods=['GET', 'POST'])
def index_page():
    if request.method == 'GET':
        # Render the index.html template for GET requests
        return render_template('index.html')
    else:
        # Redirect to the query page for POST requests
        query = request.form['query']
        return redirect(url_for('query_page', query=query))

# Route for displaying search results
@app.route('/results', methods=['POST'])
def query_page():
    # Handle the user's query and display the search results
    query = request.form['query']
    message = None

    # Make a connection to the database and retrieve search results
    try:
        search_results = index.handle_query(query.lower())
    except KeyError:
        message = "No search results found! The given search term is a stopword"

    # Render the results.html template with query, search results, and error message
    return render_template('results.html', query_str=query, results=search_results, error_message=message)  

# Run the Flask application
if __name__ == "__main__":
    app.run(debug=True)
