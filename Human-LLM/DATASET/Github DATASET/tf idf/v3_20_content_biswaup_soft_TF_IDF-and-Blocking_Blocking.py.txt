class MyDictionary:
    """
    A custom dictionary class.
    """

    def __init__(self):
        """
        Initialize an empty dictionary.
        """
        self.dictionary = {}

    def add(self, key, value):
        """
        Add a key-value pair to the dictionary.
        """
        self.dictionary[key] = value

    def update(self, key, value):
        """
        Update the value of an existing key in the dictionary.
        """
        self.dictionary[key] = value

    def get_dictionary(self):
        """
        Return the dictionary.
        """
        return self.dictionary


class Blocking:
    """
    A class for blocking functionality.
    """

    def create_hash_table(self, data):
        """
        Create a hash table from given data.
        """
        hash_table = MyDictionary()
        count = -1  # Initialize a counter for assigning unique identifiers to data
        for index, item in enumerate(data):
            key = self.create_hash_key(item)  # Generate a hash key for the data item
            count += 1  # Increment the counter
            item_with_count = f"{item},{count}"  # Append a unique identifier to the data item
            if key in hash_table.get_dictionary():
                hash_table.update(key, item_with_count)  # Update the value with the new data item
            else:
                hash_table.add(key, item_with_count)  # Add the key-value pair to the hash table
        return hash_table

    def create_hash_key(self, name):
        """
        Create a hash key from a given name.
        """
        key = ''.join(sorted(w[0] for w in name.split()))  # Extract the first character of each word and sort
        return key

    def find_keys(self, key):
        """
        Find keys based on a given key.
        """
        size = 1 << len(key)  # Calculate the size of the search space
        num = len(key)
        lst_keys = []
        for i in range(1, size - 1):  # Iterate through the search space
            val = bin(i)[2:].zfill(num)  # Convert the index to binary and pad with zeros
            tot = ''.join(key[j] for j in range(num) if val[j] == '1' and not key[j].isdigit())
            # Append the corresponding character from the original key if it meets the conditions
            lst_keys.append(tot)
        return list(set(lst_keys))  # Return a list of unique keys


# Example usage:
if __name__ == "__main__":
    data = ["John Doe", "Jane Smith", "Alice Johnson"]  # Sample data
    blocking = Blocking()  # Initialize the Blocking object
    hash_table = blocking.create_hash_table(data)  # Create a hash table from the sample data
    print("Hash Table:")
    print(hash_table.get_dictionary())  # Print the generated hash table

    key = "JhonsDoe"  # Key for blocking
    keys = blocking.find_keys(key)  # Find keys after blocking
    print("\nKeys for '{}' after blocking:".format(key))
    print(keys)  # Print the found keys
