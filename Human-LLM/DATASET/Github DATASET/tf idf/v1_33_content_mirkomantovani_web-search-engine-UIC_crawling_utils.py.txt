import os

def create_domain_directory(directory):
    """
    Create a directory if it does not exist.

    Args:
        directory (str): The path of the directory to be created.
    """
    if not os.path.exists(directory):
        print('Creating directory ' + directory)
        os.makedirs(directory)

def create_data_files(project_name, base_url):
    """
    Create data files for the project if they do not exist.

    Args:
        project_name (str): The name of the project.
        base_url (str): The base URL to start crawling.
    """
    queue = os.path.join(project_name, 'queue.txt')
    crawled = os.path.join(project_name, 'crawled.txt')
    
    if not os.path.isfile(queue):
        write_file(queue, base_url)
    
    if not os.path.isfile(crawled):
        write_file(crawled, '')

def write_file(path, data):
    """
    Write data to a file.

    Args:
        path (str): The path of the file.
        data (str): The data to be written to the file.
    """
    with open(path, 'w') as f:
        f.write(data)

def append_to_file(path, data):
    """
    Append data to a file.

    Args:
        path (str): The path of the file.
        data (str): The data to be appended to the file.
    """
    with open(path, 'a') as file:
        file.write(data + '\n')

def delete_file_contents(path):
    """
    Delete the contents of a file.

    Args:
        path (str): The path of the file.
    """
    open(path, 'w').close()

def get_set_from_file(file_name):
    """
    Read data from a file and return it as a set.

    Args:
        file_name (str): The name of the file to read from.

    Returns:
        set: The set containing data from the file.
    """
    results = set()
    with open(file_name, 'rt') as f:
        for line in f:
            results.add(line.replace('\n', ''))
    return results

def write_set_to_file(links, file_name):
    """
    Write a set of data to a file.

    Args:
        links (set): The set of data to be written to the file.
        file_name (str): The name of the file to write to.
    """
    with open(file_name, 'w') as f:
        for l in sorted(links):
            f.write(l + '\n')

    project_name = "example_project"
    base_url = "https://example.com"
    
    create_domain_directory(project_name)
    create_data_files(project_name, base_url)
