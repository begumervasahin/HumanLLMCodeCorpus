def babylonian_square_root(number, epsilon):
    """
    Calculate the square root of a number using the Babylonian method.

    Parameters:
    - number: float, the number to find the square root of
    - epsilon: float, the desired precision of the result

    Prints:
    - A message indicating the approximate square root of the number
    """
    # Initial guess for square root using Babylonian method
    x0 = (1 + number) / 2
    x1 = (x0 + number / x0) * 0.5

    # Iterate until the error is within the desired epsilon
    while abs(x1 - x0) / x0 > epsilon:
        x0 = x1
        x1 = (x0 + number / x0) * 0.5

    # Print the result
    result_message = "The square root of {} is approximately {:.5f}".format(number, x1)
    print(result_message)

def babylonian_square_root_main():
    """
    Main function to demonstrate the Babylonian square root calculation.
    """
    # Example usage
    babylonian_square_root(5, 0.00001)

# Call the main function
babylonian_square_root_main()
