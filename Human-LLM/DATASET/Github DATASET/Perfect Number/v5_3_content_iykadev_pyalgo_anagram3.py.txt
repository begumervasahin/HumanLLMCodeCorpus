def is_anagram(s1, s2):
    """
    Check if two strings are anagrams of each other.

    Parameters:
    - s1: str, first string
    - s2: str, second string

    Returns:
    - bool, True if s1 and s2 are anagrams, False otherwise
    """
    # Initialize character count lists for s1 and s2
    count_s1 = [0] * 26
    count_s2 = [0] * 26

    # Count occurrences of each character in s1
    for char in s1:
        pos = ord(char) - ord('a')
        count_s1[pos] += 1

    # Count occurrences of each character in s2
    for char in s2:
        pos = ord(char) - ord('a')
        count_s2[pos] += 1

    # Compare the counts of characters in both strings
    for i in range(26):
        if count_s1[i] != count_s2[i]:
            return False

    # Return True if all counts are equal
    return True

# Test the function
print(is_anagram('pide', 'epid'))
