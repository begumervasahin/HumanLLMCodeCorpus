import sys

# Set recursion limit to avoid "RecursionError: maximum recursion depth exceeded"
sys.setrecursionlimit(3000)

def ack(M, N):
    if M == 0:
        return N + 1
    elif N == 0:
        return ack(M - 1, 1)
    else:
        return ack(M - 1, ack(M, N - 1))

# Example usage: Calculate ackermann(3, 4)
print(ack(3, 4))
