def is_anagram(s1, s2):
    # Convert strings to lists and sort them
    s1_list = sorted(list(s1))
    s2_list = sorted(list(s2))
    
    # Initialize variables
    index = 0
    matches = True
    
    # Check if sorted lists are equal
    while index < len(s1) and matches:
        if s1_list[index] == s2_list[index]:
            index += 1
        else:
            matches = False
    
    # Return whether strings are anagrams
    return matches

# Test the function
print(is_anagram('ey edip', 'pide ye'))
