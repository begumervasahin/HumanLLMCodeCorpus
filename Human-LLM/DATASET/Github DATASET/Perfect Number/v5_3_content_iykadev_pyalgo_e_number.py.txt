import math

def calculate_e_numbers(start, end):
    """
    Calculate and print Euler's number (e) raised to the power of i,
    where i ranges from start to end.
    """
    for i in range(start, end):
        result = math.exp(1) ** i
        print(result)

def main():
    """
    Main function to calculate and print Euler's number (e) for a range of values.
    """
    start_value = 1
    end_value = 100000
    calculate_e_numbers(start_value, end_value)

if __name__ == "__main__":
    main()
