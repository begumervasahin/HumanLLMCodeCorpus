import re

def extract_replacements(grammar):
    syntax_re = r"""(?mx)
        ^(?:
          (?: (?P<comment> \
          (?: (?P<blank>   \s*  ) (?: \n | $ )  ) |
          (?: (?P<rule>    (?P<pat> .+? ) \s+ -> \s+ (?P<term> \.)? (?P<repl> .+) ) )
        )$
        """
    return [(matchobj.group('pat'), matchobj.group('repl'), bool(matchobj.group('term')))
            for matchobj in re.finditer(syntax_re, grammar)
            if matchobj.group('rule')]

def replace(text, replacements):
    while True:
        for pat, repl, term in replacements:
            if pat in text:
                text = text.replace(pat, repl, 1)
                if term:
                    return text
                break
        else:
            return text
