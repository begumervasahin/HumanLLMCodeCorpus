import random

def random_generator(seed=1):
    """
    Generate a random number generator using the Linear Congruential Generator algorithm.
    """
    max_int32 = (1 << 31) - 1
    seed = seed & max_int32
    while True:
        seed = (seed * 214013 + 2531011) & max_int32
        yield seed >> 16

def deal(seed):
    """
    Deal cards from a deck using a given seed.
    """
    num_cards = 52
    cards = list(range(num_cards - 1, -1, -1))
    rnd = random_generator(seed)
    for i, r in zip(range(num_cards), rnd):
        j = (num_cards - 1) - r % (num_cards - i)
        cards[i], cards[j] = cards[j], cards[i]
    return cards

def show(cards):
    """
    Display the shuffled deck of cards.
    """
    card_names = ["A23456789TJQK"[c // 4] + "CDHS"[c % 4] for c in cards]
    for i in range(0, len(cards), 8):
        print(" ", " ".join(card_names[i: i + 8]))

if __name__ == '__main__':
    seed = int(input("Enter seed: "))  # Input seed from the user
    print("Hand", seed)
    deck = deal(seed)
    show(deck)
