from __future__ import division
import sys
from PIL import Image

def fractional_part(x):
    """Returns the fractional part of a number."""
    return x - int(x)

def reverse_fractional_part(x):
    """Returns 1 minus the fractional part of a number."""
    return 1 - fractional_part(x)

def put_pixel(image, xy, color, alpha=1):
    """
    Paints color over the background at the point xy in the image.
    Use alpha for blending. alpha=1 means a completely opaque foreground.
    """
    background_color = image.getpixel(xy)
    blended_color = blend_colors(background_color, color, alpha)
    image.putpixel(xy, blended_color)

def blend_colors(background_color, foreground_color, alpha):
    """
    Blends two colors based on the alpha value.
    """
    blended_color = tuple(int(round(alpha * fg + (1 - alpha) * bg)) for bg, fg in zip(background_color, foreground_color))
    return blended_color

def draw_line(image, p1, p2, color):
    """
    Draws an anti-aliased line in the image from p1 to p2 with the given color.
    """
    x1, y1, x2, y2 = p1 + p2
    steep = abs(x2 - x1) < abs(y2 - y1)
    if steep:
        x1, y1, x2, y2 = y1, x1, y2, x2
    if x2 < x1:
        x1, x2, y1, y2 = x2, x1, y2, y1
    grad = (y2 - y1) / (x2 - x1) if x2 != x1 else 1
    intery = y1 + reverse_fractional_part(x1) * grad

    def draw_endpoint(pt):
        x, y = pt
        xend = round(x)
        yend = y + grad * (xend - x)
        xgap = reverse_fractional_part(x + 0.5)
        px, py = int(xend), int(yend)
        put_pixel(image, (px, py), color, reverse_fractional_part(yend) * xgap)
        put_pixel(image, (px, py + 1), color, fractional_part(yend) * xgap)
        return px

    xstart = draw_endpoint(p1 if not steep else (y1, x1)) + 1
    xend = draw_endpoint(p2 if not steep else (y2, x2))

    for x in range(xstart, xend):
        y = int(intery)
        put_pixel(image, (x, y) if not steep else (y, x), color, reverse_fractional_part(intery))
        put_pixel(image, (x, y + 1) if not steep else (y + 1, x), color, fractional_part(intery))
        intery += grad

if __name__ == '__main__':
    if len(sys.argv) != 2:
        print('Usage: python xiaolin_wus_line_algo.py [output-file]')
        sys.exit(-1)

    BLUE = (0, 0, 255)
    YELLOW = (255, 255, 0)
    image = Image.new("RGB", (500, 500), BLUE)

    for a in range(10, 431, 60):
        draw_line(image, (10, 10), (490, a), YELLOW)
        draw_line(image, (10, 10), (a, 490), YELLOW)

    draw_line(image, (10, 10), (490, 490), YELLOW)

    filename = sys.argv[1]
    image.save(filename)
    print('Image saved to', filename)
