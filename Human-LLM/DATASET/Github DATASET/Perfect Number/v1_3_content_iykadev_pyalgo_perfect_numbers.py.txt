def perfect_number(num):
    su = 0
    for i in range(1, num):
        if num % i == 0:
            su += i
    return su == num

def perfect_number_main(n):
    if perfect_number(n):
        print("%s is perfect" % n)

if __name__ == "__main__":
    for i in range(1, 10000):
        perfect_number_main(i)
