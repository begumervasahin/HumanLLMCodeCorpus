# Function to generate random numbers within a limit
def generate_numbers(limit):
    nums = []
    for letter in ["a", "b", "c", "d", "e", "f", "g", "h", "i"]:
        tmp = False
        while not tmp:
            num = random.randint((limit * -1), limit)
            if (num not in nums) and (num < limit):
                nums.append(num)
                tmp = True
    return nums

# Function to categorize results based on the number of matches
def categorize_matches(answers, numbers, results, two_matches, three_matches, four_matches, five_matches, six_matches, seven_matches, win, goal):
    occurrences = 0
    most_common_num = max(set(answers), key=answers.count)
    for i in answers:
        if i == most_common_num:
            occurrences += 1
    if occurrences == 1:
        results.append(numbers)
    elif occurrences == 2:
        two_matches.append(numbers)
    elif occurrences == 3:
        three_matches.append(numbers)
    elif occurrences == 4:
        four_matches.append(numbers)
    elif occurrences == 5:
        five_matches.append(numbers)
    elif occurrences == 6:
        six_matches.append(numbers)
    elif occurrences == 7:
        seven_matches.append(numbers)
    elif occurrences == 8:
        win.append(numbers)
    print(((len(results)) + (len(two_matches)) + (len(three_matches)) + (len(four_matches)) + (len(five_matches)) + (len(six_matches)) + (len(seven_matches)) + (len(win))), "/", goal)
    return results, two_matches, three_matches, four_matches, five_matches, six_matches, seven_matches, win

# Function to perform calculations
def calculate_results(results, two_matches, three_matches, four_matches, five_matches, six_matches, seven_matches, win, *numbers, goal):
    squares = [num ** 2 for num in numbers]
    horizontal_sums = [squares[0] + squares[1] + squares[2], squares[3] + squares[4] + squares[5], squares[6] + squares[7] + squares[8]]
    vertical_sums = [squares[0] + squares[3] + squares[6], squares[1] + squares[4] + squares[7], squares[2] + squares[5] + squares[8]]
    diagonal_sums = [squares[0] + squares[4] + squares[8], squares[6] + squares[4] + squares[2]]
    results, two_matches, three_matches, four_matches, five_matches, six_matches, seven_matches, win = categorize_matches(horizontal_sums + vertical_sums + diagonal_sums, numbers, results, two_matches, three_matches, four_matches, five_matches, six_matches, seven_matches, win, goal)
    return results, two_matches, three_matches, four_matches, five_matches, six_matches, seven_matches, win

# Function to remove duplicates from a list
def remove_duplicates(lst):
    if lst:
        lst.sort()
        last = lst[-1]
        for i in range(len(lst) - 2, -1, -1):
            if last == lst[i]:
                del lst[i]
            else:
                last = lst[i]
    return lst
