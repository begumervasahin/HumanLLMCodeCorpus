import sys

sys.setrecursionlimit(10000)

def is_hamming_number(x):
    """
    Check if a number is a Hamming number.

    Args:
        x (int): The number to check.

    Returns:
        bool: True if x is a Hamming number, False otherwise.
    """
    if x == 1:
        return True
    if x % 2 == 0:
        return is_hamming_number(x // 2)
    if x % 3 == 0:
        return is_hamming_number(x // 3)
    if x % 5 == 0:
        return is_hamming_number(x // 5)
    return False

def print_hamming_numbers(limit):
    """
    Print all Hamming numbers up to a given limit.

    Args:
        limit (int): The upper limit of the Hamming numbers to print.
    """
    for num in range(1, limit + 1):
        if is_hamming_number(num):
            print(num, end=' ')

def hamming_numbers_main():
    """
    Main function to print Hamming numbers.
    """
    sys.stdout.write("Hamming Numbers: ")
    print_hamming_numbers(9830)

if __name__ == "__main__":
    hamming_numbers_main()
