def generate_sequence(j, k, initial_sequence):
    """
    Generates a sequence based on the provided parameters and initial sequence.

    Parameters:
        j (int): The index used to retrieve a value from the sequence.
        k (int): The index used to retrieve another value from the sequence.
        initial_sequence (list): The initial sequence of integers.

    Returns:
        list: The generated sequence.
    """
    sequence = initial_sequence.copy()
    for _ in range(10):
        for i in range(len(sequence)):
            if i == 0:
                output = (sequence[j - 1] + sequence[k - 1]) % 10
            elif 0 < i < 6:
                sequence[i] = sequence[i + 1]
            else:
                sequence[i] = output
                print(sequence[i])

if __name__ == "__main__":
    j = 3
    k = 7
    initial_sequence = [8, 6, 7, 5, 3, 0, 9]
    generate_sequence(j, k, initial_sequence)
