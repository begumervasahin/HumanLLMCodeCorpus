import sys
import random

class MillerRabin:
    @staticmethod
    def is_probable_prime(n):
        """
        Check if a number is probably prime using Miller-Rabin test
        """
        if n <= 1:
            return False

        bases = [random.randrange(2, 50000) for _ in range(90)]
        for b in bases:
            if n % b == 0:
                return False

        tests, s = 0, 0
        m = n - 1
        while not m & 1:
            m >>= 1
            s += 1

        for b in bases:
            tests += 1
            is_prob = MillerRabin.miller_rabin_test(m, s, b, n)
            if not is_prob:
                break

        if is_prob:
            return (1 - (1. / (4 ** tests)))
        return False

    @staticmethod
    def miller_rabin_test(m, s, b, n):
        """
        Perform Miller-Rabin test for primality
        """
        y = pow(b, m, n)
        for _ in range(s):
            if (y == 1 and _ == 0) or (y == n - 1):
                return True
            y = pow(y, 2, n)
        return False

class BlumBlumShub:
    def __init__(self, bits):
        """
        Constructor for BlumBlumShub class
        """
        self.n = self._generate_n(bits)
        seed = random.getrandbits(self._bit_len(self.n))
        self.set_seed(seed)

    def _generate_prime(self, bits):
        """
        Generate a prime number suitable for Blum-Blum-Shub algorithm
        """
        while True:
            p = random.getrandbits(bits)
            if MillerRabin.is_probable_prime(p) and p % 4 == 3:
                return p

    def _generate_n(self, bits):
        """
        Generate 'n' value for Blum-Blum-Shub algorithm
        """
        p = self._generate_prime(bits // 2)
        while True:
            q = self._generate_prime(bits // 2)
            if p != q:
                return p * q

    def set_seed(self, seed):
        """
        Set the seed value and internal state
        """
        self.state = seed % self.n

    def _bit_len(self, x):
        """
        Get the bit length of a positive number
        """
        assert x > 0
        q = 0
        while x:
            q += 1
            x >>= 1
        return q

    def next(self, num_bits):
        """
        Generate next random number of specified bit length
        """
        result = 0
        for _ in range(num_bits):
            self.state = (self.state ** 2) % self.n
            result = (result << 1) | (self.state & 1)
        return result
