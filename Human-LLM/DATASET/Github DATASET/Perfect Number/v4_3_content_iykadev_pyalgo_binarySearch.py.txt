def binary_search(sorted_list, target):
    # Initialize pointers
    first = 0
    last = len(sorted_list) - 1
    found = False
    
    # Continue search until the pointers meet or target is found
    while first <= last and not found:
        midpoint = (first + last) // 2
        
        # Check if target is at the midpoint
        if sorted_list[midpoint] == target:
            found = True
        else:
            # Adjust pointers based on target position relative to midpoint
            if target < sorted_list[midpoint]:
                last = midpoint - 1
            else:
                first = midpoint + 1
    
    # Return whether target is found
    return found

# Example usage
test_list = [0, 1, 2, 8, 13, 17, 19, 32, 42]
print(binary_search(test_list, 3))
print(binary_search(test_list, 13))
