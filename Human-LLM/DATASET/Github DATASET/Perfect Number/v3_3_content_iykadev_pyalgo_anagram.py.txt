def is_anagram(s1, s2):
    """
    Check if two strings are anagrams of each other.

    Parameters:
        s1 (str): First string.
        s2 (str): Second string.

    Returns:
        bool: True if s1 and s2 are anagrams, False otherwise.
    """
    # Convert s2 to a list for easier manipulation
    s2_list = list(s2)

    for char in s1:
        # Check if the character from s1 exists in s2_list
        if char in s2_list:
            # If found, remove it from s2_list
            s2_list.remove(char)
        else:
            # If not found, the strings are not anagrams
            return False

    # Check if all characters from s1 were found in s2_list
    # If s2_list is empty, then s1 and s2 are anagrams
    return not s2_list

# Test the function with example inputs
print(is_anagram('ey edip', 'pide ye'))  # Output: True
