import cProfile

def fibonacci(n):
    """Compute the nth Fibonacci number recursively."""
    if n <= 1:
        return n
    else:
        return fibonacci(n - 1) + fibonacci(n - 2)

def fibonacci_main():
    """Compute and print Fibonacci numbers up to a certain limit."""
    for n in range(1, 47):
        result = fibonacci(n)
        print("%s\t%s" % (n, result))

if __name__ == "__main__":
    # Profile the main Fibonacci computation function and save results to a file
    cProfile.run("fibonacci_main()", filename="profile_results.txt")
