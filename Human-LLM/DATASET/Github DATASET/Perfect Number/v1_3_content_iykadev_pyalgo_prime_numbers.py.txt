import sys

def is_prime(num):
    for i in range(2, num):
        if num % i == 0:
            return False
    return True

def prime_numbers(low_num, high_num):
    sys.stdout.write("Prime Numbers in range (%s,%s): " % (low_num, high_num))
    for i in range(low_num, high_num):
        if is_prime(i):
            print(i, end=' ')

def prime_number_main():
    prime_numbers(2, 100000)

if __name__ == "__main__":
    prime_number_main()
