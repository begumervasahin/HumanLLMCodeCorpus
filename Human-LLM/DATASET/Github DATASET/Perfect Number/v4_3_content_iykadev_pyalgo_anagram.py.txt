def is_anagram(s1, s2):
    # Convert s2 to a list for easier manipulation
    s2_list = list(s2)
    
    # Initialize variables
    index_s1 = 0
    is_still_ok = True
    
    # Iterate through each character in s1
    while index_s1 < len(s1) and is_still_ok:
        index_s2 = 0
        found = False
        
        # Check if the current character in s1 is found in s2
        while index_s2 < len(s2_list) and not found:
            if s1[index_s1] == s2_list[index_s2]:
                found = True
            else:
                index_s2 += 1
        
        # If the character is found, remove it from s2_list
        if found:
            s2_list[index_s2] = None
        else:
            is_still_ok = False
        
        # Move to the next character in s1
        index_s1 += 1
    
    # Check if s1 and s2 are anagrams
    return is_still_ok

# Test the function
print(is_anagram('ey edip', 'pide ye'))
