def is_anagram(s1, s2):
    # Initialize lists to count occurrences of each character in s1 and s2
    count_s1 = [0] * 26
    count_s2 = [0] * 26
    
    # Count occurrences of each character in s1
    for char in s1:
        position = ord(char) - ord('a')  # Get the position of the character in the alphabet
        count_s1[position] += 1
    
    # Count occurrences of each character in s2
    for char in s2:
        position = ord(char) - ord('a')  # Get the position of the character in the alphabet
        count_s2[position] += 1
    
    # Compare the counts of characters in both strings
    for index in range(26):
        if count_s1[index] != count_s2[index]:
            return False  # If counts don't match, strings are not anagrams
    
    return True  # If all counts match, strings are anagrams

# Test the function with example inputs
print(is_anagram('pide', 'epid'))  # Output: True
