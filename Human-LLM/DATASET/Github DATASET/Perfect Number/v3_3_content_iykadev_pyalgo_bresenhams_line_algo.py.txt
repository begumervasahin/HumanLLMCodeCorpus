class Bitmap:
    def __init__(self, width, height):
        """
        Initialize a bitmap with the specified width and height.
        """
        self.width = width
        self.height = height
        self.bitmap = [[' ' for _ in range(width)] for _ in range(height)]

    def set_pixel(self, x, y):
        """
        Set a pixel at the specified (x, y) coordinates.
        """
        if 0 <= x < self.width and 0 <= y < self.height:
            self.bitmap[y][x] = '*'

    def display(self):
        """
        Display the bitmap by printing each row.
        """
        for row in self.bitmap:
            print(' '.join(row))

    def draw_line(self, x0, y0, x1, y1):
        """
        Draw a line using Bresenham's line algorithm.
        """
        dx = abs(x1 - x0)
        dy = abs(y1 - y0)
        x, y = x0, y0
        sx = -1 if x0 > x1 else 1
        sy = -1 if y0 > y1 else 1

        if dx > dy:
            err = dx / 2.0
            while x != x1:
                self.set_pixel(x, y)
                err -= dy
                if err < 0:
                    y += sy
                    err += dx
                x += sx
        else:
            err = dy / 2.0
            while y != y1:
                self.set_pixel(x, y)
                err -= dx
                if err < 0:
                    x += sx
                    err += dy
                y += sy
        self.set_pixel(x, y)

# Create a bitmap instance with a width and height of 17
bitmap = Bitmap(17, 17)

# Draw lines connecting specified points
lines = [(1, 8, 8, 16), (8, 16, 16, 8), (16, 8, 8, 1), (8, 1, 1, 8)]
for line in lines:
    bitmap.draw_line(*line)

# Display the bitmap
bitmap.display()
