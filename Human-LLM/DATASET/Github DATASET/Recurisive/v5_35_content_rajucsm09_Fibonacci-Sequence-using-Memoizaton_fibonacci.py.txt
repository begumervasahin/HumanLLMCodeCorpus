# Declare a Python dictionary to work as a cache for storing Fibonacci calculations
fibonacci_cache = {}

# Define the Fibonacci function
def fibonacci(n):
    # Check if the input is a non-negative integer
    if not isinstance(n, int) or n < 0:
        return "Invalid input: Please provide a non-negative integer."

    # Check if the Fibonacci number is already in the cache
    if n in fibonacci_cache:
        return fibonacci_cache[n]

    # Base cases for Fibonacci sequence
    if n == 0:
        result = 0
    elif n == 1:
        result = 1
    else:
        # Calculate Fibonacci recursively and store the result in the cache
        result = fibonacci(n - 1) + fibonacci(n - 2)
        fibonacci_cache[n] = result

    return result

def print_first_n_fibonacci_numbers(n):
    # Print the first n Fibonacci numbers
    for i in range(n):
        print(f"Fibonacci({i}): {fibonacci(i)}")

# Print the first 1000 Fibonacci numbers
print_first_n_fibonacci_numbers(1000)
