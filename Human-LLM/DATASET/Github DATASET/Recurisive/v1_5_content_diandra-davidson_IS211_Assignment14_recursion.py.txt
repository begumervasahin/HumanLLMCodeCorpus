def fibonacci(num):
    if num <= 1:
        return num
    else:
        return fibonacci(num - 1) + fibonacci(num - 2)

def gcd(a, b):
    r = a % b
    if b > a:
        return gcd(b, a)
    if r == 0:
        return b
    else:
        return gcd(b, r)

def compare_to(s1, s2):
    if len(s1) < len(s2):
        return compare_to(s1[1:], s2[1:]) - 1
    elif len(s1) == len(s2):
        return 0
    elif len(s1) > len(s2):
        return 1 + compare_to(s1[1:], s2[1:])

# Test cases
def test_fibonacci():
    assert fibonacci(0) == 0
    assert fibonacci(1) == 1
    assert fibonacci(2) == 1
    assert fibonacci(3) == 2
    assert fibonacci(4) == 3

def test_gcd():
    assert gcd(48, 18) == 6
    assert gcd(60, 48) == 12
    assert gcd(21, 14) == 7
    assert gcd(17, 5) == 1

def test_compare_to():
    assert compare_to("hello", "world") == -3
    assert compare_to("hello", "hi") == 2
    assert compare_to("hello", "hello") == 0

if __name__ == "__main__":
    test_fibonacci()
    test_gcd()
    test_compare_to()
    print("All tests passed!")
