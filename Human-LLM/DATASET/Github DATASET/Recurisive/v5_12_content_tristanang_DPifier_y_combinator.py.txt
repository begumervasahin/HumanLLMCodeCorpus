import inspect

class DPifier:
    def __init__(self, f, lambda_mode=False):
        self.dp = {}

        def memoized_recursive(*args):
            if args in self.dp:
                return self.dp[args]
            else:
                self.dp[args] = f(memoized_recursive)(*args)
                return self.dp[args]

        self.memoized_recursive = memoized_recursive

        if lambda_mode:
            self.f = f
        else:
            self.f = self.__parse_function(f)

    def __call__(self, *args):
        return self.memoized_recursive(*args)

    def __parse_function(self, f):
        functionName = f.__name__
        source_code = inspect.getsource(f)
        source_code = source_code.replace(functionName, 'h', 1)
        source_code = source_code.replace(functionName + '(', 'abc' + '(')

        def g(abc):
            exec(source_code, {'abc': abc}, globals())
            return h

        return g
