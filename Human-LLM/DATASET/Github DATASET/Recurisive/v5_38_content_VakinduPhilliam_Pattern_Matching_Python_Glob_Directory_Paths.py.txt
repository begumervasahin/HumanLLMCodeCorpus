import glob

# Find files starting with a single digit followed by any character
files_starting_with_digit = glob.glob('./[0-9].*')

# Find files with .gif extension in the current directory
gif_files = glob.glob('*.gif')

# Find files with single-character names and .gif extension
single_char_gif_files = glob.glob('?.gif')

# Find all .txt files recursively in all subdirectories
all_txt_files_recursive = glob.glob('**/*.txt', recursive=True)

# Find all subdirectories recursively in the current directory
all_subdirectories_recursive = glob.glob('./**/', recursive=True)
