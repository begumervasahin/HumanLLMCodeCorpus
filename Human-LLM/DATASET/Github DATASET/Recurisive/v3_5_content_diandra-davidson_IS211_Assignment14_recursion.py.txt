# Function to calculate the nth Fibonacci number recursively
def fibonacci(n):
    """Calculate the nth Fibonacci number recursively."""
    if n <= 1:
        return n
    else:
        return fibonacci(n - 1) + fibonacci(n - 2)

# Function to find the greatest common divisor (GCD) of two numbers recursively
def gcd(a, b):
    """Find the greatest common divisor (GCD) of two numbers recursively."""
    remainder = a % b
    if b > a:
        return gcd(b, a)
    if remainder == 0:
        return b
    else:
        return gcd(b, remainder)

# Function to compare the lengths of two strings recursively
def compare_string_lengths(s1, s2):
    """Compare the lengths of two strings recursively."""
    if len(s1) < len(s2):
        return compare_string_lengths(s1[1:], s2[1:]) - 1
    elif len(s1) == len(s2):
        return 0
    elif len(s1) > len(s2):
        return 1 + compare_string_lengths(s1[1:], s2[1:])

# Test cases for the functions
def test_fibonacci():
    """Test cases for the fibonacci function."""
    assert fibonacci(0) == 0
    assert fibonacci(1) == 1
    assert fibonacci(2) == 1
    assert fibonacci(3) == 2
    assert fibonacci(4) == 3

def test_gcd():
    """Test cases for the gcd function."""
    assert gcd(48, 18) == 6
    assert gcd(60, 48) == 12
    assert gcd(21, 14) == 7
    assert gcd(17, 5) == 1

def test_compare_string_lengths():
    """Test cases for the compare_string_lengths function."""
    assert compare_string_lengths("hello", "world") == -3
    assert compare_string_lengths("hello", "hi") == 2
    assert compare_string_lengths("hello", "hello") == 0

# Run test cases and print success message if all tests pass
if __name__ == "__main__":
    test_fibonacci()
    test_gcd()
    test_compare_string_lengths()
    print("All tests passed!")
