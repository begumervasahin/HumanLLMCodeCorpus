# Define global variables
a = 1000
b = 2000

# Define function f1
def f1():
    # Define local variables for function f1
    p = 3000
    q = 4000
    # Print global and local variables accessible in f1
    print("Variables accessible in f1:")
    print("a:", a)  # Global variable
    print("b:", b)  # Global variable
    print("p:", p)  # Local variable
    print("q:", q)  # Local variable

# Define function f2
def f2():
    # Define local variables for function f2
    x = 5000
    y = 6000
    # Print global and local variables accessible in f2
    print("\nVariables accessible in f2:")
    print("a:", a)  # Global variable
    print("b:", b)  # Global variable
    print("x:", x)  # Local variable
    print("y:", y)  # Local variable
    # Variables p and q are not accessible in function f2

# Call function f1
f1()
# Call function f2
f2()
