def gcd_recursive(a, b):
    """
    Compute the greatest common divisor (GCD) of two positive integers using recursion.

    Args:
        a (int): The first positive integer.
        b (int): The second positive integer.

    Returns:
        int: The greatest common divisor of the two integers.
    """
    # Base case: if b is 0, the GCD is a
    if b == 0:
        return a
    else:
        # Recursive case: compute GCD of b and the remainder of a divided by b
        return gcd_recursive(b, a % b)

if __name__ == "__main__":
    # Input two positive integers from the user
    num1 = int(input("Enter the first number: "))
    num2 = int(input("Enter the second number: "))

    # Compute and print the GCD of the two numbers
    result = gcd_recursive(num1, num2)
    print(f"The greatest common divisor of {num1} and {num2} is: {result}")
