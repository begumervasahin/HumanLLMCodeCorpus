# Initialize variables
x_value = 1
y_value = 2
z_value = 3

# Update variable values
x_value += 5  # Add 5 to x
y_value *= 10  # Multiply y by 10
z_value >>= 6  # Right shift z by 6 bits

# Print updated variable values
print("Updated value of x:", x_value)
print("Updated value of y:", y_value)
print("Updated value of z:", z_value)
