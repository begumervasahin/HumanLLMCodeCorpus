def calculate_balance(balance, min_payment_rate, annual_interest_rate, months, payments):
    '''
    Calculates the credit card balance after X months for
    a person who only pays the minimum monthly payment.
    It also builds a list of monthly payments.
    '''
    months -= 1
    if months < 0:
        return round(balance, 2), payments
    else:
        min_payment = min_payment_rate * balance
        payments.append(min_payment)
        unpaid_balance = balance - min_payment
        balance = unpaid_balance + (annual_interest_rate / 12) * unpaid_balance
        return calculate_balance(balance, min_payment_rate, annual_interest_rate, months, payments)

def pay_off_debt(balance, annual_interest_rate, months, lower_guess, upper_guess):
    '''
    Calculates the minimum fixed monthly payment needed to pay off
    a credit card balance within X months using bisection search.
    '''
    guess = (lower_guess + upper_guess) / 2
    months -= 1
    if months < 0:
        if round(balance) == 0:
            return round(guess, 2)
        elif balance < 0:
            return pay_off_debt(balance, annual_interest_rate, months, lower_guess, guess)
        else:
            return pay_off_debt(balance, annual_interest_rate, months, guess, upper_guess)
    else:
        unpaid_balance = balance - guess
        balance = unpaid_balance + (annual_interest_rate / 12) * unpaid_balance
        return pay_off_debt(balance, annual_interest_rate, months, lower_guess, upper_guess)

def main():
    initial_balance = 100000
    months = 12
    annual_interest_rate = 0.22
    monthly_payment_rate = 0.05
    annual_interest_rate_monthly = annual_interest_rate / 12
    payments = []

    # Calculate balance and payments with minimum monthly payment
    new_balance, payments = calculate_balance(initial_balance, monthly_payment_rate, annual_interest_rate_monthly, months, payments)
    average_payment = round(sum(payments) / len(payments), 2)
    loss1 = round(abs(initial_balance - new_balance - sum(payments)), 2)

    # Calculate fixed monthly payment using bisection search
    min_guess = initial_balance / 12
    max_guess = (initial_balance * (1 + annual_interest_rate_monthly)**months) / months
    payment = pay_off_debt(initial_balance, annual_interest_rate, months, min_guess, max_guess)
    loss2 = round(abs(initial_balance - 0 - payment * months), 2)

    # Print results
    print('Current Balance:', initial_balance, '€')
    print('Duration:', months, 'months')
    print('With a monthly payment rate of %.2f:' % monthly_payment_rate)
    print('   - You will pay', average_payment, '€ per month on average.')
    print('   - Your loss will be', loss1, '€ for that time period.')
    print('   - Your remaining balance will be', new_balance, '€.')
    print('If you want to pay off your balance within', months, 'months:')
    print('   - You will have to pay', payment, '€ per month.')
    print('   - Your loss will be', loss2, '€ in total.')
    print('   - Your remaining balance will be 0 €.')

if __name__ == '__main__':
    main()
