import os
import fnmatch
import logging

# Define the path to the Wowza recording directory
PATH = "/path/to/wowza/recording/dir"

# Define patterns of files to be deleted
DELPATTERN = ["*_240p*.mp4",
              "*_360p*.mp4",
              "*_480p*.mp4",
              "*_720p*.mp4",
              "*_1080p*.mp4"]

# Configure logging
logging.basicConfig(format='%(asctime)s %(message)s',
                    filename='/var/log/wowza_delete_recordings.log',
                    level=logging.INFO)

# Log script start
logging.info("Script started")

# Initialize list to store files to be deleted
FILES_TO_DELETE = []

# Recursively search input folder for files matching deletion patterns
for root, dirs, filenames in os.walk(PATH):
    for pattern in DELPATTERN:
        for name in fnmatch.filter(filenames, pattern):
            FILES_TO_DELETE.append(os.path.join(root, name))

# Check if there are files to delete
if not FILES_TO_DELETE:
    logging.info("Nothing to delete")
else:
    # Delete files and log deletion
    for deleted_file in FILES_TO_DELETE:
        os.remove(deleted_file)
        logging.info("Deleted file: %s", deleted_file)

# Log script end
logging.info("Script ended")
