class Edge:
    def __init__(self, start_node, end_node, weight):
        """
        Initialize an Edge object with its properties.

        Parameters:
        start_node (str): The starting node of the edge.
        end_node (str): The ending node of the edge.
        weight (int): The weight of the edge.
        """
        self.start_node = start_node
        self.end_node = end_node
        self.weight = weight

    def __repr__(self):
        """
        Represent the Edge object as a string.

        Returns:
        str: A string representation of the Edge object.
        """
        return f"Edge: {self.start_node} --({self.weight})--> {self.end_node}"

# Example usage
edge1 = Edge('a', 'b', 5)
edge2 = Edge('b', 'c', 7)
edge3 = Edge('c', 'd', 3)

print(edge1)  # Output: Edge: a --(5)--> b
print(edge2)  # Output: Edge: b --(7)--> c
print(edge3)  # Output: Edge: c --(3)--> d
