import sys
import argparse
from heapq import heappop, heappush

class Edge:
    def __init__(self, u, v, weight):
        self.u = u
        self.v = v
        self.weight = weight

    def __str__(self):
        return f"e {self.u} {self.v} {self.weight}"

class DisjointSet:
    def __init__(self):
        self.parent = {}

    def make_set(self, x):
        self.parent[x] = x

    def find(self, x):
        if self.parent[x] != x:
            self.parent[x] = self.find(self.parent[x])
        return self.parent[x]

    def union(self, x, y):
        self.parent[self.find(x)] = self.find(y)

class PriorityQueue:
    def __init__(self, d):
        self.heap = []
        self.d = d

    def make_heap(self, edges):
        for edge in edges:
            heappush(self.heap, (edge.weight, edge))

    def size(self):
        return len(self.heap)

    def find_min(self):
        return self.heap[0][1]

    def delete_min(self):
        heappop(self.heap)

def read_graph_from_file(file_path):
    edges = []
    vertices = []
    with open(file_path) as f:
        for line in f:
            if line.startswith("e"):
                _, u, v, weight = map(int, line.split()[1:])
                edge = Edge(u, v, weight)
                edges.append(edge)
                vertices.extend([u, v])
    return edges, list(set(vertices))

def read_graph_from_input():
    edges = []
    vertices = []
    print("Enter edge line by line in the format e u v w. Press enter each time. Press enter twice when done.")
    for line in sys.stdin:
        if not line.strip():
            break
        if line.startswith("e"):
            _, u, v, weight = map(int, line.split()[1:])
            edge = Edge(u, v, weight)
            edges.append(edge)
            vertices.extend([u, v])
    return edges, list(set(vertices))

def write_mst_to_file(mst_edges, total_cost, output_file):
    with open(output_file, "w") as f:
        f.write(f"c Total cost of tree {total_cost}\n")
        for edge in mst_edges:
            f.write(str(edge) + "\n")

def print_mst_info(mst_edges, total_cost):
    print(f"c Total cost of tree {total_cost}")
    for edge in mst_edges:
        print(edge)

def main():
    parser = argparse.ArgumentParser(description='Minimum Spanning Tree (MST) Solver')
    parser.add_argument("-d", "--heap_d", default=2, type=int, help="balanced head width")
    parser.add_argument("-o", "--output", help="Output file to write solution")
    parser.add_argument("-i", "--infile", help="Input graph file. Graph must be in DIMACS format")
    args = parser.parse_args()

    d = args.heap_d
    priority_queue = PriorityQueue(d)

    if args.infile:
        edges, vertices = read_graph_from_file(args.infile)
    else:
        edges, vertices = read_graph_from_input()

    if not vertices or not edges:
        print("Error: Invalid graph")
        sys.exit(1)

    priority_queue.make_heap(edges)

    disjoint_set = DisjointSet()
    for i in vertices:
        disjoint_set.make_set(i)

    mst_edges = []
    total_cost = 0

    while priority_queue.size() != 0:
        _, edge = priority_queue.find_min()
        if disjoint_set.find(edge.u) != disjoint_set.find(edge.v):
            disjoint_set.union(edge.u, edge.v)
            mst_edges.append(edge)
            total_cost += edge.weight
        priority_queue.delete_min()

    if args.output:
        write_mst_to_file(mst_edges, total_cost, args.output)
    else:
        print_mst_info(mst_edges, total_cost)

if __name__ == "__main__":
    main()
