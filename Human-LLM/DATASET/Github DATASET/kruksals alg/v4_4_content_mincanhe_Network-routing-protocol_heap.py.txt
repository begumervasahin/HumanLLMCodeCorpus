def max_heapify(A, i):
    n = len(A)
    l = 2 * i
    r = 2 * i + 1
    largest = i
    
    if l <= n and A[l] > A[i]:
        largest = l
    if r <= n and A[r] > A[largest]:
        largest = r
    
    if largest != i:
        A[i], A[largest] = A[largest], A[i]
        max_heapify(A, largest)

def build_max_heap(A):
    n = len(A)
    for i in range(n // 2, 0, -1):
        max_heapify(A, i)

def heapsort(A):
    build_max_heap(A)
    n = len(A)
    for i in range(n, 1, -1):
        A[0], A[i - 1] = A[i - 1], A[0]
        n -= 1
        max_heapify(A[:n], 0)

def heap_maximum(A):
    return A[0]

def heap_insert(A, a):
    A.append(a)
    n = len(A)
    max_heapify(A, n // 2)

def heap_delete(A, h):
    n = len(A)
    A[h - 1] = A[n - 1]
    del A[n - 1]
    max_heapify(A, h - 1)
