class Node:
    """
    Represents a node in a disjoint set.

    Attributes:
        value: The value stored in the node.
        parent: The parent node of the current node in the disjoint set.
        rank: The rank of the node used for union-by-rank heuristic.
    """

    def __init__(self, value):
        """
        Initializes a node with the given value.

        Parameters:
            value: The value to be stored in the node.
        """
        self.value = value
        self.parent = self  # Initially, each node is its own parent
        self.rank = 0  # Initially, each node has rank 0

    def __repr__(self):
        """
        Returns a string representation of the node.

        Returns:
            str: A string representation of the node.
        """
        parent_value = self.parent.value if self.parent else None
        return f"Node(value={self.value}, rank={self.rank}, parent={parent_value})"
