def max_heapify(A, i):
    """
    Maintain the max-heap property.
    """
    n = len(A)
    l = 2 * i
    r = 2 * i + 1
    largest = i

    if l <= n and A[l - 1] > A[i - 1]:
        largest = l
    if r <= n and A[r - 1] > A[largest - 1]:
        largest = r

    if largest != i:
        A[i - 1], A[largest - 1] = A[largest - 1], A[i - 1]
        max_heapify(A, largest)

def build_max_heap(A):
    """
    Build a max heap from an unsorted array.
    """
    n = len(A)
    for i in range(n // 2, 0, -1):
        max_heapify(A, i)

def heapsort(A):
    """
    Sort an array using heap sort algorithm.
    """
    build_max_heap(A)
    n = len(A)
    for i in range(n, 1, -1):
        A[0], A[i - 1] = A[i - 1], A[0]
        n -= 1
        max_heapify(A[:n], 1)

def heap_maximum(A):
    """
    Return the maximum element of the heap.
    """
    return A[0]

def heap_insert(A, a):
    """
    Insert a new element into the heap.
    """
    A.append(a)
    n = len(A)
    max_heapify(A, n // 2)

def heap_delete(A, h):
    """
    Delete an element from the heap.
    """
    n = len(A)
    A[h - 1] = A[n - 1]
    del A[n - 1]
    max_heapify(A, h)
