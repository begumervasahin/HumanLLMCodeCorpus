from operator import itemgetter
import statistics
import time as ti

def read_coordinates(filename):
    """Read coordinates from a file and return them as a list of tuples."""
    coordinates = []
    with open(filename, "r") as file:
        for line in file:
            x, y = map(float, line.split(","))
            coordinates.append((x, y))
    return coordinates

def separate_lists(coordinates):
    """Separate x and y coordinates into separate lists."""
    y_coordinates = [i[1] for i in coordinates]
    aux = coordinates[:]
    aux.sort(key=itemgetter(0))
    x_coordinates = [i[0] for i in aux]
    return x_coordinates, y_coordinates

def create_quadrants(coordinates):
    """Create quadrants based on the given coordinates."""
    x_coordinates, y_coordinates = separate_lists(coordinates)
    y_median = statistics.median(y_coordinates)
    x_median = statistics.median(x_coordinates)
    q1, q2, q3, q4 = [], [], [], []
    for i in coordinates:
        if i[0] <= x_median and i[1] >= y_median:
            q1.append(i)
        if i[0] > x_median and i[1] >= y_median:
            q2.append(i)
        if i[0] <= x_median and i[1] < y_median:
            q3.append(i)
        if i[0] > x_median and i[1] < y_median:
            q4.append(i)
    return q1, q2, q3, q4, x_median, y_median
