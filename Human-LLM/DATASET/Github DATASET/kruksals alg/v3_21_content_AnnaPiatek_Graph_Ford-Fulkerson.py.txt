import graphviz as gv

class Edge:
    def __init__(self, node_0, node_1, capacity):
        self.node_0 = node_0
        self.node_1 = node_1
        self.capacity = capacity
        self.flow = 0

class Graph:
    def __init__(self):
        self.edges = []

    def add_edge(self, start, end, capacity):
        self.edges.append(Edge(start, end, capacity))
        return self

    def nodes(self):
        return list(set([edge.node_0 for edge in self.edges] + [edge.node_1 for edge in self.edges]))

    def neighbors(self, node):
        return [edge for edge in self.edges if edge.node_0 == node]

    def print_edges(self):
        return [[edge.node_0, edge.node_1, edge.flow, edge.capacity] for edge in self.edges]

    def find_path(self, start_node, end_node):
        antecedents = {node: 0 for node in self.nodes()}
        visited_nodes = [start_node]
        visited_edges = []
        path_found = False

        while not path_found:
            for node in visited_nodes:
                neighbors = self.neighbors(node)
                for edge in neighbors:
                    if edge not in visited_edges and edge.node_1 not in visited_nodes and edge.node_0 != end_node and edge.capacity != edge.flow:
                        antecedents[edge.node_1] = edge.node_0
                visited_edges.extend(neighbors)
                visited_nodes.extend([edge.node_0 for edge in neighbors] + [edge.node_1 for edge in neighbors if edge.node_1 not in visited_nodes])
                if end_node in visited_nodes:
                    path_found = True
                    break

        node = end_node
        path = []
        max_flow = 0

        while antecedents[node] != 0:
            for edge in self.edges:
                if edge.node_1 == node and edge.node_0 == antecedents[node]:
                    path.insert(0, edge)
                    node = edge.node_0

        min_capacity_edge = min(path, key=lambda x: x.capacity - x.flow)
        min_capacity = min_capacity_edge.capacity - min_capacity_edge.flow

        for edge in path:
            edge.flow += min_capacity

        max_flow += min_capacity

        return [path, max_flow]
