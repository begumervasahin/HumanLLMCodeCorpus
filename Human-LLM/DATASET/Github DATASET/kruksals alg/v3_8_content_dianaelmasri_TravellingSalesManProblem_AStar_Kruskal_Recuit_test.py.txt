import queue as Queue

class ComparableString(str):
    """
    A subclass of str that enables custom comparison based on string length.
    """
    def __lt__(self, other):
        """
        Custom less-than comparison based on string length.

        Parameters:
        other (ComparableString): Another ComparableString object to compare.

        Returns:
        bool: True if the length of self is less than the length of other, False otherwise.
        """
        return len(self) < len(other)

# Create a PriorityQueue instance
priority_queue = Queue.PriorityQueue()

# Enqueue items into the PriorityQueue
items_to_enqueue = ['jeej', 'kek', 'topkek', 'non']
for item in items_to_enqueue:
    priority_queue.put(ComparableString(item))

# Retrieve and print the smallest item in the PriorityQueue
smallest_item = priority_queue.get()
print("The smallest item in the PriorityQueue is:", smallest_item)
