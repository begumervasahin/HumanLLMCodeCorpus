from DisJSets import DisJSet
from heapqriorityqueue import HeapPriorityQueue

def MST_Kruskal(graph):
    """
    Computes a minimum spanning tree of a graph using Kruskal's algorithm.
    
    Returns:
    - A list of edges that comprise the MST.
    - A set of vertices with their associated edges.
    - The total weight of the MST.
    
    Assumes that the elements of the graph's edges are weights.
    """
    tree = []
    priority_queue = HeapPriorityQueue()
    forest = DisJSet()
    position = {}

    # Create groups for each vertex in the forest
    for vertex in graph.vertices():
        position[vertex] = forest.make_group(vertex)

    # Insert all edges into the priority queue
    for edge in graph.edges():
        priority_queue.insert(edge)

    # Main Kruskal's algorithm loop
    size = graph.vertex_count()
    while len(tree) != size - 1 and not priority_queue.is_empty():
        edge = priority_queue.deleteMin()
        u, v = edge.endpoints()
        a = forest.find(position[u])
        b = forest.find(position[v])
        if a != b:
            tree.append(edge)
            forest.unionSets(a, b)

    return tree
