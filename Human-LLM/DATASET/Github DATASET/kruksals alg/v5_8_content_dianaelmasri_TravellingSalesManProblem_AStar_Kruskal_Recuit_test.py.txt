import queue as Q

class CustomComparator:
    """
    Defines a custom comparator for comparing the lengths of strings.
    """

    def __lt__(self, other):
        """
        Defines the less-than comparison operation based on string length.

        Parameters:
            other (str): The other string to compare with.

        Returns:
            bool: True if the length of this string is less than the length of the other string, False otherwise.
        """
        return len(self) < len(other)

# Create a priority queue with the custom comparator
priority_queue = Q.PriorityQueue()

# Add elements to the priority queue
elements = ['jeej', 'kek', 'topkek', 'non']
for element in elements:
    priority_queue.put(element)

# Get the smallest element by removing it from the queue
smallest = priority_queue.get()

# Print the smallest element
print(smallest)
