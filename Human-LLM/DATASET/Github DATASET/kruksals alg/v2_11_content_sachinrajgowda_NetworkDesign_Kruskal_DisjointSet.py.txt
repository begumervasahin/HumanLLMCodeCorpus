class DisjointSet(dict):
    def add(self, item):
        # Add a new item to the disjoint set
        self[item] = item

    def find(self, item):
        # Find the root parent of the item in the disjoint set
        parent = self[item]
        while self[parent] != parent:
            parent = self[parent]
        self[item] = parent  # Path compression for optimization
        return parent

    def union(self, item1, item2):
        # Union operation to merge two sets
        self[item2] = self[item1]

def kruskal(nodes, sorted_edges):
    # Initialize the disjoint set forest
    forest = DisjointSet()
    mst = []  # Minimum Spanning Tree

    # Add nodes to the forest
    for node in nodes:
        forest.add(node)

    # Number of edges required for the minimum spanning tree
    no_edges = len(nodes) - 1

    # Iterate through sorted edges
    for edge in sorted_edges:
        src, dest, _ = edge
        root_src = forest.find(src)
        root_dest = forest.find(dest)

        # Check if adding the edge creates a cycle
        if root_src != root_dest:
            mst.append(edge)
            no_edges -= 1

            # Check if minimum spanning tree is complete
            if no_edges == 0:
                return mst

            # Merge the sets containing src and dest
            forest.union(root_src, root_dest)

    return mst

# Example usage
nodes = ['A', 'B', 'C', 'D', 'E']
sorted_edges = [('A', 'B', 1), ('A', 'C', 3), ('B', 'C', 2), ('B', 'D', 5), ('C', 'D', 4), ('C', 'E', 6), ('D', 'E', 7)]
minimum_spanning_tree = kruskal(nodes, sorted_edges)
print("Minimum Spanning Tree:", minimum_spanning_tree)
