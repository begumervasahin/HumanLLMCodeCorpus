def max_heapify(A, i):
    """
    Maintains the max-heap property.

    Args:
        A (list): The input list representing the heap.
        i (int): Index of the element to heapify.

    Returns:
        None
    """
    n = len(A)
    l = 2 * i
    r = 2 * i + 1
    largest = i

    if l <= n and A[l - 1] > A[i - 1]:
        largest = l
    if r <= n and A[r - 1] > A[largest - 1]:
        largest = r
    if largest != i:
        A[i - 1], A[largest - 1] = A[largest - 1], A[i - 1]
        max_heapify(A, largest)

def build_max_heap(A):
    """
    Builds a max-heap from an unordered array.

    Args:
        A (list): The input list representing the heap.

    Returns:
        None
    """
    n = len(A)
    for i in range(n // 2, 0, -1):
        max_heapify(A, i)

def heapsort(A):
    """
    Sorts a list using the heapsort algorithm.

    Args:
        A (list): The input list to be sorted.

    Returns:
        None
    """
    build_max_heap(A)
    n = len(A)
    for i in range(n, 1, -1):
        A[0], A[i - 1] = A[i - 1], A[0]
        n -= 1
        max_heapify(A[:n], 1)

def heap_maximum(A):
    """
    Returns the maximum element in the heap.

    Args:
        A (list): The input list representing the heap.

    Returns:
        int: The maximum element in the heap.
    """
    return A[0]

def heap_insert(A, a):
    """
    Inserts a new element into the heap.

    Args:
        A (list): The input list representing the heap.
        a (int): The element to be inserted.

    Returns:
        None
    """
    A.append(a)
    i = len(A)
    while i > 1 and A[i // 2 - 1] < A[i - 1]:
        A[i // 2 - 1], A[i - 1] = A[i - 1], A[i // 2 - 1]
        i //= 2

def heap_delete(A, h):
    """
    Deletes an element from the heap.

    Args:
        A (list): The input list representing the heap.
        h (int): The index of the element to be deleted.

    Returns:
        None
    """
    A[h - 1] = A[-1]
    del A[-1]
    max_heapify(A, h)

# Example usage
arr = [4, 1, 3, 2, 16, 9, 10, 14, 8, 7]
print("Original array:", arr)
heapsort(arr)
print("Array after heapsort:", arr)
heap_insert(arr, 20)
print("Array after inserting 20:", arr)
heap_delete(arr, 2)
print("Array after deleting element at index 2:", arr)
print("Maximum element in the heap:", heap_maximum(arr))
