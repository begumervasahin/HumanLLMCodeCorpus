class Node:
    def __init__(self, xloc, yloc, kval, adjList):
        self.xloc = xloc
        self.yloc = yloc
        self.kval = kval
        self.adjList = adjList

    def __eq__(self, rhs):
        return self.xloc == rhs.xloc and self.yloc == rhs.yloc and self.kval == rhs.kval

# Example usage
node1 = Node(1, 2, 3, [4, 5, 6])
node2 = Node(1, 2, 3, [4, 5, 6])
node3 = Node(4, 5, 6, [7, 8, 9])

print("Are node1 and node2 equal?", node1 == node2)  # Should print True
print("Are node1 and node3 equal?", node1 == node3)  # Should print False
