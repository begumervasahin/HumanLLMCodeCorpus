import numpy as np

# Define the matrix as a string
matrix_string = '1 2 3 4; 5 6 7 8; 0 1 2 3'

# Create a numpy matrix from the string
graph_matrix = np.matrix(matrix_string)

# Get the shape of the matrix
matrix_shape = graph_matrix.shape

# Access elements of the matrix by index
element_1_2 = graph_matrix[0, 1]  # Access element at row 0, column 1
element_2_3 = graph_matrix[1, 2]  # Access element at row 1, column 2

# Modify an element of the matrix
graph_matrix[0, 1] = -8  # Set the element at row 0, column 1 to -8

# Access a specific row or column of the matrix
first_row = graph_matrix[0, :]  # Get the entire first row
second_column = graph_matrix[:, 1]  # Get the entire second column
