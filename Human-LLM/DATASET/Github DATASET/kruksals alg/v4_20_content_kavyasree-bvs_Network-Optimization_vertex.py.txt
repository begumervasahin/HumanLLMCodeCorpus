class Vertex:
    """
    Represents a vertex in a graph.
    """

    def __init__(self, name):
        """
        Initializes a vertex with a given name.
        """
        self.name = name
        self.neighbors = []

    def add_neighbor(self, neighbor):
        """
        Adds a neighboring vertex to this vertex.
        """
        if isinstance(neighbor, Vertex):
            if neighbor.name not in self.neighbors:
                # Add neighbor to this vertex's list of neighbors
                self.neighbors.append(neighbor.name)
                # Also add this vertex to the neighbor's list of neighbors
                neighbor.neighbors.append(self.name)
                # Sort the list of neighbors for both vertices
                self.neighbors.sort()
                neighbor.neighbors.sort()
        else:
            return False

    def __repr__(self):
        """
        Returns a string representation of the vertex's neighbors.
        """
        return str(self.neighbors)
