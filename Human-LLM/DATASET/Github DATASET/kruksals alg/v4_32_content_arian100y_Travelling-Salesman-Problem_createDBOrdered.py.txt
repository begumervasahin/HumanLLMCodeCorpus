# Open the file "db2.txt" with UTF-8 encoding
text_file = open("db2.txt", encoding="utf-8")

# Initialize an empty list to store lines
lines = []

# Loop through each line in the file
for line in text_file:
    # Split the line by comma to get individual fields
    fields = line.split(",")
    # Check if the fourth field is "LOS OLIVOS"
    if fields[3] == "LOS OLIVOS":
        # Append the latitude and longitude to the list
        lines.append((float(fields[5]), float(fields[6])))

# Function to partition the array for quicksort
def partition(arr, low, high):
    i = low - 1
    pivot = arr[high][1]
    for j in range(low, high):
        if arr[j][1] <= pivot:
            i += 1
            arr[i], arr[j] = arr[j], arr[i]
    arr[i + 1], arr[high] = arr[high], arr[i + 1]
    return i + 1

# Function to perform quicksort
def quickSort(arr, low, high):
    if low < high:
        pi = partition(arr, low, high)
        quickSort(arr, low, pi - 1)
        quickSort(arr, pi + 1, high)

# Sort the lines based on longitude using quicksort
quickSort(lines, 0, len(lines) - 1)

# Print the sorted lines
for line in lines:
    print(line)

# Write the sorted lines to a new file "testSET4.txt"
write_file = open("testSET4.txt", "w")
for line in lines:
    write_file.write(str(line[0]) + "," + str(line[1]) + "\n")
write_file.close()

# Print the number of lines
print("Number of lines:", len(lines))
