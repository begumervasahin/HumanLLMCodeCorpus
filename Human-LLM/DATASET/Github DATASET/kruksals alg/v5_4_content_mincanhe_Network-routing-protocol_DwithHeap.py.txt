from collections import defaultdict
from heapq import heappop, heappush

def dijkstra(edges, start, end):
    # Create the adjacency list representation of the graph
    graph = defaultdict(list)
    for left, right, cost in edges:
        graph[left].append((cost, right))

    # Initialize priority queue and set to track visited vertices
    priority_queue = [(0, start, ())]
    visited = set()

    # Dijkstra's algorithm
    while priority_queue:
        cost, vertex, path = heappop(priority_queue)

        # Check if vertex is visited
        if vertex not in visited:
            visited.add(vertex)
            path = (vertex, path)

            # Check if destination vertex is reached
            if vertex == end:
                return (cost, path)

            # Explore neighboring vertices
            for next_cost, next_vertex in graph.get(vertex, ()):
                if next_vertex not in visited:
                    heappush(priority_queue, (cost + next_cost, next_vertex, path))

    # Return infinity if no path is found
    return float("inf")
