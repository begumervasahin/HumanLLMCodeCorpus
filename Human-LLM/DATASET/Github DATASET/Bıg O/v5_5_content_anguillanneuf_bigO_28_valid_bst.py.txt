class BinaryTreeNode:
    """
    Represents a node in a binary search tree (BST).
    """

    def __init__(self, value):
        """
        Initializes a BinaryTreeNode object with a given value.
        """
        self.value = value
        self.left = None
        self.right = None

    def insert_left(self, value):
        """
        Inserts a new node with the given value as the left child of this node.
        """
        self.left = BinaryTreeNode(value)
        return self.left

    def insert_right(self, value):
        """
        Inserts a new node with the given value as the right child of this node.
        """
        self.right = BinaryTreeNode(value)
        return self.right

    def is_binary_search_tree_iterative(self):
        """
        Checks if the binary tree rooted at this node is a valid binary search tree iteratively.
        """
        stack = []
        stack.append((self, float('-inf'), float('inf')))

        while stack:
            node, lower_bound, upper_bound = stack.pop()

            if not lower_bound < node.value < upper_bound:
                return False

            if node.left:
                stack.append((node.left, lower_bound, node.value))
            if node.right:
                stack.append((node.right, node.value, upper_bound))

        return True

    @staticmethod
    def is_binary_search_tree_recursive(node, lower=float('-inf'), upper=float('inf')):
        """
        Checks if the binary tree rooted at the given node is a valid binary search tree recursively.
        """
        if not node:
            return True

        if not lower < node.value < upper:
            return False

        return (
            BinaryTreeNode.is_binary_search_tree_recursive(node.left, lower, node.value)
            and BinaryTreeNode.is_binary_search_tree_recursive(node.right, node.value, upper)
        )


# Example usage:
bt = BinaryTreeNode(5)
bt.insert_left(3)
bt.insert_right(7)
bt.left.insert_left(1)
bt.left.insert_right(4)

# Check if the binary tree is a valid binary search tree iteratively
print("Iterative approach:", bt.is_binary_search_tree_iterative())

# Check if the binary tree is a valid binary search tree recursively
print("Recursive approach:", BinaryTreeNode.is_binary_search_tree_recursive(bt))
