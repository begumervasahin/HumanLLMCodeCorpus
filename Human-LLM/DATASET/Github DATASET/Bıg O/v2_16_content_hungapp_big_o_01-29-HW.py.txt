# Function to solve the second problem
def solve_second_problem():
    # Initialize lists
    a = []
    b = []

    # Read input count
    n = int(input())

    # Read and store input values
    for i in range(n):
        x, y = map(int, input().split())
        a.append(x)
        b.append(y)

    # Find answer
    answer = -1
    min_value, max_value = min(a), max(b)
    for i in range(n):
        if a[i] == min_value and b[i] == max_value:
            answer = i + 1
            break

    # Print answer
    print(answer)

# Function to solve the third problem

    a = []
    b = []

    for i in range(n):
        x, y = map(int, input().split())
        a.append(x)
        b.append(y)

    # Initialize final choices
    final_1 = [False] * n
    final_2 = [False] * n
    i = j = 0
    chosen = 0

    # Choose teams based on the condition
    while chosen < n:
        if a[i] < b[j]:
            final_1[i] = True
            i += 1
        else:
            final_2[j] = True
            j += 1
        chosen += 1

    # Print final teams
    for i in range(n):
        if final_1[i] or i < n // 2:
            print(1, end='')

# Function to solve the fourth problem
def solve_fourth_problem():
    # Read inputs
    n, m = map(int, input().split())
    a = list(map(int, input().split()))

    # Count the occurrences of each genre
    count = [0] * (m + 1)
    for genre in a:
        count[genre] += 1

    # Calculate the total number of ways
    ways = 0
    for i in range(1, m):
        for j in range(i + 1, m + 1):
            ways += count[i] * count[j]

    # Print the total number of ways
    print(ways)
