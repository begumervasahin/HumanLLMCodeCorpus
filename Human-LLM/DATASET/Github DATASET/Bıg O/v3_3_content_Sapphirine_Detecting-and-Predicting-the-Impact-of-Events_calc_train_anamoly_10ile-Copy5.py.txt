import psycopg2
import pandas as pd
import datetime

def establish_connection():
    """Establishes connection to the PostgreSQL database."""
    return psycopg2.connect(database="postgres", user="postgres", password="swapnil", host="35.190.146.57", port="5432")

def fetch_security_data(connection):
    """Fetches security data from the domain_security table."""
    cursor = connection.cursor()
    cursor.execute('SELECT * FROM domain_security')
    return pd.DataFrame(cursor.fetchall(), columns=[desc[0] for desc in cursor.description])

def fetch_tick_anomalies(connection, from_id, to_id):
    """Fetches tick anomalies data within the specified range."""
    cursor = connection.cursor()
    cursor.execute("SELECT tickdate, security, id, anamoly_score FROM tick_anomalies WHERE id BETWEEN %s AND %s ORDER BY id;", (from_id, to_id))
    return cursor.fetchall()

def fetch_raw_ticks_data(connection, security, tickdate):
    """Fetches relevant raw ticks data from import_raw_ticks table."""
    cursor = connection.cursor()
    sql_stmt = "SELECT tickdate, field, value FROM import_raw_ticks " \
               "WHERE security = %s AND field IN ('PX_LAST', 'VOLUME') AND " \
               "tickdate BETWEEN %s - interval '30 day' AND %s + interval '10 day' " \
               "ORDER BY field, tickdate;"
    cursor.execute(sql_stmt, (security, tickdate, tickdate))
    return pd.DataFrame(cursor.fetchall(), columns=[desc[0] for desc in cursor.description])

def process_anomalies(from_id, to_id):
    """Processes tick anomalies within the specified range."""
    conn = establish_connection()
    sec_df = fetch_security_data(conn)
    cur_write = conn.cursor()

    anomalies = fetch_tick_anomalies(conn, from_id, to_id)
    count = 0

    for anom in anomalies:
        anodate, security, anom_id, ano_score = anom
        security_id = int(sec_df[sec_df['security'] == security]['security_id'])

        raw_ticks_data = fetch_raw_ticks_data(conn, security, anodate)
        price_df = raw_ticks_data.loc[raw_ticks_data['field'] == 'PX_LAST']
        f_idx = price_df.index[price_df['tickdate'] == anodate][0] if price_df.shape[0] > 0 else 0

        if 10 < f_idx < price_df.shape[0] - 2:
            label = +1 if price_df['value'][f_idx+1] > price_df['value'][f_idx] else -1
            vol_df = raw_ticks_data.loc[raw_ticks_data['field'] == 'VOLUME']
            v_idx = vol_df.index[vol_df['tickdate'] == anodate][0] if vol_df.shape[0] > 0 else 0
            if v_idx > 10:
                sql_insert = "INSERT INTO train_tick_data VALUES (%s);"  # Adjust this query according to your table structure
                values = [anom_id, anodate, ano_score, security, security_id, label] + price_df['value'].loc[f_idx-9:f_idx].tolist() + vol_df['value'].loc[v_idx-9:v_idx].tolist()
                cur_write.execute(sql_insert, values)

        count += 1
        if count % 1000 == 0:
            conn.commit()
            print(str(datetime.datetime.now().time()), 'Count:', count)

    conn.commit()
    conn.close()
    print("DONE..........")
