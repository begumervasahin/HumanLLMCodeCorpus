from BubbleSort import Bubble_Sort
from ShakerSort import Shaker_Sort
from SelectionSort import SelectionSort
from QuickSort import QuickSort, ModifiedQuickSort
from MergeSort import MergeSort
from HashSort import HashSort
from random import shuffle, randrange, seed

def test(iterations=1000, debug=False):
    # Set seed for reproducibility if debug mode is enabled
    if debug:
        seed(10)

    # Generate a random test case
    case = [randrange(0, 7) for _ in range(7)]
    testCase = sorted(case[:])

    # Track failures
    failure = []

    # Increase recursion limit to avoid recursion errors
    import sys
    sys.setrecursionlimit(12000)

    # Run tests
    for _ in range(iterations):
        # Bubble Sort
        case_copy = case[:]
        Bubble_Sort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        # Shaker Sort
        case_copy = case[:]
        shuffle(case_copy)
        Shaker_Sort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        # Selection Sort
        case_copy = case[:]
        shuffle(case_copy)
        SelectionSort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        # Quick Sort
        case_copy = case[:]
        shuffle(case_copy)
        QuickSort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        # Modified Quick Sort
        case_copy = case[:]
        shuffle(case_copy)
        ModifiedQuickSort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        # Merge Sort
        case_copy = case[:]
        shuffle(case_copy)
        MergeSort(case_copy, tabulations)
        failure.append(case_copy == testCase)

        # Hash Sort
        case_copy = case[:]
        shuffle(case_copy)
        HashSort(case_copy, tabulations)
        failure.append(case_copy == testCase)

    # Print test result
    if False in failure:
        print(False)
    else:
        print(True)

# Metrics for tracking comparisons and swaps
metrics = {"comparisons": 0, "swaps": 0}

# Tabulations for different sorting algorithms
tabulations = {
    "bubbleSort": metrics,
    "shakerSort": metrics,
    "selectionSort": metrics,
    "quickSort": metrics,
    "modifiedQuickSort": metrics,
    "mergeSort": metrics,
    "hashSort": metrics
}

# Run the test
test()
