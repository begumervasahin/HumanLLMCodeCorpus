from BubbleSort import Bubble_Sort
from ShakerSort import Shaker_Sort
from SelectionSort import SelectionSort
from QuickSort import QuickSort, ModifiedQuickSort
from MergeSort import MergeSort
from HashSort import HashSort
from random import shuffle, randrange, seed

# Function to perform sorting algorithm tests
def perform_sorting_tests(iterations=1000, debug=False):
    # Set random seed for reproducibility if debugging
    if debug:
        seed(10)

    # Generate a random test case
    test_case = [randrange(0, 7) for _ in range(7)]

    # Sort the test case for comparison
    sorted_test_case = sorted(test_case)

    # List to store failure flags
    failures = []

    # Set recursion limit for sorting algorithms
    import sys
    sys.setrecursionlimit(12000)

    # Perform sorting algorithm tests
    for _ in range(iterations):
        # Test each sorting algorithm
        for sorting_algorithm in sorting_algorithms:
            case_copy = test_case[:]  # Create a copy of the test case
            sorting_algorithm(case_copy, tabulations)  # Sort the test case
            failures.append(case_copy == sorted_test_case)  # Check if the sorted case matches the expected result
            shuffle(test_case)  # Shuffle the test case for the next iteration

    # Print test result
    print(all(failures))

# Define sorting algorithms
sorting_algorithms = [
    Bubble_Sort,
    Shaker_Sort,
    SelectionSort,
    QuickSort,
    ModifiedQuickSort,
    MergeSort,
    HashSort
]

# Metrics dictionary for tracking comparisons and swaps
metrics = {"comparisons": 0, "swaps": 0}

# Dictionary to store metrics for each sorting algorithm
tabulations = {algorithm.__name__: metrics.copy() for algorithm in sorting_algorithms}

# Run the sorting tests
perform_sorting_tests()
