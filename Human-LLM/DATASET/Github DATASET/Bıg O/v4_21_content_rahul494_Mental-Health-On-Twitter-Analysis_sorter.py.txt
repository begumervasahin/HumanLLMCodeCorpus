import sys
import csv
import operator

# Initialize an empty dictionary to store data
storage = {}

# Read input from standard input and populate the dictionary
for line in sys.stdin:
    # Split each line into value and key
    value, key = line.split()
    # Convert the value to an integer and store it in the dictionary
    storage[key] = int(value)

# Sort the dictionary by values in descending order
sorted_storage = sorted(storage.items(), key=operator.itemgetter(1), reverse=True)

# Print the sorted key-value pairs
for key, value in sorted_storage:
    # Output each key-value pair as a comma-separated string
    print(key + "," + str(value))
