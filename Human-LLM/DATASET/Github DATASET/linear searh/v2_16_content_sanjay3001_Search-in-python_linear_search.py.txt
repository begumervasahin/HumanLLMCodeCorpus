# Linear Search

# Prompting the user to input the number of elements
num_elements = int(input("Enter the number of elements: "))

# Initializing an empty list to store elements
elements_list = []

# Gathering elements from the user
for i in range(num_elements):
    element = input("Enter element {}: ".format(i + 1))
    elements_list.append(element)

# Displaying the list of elements entered by the user
print("THE LIST IS", elements_list)

# Prompting the user to input the item to find
item_to_find = input("Enter the item to find: ")

# Initializing variables for search
position = 0
found = False
iteration_count = 0

# Performing linear search
while position < len(elements_list) and not found:
    iteration_count += 1
    if elements_list[position] == item_to_find:
        found = True
        print("Found")
    position += 1

# Displaying the number of iterations performed during the search
print("Number of iterations:", iteration_count)
