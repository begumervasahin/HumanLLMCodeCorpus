def calculate_yearly_payment(balance, annual_interest):
    """
    Calculates the minimum monthly payment needed to pay off a yearly debt with annual interest.
    """
    remaining_balance = balance
    monthly_payment = 0
    
    while remaining_balance >= 0:
        remaining_balance = balance
        
        for _ in range(12):
            remaining_balance -= monthly_payment
            monthly_interest = (annual_interest / 12) * remaining_balance
            remaining_balance += monthly_interest
        
        monthly_payment += 10
    
    return monthly_payment - 10

# Example usage:
total_debt = float(input("Enter the total debt for the year: "))
annual_interest_rate = float(input("Enter the annual interest rate (as a decimal): "))

minimum_payment = calculate_yearly_payment(total_debt, annual_interest_rate)
print(f"The minimum monthly payment required to pay off the debt in a year is: ${minimum_payment}")
