class HashTable:
    def __init__(self, size=31):
        # Initialize the hash table with a given size
        self.nbuckets = size
        self.the_table = []
        # Create empty buckets for each index in the table
        for i in range(self.nbuckets):
            self.the_table.append([])

    def hashcode(self, o):
        # Calculate the hash code for an object
        if isinstance(o, int):
            return o
        elif isinstance(o, str):
            h = 0
            # Apply a hashing algorithm for strings
            for char in o:
                h = h * 31 + ord(char)
            return h
        else:
            return None

    def buckets_str(self):
        # Represent the hash table as a string showing bucket contents
        if self.the_table is None:
            return None
        output = ""
        for i in range(len(self.the_table)):
            output += str(i).zfill(4)
            output += '->'
            # Show each key-value pair in the bucket
            for node in self.the_table[i]:
                output += str(node[0]) + ':' + str(node[1])
                output += ', '
            output = output.rstrip(', ')
            output += '\n'
        return output

    def __str__(self):
        # Represent the hash table as a string
        if self.the_table is None:
            return None
        output = '{'
        for i in range(len(self.the_table)):
            for node in self.the_table[i]:
                output += str(node[0]) + ':' + str(node[1])
                output += ', '
        output = output.rstrip(', ')
        output += '}'
        return output

    def put(self, key, value):
        # Insert or update a key-value pair in the hash table
        if self.the_table is None or len(self.the_table) == 0:        
            return
        # Calculate the bucket index for the given key
        bucket_index = self.hashcode(key) % len(self.the_table)
        bucket = self.the_table[bucket_index]
        new_item_flag = True
        # Check if the key already exists in the bucket
        for i in range(len(bucket)):
            if bucket[i][0] == key:
                # Update the value if the key exists
                new_value = bucket[i][1] | value
                bucket[i] = (key, new_value)
                new_item_flag = False
                break
        # If the key doesn't exist, add a new key-value pair to the bucket
        if new_item_flag:
            bucket.append((key, value))

    def get(self, key):
        # Retrieve the value associated with a given key
        if self.the_table is None or len(self.the_table) == 0:        
            return None
        bucket_index = self.hashcode(key) % len(self.the_table)
        bucket = self.the_table[bucket_index]
        # Search for the key in the bucket and return its value
        for node in bucket:
            if node[0] == key:
                return node[1]
        return None

    def bucket_indexof(self, key):
        # Retrieve the elements within a specific bucket
        if self.the_table is None or len(self.the_table) == 0:        
            return None
        output = '{'
        for node in self.the_table[key]:
            output += str(node[0]) + ':' + str(node[1])
            output += ', '
        output = output.rstrip(', ')
        output += '}'
        return output

    def __getitem__(self, key):
        # Retrieve the value associated with a key using indexing
        return self.get(key)

    def __setitem__(self, key, value):
        # Insert or update a key-value pair using indexing
        self.put(key, value)
