# Function to perform linear search for files containing all search terms
def linear_search(files, search_terms):
    # Check if files or search_terms are None or empty
    if not files or not search_terms:
        return None
    
    # List to store filenames containing all search terms
    matching_files = []
    
    # Iterate through each file
    for file_path in files:
        # Retrieve text content from the file
        file_content = get_text(file_path)
        
        # Check if file_content is not None
        if file_content is not None:
            # Extract words from file_content
            file_words = extract_words(file_content)
            
            # Check if all search terms are found in the file words
            if all(term in file_words for term in search_terms):
                matching_files.append(file_path)
    
    # Return the list of filenames containing all search terms
    return matching_files
