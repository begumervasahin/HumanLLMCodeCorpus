from myFunctions import *

# Initialize variables
guesses = 1
lower_bound, upper_bound = 0, 1000

# Initial guess by the computer
cpu_guess = mid(lower_bound, upper_bound)
computer_says(cpu_guess)

# Continue the guessing game until the correct number is found
while True:
    # Get user feedback and adjust the guess accordingly
    adjustment = get_adjustment()
    
    if adjustment == 'c':
        break

    # Update the bounds and recalculate the guess
    lower_bound, upper_bound = make_adjustment(adjustment, cpu_guess, lower_bound, upper_bound)
    cpu_guess = mid(lower_bound, upper_bound)
    
    # Increment the number of guesses and display the computer's new guess
    guesses += 1
    computer_says(cpu_guess)

# Display the final result once the correct number is found
found(cpu_guess, guesses)
