import time
from matplotlib import pyplot as plt

# Generate a test case of integers from 0 to 999
test_case = [y for y in range(1000)]
result = []  # List to store the execution times

# Function to perform linear search
def search(L, e):
    for i in range(len(L)):
        if L[i] == e:
            return True  # Element found
        if L[i] > e:
            return False  # Element not in the list
    return False  # Element not in the list

# Function to test the performance of linear search on the test case
def test(L):
    for j in range(0, len(L)):
        start = time.time()  # Record start time
        flg = search(L, j)   # Perform linear search
        end = time.time()    # Record end time
        if flg:
            result.append(end - start)  # If element found, record execution time
    return result

# Function to display the results using Matplotlib
def show_it(L, res):
    plt.plot(L, res, color='green', marker='o', linestyle='solid')
    plt.title("Linear Search Performance")
    plt.xlabel('Test Case')
    plt.ylabel('Execution Time (s)')
    plt.show()

# Perform the test
test(test_case)

# Display the results
show_it(test_case, result)
