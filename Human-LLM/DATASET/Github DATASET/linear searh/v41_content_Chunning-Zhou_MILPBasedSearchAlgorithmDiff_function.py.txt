def generate_file_folder(cipher_name, goal):
    file_folder = f"model/{cipher_name}/LBAS/"
    if not os.path.isdir(file_folder):
        os.makedirs(file_folder)

    file_folder = f"model/{cipher_name}/{goal}/"
    if not os.path.isdir(file_folder):
        os.makedirs(file_folder)

    file_folder = f"result/{cipher_name}/{goal}/"
    if not os.path.isdir(file_folder):
        os.makedirs(file_folder)

    file_folder = f"txt/{cipher_name}/{goal}/optimal_solution_of_submodel/"
    if not os.path.isdir(file_folder):
        os.makedirs(file_folder)
