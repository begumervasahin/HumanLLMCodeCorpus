import cv2
import cv2.aruco as aruco

# Initialize the video capture device
cap = cv2.VideoCapture(1)

# Uncomment the lines below to adjust camera settings
'''
cap.set(cv2.CAP_PROP_AUTOFOCUS, 0)
cap.set(3, 720)  # Set width
cap.set(4, 720)  # Set height
'''

# Counter for frames processed
frame_counter = 0

# Main loop for capturing and processing frames
while True:
    # Read a frame from the video capture device
    ret, frame = cap.read()
    
    # Convert the frame to grayscale
    gray = frame
    
    # Define the ArUco dictionary and parameters
    aruco_dict = aruco.Dictionary_get(aruco.DICT_6X6_250)
    parameters = aruco.DetectorParameters_create()
    
    # Detect ArUco markers in the grayscale frame
    corners, ids, rejectedImgPoints = aruco.detectMarkers(gray, aruco_dict, parameters=parameters)
    
    # Draw detected markers on the grayscale frame
    gray = aruco.drawDetectedMarkers(gray, corners, ids)
    
    # Display the frame with detected markers
    cv2.imshow('frame', gray)
    
    # Check for user input to quit the application
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break
    
    # Increment frame counter
    frame_counter += 1

# Release the video capture device and close all windows
cap.release()
cv2.destroyAllWindows()
