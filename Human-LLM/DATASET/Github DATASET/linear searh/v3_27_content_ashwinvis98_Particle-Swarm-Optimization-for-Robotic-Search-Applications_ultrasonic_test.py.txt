import os
import time
import threading
import nxt
from nxt.sensor import *

# Connect to the NXT brick
brick = nxt.locator.find_one_brick('00:16:53:0C:1D:29')

# Function to continuously read ultrasonic sensor
def read_ultrasonic_sensor():
    while True:
        distance = Ultrasonic(brick, PORT_2, check_compatible=False).get_sample()
        print(f"Distance from obstacle: {distance} cm")

        # Check if the distance is less than 50 cm
        if distance < 50:
            print('Warning: An obstacle is detected! Taking necessary action...')
            # Implement your logic here for handling obstacles, such as capturing an image
            time.sleep(1)

# Function to continuously increment a counter
def increment_counter():
    counter = 0
    while True:
        counter += 1
        print(f"Counter value: {counter}")
        time.sleep(1)

# Create threads for ultrasonic sensor reading and counter increment  
ultrasonic_thread = threading.Thread(target=read_ultrasonic_sensor)   
counter_thread = threading.Thread(target=increment_counter)

# Start the threads
ultrasonic_thread.start()
counter_thread.start()

# Join the threads to the main thread
ultrasonic_thread.join()
counter_thread.join()
