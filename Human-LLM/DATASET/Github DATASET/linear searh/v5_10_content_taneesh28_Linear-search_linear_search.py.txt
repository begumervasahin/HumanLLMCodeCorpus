def linear_search(arr, val):
    """
    Perform linear search to find the index of a value in an array.

    Parameters:
        arr (list): The array to search within.
        val (int): The value to search for.

    Returns:
        int: The index of the value if found, otherwise -1.
    """
    for index, element in enumerate(arr):
        if element == val:
            return index
    return -1

def main():
    # Input size of the array
    n = int(input("Enter the size of the array: "))

    # Input array elements
    arr = []
    for i in range(n):
        element = int(input("Enter element {} of the array: ".format(i+1)))
        arr.append(element)

    # Display the array
    print("Array:", arr)

    # Input value to search for
    search_value = int(input("Enter the value you want to search for: "))

    # Perform linear search
    index = linear_search(arr, search_value)

    # Display search result
    if index != -1:
        print("The value", search_value, "is found at index", index)
    else:
        print("Value not found in the array")

if __name__ == "__main__":
    main()
