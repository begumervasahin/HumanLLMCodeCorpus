def calculate_yearly_balance(initial_balance, annual_interest, monthly_rate):
    """
    Calculate the remaining debt at the end of the year.

    Parameters:
    - initial_balance (float): The initial balance of the debt.
    - annual_interest (float): The annual interest rate (as a decimal).
    - monthly_rate (float): The monthly interest rate (as a decimal).

    Returns:
    - float: The remaining debt at the end of the year.
    """
    unpaid_balance = initial_balance

    # Iterate through 12 months to simulate one year
    for month in range(12):
        # Reduce the unpaid balance by the monthly interest
        unpaid_balance -= monthly_rate * unpaid_balance

        # Calculate the interest amount for the current month
        interest_amount = (annual_interest / 12) * unpaid_balance

        # Add the interest amount back to the unpaid balance
        unpaid_balance += interest_amount

    return unpaid_balance

# Example usage:
initial_balance = float(input("Enter the initial balance: "))
annual_interest_rate = float(input("Enter the annual interest rate (as a decimal): "))
monthly_interest_rate = float(input("Enter the monthly interest rate (as a decimal): "))

final_debt = calculate_yearly_balance(initial_balance, annual_interest_rate, monthly_interest_rate)
print(f"The remaining debt at the end of the year is: {final_debt:.2f}")
