# Prompting the user to input the number of elements in the list
n = int(input("Enter the number of elements: "))

# Creating an empty list to store the elements
li = []

# Looping through the range of numbers provided by the user
for i in range(0, n):
    # Prompting the user to input each element and adding it to the list
    li.append(input("Enter the element: "))

# Printing the list
print("THE LIST IS", li)

# Prompting the user to input the item to find in the list
item = input("Enter the item to find: ")

# Initializing variables
position = 0
found = False
counter = 0

# Looping until either the item is found or we reach the end of the list
while position < len(li) and not found:
    # Counting the iterations
    counter += 1
    # Checking if the current element is equal to the item
    if li[position] == item:
        # If found, setting the 'found' flag to True and printing a message
        found = True
        print("Item found!")
    # Moving to the next position in the list
    position += 1

# Printing the number of iterations performed
print("Number of iterations:", counter)
