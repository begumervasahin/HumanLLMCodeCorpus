def search(unordered_list, term):
    # Iterate through the list along with their indices
    for index, item in enumerate(unordered_list):
        # Check if the current item is equal to the search term
        if term == item:
            # If a match is found, provide information about the index and the searched number
            return (
                f'Index of the search item: {index}\n'
                f'Number you searched for: {term}\n'
            )
    
    # If no match is found, indicate that the term was not found
    return f'The number {term} was not found in the list\n'

# Example usage with a list named 'mylist'
mylist = [5, 87, 2, 100, -1, 22, 4000, 34, 23, 19, 567]

# Test cases
print(search(mylist, -100))
print(search(mylist, 5))
print(search(mylist, 567))
