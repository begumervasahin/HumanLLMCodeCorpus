def linear_search(lst, num):
    # Initialize index variable
    i = 0
    # Iterate through the list
    while i < len(lst):
        # Check if the current element equals the search number
        if lst[i] == int(num):
            # Store the index of the found number
            globals()['pos'] = i
            # Return True to indicate the number is found
            return True
        # Move to the next element in the list
        i = i + 1
    # Return False if the number is not found in the list
    return False

# Initialize variable to store the index of the found number
pos = -1
# Define the list to search within
lst = [21, 33, 45, 9, 24, 9, 34, 7, 88, 96, 54, 35, 66, 77]

# Prompt the user to input the number to search for
num = input("Enter the number you are searching for: ")

# Call the linear_search function and check if the number is found
if linear_search(lst, num):
    # If the number is found, print its index
    print("The number is found at index:", pos)
else:
    # If the number is not found, print a message indicating so
    print("Number not found")
