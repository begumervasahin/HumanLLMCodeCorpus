import matplotlib.pyplot as plt
import numpy as np
from random import randint
from time import time

# Binary search algorithm
def binary_search(arr, target):
    left, right = 0, len(arr) - 1
    while left <= right:
        mid = (left + right) // 2
        if arr[mid] < target:
            left = mid + 1
        elif arr[mid] > target:
            right = mid - 1
        else:
            return mid
    return -1

# Linear search algorithm
def linear_search(arr, target):
    for i, num in enumerate(arr):
        if num == target:
            return i
        if num > target:
            return -1
    return -1

# Generate an array of random integers
def generate_random_array(size):
    return [randint(-np.iinfo(np.int64).max, np.iinfo(np.int64).max) for _ in range(size)]

# Calculate average CPU time for a search function
def calculate_avg_cpu_time(search_func, arr, num_trials):
    total_time = 0
    for _ in range(num_trials):
        start_time = time()
        search_func(arr)
        end_time = time()
        total_time += end_time - start_time
    return total_time / num_trials

# Run performance trials for binary and linear search
def run_performance_trials(num_trials=5, array_size=None):
    if array_size is None:
        array_size = 2
    while True:
        array = sorted(generate_random_array(array_size))
        binary_search_time = calculate_avg_cpu_time(binary_search, array, num_trials)
        linear_search_time = calculate_avg_cpu_time(linear_search, array, num_trials)
        
        if binary_search_time < linear_search_time:
            return array_size
        array_size += 1

# Find the threshold at which binary performs better than linear search
def find_threshold():
    num_trials = 250
    num_iterations = 100
    total_threshold = sum(run_performance_trials(num_trials) for _ in range(num_iterations))
    return total_threshold / num_iterations

# Get the model for performance comparison
def get_model(linear, data):
    if linear:
        min_val = min(data)
        min_idx = data.index(min_val)
        max_val = max(data)
        max_idx = data.index(max_val)
        slope = (max_val - min_val) / (max_idx - min_idx)
        return lambda array_size: slope * array_size + min_val
    else:
        b, a = np.polyfit(np.log(np.arange(1, len(data) + 1)), np.array(data), 1)
        return lambda array_size: a + (b * np.log(array_size))

# Plot performance of linear and binary search
def plot_performance(max_array_size=50, num_trials=250):
    linear_y = []
    binary_y = []
    for size in range(1, max_array_size + 1):
        linear_perf = calculate_avg_cpu_time(linear_search, generate_random_array(size), num_trials)
        binary_perf = calculate_avg_cpu_time(binary_search, generate_random_array(size), num_trials)
        linear_y.append(linear_perf)
        binary_y.append(binary_perf)

    linear_model = get_model(True, linear_y)
    binary_model = get_model(False, binary_y)

    plt.plot(linear_y, 'o', label='Linear search')
    plt.plot(binary_y, 'o', label='Binary search')
    plt.plot(np.arange(1, max_array_size + 1), [linear_model(i) for i in range(1, max_array_size + 1)])
    plt.plot(np.arange(1, max_array_size + 1), [binary_model(i) for i in range(1, max_array_size + 1)])

    plt.legend()
    plt.xlim(left=1)
    plt.title('Linear versus binary search runtime')
    plt.xlabel('Size of array')
    plt.ylabel('CPU time')
    plt.show()

    return linear_model, binary_model

# Run the performance plot
linear_model, binary_model = plot_performance(350)
