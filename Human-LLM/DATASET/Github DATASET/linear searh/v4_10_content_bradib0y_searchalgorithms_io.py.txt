import re

# Open the file containing names
names_file = open("names.txt", encoding="utf-8")
data = names_file.read()
names_file.close()

# Check if a string starts with 'Love'
print(re.match(r'Love', data))

# Search for 'Chalkley' in the data
print(re.search(r'Chalkley', data))

# Find all phone numbers in the data
phones = re.findall(r'\(?\d{3}\)?-?\s?\d{3}-?\s?\d{4}', data)
print("There are this many phone numbers:", len(phones))

# Find all names and titles in the data
namesAndTitles = re.findall(r'\w*, \w*', data)
print("There are this many names and titles:", len(namesAndTitles))

# Define a function to find words of a specific character length in a string
def find_words(length, string):
    pattern = r'\w{' + str(length) + r'}\w*'
    return re.findall(pattern, string)

print(find_words(12, data))

# Find all emails in the data
emails = re.findall(r'[\w\d\-\+\.]+@[\w\d\-\+\.]+', data)
print("There are this many emails:", len(emails))

# Find emails without a '.gov' ending
emailsWithoutGov = re.findall(r'[\w\d\-\+\.]+@[\w\d\-\+\.]+', data)   
print("There are this many emails without .gov ending:", len(emails)) 

# Extract structured data (name, email, phone, etc.) using regex
print(re.findall(r'''
^([-\w ]+,\s[-\w ]+)\t
([-\w\d.+]+@[-\w\d.]+)\t
(\(?\d{3}\)?-?\s?\d{3}-\d{4})\t
([\w\s]+,\s[\w\s]+)\t
(@[\w\d]+)$
''', data, re.X))

# Function to remember and store something in a database
def remember(thing):
    file = open("database.txt", "a")
    file.write(thing + "\n")
    file.close()

# Prompt the user for input to remember something
remember(input("What should I remember?"))
