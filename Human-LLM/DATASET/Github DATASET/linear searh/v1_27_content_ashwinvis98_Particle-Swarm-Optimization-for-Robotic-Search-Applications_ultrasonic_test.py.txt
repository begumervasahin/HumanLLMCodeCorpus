import os
import time
import threading
import nxt
from nxt.sensor import *

# Connect to the NXT brick
b = nxt.locator.find_one_brick('00:16:53:0C:1D:29')

# Function to continuously read ultrasonic sensor
def ultra():
    while True:
        distance = Ultrasonic(b, PORT_2, check_compatible=False).get_sample()
        print("Obstacle at", distance)
        if distance < 50:
            print('You are close, I am going to take your picture')
            # You can add your camera capture logic here
            time.sleep(1)

# Function to continuously increment a counter
def cal():
    i = 0
    while True:
        i += 1
        print("i =", i)
        time.sleep(1)

# Create threads for ultrasonic sensor reading and counter increment
t1 = threading.Thread(target=ultra)
t2 = threading.Thread(target=cal)

# Start the threads
t1.start()
t2.start()

# Join the threads to the main thread
t1.join()
t2.join()
