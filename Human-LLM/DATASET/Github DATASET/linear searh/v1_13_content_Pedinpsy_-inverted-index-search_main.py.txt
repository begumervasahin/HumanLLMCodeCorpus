import os
import time

class HashTable:
    def __init__(self, size, method='division', collision='linear'):
        self.size = size
        self.table = [[] for _ in range(size)]
        self.method = method
        self.collision = collision

    def hash_function(self, key):
        if self.method == 'multiplication':
            return self.hash_multiplication(key)
        else:  # default to division method
            return self.hash_division(key)

    def hash_division(self, key):
        return hash(key) % self.size

    def hash_multiplication(self, key):
        A = 0.6180339887  # Fractional part of the golden ratio
        return int(self.size * ((hash(key) * A) % 1))

    def insert(self, key, value):
        index = self.hash_function(key)
        if not self.table[index]:
            self.table[index].append((key, value))
        elif self.collision == 'linear':
            next_index = (index + 1) % self.size
            while next_index != index:
                if not self.table[next_index]:
                    self.table[next_index].append((key, value))
                    return
                next_index = (next_index + 1) % self.size
            print("Table is full. Couldn't insert:", key)
        elif self.collision == 'quadratic':
            i = 1
            next_index = (index + i**2) % self.size
            while next_index != index:
                if not self.table[next_index]:
                    self.table[next_index].append((key, value))
                    return
                i += 1
                next_index = (index + i**2) % self.size
            print("Table is full. Couldn't insert:", key)

    def get_value(self, key):
        index = self.hash_function(key)
        if self.table[index]:
            for k, v in self.table[index]:
                if k == key:
                    return v
        return None

def read_folder(folder):
    paths = [os.path.join(folder, name) for name in os.listdir(folder)]
    files = [path for path in paths if os.path.isfile(path)]
    return files

def generate_index(file_list, hashtable):
    for i, file in enumerate(file_list):
        with open(file, 'r') as f:
            words = f.read().lower().replace(",", "").replace(".", "").replace("!", "").replace("?", "").replace("\r", "").replace("\t", "").replace("\n", "").split(" ")
            for word in words:
                count = words.count(word)
                hashtable.insert(word, [count, i + 1])

def main():
    hashtable_multiplication_linear = HashTable(1000000, 'multiplication', 'linear')
    hashtable_multiplication_quadratic = HashTable(1000000, 'multiplication', 'quadratic')
    hashtable_division_linear = HashTable(1000000, 'division', 'linear')
    hashtable_division_quadratic = HashTable(1000000, 'division', 'quadratic')

    folder = 'base'
    files = read_folder(folder)

    start = time.time()
    generate_index(files, hashtable_multiplication_linear)
    end = time.time()
    print("Time for Multiplication Hashing with Linear Collision:", end - start)

    start = time.time()
    generate_index(files, hashtable_multiplication_quadratic)
    end = time.time()
    print("Time for Multiplication Hashing with Quadratic Collision:", end - start)

    start = time.time()
    generate_index(files, hashtable_division_linear)
    end = time.time()
    print("Time for Division Hashing with Linear Collision:", end - start)

    start = time.time()
    generate_index(files, hashtable_division_quadratic)
    end = time.time()
    print("Time for Division Hashing with Quadratic Collision:", end - start)

    while True:
        word = input("Enter a word:\n")
        print("Multiplication Linear:", hashtable_multiplication_linear.get_value(word))
        print("Multiplication Quadratic:", hashtable_multiplication_quadratic.get_value(word))
        print("Division Linear:", hashtable_division_linear.get_value(word))
        print("Division Quadratic:", hashtable_division_quadratic.get_value(word))

if __name__ == "__main__":
    main()
