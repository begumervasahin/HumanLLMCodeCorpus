import time

# Function to add corresponding elements of two lists        
def add_lists(list1, list2):
    result = []  # Initialize an empty list to store the result       
    # Iterate over the indices of the lists
    for num1, num2 in zip(list1, list2):
        # Add corresponding elements from both lists and append to result
        result.append(num1 + num2)
    return result  # Return the resulting list

# Create two lists containing ranges of numbers
nums1 = range(0, 9999999)
nums2 = range(100, 10000100)

# Record the start time
start_time = time.time()

# Call the add_lists function to add the two lists
result_list = add_lists(nums1, nums2)

# Calculate the elapsed time
elapsed_time = time.time() - start_time

# Print the elapsed time and length of the resulting list
print("Time elapsed:", elapsed_time)
print("Result list length:", len(result_list))

# Iterate over the resulting list
for item in result_list:
    # Check if the item is divisible by 777777
    if item % 777777 == 0:
        print(item)  # Print the item if divisible by 777777
