import time

def linear_search(target, lst):
    found = False

    for number in lst:
        if number == target:
            found = True
            break

    return found

def calculate_runtime(search_function, target, lst):
    start_time = time.perf_counter()
    result = search_function(target, lst)
    end_time = time.perf_counter()
    elapsed_time = (end_time - start_time) * 100000

    return elapsed_time

if __name__ == "__main__":
    numbers = [-125, -50, -1, 0, 1, 5, 7, 10, 12, 20, 26, 30, 50, 127, 500, 833, 900, 1000, 12345, 123456]
    target_number = int(input("What number do you want to find? "))

    is_found = linear_search(target_number, numbers)
    time_to_find = calculate_runtime(linear_search, target_number, numbers)

    if is_found:
        print("Your item is in the list")
    else:
        print("Your item is not in the list")

    print("Time to find: ", time_to_find)
