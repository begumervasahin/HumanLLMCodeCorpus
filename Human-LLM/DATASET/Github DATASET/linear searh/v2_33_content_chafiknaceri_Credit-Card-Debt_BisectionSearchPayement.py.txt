def calculate_yearly_balance(balance, annual_interest):
    """
    This function calculates the minimum monthly payment required to pay off a debt
    with given balance and annual interest, using the bisection method.
    
    Parameters:
        balance (float): The balance of the debt.
        annual_interest (float): The annual interest rate (as a decimal).
        
    Returns:
        float: The minimum monthly payment required to pay off the debt.
    """
    # Initialize variables
    unpaid_balance = balance
    monthly_interest = annual_interest / 12
    monthly_lower_bound = unpaid_balance / 12
    upper_monthly_bound = (unpaid_balance * ((1 + monthly_interest) ** 12)) / 12

    # Use bisection method to find the minimum monthly payment
    while abs(unpaid_balance) > 0.01:
        result = (monthly_lower_bound + upper_monthly_bound) / 2
        unpaid_balance = balance

        # Calculate unpaid balance for each month
        for i in range(12):
            unpaid_balance -= result
            interest_amount = (annual_interest / 12) * unpaid_balance 
            unpaid_balance += interest_amount

        # Update bounds based on the unpaid balance
        if unpaid_balance > 0.01:
            monthly_lower_bound = result
        elif unpaid_balance < -0.01:
            upper_monthly_bound = result
        else:
            break
            upper_monthly_bound = result

    # Return the rounded result
    return round(result, 2)

# Example usage:
balance = 5000
annual_interest = 0.18  # 18%
monthly_payment = calculate_yearly_balance(balance, annual_interest)  
print("Monthly Payment:", monthly_payment)
