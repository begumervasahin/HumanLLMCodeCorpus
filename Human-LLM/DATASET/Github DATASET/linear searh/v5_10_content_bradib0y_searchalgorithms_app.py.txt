import time
import math

def recursive_binary_search(arr, target):
    """
    Recursive binary search function.
    Returns True if the target is found in the array, else False.
    """
    if len(arr) == 0:
        return False
    else:
        midpoint = len(arr) // 2
        if arr[midpoint] == target:
            return True
        elif arr[midpoint] < target:
            return recursive_binary_search(arr[midpoint + 1:], target)
        else:
            return recursive_binary_search(arr[:midpoint], target)

def binary_search(arr, target):
    """
    Iterative binary search function.
    Returns the index position of the target if found, else None.
    """
    first = 0
    last = len(arr) - 1
    while first <= last:
        midpoint = (first + last) // 2
        if arr[midpoint] == target:
            return midpoint
        elif arr[midpoint] < target:
            first = midpoint + 1
        else:
            last = midpoint - 1
    return None

def linear_search(arr, target):
    """
    Linear search function.
    Returns the index position of the target if found, else None.
    """
    for i, num in enumerate(arr):
        if target == num:
            return i
    return None

def test_log():
    """
    Function to test and log search algorithms.
    """
    for i in range(10, 19):
        numbers = range(0, int(math.pow(10, i)))
        target = (int(math.pow(10, i)) - 3)
        
        data_string = f"{len(numbers)};"
        print(f"\n-> Quantity: {target + 3} (10 to the power of {i})")

        print("Linear search: null")
        data_string += "null;"

        log_start = time.time()
        print("Binary search:", binary_search(numbers, target))
        result = time.time() - log_start
        print("Time taken for Binary search:", result)
        data_string += f"{result};"

        log_start = time.time()
        print("Recursive binary search:", recursive_binary_search(numbers, target))
        result = time.time() - log_start
        print("Time taken for Recursive binary search:", result)
        data_string += f"{result}"

        with open("data.csv", "a") as file:
            file.write(data_string + "\n")

test_log()
