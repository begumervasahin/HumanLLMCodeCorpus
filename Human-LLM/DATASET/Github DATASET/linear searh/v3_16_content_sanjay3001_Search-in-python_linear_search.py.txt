def linear_search(elements_list, item_to_find):
    """
    Perform linear search to find the given item in the list.

    Parameters:
        elements_list (list): List of elements to search through.
        item_to_find (any): Item to find in the list.

    Returns:
        int: Number of iterations performed during the search.
        bool: True if the item is found, False otherwise.
    """
    iteration_count = 0
    found = False

    for position, element in enumerate(elements_list):
        iteration_count += 1
        if element == item_to_find:
            found = True
            print("Found")
            break

    return iteration_count, found


def main():
    # Prompting the user to input the number of elements
    num_elements = int(input("Enter the number of elements: "))

    # Gathering elements from the user
    elements_list = [input("Enter element {}: ".format(i + 1)) for i in range(num_elements)]

    # Displaying the list of elements entered by the user
    print("THE LIST IS", elements_list)

    # Prompting the user to input the item to find
    item_to_find = input("Enter the item to find: ")

    # Performing linear search
    iteration_count, found = linear_search(elements_list, item_to_find)

    # Displaying the search results
    print("Number of iterations:", iteration_count)
    if not found:
        print("Item not found.")


if __name__ == "__main__":
    main()
