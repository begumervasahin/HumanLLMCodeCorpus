import time
from matplotlib import pyplot as plt

# Generate a test case of integers from 0 to 999
test_case = list(range(1000))
execution_times = []  # List to store the execution times

# Function to perform linear search
def linear_search(lst, target):
    for num in lst:
        if num == target:
            return True  # Element found
        if num > target:
            return False  # Element not in the list
    return False  # Element not in the list

# Function to test the performance of linear search on the test case  
def test_linear_search(lst):
    for target in lst:
        start_time = time.time()  # Record start time
        found = linear_search(lst, target)  # Perform linear search
        end_time = time.time()  # Record end time
        if found:
            execution_times.append(end_time - start_time)  # Record execution time if element found
    return execution_times

# Function to display the results using Matplotlib
def plot_execution_times(lst, times):
    plt.plot(lst, times, color='green', marker='o', linestyle='solid')    
    plt.title("Linear Search Performance")
    plt.xlabel('Test Case')
    plt.ylabel('Execution Time (s)')
    plt.show()

# Perform the test
execution_times = test_linear_search(test_case)

# Display the results
plot_execution_times(test_case, execution_times)
