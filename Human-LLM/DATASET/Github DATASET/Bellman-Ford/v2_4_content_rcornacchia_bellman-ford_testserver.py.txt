import socket
import sys

# Check if the command-line argument for UDP port is provided
if len(sys.argv) < 2:
    print("Usage: python udp_server.py <UDP_PORT>")
    sys.exit(1)

# Get the UDP port from the command-line argument
UDP_PORT = int(sys.argv[1])

# Create a UDP socket
udp_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)

# Bind the UDP socket to the specified IP address and port
udp_socket.bind(("160.39.231.6", UDP_PORT))

print(f"UDP server is running on port {UDP_PORT}")

# Listen for incoming messages and print them
while True:
    data, addr = udp_socket.recvfrom(1024)
    print(f"Received message from {addr}: {data.decode('utf-8')}")
