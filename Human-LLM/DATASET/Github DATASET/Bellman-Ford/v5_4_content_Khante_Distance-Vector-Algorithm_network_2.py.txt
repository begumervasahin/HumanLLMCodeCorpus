import queue
import threading
from collections import defaultdict

class Interface:
    def __init__(self, maxsize=0):
        self.in_queue = queue.Queue(maxsize)
        self.out_queue = queue.Queue(maxsize)

    def get(self, direction):
        try:
            if direction == 'in':
                return self.in_queue.get(False)
            else:
                return self.out_queue.get(False)
        except queue.Empty:
            return None

    def put(self, packet, direction, block=False):
        if direction == 'out':
            self.out_queue.put(packet, block)
        else:
            self.in_queue.put(packet, block)

class NetworkPacket:
    DST_LENGTH = 5
    PROT_LENGTH = 1

    def __init__(self, dst, prot, data):
        self.dst = dst
        self.data = data
        self.prot = prot

    def __str__(self):
        return self.to_byte_string()

    def to_byte_string(self):
        byte_string = str(self.dst).zfill(self.DST_LENGTH)
        if self.prot == 'data':
            byte_string += '1'
        elif self.prot == 'control':
            byte_string += '2'
        else:
            raise ValueError('Unknown protocol option: {}'.format(self.prot))
        byte_string += self.data
        return byte_string

    @classmethod
    def from_byte_string(cls, byte_string):
        dst = byte_string[:cls.DST_LENGTH].strip('0')
        prot = byte_string[cls.DST_LENGTH: cls.DST_LENGTH + cls.PROT_LENGTH]
        if prot == '1':
            prot = 'data'
        elif prot == '2':
            prot = 'control'
        else:
            raise ValueError('Unknown protocol field: {}'.format(prot))
        data = byte_string[cls.DST_LENGTH + cls.PROT_LENGTH:]
        return cls(dst, prot, data)

class Host:
    def __init__(self, addr):
        self.addr = addr
        self.interfaces = [Interface()]
        self.stop = False

    def __str__(self):
        return self.addr

    def udt_send(self, dst, data):
        packet = NetworkPacket(dst, 'data', data)
        print('{}: SENDING PACKET "{}"'.format(self, packet))
        self.interfaces[0].put(packet.to_byte_string(), 'out')

    def udt_receive(self):
        packet_string = self.interfaces[0].get('in')
        if packet_string is not None:
            print('{}: RECEIVED PACKET "{}"'.format(self, packet_string))
            if self.addr == 'H2':
                self.udt_send('H1', 'The way you look should be a sin, you my sensation')

    def run(self):
        print('{}: Starting'.format(threading.currentThread().getName()))
        while True:
            self.udt_receive()
            if self.stop:
                print('{}: Ending'.format(threading.currentThread().getName()))
                return

class Router:
    def __init__(self, name, cost_D, max_queue_size):
        self.stop = False
        self.name = name
        self.interfaces = [Interface(max_queue_size) for _ in range(len(cost_D))]       
        self.neighbors = cost_D
        self.routing_table = cost_D.copy()
        self.total_routing_table = defaultdict(dict)
        for neighbor_name, neighbor_info in self.neighbors.items():
            for interface, cost in neighbor_info.items():
                self.total_routing_table[name][neighbor_name] = [cost]
        print('{}: Initialized routing table'.format(self))
        self.print_routing_table()

    def __str__(self):
        return self.name

    def process_queues(self):
        for i in range(len(self.interfaces)):
            packet_string = self.interfaces[i].get('in')
            if packet_string is not None:
                packet = NetworkPacket.from_byte_string(packet_string)
                if packet.prot == 'data':
                    self.forward_packet(packet, i)
                elif packet.prot == 'control':
                    self.update_routing_table(packet, i)
                else:
                    raise ValueError('Unknown packet type in packet {}'.format(packet)) 

    def forward_packet(self, packet, i):
        try:
            route = self.routing_table.get(str(packet.dst))
            for interface, cost in route.items():
                j = int(interface)
                break
            self.interfaces[j].put(packet.to_byte_string(), 'out', True)
            print('{}: forwarding packet "{}" from interface {} to {}'.format(self, packet, i, j))
        except queue.Full:
            print('{}: packet "{}" lost on interface {}'.format(self, packet, i))       

    def send_routing_table(self, i):
        print('sending routing table')
        contents = '{}->'.format(self.name)
        for k, j in self.routing_table.items():
            for link, cost in j.items():
                contents += '{}_{}_{} '.format(k, link, cost)
        packet = NetworkPacket(0, 'control', contents)
        self.interfaces[i].put(packet.to_byte_string(), 'out', True)

    def update_routing_table(self, packet, i):
        update_flag = 0
        contents = packet.data.split('->')
        name = contents[0]
        route_list = contents[1].split(' ')
        for item in route_list:
            if item != '':
                node_list = item.split('_')
                current_node = node_list[0]
                cost = int(node_list[2])
                self.total_routing_table[name][current_node] = [cost]
                if current_node == self.name:
                    neighbor_interface = int(list(self.neighbors[name].keys())[0])      
                    self.routing_table[current_node] = {neighbor_interface: 0}
                    self.total_routing_table[self.name][current_node] = [0]
                    continue
                else:
                    if current_node not in self.neighbors and current_node not in self.routing_table:
                        neighbor_cost = list(self.neighbors[name].values())[0]
                        neighbor_interface = int(list(self.neighbors[name].keys())[0])  
                        self.routing_table[current_node] = {neighbor_interface: cost + neighbor_cost}
                        self.total_routing_table[self.name][current_node] = [cost + neighbor_cost]
                        update_flag = 1
                    elif current_node not in self.neighbors and current_node in self.routing_table:
                        neighbor_cost = list(self.neighbors[name].values())[0]
                        neighbor_interface = int(list(self.neighbors[name].keys())[0])  
                        current_cost = list(self.routing_table[current_node].values())[0]
                        if current_cost > neighbor_cost + cost:
                            self.routing_table[current_node] = {neighbor_interface: cost + neighbor_cost}
                            self.total_routing_table[self.name][current_node] = [cost + neighbor_cost]
                            update_flag = 1
        if update_flag == 1:
            for neighbor_name, neighbor_info in self.neighbors.items():
                for interface, cost in neighbor_info.items():
                    if 'R' in neighbor_name:
                        self.send_routing_table(interface)

    def print_routing_table(self):
        cost_table = []
        print('cost table for {}:'.format(self))
        sorted_table = sorted(self.total_routing_table.items())
        interfaces = sorted(set(link for _, neighbor_info in sorted_table for link in neighbor_info))
        for interface in interfaces:
            print('{}\t'.format(interface), end='')
            for item, neighbor_info in sorted_table:
                for link, cost_info in neighbor_info.items():
                    if link == interface:
                        print('{}\t'.format(cost_info[0]), end='')
                        cost_table.append(cost_info[0])
        print("\n")
        table_str1 = ''
        table_str2 = ''
        for i in range(int(len(cost_table) / 2 + 1)):
            table_str1 += '| '
            table_str1 += str(cost_table[(i - 1) * 2])
            if i < len(cost_table):
                table_str2 += '| '
                table_str2 += str(cost_table[(i - 1) * 2 + 1])
        table_str1 += ' | '
        table_str2 += ' | '
        print(table_str1)
        print(table_str2)
        print("\n")

    def run(self):
        print('{}: Starting'.format(threading.currentThread().getName()))
        while True:
            self.process_queues()
            if self.stop:
                print('{}: Ending'.format(threading.currentThread().getName()))
                return
