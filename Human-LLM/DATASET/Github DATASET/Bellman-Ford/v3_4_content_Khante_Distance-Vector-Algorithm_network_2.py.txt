import queue
import threading
from collections import OrderedDict, defaultdict

class Interface:
    def __init__(self, maxsize=0):
        self.in_queue = queue.Queue(maxsize)
        self.out_queue = queue.Queue(maxsize)

    def get(self, direction):
        try:
            if direction == 'in':
                return self.in_queue.get(False)
            else:
                return self.out_queue.get(False)
        except queue.Empty:
            return None

    def put(self, pkt, direction, block=False):
        if direction == 'out':
            self.out_queue.put(pkt, block)
        else:
            self.in_queue.put(pkt, block)

class NetworkPacket:
    DST_LENGTH = 5
    PROTOCOL_LENGTH = 1

    def __init__(self, dst, protocol, data):
        self.dst = dst
        self.data = data
        self.protocol = protocol

    def __str__(self):
        return self.to_byte_S()

    def to_byte_S(self):
        byte_S = str(self.dst).zfill(self.DST_LENGTH)
        if self.protocol == 'data':
            byte_S += '1'
        elif self.protocol == 'control':
            byte_S += '2'
        else:
            raise ValueError(f'Unknown protocol: {self.protocol}')
        byte_S += self.data
        return byte_S

    @classmethod
    def from_byte_S(cls, byte_S):
        dst = byte_S[:cls.DST_LENGTH].strip('0')
        protocol = byte_S[cls.DST_LENGTH: cls.DST_LENGTH + cls.PROTOCOL_LENGTH]
        if protocol == '1':
            protocol = 'data'
        elif protocol == '2':
            protocol = 'control'
        else:
            raise ValueError(f'Unknown protocol: {protocol}')
        data = byte_S[cls.DST_LENGTH + cls.PROTOCOL_LENGTH:]
        return cls(dst, protocol, data)

class Host:
    def __init__(self, addr):
        self.addr = addr
        self.interfaces = [Interface()]
        self.stop = False

    def __str__(self):
        return self.addr

    def udt_send(self, dst, data):
        packet = NetworkPacket(dst, 'data', data)
        print(f'{self}: SENDING PACKET "{packet}"')
        self.interfaces[0].put(packet.to_byte_S(), 'out')

    def udt_receive(self):
        packet_S = self.interfaces[0].get('in')
        if packet_S is not None:
            print(f'{self}: RECEIVED PACKET "{packet_S}"')
            if self.addr == 'H2':
                self.udt_send('H1', 'The way you look should be a sin, you my sensation')

    def run(self):
        print(f'{threading.currentThread().getName()}: Starting')
        while True:
            self.udt_receive()
            if self.stop:
                print(f'{threading.currentThread().getName()}: Ending')
                return

class Router:
    def __init__(self, name, neighbor_info, max_queue_size):
        self.stop = False
        self.name = name
        self.interfaces = [Interface(max_queue_size) for _ in range(len(neighbor_info))]
        self.neighbors = neighbor_info
        self.routing_table = neighbor_info.copy()
        self.total_routing_table = defaultdict(dict)
        for neighbor_name, cost_info in self.neighbors.items():
            for interface, cost in cost_info.items():
                self.total_routing_table[name][neighbor_name] = [cost]
        print(f'{self}: Initialized routing table')
        self.print_routing_table()

    def process_queues(self):
        for i, interface in enumerate(self.interfaces):
            packet_S = interface.get('in')
            if packet_S is not None:
                packet = NetworkPacket.from_byte_S(packet_S)
                if packet.protocol == 'data':
                    self.forward_packet(packet, i)
                elif packet.protocol == 'control':
                    self.update_routing_table(packet, i)
                else:
                    raise ValueError(f'Unknown packet type in packet: {packet}')

    def forward_packet(self, packet, interface_index):
        try:
            route = self.routing_table.get(str(packet.dst))
            for interface, cost in route.items():
                neighbor_interface_index = int(interface)
                break
            self.interfaces[neighbor_interface_index].put(packet.to_byte_S(), 'out', True)
            print(f'{self}: forwarding packet "{packet}" from interface {interface_index} to {neighbor_interface_index}')
        except queue.Full:
            print(f'{self}: packet "{packet}" lost on interface {interface_index}')

    def send_routing_table(self, interface_index):
        print('sending routing table')
        contents = f'{self.name}->'
        for dest, cost_info in self.routing_table.items():
            for link, cost in cost_info.items():
                contents += f'{dest}_{link}_{cost} '
        packet = NetworkPacket(0, 'control', contents)
        self.interfaces[interface_index].put(packet.to_byte_S(), 'out', True)

    def update_routing_table(self, packet, interface_index):
        update_flag = 0
        contents = packet.data.split("->")
        sender_name = contents[0]
        route_list = contents[1].split(" ")
        for item in route_list:
            if item != '':
                node_list = item.split("_")
                current_node = node_list[0]
                cost = int(node_list[2])
                self.total_routing_table[sender_name][current_node] = [cost]
                if current_node == self.name:
                    neighbor_interface_index = int(list(self.neighbors[sender_name].keys())[0])
                    self.routing_table[current_node] = {neighbor_interface_index: 0}
                    self.total_routing_table[self.name][current_node] = [0]
                    continue
                else:
                    if current_node not in self.neighbors and current_node not in self.routing_table:
                        neighbor_cost = list(self.neighbors[sender_name].values())[0]
                        neighbor_interface_index = int(list(self.neighbors[sender_name].keys())[0])
                        self.routing_table[current_node] = {neighbor_interface_index: cost + neighbor_cost}
                        self.total_routing_table[self.name][current_node] = [cost + neighbor_cost]
                        update_flag = 1
                    elif current_node not in self.neighbors and current_node in self.routing_table:
                        neighbor_cost = list(self.neighbors[sender_name].values())[0]
                        neighbor_interface_index = int(list(self.neighbors[sender_name].keys())[0])
                        current_cost = list(self.routing_table[current_node].values())[0]
                        if current_cost > neighbor_cost + cost:
                            self.routing_table[current_node] = {neighbor_interface_index: cost + neighbor_cost}
                            self.total_routing_table[self.name][current_node] = [cost + neighbor_cost]
                            update_flag = 1
        if update_flag == 1:
            for neighbor_name, neighbor_info in self.neighbors.items():
                for interface, cost in neighbor_info.items():
                    if 'R' in neighbor_name:
                        self.send_routing_table(interface)

    def print_routing_table(self):
        cost_table = []
        print(f'cost table for {self}:')
        sorted_table = OrderedDict(sorted(self.total_routing_table.items()))
        interfaces = []
        for item, neighbor_info in sorted_table.items():
            for link, cost_info in neighbor_info.items():
                if link not in interfaces:
                    interfaces.append(link)
        interfaces = sorted(interfaces)
        for interface in interfaces:
            print(f'{interface}\t', end='')
            for item, neighbor_info in sorted_table.items():
                for link, cost_info in neighbor_info.items():
                    if link == interface:
                        print(f'{cost_info[0]}\t', end='')
                        cost_table.append(cost_info[0])
        print("\n")
        table_str1 = ''
        table_str2 = ''
        for i in range(int(len(cost_table) / 2 + 1)):
            table_str1 += '| '
            table_str1 += str(cost_table[(i - 1) * 2])
            if i < len(cost_table):
                table_str2 += '| '
                table_str2 += str(cost_table[(i - 1) * 2 + 1])
        table_str1 += ' | '
        table_str2 += ' | '
        print(table_str1)
        print(table_str2)
        print("\n")

    def run(self):
        print(f'{threading.currentThread().getName()}: Starting')
        while True:
            self.process_queues()
            if self.stop:
                print(f'{threading.currentThread().getName()}: Ending')
                return
