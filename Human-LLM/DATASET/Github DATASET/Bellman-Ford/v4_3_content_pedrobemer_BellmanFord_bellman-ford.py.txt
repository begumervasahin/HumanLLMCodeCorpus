import argparse
import string

class BellmanFord:
    def __init__(self, start_node, graph):
        self.graph = graph
        self.distances = {}
        self.distances[start_node] = 0
        self.start_node = start_node
        self.nodes = list(string.ascii_uppercase[:len(self.graph)])

        for node in self.nodes[1:]:
            self.distances[node] = float('inf')

        self.relax()
        self.detect_negative_cycle()

    def relax(self):
        for _ in range(1, len(self.graph) - 1):
            for node in self.nodes:
                neighbors = self.graph[node].keys()

                for neighbor in neighbors:
                    if self.distances[node] == float('inf'):
                        break
                    elif self.distances[neighbor] == float('inf'):
                        self.distances[neighbor] = self.graph[node][neighbor] + self.distances[node]
                    elif self.graph[node][neighbor] + self.distances[node] < self.distances[neighbor]:
                        self.distances[neighbor] = self.graph[node][neighbor] + self.distances[node]

        print(self.distances)

    def detect_negative_cycle(self):
        for node in self.nodes:
            neighbors = self.graph[node].keys()

            for neighbor in neighbors:
                if self.graph[node][neighbor] + self.distances[node] < self.distances[neighbor]:
                    print('The graph has a negative-weight cycle')
