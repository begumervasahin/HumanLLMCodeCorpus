# Import necessary functions from the utils module
from utils import initialize_single_source, extract_min, relax

def dijkstra(graph, source):
    """Implements Dijkstra's algorithm to find the shortest paths from a source vertex."""
    initialize_single_source(graph, source)
    
    S = []  # List to store visited vertices
    Q = [v for v in graph.get_vertices()]  # List to store vertices

    while len(Q) > 0:
        u = extract_min(Q)  # Extract the vertex with the minimum distance from Q
        u.set_visitado()  # Mark u as visited
        S.append(u)  # Add u to the list of visited vertices

        # Relax edges going from u to adjacent vertices
        for v in u.get_vertices_adjacentes():
            if v.get_visitado():
                continue
            relax(u, v)
