import queue
import threading
import time

class Link:
    def __init__(self, node1, intf1, node2, intf2):
        self.node1 = node1
        self.intf1 = intf1
        self.node2 = node2
        self.intf2 = intf2
        print(f'Created link: {self}')

    def __str__(self):
        return f'Link {self.node1}-{self.intf1} - {self.node2}-{self.intf2}'

    def transmit_packet(self):
        for src_node, src_intf, dest_node, dest_intf in [(self.node1, self.intf1, self.node2, self.intf2), (self.node2, self.intf2, self.node1, self.intf1)]:
            src_intf_obj = src_node.intf_L[src_intf]
            dest_intf_obj = dest_node.intf_L[dest_intf]
            packet = src_intf_obj.get('out')
            if packet is None:
                continue
            try:
                dest_intf_obj.put(packet, 'in')
                print(f'{self}: transmitting packet "{packet}" from {src_node}-{src_intf} to {dest_node}-{dest_intf}')
            except queue.Full:
                print(f'{self}: packet lost in transmission')

class LinkLayer:
    def __init__(self):
        self.links = []
        self.stop = False

    def __str__(self):
        return 'Link Layer'

    def add_link(self, link):
        self.links.append(link)

    def transfer_packets(self):
        for link in self.links:
            link.transmit_packet()

    def run(self):
        print(f'{threading.currentThread().getName()}: Link layer started')
        while not self.stop:
            self.transfer_packets()
        print(f'{threading.currentThread().getName()}: Link layer stopped')
