import sys
import socket
import json
import threading
import time
import os

# Global variables
current_node = sys.argv[1]
current_port = int(sys.argv[2])
config = sys.argv[3]
d = {}
dv = {}
nexthop = {}
displayLock = threading.Lock()
routing_table = {}
pre_time = {}

class Nodes:
    def __init__(self, current_node, current_port, config):
        self.current_node = current_node
        self.current_port = current_port
        self.direct_links = {}
        self.neighbours = []
        self.load_config(config)

    def load_config(self, config_file):
        with open(config_file, "r+") as fo:
            data = fo.read()
            nodes = data.split()
            total_nodes = nodes[0]
            i = 1
            while i < len(nodes):
                d[nodes[i]] = [float(0), int(nodes[i + 2])]
                i += 3
            for key, value in d.items():
                self.direct_links[key] = value[0]
                if key != self.current_node:
                    self.neighbours.append(key)
            dv[self.current_node] = self.direct_links.copy()

def recv():
    recvIP = "localhost"
    recvPort = current_port
    recv = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
    recv.bind((recvIP, recvPort))
    while True:
        message, client_address = recv.recvfrom(2048)
        dist_vctr = json.loads(message)
        for key, val in dist_vctr.items():
            neighbor = key.encode('ascii', 'ignore')
            for k, v in dist_vctr[neighbor].items():
                k = k.encode('ascii', 'ignore')
                routing_table[k] = v
        threading.Thread(target=bellman_ford).start()
    recv.close()

def sender():
    while True:
        port_list = [value[1][1] for value in d.items()]
        serverIP = "localhost"
        sender_socket = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
        strings = json.dumps(dv)
        for item in port_list:
            if int(item) != current_port:
                sender_socket.sendto(strings.encode(), (serverIP, int(item)))
        time.sleep(5)
        sender_socket.close()

def bellman_ford():
    my_neighbour = [node for node in routing_table if node in node1.neighbours]
    all_routers = set([neighbour for node in routing_table for neighbour in routing_table[node]])
    for node in routing_table:
        for router in all_routers:
            if router not in routing_table[node]:
                routing_table[node][router] = float('inf')
    for router in all_routers:
        for neighbour in my_neighbour:
            if routing_table[current_node][router] > routing_table[neighbour][router] + routing_table[current_node][neighbour]:
                min_distance = routing_table[neighbour][router] + routing_table[current_node][neighbour]
                routing_table[current_node][router] = min_distance
                if router in my_neighbour and node1.direct_links[router] <= min_distance:
                    routing_table[current_node][router] = node1.direct_links[router]
                    nexthop[router] = 'direct'
                else:
                    nexthop[router] = neighbour
            elif len(nexthop) != len(all_routers):
                nexthop[router] = 'direct'
            elif router in my_neighbour and node1.direct_links[router] <= routing_table[current_node][router]:
                routing_table[current_node][router] = node1.direct_links[router]
                nexthop[router] = 'direct'
    dv[current_node] = routing_table[current_node]
    display()

def display():

            print("Least cost path to router " + node + ": through " + nexthop[node] + " with cost " + str("{0:.1f}".format(dv[current_node][node]) + "\n"))
