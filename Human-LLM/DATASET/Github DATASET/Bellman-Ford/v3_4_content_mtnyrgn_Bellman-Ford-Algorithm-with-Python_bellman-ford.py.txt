class Graph:
    def __init__(self, vertices):
        """Initialize the graph with a given number of vertices."""
        self.V = vertices
        self.graph = []

    def add_edge(self, u, v, w):
        """Add an edge to the graph."""
        self.graph.append([u, v, w])

    def print_graph(self, dist):
        """Print the distances of vertices from the source."""
        print("Distance:")
        for i in range(self.V):
            print("%d \t\t %d" % (i, dist[i]))

    def bellman_ford(self, src):
        """
        Apply the Bellman-Ford algorithm to find shortest paths.

        Parameters:
        - src: The source vertex.
        """
        # Initialize distances with infinity
        dist = [float("Inf")] * self.V
        dist[src] = 0

        # Relax edges repeatedly for V-1 iterations
        for _ in range(self.V - 1):
            for u, v, w in self.graph:
                if dist[u] != float("Inf") and dist[u] + w < dist[v]:
                    dist[v] = dist[u] + w

        # Check for negative cycles
        for u, v, w in self.graph:
            if dist[u] != float("Inf") and dist[u] + w < dist[v]:
                print("Graph contains negative cycle!!")
                return

        # Print the distances of vertices from the source
        self.print_graph(dist)

# Get input from the user
vertices = int(input("Enter number of vertices: "))
edges = int(input("Enter number of edges: "))

# Create a graph object and add edges
g = Graph(vertices)
for i in range(edges):
    u = int(input("Source vertex: "))
    v = int(input("Destination vertex: "))
    w = int(input("Weight of Edge: "))
    g.add_edge(u, v, w)

# Apply Bellman-Ford algorithm from source vertex 0
g.bellman_ford(0)
