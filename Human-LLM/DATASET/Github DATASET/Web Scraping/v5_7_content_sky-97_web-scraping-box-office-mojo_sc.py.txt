import requests
from bs4 import BeautifulSoup
import sqlite3

# Connect to SQLite database
conn = sqlite3.connect('movie.db')
c = conn.cursor()

# Create a table named 'movie' if it doesn't exist
c.execute("""CREATE TABLE IF NOT EXISTS movie (
                Rank TEXT,
                MovieTitle TEXT,
                Total TEXT,
                Opening TEXT,
                Open TEXT
            )""")

def fetch_movie_data(year):
    # Generate URL for the given year
    url = f'https://www.boxofficemojo.com/yearly/chart/?yr={year}&p=.htm'
    
    # Send request to fetch the webpage
    source = requests.get(url)
    
    # Parse the HTML content
    soup = BeautifulSoup(source.text, 'lxml')
    
    # Extract the table containing movie data
    movie_table = soup.find_all('table')[6]
    
    # Extract column headers from the table
    headers = movie_table.find_all('tr')[0].find_all('td')
    header_row_list = [header.text for header in headers]
    page_data_list = [header_row_list]
    
    # Loop through each movie row in the table
    for movie_row in movie_table.find_all('tr')[2:-4]:
        movie_data_items = movie_row.find_all('td')
        
        # Extract movie details from each row
        movie_details = [item.text for item in movie_data_items]
        
        # Append movie details to the list of data for the page
        page_data_list.append(movie_details)
        
        # Insert movie details into the SQLite database
        c.execute("INSERT INTO movie VALUES (?,?,?,?,?)", movie_details)

    # Return the data list for the page
    return page_data_list
