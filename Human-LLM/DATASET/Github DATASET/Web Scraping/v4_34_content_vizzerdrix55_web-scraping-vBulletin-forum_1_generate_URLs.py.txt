import requests
from bs4 import BeautifulSoup
import re
import random
import datetime
import logging
import sys
import time

# Configuring logging
logging.basicConfig(filename='log.log',
                    filemode='w',
                    format='%(asctime)s %(message)s',
                    level=logging.DEBUG)
logging.info('Script started')

# List of delays for simulating human-like behavior
delays = [i/100 for i in range(80, 121)]

# URL of the website to scrape
url = 'http://forum.pcgames.de/'

# Retrieving the page content
page = requests.get(url)
soup = BeautifulSoup(page.text, 'html.parser')

# Finding forum titles
h2_forumtitles = soup.find_all('h2', class_='forumtitle')
forumtitles = []

if h2_forumtitles:
    for item in h2_forumtitles:
        a_forumtitle = item.find('a')
        if a_forumtitle:
            forumtitles.append(a_forumtitle.get('href'))
else:
    logging.error('Unexpected HTML structure for forum titles')

# Writing forum titles to a file
with open('forumtitles.txt', 'w') as file:
    for title in forumtitles:
        file.write(title + '\n')
logging.info(f'Printed {len(forumtitles)} forum titles to forumtitles.txt')

# Finding subforum titles
li_subforum = soup.find_all('li', class_='subforum')
subforumtitles = []

if li_subforum:
    for item in li_subforum:
        a_subforumtitle = item.find('a')
        if a_subforumtitle:
            subforumtitles.append(a_subforumtitle.get('href'))
else:
    logging.error('Unexpected HTML structure for subforum titles')

# Writing subforum titles to a file
with open('subforumtitles.txt', 'w') as file:
    for title in subforumtitles:
        file.write(title + '\n')
logging.info(f'Printed {len(subforumtitles)} subforum titles to subforumtitles.txt')

# Scanning forum pages for additional pages
forumtitles_new = []

for item in forumtitles:
    delay = random.choice(delays)
    time.sleep(delay)
    try:
        page = requests.get(item)
        soup = BeautifulSoup(page.text, 'html.parser')
        x = soup.find('form', class_='pagination')
        if x is not None:
            pattern = re.compile(r'Seite\s\d+\svon\s(\d+)', re.I)
            pages = soup.find('a', text=pattern).text.strip()
            pages = int(pattern.match(pages).group(1))
            page_urls = [f'{item}/index{p}.html' for p in range(1, pages + 1)]
            forumtitles_new.extend(page_urls)
        else:
            forumtitles_new.append(item)
    except Exception as e:
        logging.error(f'ERROR: Skipped {item} because of {e}')

# Writing updated forum titles to a file
with open('forumtitles_new.txt', 'w') as file:
    for title in forumtitles_new:
        file.write(title + '\n')
logging.info(f'Printed {len(forumtitles_new)} updated forum titles to forumtitles_new.txt')

# Scanning subforum pages for additional pages
subforumtitles_new = []

for item in subforumtitles:
    delay = random.choice(delays)
    time.sleep(delay)
    try:
        page = requests.get(item)
        soup = BeautifulSoup(page.text, 'html.parser')
        x = soup.find('form', class_='pagination')
        if x is not None:
            pattern = re.compile(r'Seite\s\d+\svon\s(\d+)', re.I)
            pages = soup.find('a', text=pattern).text.strip()
            pages = int(pattern.match(pages).group(1))
            page_urls = [f'{item}/index{p}.html' for p in range(1, pages + 1)]
            subforumtitles_new.extend(page_urls)
        else:
            subforumtitles_new.append(item)
    except Exception as e:
        logging.error(f'ERROR: Skipped {item} because of {e}')

# Writing updated subforum titles to a file
with open('subforumtitles_new.txt', 'w') as file:
    for title in subforumtitles_new:
        file.write(title + '\n')
logging.info(f'Printed {len(subforumtitles_new)} updated subforum titles to subforumtitles_new.txt')

# Combining forum and subforum titles
all_forumtitles_new = subforumtitles_new + forumtitles_new

# Writing all updated titles to a file
with open('all_forumtitles_new.txt', 'w') as file:
    for title in all_forumtitles_new:
        file.write(title + '\n')
logging.info(f'Printed {len(all_forumtitles_new)} combined updated titles to all_forumtitles_new.txt')

# Extracting thread URLs from forum and subforum pages
threadurls = []

for item in all_forumtitles_new:
    delay = random.choice(delays)
    time.sleep(delay)
    try:
        page = requests.get(item)
        soup = BeautifulSoup(page.text, 'html.parser')
        threadlist = soup.find_all('h3', class_='threadtitle')
        for thread in threadlist:
            a_thread = thread.find('a')
            if a_thread:
                threadurls.append(a_thread.get('href'))
    except Exception as e:
        logging.error(f'ERROR: Skipped {item} because of {e}')

# Writing thread URLs to a file
with open('threadurls.txt', 'w') as file:
    for url in threadurls:
        file.write(url + '\n')
logging.info(f'Printed {len(threadurls)} thread URLs to threadurls.txt')

# Scanning thread pages for additional pages
threadurls_new = []

for item in threadurls:
    delay = random.choice(delays)
    time.sleep(delay)
    try:
        page = requests.get(item)
        soup = BeautifulSoup(page.text, 'html.parser')
        x = soup.find('form', class_='pagination')
        if x is not None:
            pattern = re.compile(r'Seite\s\d+\svon\s(\d+)', re.I)
            pages = soup.find('a', text=pattern).text.strip()
            pages = int(pattern.match(pages).group(1))
            page_urls = [f'{item[:-5]}-{p}.html' for p in range(1, pages + 1)]
            threadurls_new.extend(page_urls)
        else:
            threadurls_new.append(item)
    except Exception as e:
        logging.error(f'ERROR: Skipped {item} because of {e}')

# Writing updated thread URLs to a file
with open('threadurls_new.txt', 'w') as file:
    for url in threadurls_new:
        file.write(url + '\n')
logging.info(f'Printed {len(threadurls_new)} updated thread URLs to threadurls_new.txt')

logging.info('Script ended')
