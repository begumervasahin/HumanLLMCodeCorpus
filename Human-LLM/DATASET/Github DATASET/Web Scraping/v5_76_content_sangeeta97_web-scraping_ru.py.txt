import requests
from bs4 import BeautifulSoup
import pandas as pd
import numpy as np
import multiprocessing as mp
from time import sleep
from warnings import warn

# Read data from an Excel file
df = pd.read_excel('eunew1.xlsx')

# Function to fetch data from a URL
def fetch_data(url):
    max_tries = 3
    for attempt in range(max_tries):
        try:
            response = requests.get(url, verify=False)
            response.raise_for_status()  # Raise exception for bad status
            break
        except (requests.ConnectionError, requests.HTTPError) as e:
            if attempt < max_tries - 1:
                sleep(2)
                continue
            else:
                raise

    # Parse HTML content
    page_html = BeautifulSoup(response.text, 'html.parser')
    # Extract data from specific HTML elements
    data = [str(td.text).splitlines() for td in page_html.find_all("td", "third", limit=8)]
    return data

# Using multiprocessing for concurrent URL fetching
pool = mp.Pool(mp.cpu_count())

# Get URLs from the dataframe
urls = df['url'].tolist()

# Fetch data from multiple URLs concurrently
results = pool.map(fetch_data, urls)

# Close the pool of workers
pool.close()
pool.join()

# Create a dataframe from the fetched results
df_fetched = pd.DataFrame(np.array(results), columns=['all'])

# Merge the fetched data with the original dataframe
df_merged = pd.concat([df, df_fetched], axis=1)

# Save the results to an Excel file
df_merged.to_excel('results_oct.xlsx', index=False)
