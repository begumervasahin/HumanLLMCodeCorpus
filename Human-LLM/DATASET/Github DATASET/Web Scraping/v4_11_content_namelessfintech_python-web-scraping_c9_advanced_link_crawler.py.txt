import re
import time
import urllib.request
import urllib.error
import urllib.parse
from urllib import robotparser
from lxml.html import fromstring

class Throttle:
    """Prevents quick repeated requests to the same domain by sleeping between requests."""
    def __init__(self, delay):
        self.delay = delay
        self.domains_last_accessed = {}

    def wait(self, url):
        """Ensure delay between requests to the same domain."""
        domain = urllib.parse.urlparse(url).netloc
        last_accessed = self.domains_last_accessed.get(domain)

        if self.delay > 0 and last_accessed is not None:
            sleep_duration = self.delay - (time.time() - last_accessed)
            if sleep_duration > 0:
                time.sleep(sleep_duration)

        self.domains_last_accessed[domain] = time.time()

def download(url, num_retries=2, user_agent='wswp', charset='utf-8', proxy=None):
    """Attempts to download a URL and returns its HTML content. Retries on 5xx errors."""
    print(f'Downloading: {url}')
    request = urllib.request.Request(url)
    request.add_header('User-agent', user_agent)

    try:
        if proxy:
            proxy_support = urllib.request.ProxyHandler({'http': proxy})
            opener = urllib.request.build_opener(proxy_support)
            urllib.request.install_opener(opener)
        
        response = urllib.request.urlopen(request)
        content_type = response.headers.get_content_charset() or charset
        return response.read().decode(content_type)
    
    except (urllib.error.URLError, urllib.error.HTTPError, urllib.error.ContentTooShortError) as e:
        print(f'Download error: {e.reason}')
        html = None
        if num_retries > 0 and hasattr(e, 'code') and 500 <= e.code < 600:
            return download(url, num_retries - 1, user_agent, charset, proxy)
    return None

def get_robots_parser(robots_url):
    """Returns the robots parser object for the specified robots.txt URL."""
    rp = robotparser.RobotFileParser()
    rp.set_url(robots_url)
    rp.read()
    return rp

def get_links(html):
    """Returns a list of links extracted from the HTML content."""
    webpage_regex = re.compile(r'<a[^>]+href=["\'](.*?)["\']', re.IGNORECASE)
    return webpage_regex.findall(html)

def link_crawler(start_url, link_regex, user_agent='wswp', delay=3, max_depth=4, scrape_callback=None):
    """Crawls a website starting from a URL, following links that match a given regex."""
    crawl_queue = [start_url]
    seen = {}
    robots_url = f'{start_url}/robots.txt'
    rp = get_robots_parser(robots_url)
    throttle = Throttle(delay)

    while crawl_queue:
        url = crawl_queue.pop()
        if not rp.can_fetch(user_agent, url):
            print(f'Blocked by robots.txt: {url}')
            continue

        depth = seen.get(url, 0)
        if depth == max_depth:
            print(f'Skipping {url} due to depth')
            continue

        throttle.wait(url)
        html = download(url, user_agent=user_agent)
        if not html:
            continue

        if scrape_callback:
            scrape_callback(url, html)

        for link in get_links(html):
            if re.match(link_regex, link):
                abs_link = urljoin(start_url, link)
                if abs_link not in seen:
                    seen[abs_link] = depth + 1
                    crawl_queue.append(abs_link)

def simple_scrape_callback(url, html):
    """A simple example of a callback function for scraping content."""
    print(f'Scraped URL: {url}')

# Example of how to use the crawler with a simple scrape callback
link_crawler(
    start_url='http://example.webscraping.com/',
    link_regex=r'/(index|view)',
    delay=1,
    max_depth=1,
    scrape_callback=simple_scrape_callback
)
