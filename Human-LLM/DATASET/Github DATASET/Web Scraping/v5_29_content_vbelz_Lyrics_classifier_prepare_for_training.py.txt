import spacy as spacy_en
import pandas as pd
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer
import pickle
import os

# Load English language model from spaCy
model = spacy_en.load('en_core_web_md')

# Function to clean the text data
def clean_my_text(song):
    """ 
    Cleans the song text by removing punctuation, numbers, and stop words,
    and returns lemmatized words.
    """
    doc = model(song)
    clean_text = ' '.join(word.lemma_ for word in doc if not (word.is_stop or word.pos_ in ['PUNCT', 'NUM']))
    return clean_text

# Function to filter DataFrame to keep only English language songs
def keep_english_for_spacy_nn(df):
    """ 
    Filters the DataFrame to keep only songs with English as the main language,
    for use with the English version of spaCy's neural network for word processing.
    """
    english_songs = df[df['Main Language'] == 'en'].drop(columns=['English Translation Lyrics'])
    return english_songs

# Apply spaCy's neural network to DataFrame for text processing
def apply_spacy_nn_to_DataFrame(df):
    """ 
    Apply word reduction using the clean_my_text function to the 'Text Lyrics' column.
    """
    df['Text Lyrics'] = df['Text Lyrics'].apply(clean_my_text)
    return df

# Save CountVectorizer and TfidfTransformer to disk
def save_transform_to_disk(cv, tf, folder_save):
    """ 
    Save CountVectorizer and TfidfTransformer objects to disk.
    """
    countvectorfile = os.path.join(folder_save, 'countvector.sav')
    pickle.dump(cv, open(countvectorfile, 'wb'))
    Tfidfile = os.path.join(folder_save, 'Tfidfile.sav')
    pickle.dump(tf, open(Tfidfile, 'wb'))

# Prepare data for training
def prepare_training(df_read, folder_save):
    """ 
    Prepare data for training by filtering songs to English language,
    applying text processing, and transforming text data.
    """
    df_prep = keep_english_for_spacy_nn(df_read)
    df_prep = apply_spacy_nn_to_DataFrame(df_prep)
    
    # Convert text data into numerical feature vectors
    cv = CountVectorizer()
    corpus_vec = cv.fit_transform(df_prep['Text Lyrics'])
    
    # Transform count matrix to a normalized term-frequency or term-frequency times inverse document-frequency representation
    tf = TfidfTransformer()
    transform_vec = tf.fit_transform(corpus_vec)
    
    # Save transformations to disk
    save_transform_to_disk(cv, tf, folder_save)
    
    # Convert transformed data into DataFrame
    df_word_vec = pd.DataFrame(transform_vec.todense(), columns=cv.get_feature_names())
    
    # Extract target variable (y) and features (X)
    y = df_prep['Name']
    X = df_word_vec
    return X, y
