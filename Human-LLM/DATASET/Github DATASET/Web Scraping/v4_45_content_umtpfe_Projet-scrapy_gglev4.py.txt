from selenium import webdriver
from selenium.common.exceptions import NoSuchElementException, WebDriverException, TimeoutException
from selenium.webdriver.common.by import By
from selenium.webdriver.support.ui import WebDriverWait
from selenium.webdriver.support import expected_conditions as EC
import time

# Function to handle CAPTCHA
def handle_captcha(driver):
    try:
        # Locate CAPTCHA frame
        frame = driver.find_element(By.XPATH, '//iframe[contains(@src, "recaptcha")]')
        # Switch to the CAPTCHA frame
        driver.switch_to.frame(frame)
        # Click on the CAPTCHA checkbox
        driver.find_element(By.XPATH, "//*[@id='recaptcha-anchor']").click()
        print("CAPTCHA detected. Waiting for 120 seconds for manual solving.")
        # Wait for manual solving
        time.sleep(120)
        # Switch back to default content
        driver.switch_to.default_content()
        return True
    except NoSuchElementException:
        return False

# Set up the WebDriver
driver = webdriver.Firefox()
url = "http://www.google.com/search?q=chanel"

while True:
    try:
        # Open the URL
        driver.get(url)
        val = True
        while val:
            try:
                # Find search results
                results = driver.find_elements(By.XPATH, '//*[@id="rso"]')
                for result in results:
                    print(result.text)
                # Find and click on the next page button
                next_button = driver.find_element(By.XPATH, '//*[@id="pnnext"]')
                next_button.click()
            except NoSuchElementException:
                # If CAPTCHA encountered, handle it
                if handle_captcha(driver):
                    continue
                else:
                    val = False
            except WebDriverException:
                val = False
    except TimeoutException:
        print("Timeout occurred while loading page.")
    except Exception as e:
        print("An error occurred:", e)

# Close the WebDriver
driver.quit()
