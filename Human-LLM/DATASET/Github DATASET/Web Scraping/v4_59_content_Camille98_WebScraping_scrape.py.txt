import requests
from bs4 import BeautifulSoup
from collections import deque
import datetime

# Initialize variables and data structures
visited_urls = set(["http://toscrape.com/"])
queue = deque([
    ["http://toscrape.com/", "", 0],
    ["http://www.fabpedigree.com/james/mathmen.htm", "", 0],
    ["http://scratchpads.eu/explore/sites-list", "", 0]
])
max_depth = 2
index = 0

# Open a file to store index data
with open("index.txt", "w+") as index_file:
    # Loop until the queue is empty
    while queue:
        # Get the next URL to process from the queue
        base_url, path, depth = queue.popleft()
        
        # Check if the current depth is within the limit
        if depth < max_depth:
            try:
                # Fetch the HTML content of the page
                response = requests.get(base_url + path)
                soup = BeautifulSoup(response.text, "html.parser")
                
                # Find all links on the page
                for link in soup.find_all("a"):
                    href = link.get("href")
                    
                    # Check if the link has not been visited before
                    if href not in visited_urls:
                        visited_urls.add(href)
                        
                        # Generate a line for the index file
                        if href.startswith("http"):
                            timestamp = datetime.datetime.now().strftime("%m/%d/%Y")
                            line = f"{index}.html {timestamp} {href}\n"
                            index_file.write(line)
                            index += 1
                            
                        # Add the new URL to the queue for further processing
                        if href.startswith("http"):
                            queue.append([href, "", depth + 1])
                        else:
                            queue.append([base_url, href, depth + 1])
            except Exception as e:
                print(f"Error processing URL: {base_url + path}, {e}")
