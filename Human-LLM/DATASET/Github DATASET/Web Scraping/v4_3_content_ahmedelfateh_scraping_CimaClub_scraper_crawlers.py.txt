import threading
from queue import Queue
from workfiles.createdir import create_project_directory, get_domain_name
from workfiles.linkgetter import file_to_set
from workfiles.crawler import Crawler

# Project details
proj_name = 'CimaClub'
base_url = 'http://cimaclub.com/'

# Determine domain name
domain_name = get_domain_name(base_url)

# File paths
queue_file = proj_name + '/getURL.txt'
scraped_file = proj_name + '/setURL.tex'

# Number of threads
thread_num = 8

# Initialize queue
queue = Queue()

# Create project directory if it doesn't exist
create_project_directory(proj_name)

# Initialize crawler
Crawler(proj_name, base_url, domain_name)

def crawl():
    # Retrieve queued links
    queued_links = file_to_set(queue_file)
    if len(queued_links) > 0:
        print(str(len(queued_links)) + " links in the queue")
        create_jobs()

def create_jobs():
    # Put links into the queue and start crawling
    for link in file_to_set(queue_file):
        queue.put(link)
        queue.join()  # Wait for queue to be empty
        crawl()

def create_crawlers():
    # Create threads for crawling
    for _ in range(thread_num):
        t = threading.Thread(target=work)
        t.daemon = True  # Mark thread as daemon
        t.start()

def work():
    # Execute crawling work
    while True:
        url = queue.get()
        Crawler.crawl_page(threading.current_thread().name, url)
        queue.task_done()

# Create crawler threads
create_crawlers()

# Start crawling
crawl()
