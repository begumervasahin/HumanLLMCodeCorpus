import pandas as pd
import numpy as np
import multiprocessing as mp
from multiprocessing.dummy import Pool as ThreadPool
from requests.exceptions import ConnectionError
from warnings import warn
from time import sleep
from bs4 import BeautifulSoup
import requests

# Read the input Excel file
df1 = pd.read_excel('eunew1.xlsx')

# Define a function to scrape data from a single URL
def scrape_url(url):
    tries = 3
    for i in range(tries):
        try:
            response = requests.get(url, verify=False)
            break
        except ConnectionError as e:
            if i < tries - 1:
                sleep(2)
                continue
            else:
                raise

    if response.status_code != 200:
        warn('Request: {}; Status code: {}'.format(requests, response.status_code))
    
    page_html = BeautifulSoup(response.text, 'html.parser')
    result = [(str(td.text)).splitlines() for td in page_html.find_all("td", "third", limit=8)]
    return result

# Number of threads to use for parallel processing
num_threads = mp.cpu_count()

# Use ThreadPool for parallel processing
pool = ThreadPool(num_threads)

# Extract URLs from DataFrame
urls = df1['url'].tolist()

# Scrape data from URLs in parallel
results = pool.map(scrape_url, urls)

# Close the pool of threads
pool.close()
pool.join()

# Create DataFrame from results
df2 = pd.DataFrame(np.array(results), columns=['all'])

# Merge with original DataFrame
df2 = pd.concat([df1, df2], axis=1)

# Save the results to Excel
df2.to_excel('results_oct.xlsx', index=False)
