# Importing necessary libraries
import plotly.graph_objs as go
import pandas as pd
import plotly.offline as pyo
import dash
import dash_html_components as html
import dash_core_components as dcc

# Creating a Dash web application instance
app = dash.Dash()

# Taking user input for stock symbol
symbol = input('Enter Stock Symbol: \n')

# Setting data type for API query
datatype = 'datatype=csv'

# API Key for Alpha Vantage
API_KEY = 'VW506K51LFXGUT1C'

# Constructing API query URL
API_QUERY = f'https://www.alphavantage.co/query?function=TIME_SERIES_DAILY_ADJUSTED&symbol={symbol}&apikey={API_KEY}&{datatype}'

# Printing API query URL
print(API_QUERY)

# Reading data from the API response into a Pandas DataFrame
df = pd.read_csv(API_QUERY)

# Defining the layout of the Dash application
app.layout = html.Div([
    dcc.Graph(
        id='linegraph',
        figure={
            'data': [
                go.Scatter(
                    x=df['timestamp'],
                    y=df['close'],
                    mode='lines',
                    marker={'size': 12}
                )
            ],
            'layout': go.Layout(title='Stock prices over time')
        }
    )
])

# Running the Dash application
if __name__ == '__main__':
    app.run_server()
