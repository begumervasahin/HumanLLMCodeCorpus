def represents_int(s):
    try:
        int(s)
        return True
    except ValueError:
        return False

def main():
    # Initialize the database helper
    DB = PerfumesDBHelper("perfumes.sqlite")
    # Set up the database
    DB.setup()
    
    # Read the list of perfume URLs from a file
    with open("perfumes_manifest.txt", "r") as f:
        perfume_urls = f.readlines()
    
    # Get the total number of perfumes
    number_of_perfumes = len(perfume_urls)
    # Get the number of records already in the database
    flag = DB.number_of_records()
    
    # Loop through each perfume URL
    for index in range(flag, number_of_perfumes):
        try:
            url = perfume_urls[index][:-1]
            page = requests.get(url)
            tree = html.fromstring(page.content)
            
            # Extract brand and perfume title from the URL
            brand_and_perfume = url[35:-6].split("/")
            brand = brand_and_perfume[0].replace("-", " ")
            perfume_title = " ".join(brand_and_perfume[1].split("-")[:-1])
            
            # Extract perfume image URL
            perfume_image_url = tree.xpath("//meta[@property='og:image']/@content")[0] if tree.xpath("//meta[@property='og:image']/@content") else None
            
            # Download and save perfume image
            if perfume_image_url:
                perfume_image_file_name = perfume_image_url.split("/")[-1]
                if not os.path.exists("fragrantica_images/perfumes"):
                    os.makedirs("fragrantica_images/perfumes")
                request.urlretrieve(perfume_image_url, "fragrantica_images/perfumes/" + perfume_image_file_name)
            else:
                perfume_image_file_name = None
            
            # Create perfume dictionary
            perfume = {"title": perfume_title, "image": perfume_image_file_name}
            
            # Extract launch year from page title
            page_title = tree.xpath("//title/text()")[0] if tree.xpath("//title/text()") else None
            launch_year = page_title[-4:] if page_title and represents_int(page_title[-4:]) else None
            
            # Extract main accords
            main_accords = tree.xpath("//div[@id='prettyPhotoGallery']/div/div/span/text()")
            if main_accords:
                main_accords = [accord for accord in main_accords if accord not in ["main accords", "Videos", "Pictures"]]
            else:
                main_accords = None
            
            # Extract fragrance notes
            notes_captions = tree.xpath("//h3/text()")
            notes = {}
            if "Fragrance Notes" in notes_captions:
                note_tags = tree.xpath("//span[@class='rtgNote']/img")
                notes["general"] = [note_tag.get("title") for note_tag in note_tags]
            elif "Perfume Pyramid" in notes_captions:
                note_categories = ["Top Notes", "Middle Notes", "Base Notes"]
                for category in note_categories:
                    note_tags = tree.xpath(f"//p[b/text()='{category}']/span[@class='rtgNote']/img")
                    notes[category.lower().replace(" ", "_")] = [note.get("title") for note in note_tags] if note_tags else None
            else:
                notes = None
            
            # Extract longevity votes
            longevity_votes = tree.xpath("//table[@class='voteLS long']/tr/td[@class='ndSum']/text()")
            longevity = {"poor": longevity_votes[0], "weak": longevity_votes[1], "moderate": longevity_votes[2],
                         "long lasting": longevity_votes[3], "very long lasting": longevity_votes[4]} if longevity_votes else None
            
            # Extract sillage votes
            sillage_votes = tree.xpath("//table[@class='voteLS sil']/tr/td[@class='ndSum']/text()")
            sillage = {"soft": sillage_votes[0], "moderate": sillage_votes[1], "heavy": sillage_votes[2],
                       "enormous": sillage_votes[3]} if sillage_votes else None
            
            # Print information about the perfume
            print({"count": f"{index}/{number_of_perfumes}", "brand": brand, "perfume": perfume})
            
            # Add the record to the database
            DB.add_record((brand, json.dumps(perfume), launch_year, json.dumps(main_accords), json.dumps(notes),
                           json.dumps(longevity), json.dumps(sillage)))
        
        except error.HTTPError:
            continue

if __name__ == "__main__":
    main()
