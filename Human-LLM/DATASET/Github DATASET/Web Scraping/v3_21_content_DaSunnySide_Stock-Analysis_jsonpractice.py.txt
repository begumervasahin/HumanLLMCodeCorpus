# Import necessary libraries
import plotly.graph_objs as go
import pandas as pd
import dash
import dash_html_components as html
import dash_core_components as dcc

# Initialize Dash app
app = dash.Dash()

def get_stock_data():
    # Prompt user for stock symbol
    symbol = input('Enter the stock symbol: \n')

    # Set API parameters
    datatype = 'datatype=csv'
    API_KEY = 'VW506K51LFXGUT1C'
    API_QUERY = f'https://www.alphavantage.co/query?function=TIME_SERIES_DAILY_ADJUSTED&symbol={symbol}&apikey={API_KEY}&{datatype}'
    print(f'API Query: {API_QUERY}')

    # Retrieve data from Alpha Vantage API and load into DataFrame
    df = pd.read_csv(API_QUERY)
    return df

def create_stock_price_graph(df):
    # Create a line graph displaying stock prices over time
    graph = dcc.Graph(
        id='linegraph',
        figure={
            'data': [
                go.Scatter(
                    x=df['timestamp'],
                    y=df['close'],
                    mode='lines',
                    marker={'size': 12}
                )
            ],
            'layout': go.Layout(
                title='Stock Prices Over Time'
            )
        }
    )
    return graph

# Define Dash layout
app.layout = html.Div([
    # Display the stock price graph
    create_stock_price_graph(get_stock_data())
])

# Run the Dash app
if __name__ == '__main__':
    app.run_server()
