import requests
from bs4 import BeautifulSoup
import os
import traceback

# Function to download images from a given URL
def download(url, filename):
    # Check if the file already exists
    if os.path.exists(filename):
        print('File already exists!')
        return

    try:
        # Fetch the content from the URL
        response = requests.get(url, stream=True, timeout=60)
        response.raise_for_status()

        # Write the content to the file
        with open(filename, 'wb') as file:
            for chunk in response.iter_content(chunk_size=1024):
                if chunk:
                    file.write(chunk)
                    file.flush()
        
        return filename
    
    # Handling KeyboardInterrupt separately
    except KeyboardInterrupt:
        # Remove the file if it was partially downloaded
        if os.path.exists(filename):
            os.remove(filename)
        raise KeyboardInterrupt
    
    # Catching any other exceptions
    except Exception as e:
        traceback.print_exc()
        # Remove the file if an exception occurred
        if os.path.exists(filename):
            os.remove(filename)

# Create a directory to store downloaded images if it doesn't exist
if not os.path.exists('data/anime'):
    os.makedirs('data/anime')

# Define the range of pages to scrape
start_page = 1
end_page = 8000

# Loop through each page
for page_number in range(start_page, end_page + 1):
    # Construct the URL for the page
    url = 'http://konachan.net/post?page=%d&tags=' % page_number
    
    # Fetch the HTML content of the page
    html_content = requests.get(url).text
    
    # Parse the HTML content using BeautifulSoup
    soup = BeautifulSoup(html_content, 'html.parser')
    
    # Extract image URLs from the page and download them
    for image_tag in soup.find_all('img', class_="preview"):
        # Get the source URL of the image
        image_source_url = 'http:' + image_tag['src']
        
        # Generate the filename for the image
        filename = os.path.join('data/anime', image_source_url.split('/')[-1])
        
        # Download the image
        download(image_source_url, filename)
    
    # Print progress
    print('Processed page %d out of %d' % (page_number, end_page))
