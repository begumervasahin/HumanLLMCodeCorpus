import csv
from selenium import webdriver
import time

# Proxy setup for the webdriver
PROXY = "5.58.85.32:43214"
chrome_options = webdriver.ChromeOptions()
chrome_options.add_argument('--proxy-server=%s' % PROXY)
chrome_options.add_argument('--ignore_certificate-errors')
chrome_options.add_argument('--ignore-ssl-errors')

# Constants
MAX_PAGE_NUM = 16
MAX_PAGE_DIG = 3
CSV_FILENAME = "contacts.csv"
PAGE_LOAD_DELAY = 10

def construct_page_url(page_num):
    """
    Construct URL for each page based on the page number.
    """
    return f"https://www.pagesjaunes.fr/annuaire/chercherlespros?quoiqui=avocat%20droit%20du%20travail&ou=paris-75&proximite=0&quoiQuiInterprete=avocat%20droit%20du%20travail&contexte=V0RcB2CgpYFodNUZBe%2BLCQ%3D%3D&idOu=L07505600&page={page_num:03}"

# Open CSV file to write data
with open(CSV_FILENAME, "w") as csv_file:
    # Initialize webdriver
    driver = webdriver.Chrome(options=chrome_options)

    # Iterate through page numbers
    for page_num in range(1, MAX_PAGE_NUM + 1):
        # Format page number with leading zeros
        formatted_page_num = f"{page_num:03}"

        # Construct URL for the page
        url = construct_page_url(formatted_page_num)

        # Open the webpage
        driver.get(url)

        # Pause for page load
        time.sleep(PAGE_LOAD_DELAY)

        # Extract contact elements from the webpage
        contacts = driver.find_elements_by_xpath('//*[@id="bi-contact-06253114000031C0001"]')

        # Write contact information to CSV file
        for contact in contacts:
            csv_file.write(contact.text + ",\n")

    # Close the webdriver
    driver.close()

# Print completion message
print("Finished writing contacts to CSV.")
