import requests
from bs4 import BeautifulSoup

def injury_scrape():
    # Define the URL to scrape
    url = 'https://www.online-betting.me.uk/germany-bundesliga-injuries-and-suspensions.php'

    # Send a GET request to the URL and retrieve the HTML content
    response = requests.get(url)
    
    # Parse the HTML content using BeautifulSoup
    soup = BeautifulSoup(response.content, 'html.parser')

    # Find the table containing injury information
    injury_table = soup.find('div', {'id': 'main-content'})\
                       .find('div', {'class': 'injuries-table-container'})\
                       .find('table', {'class': 'injurytable'})

    # Initialize a list to store injury data
    injury_list = []

    # Loop through each row in the injury table, starting from index 1 to skip the header row
    for row in injury_table.find_all('tr')[1:]:
        # Extract data from each column in the row
        columns = row.find_all('td')
        team = columns[0].text.strip()   # Extract the team name
        player = columns[1].text.strip() # Extract the player name

        # Create a dictionary to store the extracted data and add it to the injury list
        injury_data = {'team': team, 'player': player}
        injury_list.append(injury_data)

    # Return the list of injury data
    return injury_list

# Example usage
injuries = injury_scrape()

# Print each injury record
for injury in injuries:
    print(injury)
