from bs4 import BeautifulSoup as soup
from urllib.request import urlopen as uReq

# Define the URL of the Flipkart page to scrape
my_url = "https://www.flipkart.com/search?q=iphone&otracker=search&otracker1=search&marketplace=FLIPKART&as-show=off&as=off"

# Open the URL and read its content
uClient = uReq(my_url)
page_html = uClient.read()
uClient.close()

# Parse the HTML content using BeautifulSoup
page_soup = soup(page_html, "html.parser")

# Find all containers holding product information
containers = page_soup.findAll("div", {"class": "_3O0U0u"})

# Open a CSV file to write the scraped data
filename = "products.csv"
f = open(filename, "w")

# Write headers to the CSV file
headers = "Product_name,Pricing,Ratings\n"
f.write(headers)

# Loop through each container to extract product details
for container in containers:
    # Extract product name
    product_name = container.div.img["alt"]

    # Extract product price
    price_container = container.findAll("div", {"class": "col col-5-12 _2o7WAb"})
    price = price_container[0].text.strip()
    trim_price = "".join(price.split(","))
    rm_rupee = trim_price.split("₹")
    add_rs_price = "Rs." + rm_rupee[1]
    split_price = add_rs_price.split("E")
    final_price = split_price[0]

    # Extract product rating
    rating_container = container.findAll("div", {"class": "hGSR34"})
    rating = rating_container[0].text
    split_rating = rating.split(" ")
    final_rating = split_rating[0]

    # Write product details to the CSV file
    f.write(product_name.replace(",", "|") + "," + final_price + "," + final_rating + "\n")

    # Print product details to the console
    print(product_name.replace(",", "|") + "," + final_price + "," + final_rating + "\n")

# Close the CSV file
f.close()
