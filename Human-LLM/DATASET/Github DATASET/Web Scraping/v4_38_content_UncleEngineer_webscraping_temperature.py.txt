# Importing necessary libraries
from urllib.request import urlopen as req
from bs4 import BeautifulSoup as soup
import songline

# Defining a function to fetch temperature data for a specific province
def get_temperature(province_id):
    try:
        # Constructing the URL for the province's weather data
        url = 'https://www.tmd.go.th/province.php?id=' + str(province_id)
        
        # Opening the URL and reading the HTML content
        web_open = req(url)
        page_html = web_open.read()
        web_open.close()
        
        # Parsing the HTML content using BeautifulSoup
        data = soup(page_html, 'html.parser')
        
        # Extracting temperature and province name
        temperature = data.findAll('td', {'class': 'strokeme'})[0].text
        province_name = data.findAll('span', {'class': 'title'})[0].text.replace('Â ', '')
        
        # Printing the province name and temperature
        print('Province:', province_name, 'Temperature:', temperature)
        
        # Constructing a message
        message = 'Province: {} Temperature: {}'.format(province_name, temperature)
        
        # Returning the message
        return message
    except Exception as e:
        print('Error occurred:', e)
        return 'No Result'
