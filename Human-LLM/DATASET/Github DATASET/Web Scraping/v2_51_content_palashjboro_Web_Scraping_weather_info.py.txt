# Importing necessary libraries
import pandas as pd
import requests
from bs4 import BeautifulSoup

# Fetching the webpage content
page = requests.get('https://weather.com/en-IN/weather/tenday/l/8ecdb120c02061d6fe17ec5fd60155f9da5bb988e79dd6d6e3db71248dac9b61')

# Parsing the webpage content
soup = BeautifulSoup(page.content, 'html.parser')

# Finding the main section containing weather information
main_section = soup.find(id='main-DailyForecast-1bbda948-59cc-4040-9a36-d9c1ed37a806')

# Extracting various weather components
day_elements = main_section.find_all(class_='date-time')  # Extracting day and time
date_elements = main_section.find_all(class_='day-detail clearfix')  # Extracting date
temp_elements = main_section.find_all('td', class_='temp')  # Extracting temperature
desc_elements = main_section.find_all('td', class_='description')  # Extracting weather description
rain_elements = main_section.find_all('td', class_='precip')  # Extracting precipitation
wind_elements = main_section.find_all('td', class_='wind')  # Extracting wind speed
humi_elements = main_section.find_all('td', class_='humidity')  # Extracting humidity

# Extracting text data from each component
days = [element.get_text() for element in day_elements]
dates = [element.get_text() for element in date_elements]
temperatures = [element.get_text() for element in temp_elements]
descriptions = [element.get_text() for element in desc_elements]
precipitations = [element.get_text() for element in rain_elements]
winds = [element.get_text() for element in wind_elements]
humidities = [element.get_text() for element in humi_elements]

# Creating a pandas DataFrame to store the weather information
weather_info = pd.DataFrame({
    'Day': days,
    'Date': dates,
    'Description': descriptions,
    'Temperature': temperatures,
    'Rain Percip': precipitations,
    'Wind': winds,
    'Humidity': humidities
})

# Printing the weather information
print(weather_info)
