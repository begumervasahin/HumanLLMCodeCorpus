# Import necessary libraries and functions
import ScrapFunction
from urllib.request import urlopen
from bs4 import BeautifulSoup

# Define the URL to scrape
url = "https://www.pea.co.th/"

# Get the title of the webpage
title = ScrapFunction.getTitle(url)

# Check if title is found
if title == None:
    print("Title not found")
else:
    print("***Title below here***")
    print(title)

# Get the body of the webpage
body = ScrapFunction.getBody(url)

# Check if body is found
if body == None:
    print("Body not found")
else:
    print("***Body below here***")
    print(body)

# Get the text content of the webpage
textList = ScrapFunction.getText(url)

# Check if text content is found
if textList == None:
    print("Text not found")
else:
    print("***Text below here***")
    for text in textList:
        print(text.get_text())

# Get the head section of the webpage
head = ScrapFunction.getHead(url)

# Check if head section is found
if head == None:
    print("Head not found")
else:
    print("***Head below here***")
    print(head)

# Get the list of links from the webpage
linkList = ScrapFunction.getLink(url)

# Check if links are found
if linkList == None:
    print("Links not found")
else:
    print("***Links below here***")
    for link in linkList:
        if "href" in link.attrs:
            print(link.attrs["href"])

# Get the list of images from the webpage
imgList = ScrapFunction.getImg(url)

# Check if images are found
if imgList == None:
    print("Images not found")
else:
    print("***Image links below here***")
    for img in imgList:
        print(img.attrs["data-src"])
        print(img.attrs["src"])
