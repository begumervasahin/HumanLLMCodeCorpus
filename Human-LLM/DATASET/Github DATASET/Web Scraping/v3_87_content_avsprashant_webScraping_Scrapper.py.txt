import time
from selenium import webdriver
from selenium.webdriver.support.select import Select

# WebDriver setup
chrome_driver_path = "C:\\Users\\raghu\\Desktop\\Chromedriver\\chromedriver.exe"
url = "http://court.mah.nic.in/courtweb/index_eng.php"
driver = webdriver.Chrome(chrome_driver_path)
driver.get(url)
driver.maximize_window()

# Selecting District
district_dropdown = Select(driver.find_element_by_class_name("selectdistrict"))
district_dropdown.select_by_value("25")

# Navigation
time.sleep(3)  # Wait for page to load
driver.find_element_by_xpath("//*[@id='fourlinks']/ul/li[2]/div[1]/a").click()
time.sleep(5)  # Wait for page to load
driver.find_element_by_xpath("//*[@id='fourlinks']/ul/li[2]/div[2]/ul/li[4]/a").click()
time.sleep(5)  # Wait for page to load

# Switch to iframe
driver.switch_to.frame('data')

# Selecting Court Complex
driver.find_element_by_id("radCourtComplex").click()
court_complex_dropdown = Select(driver.find_element_by_xpath("//*[@id='court_complex_code']"))
court_complex_dropdown.select_by_visible_text("District and Sessions Court, Shivajinagar, Pune - 411 005")

# Setting From Date
driver.find_element_by_id("from_date").click()
month_year_dropdown = Select(driver.find_element_by_class_name("datepick-month-year"))
month_year_dropdown.select_by_value("1/2019")
driver.find_element_by_xpath("/html/body/div[2]/div/div[2]/div/table/tbody/tr[1]/td[3]/a").click()
time.sleep(2)

# Setting To Date
driver.find_element_by_id("to_date").click()
time.sleep(2)
driver.find_element_by_xpath("/html/body/div[2]/div/div[2]/div/div/select[1]/option[1]").click()
time.sleep(2)
driver.find_element_by_xpath("/html/body/div[2]/div/div[2]/div/table/tbody/tr[1]/td[4]/a").click()

print("Please enter CAPTCHA and wait for 10 seconds...")
time.sleep(10)

# Clicking on Submit
driver.find_element_by_xpath("//*[@id='caseNoDet']/div[7]/span[3]/input[1]").click()
time.sleep(10)

# Scraping data
start_row = 2
for page_num in [3, 4, 5, 6]:
    page_element_xpath = f'//*[@id="showList2"]/div[{page_num}]/a'
    page_info = driver.find_element_by_xpath(page_element_xpath).text
    filename, rows = page_info.split(':')[0], int(page_info.split(':')[1][1:])
    
    with open(f"{filename}.txt", "w") as file:
        file.write("(Sr No, Case Type/Case Number/Case Year, Order Date, Order No.)\n")
        for row_num in range(start_row, rows + start_row):
            row_data = []
            for col_num in [1, 2, 3, 4]:
                data_xpath = f'//*[@id="showList1"]/tr[{row_num}]/td[{col_num}]'
                data = driver.find_element_by_xpath(data_xpath).text
                row_data.append(data)
            file.write('\t'.join(row_data) + '\n')
        start_row += 2

# Exiting WebDriver
driver.quit()
print("Scraping Completed.")
