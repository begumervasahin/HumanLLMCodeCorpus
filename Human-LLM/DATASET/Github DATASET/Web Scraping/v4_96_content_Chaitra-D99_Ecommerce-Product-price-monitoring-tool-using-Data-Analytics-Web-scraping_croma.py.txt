from bs4 import BeautifulSoup as soup
from urllib.request import urlopen as uReq
import numpy as np
import datetime
import csv

# URLs to scrape
my_urls = ['https://www.croma.com/phones/c/1',
           'https://www.croma.com/phones/c/1?q=%3Arelevance%3AskuStockFlag%3Atrue&page=2',
           'https://www.croma.com/phones/c/1?q=%3Arelevance%3AskuStockFlag%3Atrue&page=3',
           'https://www.croma.com/phones/c/1?q=%3Arelevance%3AskuStockFlag%3Atrue&page=4',
           'https://www.croma.com/phones/c/1?q=%3Arelevance%3AskuStockFlag%3Atrue&page=5',
           'https://www.croma.com/phones/c/1?q=%3Arelevance%3AskuStockFlag%3Atrue&page=6',
           'https://www.croma.com/phones/c/1?q=%3Arelevance%3AskuStockFlag%3Atrue&page=7',
           'https://www.croma.com/phones/c/1?q=%3Arelevance%3AskuStockFlag%3Atrue&page=8',
           'https://www.croma.com/phones/c/1?q=%3Arelevance%3AskuStockFlag%3Atrue&page=9',
           'https://www.croma.com/phones/c/1?q=%3Arelevance%3AskuStockFlag%3Atrue&page=10']

# Output file names
filename = "croma.csv"
filename1 = "past_data_croma.csv"

# Open CSV files for writing
f = open(filename, "w", encoding='utf-8')
f1 = open(filename1, "a", encoding='utf-8')

# Write headers for the main CSV file
headers = "number,model,rating,offer_price,original_price,price_difference\n"
f.write(headers)

# Initialize counters
count = 1
no_of_pages = 0

# Loop through each URL
for url in my_urls:
    uClient = uReq(url)
    page_html = uClient.read()
    uClient.close()

    # Parse the HTML content
    page_soup = soup(page_html, 'lxml')
    products = page_soup.find_all('li', {'class': 'product__list--item'})
    no_of_pages += 1

    # Loop through each product on the page
    for product in products:
        if count == 181:
            break

        # Extract product details
        title = product.find('a', class_='product__list--name').text.strip().lower().replace(",", "")
        if title.startswith('xiaomi'):
            title = title.replace('xiaomi', 'redmi')

        final_price = product.find('div', class_='col-md-4 col-xs-8').find('span', class_='pdpPrice').text.strip().replace("₹", "").replace(",", "")

        try:
            original_price = product.find('span', class_='pdpPriceMrp').text.strip().replace("₹", "").replace(",", "")
        except AttributeError:
            original_price = final_price

        try:
            rating = len(product.find('div', class_='col-xs-12 col-sm-4 col-md-3').find('div', class_='rating').find_all('span', class_='glyphicon glyphicon-star active'))
        except AttributeError:
            rating = 0

        price_diff = float(original_price) - float(final_price)

        # Write data to the main CSV file
        f.write(f"{count},{title},{rating},{final_price},{original_price},{price_diff}\n")
        count += 1

# Calculate mean price
data = list(csv.reader(open(filename, 'r')))
data.pop(0)  # Remove header
price_diffs = [float(row[5]) for row in data]

mean_price = np.mean(price_diffs)

print('Mean price of Croma:', mean_price)

# Write data to the past data CSV file
now = datetime.datetime.now()
date = now.strftime("%Y-%m-%d")
f1.write(f"{date},{mean_price}\n")

# Close files
f.close()
f1.close()
