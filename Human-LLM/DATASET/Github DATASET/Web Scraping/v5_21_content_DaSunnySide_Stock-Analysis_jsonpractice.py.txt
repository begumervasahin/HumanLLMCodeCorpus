import plotly.graph_objs as go
import pandas as pd
import dash
import dash_html_components as html
import dash_core_components as dcc

# Create a Dash web application instance
app = dash.Dash()

# Prompt user for stock symbol input
symbol = input('Enter Stock Symbol: \n')

# Set data type for API query
datatype = 'datatype=csv'

# Define API Key for Alpha Vantage
API_KEY = 'VW506K51LFXGUT1C'

# Construct API query URL
API_QUERY = f'https://www.alphavantage.co/query?function=TIME_SERIES_DAILY_ADJUSTED&symbol={symbol}&apikey={API_KEY}&{datatype}'

# Print API query URL for reference
print(f"API Query URL: {API_QUERY}")

# Read data from API response into a Pandas DataFrame
df = pd.read_csv(API_QUERY)

# Define the layout of the Dash application
app.layout = html.Div([
    dcc.Graph(
        id='linegraph',
        figure={
            'data': [
                go.Scatter(
                    x=df['timestamp'],
                    y=df['close'],
                    mode='lines',
                    marker={'size': 12}
                )
            ],
            'layout': go.Layout(title='Stock Prices Over Time')
        }
    )
])

# Run the Dash application
if __name__ == '__main__':
    app.run_server()
