from bs4 import BeautifulSoup as soup
from urllib.request import urlopen as uReq

# URL of the webpage to scrape
my_url = "https://www.flipkart.com/search?q=iphone&otracker=search&otracker1=search&marketplace=FLIPKART&as-show=off&as=off"

# Opening connection, grabbing the page
uClient = uReq(my_url)
page_html = uClient.read()
uClient.close()

# Parsing the HTML content
page_soup = soup(page_html, "html.parser")

# Finding all containers that hold product information
containers = page_soup.findAll("div", {"class": "_3O0U0u"})

# Name of the file to write the scraped data
filename = "products.csv"

# Writing data to CSV file
with open(filename, "w", encoding="utf-8") as f:
    # Writing headers to CSV
    headers = "Product_name,Pricing,Ratings\n"
    f.write(headers)

    # Iterating through each container to extract product details
    for container in containers:
        # Extracting product name
        product_name = container.find("div", {"class": "_4rR01T"}).text.strip()

        # Extracting product price
        price_container = container.find("div", {"class": "_30jeq3 _1_WHN1"})
        price = price_container.text.strip() if price_container else "N/A"

        # Extracting product rating
        rating_container = container.find("div", {"class": "_3LWZlK"})
        rating = rating_container.text.strip() if rating_container else "N/A"

        # Writing extracted data to console
        print(product_name.replace(",", "|") + "," + price.replace(",", "") + "," + rating + "\n")

        # Writing extracted data to CSV file
        f.write(product_name.replace(",", "|") + "," + price.replace(",", "") + "," + rating + "\n")
