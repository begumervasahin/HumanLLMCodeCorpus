import xlrd
import xlwt
import pandas as pd
import os

def calculate_grade(marks):
    grade_point = 0
    grade_letter = ''

    if marks >= 90:
        grade_letter += ',S+,'
        grade_point = 10
    elif marks >= 80:
        grade_letter += ',S,'
        grade_point = 9
    elif marks >= 70:
        grade_letter += ',A,'
        grade_point = 8
    elif marks >= 60:
        grade_letter += ',B,'
        grade_point = 7
    elif marks >= 50:
        grade_letter += ',C,'
        grade_point = 6
    elif marks >= 45:
        grade_letter += ',D,'
        grade_point = 5
    elif marks >= 40:
        grade_letter += ',E,'
        grade_point = 4
    elif marks >= 0:
        grade_letter += ',F,'
        grade_point = 0
    elif marks == -1:
        grade_letter += ',Ab,'
        grade_point = 0

    return grade_letter, grade_point

def calculate_gpa(college, year, branch, low, high, semester):
    # Define credit points for different grades
    if int(semester) == 1 or int(semester) == 2:
        credit_4 = 2
        credit_2 = 0
        credit_3 = 3
        credit_1 = 3
    elif int(semester) == 5 or int(semester) == 6:
        credit_4 = 4
        credit_2 = 2
        credit_3 = 2
        credit_1 = 0
    elif int(semester) == 7 or int(semester) == 8:
        credit_4 = 3
        credit_2 = 3
        credit_3 = 2
        credit_1 = 0
    else:
        credit_4 = 6
        credit_2 = 2
        credit_3 = 0
        credit_1 = 0

    # Path to Excel files
    file_path = 'ExcelFiles/'
    
    # Open Excel file
    workbook = xlrd.open_workbook(file_path + '1' + college + year + branch + str(low) + '-' + str(high - 1) + '.xls')
    sheet = workbook.sheet_by_name('Sheet1')
    
    # Lists to store subjects and marks
    subjects, marks_list = [], []

    # Process data from Excel file
    with open('gpa.txt', 'w+') as file:
        for i in range(sheet.nrows):
            record = ''
            record += sheet.cell_value(i, 0) + ',' + sheet.cell_value(i, 1) + ','

            for j in range(5, 41, 5):
                if sheet.cell_value(i, j + 1) == 'P':
                    subjects.append(sheet.cell_value(i, j - 3))
                    marks_list.append(int(sheet.cell_value(i, j)))
                elif sheet.cell_value(i, j + 1) == 'A':
                    subjects.append(sheet.cell_value(i, j - 3))
                    marks_list.append(-1)
                else:
                    subjects.append(sheet.cell_value(i, j - 3))
                    marks_list.append(0)

            record, semester_gpa = calculate_semester_gpa(marks_list, subjects, credit_4, credit_2, credit_3, credit_1, record)

            # Calculate percentage based on GPA
            percentage = str(round((float(semester_gpa) - 0.750) * 10, 2))
            record += semester_gpa + ',' + percentage + ','
            print(record, end='\t')
            print('\n')
            file.write(record + '\n')
    
    # Close the file
    file.close()

    # Write data to a new Excel file
    workbook_new = xlwt.Workbook()
    worksheet = workbook_new.add_sheet('Sheet1')
    file = open('gpa.txt', 'r+')
    data = file.readlines()
    
    # Write data to the new Excel file
    for i in range(len(data)):
        row_data = data[i].split(',')
        for j in range(len(row_data)):
            if row_data[j].replace('.', '', 1).isdigit():
                worksheet.write(i, j, float(row_data[j]))
            else:
                worksheet.write(i, j, row_data[j])

    # Save the new Excel file
    workbook_new.save(file_path + '1' + college + year + branch + str(low) + '-' + str(high - 1) + 'GPA.xls')
    file.close()

    # Create a DataFrame from the GPA data
    columns = pd.read_csv("gpa.txt").columns
    relevant_cols = [0, 1, -3, -2]
    df = pd.read_csv("gpa.txt", sep=",", usecols=columns[relevant_cols])

    # Save the DataFrame to a new CSV file
    df.to_csv("gpar.txt", sep=",", index=False)

    # Read the new CSV file into a DataFrame
    df_rank = pd.read_csv("gpar.txt", sep=",", header=None)
    df_rank.columns = ['USN', 'Name', 'GPA', 'Percentage']

    # Sort DataFrame by GPA
    try:
        df_rank = df_rank.sort_values(by=[df_rank.columns[2]], ascending=False)
    except AttributeError:
        print(" ")

    # Write ranked data to a new Excel file
    writer = pd.ExcelWriter(file_path + '1' + college + year + branch + str(low) + '-' + str(high - 1) + 'RANK.xls')
    df_rank.to_excel(writer, sheet_name='Sheet1', index=False)
    writer.save()

    # Clean up temporary files
    files = ['gpa.txt', 'gpar.txt']
    for file_name in files:
        try:
            os.remove(file_name)
        except:
            pass
