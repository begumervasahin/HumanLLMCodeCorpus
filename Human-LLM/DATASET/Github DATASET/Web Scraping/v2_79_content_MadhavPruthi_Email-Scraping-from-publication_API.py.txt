import re
import urllib.parse
import subprocess
import requests
from bs4 import BeautifulSoup
from crossref.restful import Works
import metainfoscrapper

# Global variables
citation_urls = []
works = Works()

def get_doi_by_title(title):
    # Constructing the search URL
    args = {"q": title}
    search_url = "https://search.crossref.org/?" + urllib.parse.urlencode(args)

    # Fetching search results
    search_page = requests.get(search_url)
    soup = BeautifulSoup(search_page.content, 'html.parser')

    # Extracting DOI links from search results
    doi_links = soup.find_all(href=re.compile("https://doi.org/"))
    if doi_links:
        match = re.search("(?P<url>https?://[^\s]+)", str(doi_links[0]))
        if match:
            return match.group(0)[16:-1]
    return None

def get_citations(title):
    # Building the command to fetch citations using scholar.py
    command = f"python scholar.py -c 1 -A \"{title}\""

    # Executing the command to fetch citations
    try:
        result = subprocess.check_output(command, shell=True)
    except subprocess.CalledProcessError:
        print("Error occurred while fetching results! IP might be blocked.")
        return None

    # Processing citation URLs
    citation_urls.clear()
    decoded_result = result.decode('windows-1252')
    strings = decoded_result.split(" ")
    for string in strings:
        if string.startswith('http://scholar.google.com/scholar?cites'):
            citation_urls.append(string.strip())

    # Scraping citation details
    main_url = citation_urls[0] if citation_urls else None
    i = 0
    while main_url:
        url_parts = main_url.split('?')
        url = f"{url_parts[0]}?start={i}&{url_parts[1]}"
        page = requests.get(url)
        soup = BeautifulSoup(page.content, 'html.parser')
        citation_list = soup.find_all('h3', class_='gs_rt')
        if not citation_list:
            break
        for paper in citation_list:
            name = paper.get_text().split(' ', 1)[1] if paper.get_text()[0] == '[' else paper.get_text()
            _doi = get_doi_by_title(name)
            if _doi:
                email = metainfoscrapper.getEmail(_doi)
        i += 10

def main_search(doi):
    # Fetching citations for a given DOI
    title = works.doi(doi)['title']
    get_citations(title)
