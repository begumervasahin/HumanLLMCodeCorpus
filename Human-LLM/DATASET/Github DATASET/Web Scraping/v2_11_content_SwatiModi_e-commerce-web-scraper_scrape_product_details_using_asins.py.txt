import pandas as pd
from bs4 import BeautifulSoup
import requests

# Standard headers for HTTP requests
headers_std = {
    'User-Agent': 'Mozilla/5.0 (Windows NT 6.1; WOW64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/46.0.2490.80 Safari/537.36',
    'Content-Type': 'text/html',
}

# Function to extract product details from Amazon
def get_product_details(asin_list):
    # Initialize lists to store product details
    product_names = []
    brand_names = []
    ratings = []
    no_of_ratings = []
    product_images = []
    product_prices = []

    # Loop through each ASIN
    for asin in asin_list:
        # Construct URL for the product
        url = "https://www.amazon.in/dp/" + str(asin)
        
        # Fetch HTML content of the product page
        html = requests.get(url, headers=headers_std).text
        soup = BeautifulSoup(html, 'lxml')
        
        # Extract relevant elements from the HTML
        image_list = soup.find_all('span', {'class': 'a-button-text'})
        brand_name = soup.find('a', {'id': 'bylineInfo'})
        product_name = soup.find('span', {'id': 'productTitle'})
        rating = soup.find('span', {'class': 'a-icon-alt'})
        no_of_rat = soup.find('span', {'id': 'acrCustomerReviewText'})
        sell_price_range = soup.find('span', {'id': 'priceblock_ourprice'})

        # Extract data from the elements
        image_urls = [image.find('img').get('src') for image in image_list if image.find('img')]
        product_images.append(str(image_urls))
        brand_names.append(brand_name.text.strip() if brand_name else '')
        product_names.append(product_name.text.strip() if product_name else '')
        ratings.append(rating.text.strip() if rating else '')
        no_of_ratings.append(no_of_rat.text.strip() if no_of_rat else '')
        product_prices.append(sell_price_range.text.strip() if sell_price_range else '')

    # Create a DataFrame to store the collected data
    df = pd.DataFrame({
        'product_name': product_names,
        'brand_name': brand_names,
        'images': product_images,
        'price': product_prices,
        'product_rating': ratings,
        'no. of ratings': no_of_ratings
    })
    
    return df

if __name__ == "__main__":
    # Read ASINs from a CSV file
    asins_df = pd.read_csv('ASINs.csv')
    asin_list = asins_df['ASIN']
    
    # Extract product details and save them to a CSV file
    df = get_product_details(asin_list)
    df.to_csv('ASIN_product_details.csv', index=False)
