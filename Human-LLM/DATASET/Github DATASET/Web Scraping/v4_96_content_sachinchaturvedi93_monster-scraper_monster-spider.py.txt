import scrapy
import json
from pprint import pprint
from bs4 import BeautifulSoup
from bs4.element import Comment
import urllib.request

# Function to check if a tag is visible
def tag_visible(element):
    # Check if the parent tag is in the list of non-visible tags
    if element.parent.name in ['style', 'script', 'head', 'title', 'meta', '[document]']:
        return False
    # Check if the element is a comment
    if isinstance(element, Comment):
        return False
    return True

# Function to extract visible text from HTML
def text_from_html(body):
    soup = BeautifulSoup(body, 'html.parser')
    # Find all text elements
    texts = soup.findAll(text=True)
    # Filter out non-visible text
    visible_texts = filter(tag_visible, texts)
    # Join visible text elements into a single string
    return u" ".join(t.strip() for t in visible_texts)

# Define a Spider class for scraping Monster job listings
class MonsterSpiderSpider(scrapy.Spider):
    # Name of the Spider
    name = 'monster-spider'
    # Domains to be scraped
    allowed_domains = ['monster.com']
    # URLs to start scraping from
    start_urls = [
        'https://www.monster.com/jobs/search/pagination/'
        '?q=Product-Manager&where=USA&isDynamicPage=true&'
        'isMKPagination=true&page={}'.format(i + 1) for i in range(2)
    ]

    # Parse function to extract data from the response
    def parse(self, response):
        # Load the JSON response
        results = json.loads(response.body)
        # Loop through each job listing
        for result in results:
            try:
                # Extract job ID
                job_id = result['MusangKingId']
                # Construct URL for detailed job information
                next_url = ('https://job-openings.monster.com/v2/job/pure-json-view?jobid={}').format(job_id)
                # Follow the URL and parse detailed information
                yield response.follow(next_url, callback=self.parse_detail)
            except:
                continue

    # Parse function for detailed job information
    def parse_detail(self, response):
        # Load the JSON response
        result = json.loads(response.body)
        # Initialize a dictionary to store job details
        detail = {}
        # Extract job ID
        detail["JobID"] = result['jobId']
        # Extract job title
        detail["Title"] = result['companyInfo']['companyHeader']
        # Extract job location
        detail["Location"] = result['companyInfo']['jobLocation']
        try:
            # Extract company name
            detail["Company"] = result['companyInfo']['name']
        except:
            detail["Company"] = ''
        # Extract job category
        detail['Job Category'] = result["jobCategory"]
        # Extract job description (cleaned from HTML)
        detail["Description"] = text_from_html(result["jobDescription"])
        try:
            # Extract company size
            detail["Company Size"] = result['companyInfo']["companySizeName"]
        except:
            detail["Company Size"] = ''
        # Extract additional job information
        info = result['summary']['info']
        for i in info:
            detail[i['title']] = i['text']
        return detail
