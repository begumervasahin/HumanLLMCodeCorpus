# Import necessary libraries
from selenium import webdriver
from selenium.webdriver.common.keys import Keys
import time
from bs4 import BeautifulSoup
import pandas as pd

# Function to prompt user for LinkedIn credentials
def get_user_input():
    company_id = input("Enter Company ID: ")
    username = input("Enter username: ")
    password = input("Enter password: ")
    return company_id, username, password

# Function to initialize web driver and login to LinkedIn
def login_to_linkedin(username, password):
    browser = webdriver.Firefox()
    browser.get('https://www.linkedin.com')
    time.sleep(3)  # Pause for page loading
    email_field = browser.find_element_by_name('session_key')
    password_field = browser.find_element_by_name('session_password')
    email_field.send_keys(username + Keys.RETURN)
    password_field.send_keys(password + Keys.RETURN)
    time.sleep(3)  # Pause for login process
    return browser

# Function to construct search URL for employees of specified company
def build_search_url(company_id):
    return f"https://www.linkedin.com/search/results/people/?facetCurrentCompany=%5B%22{company_id}%22%5D"

# Function to scroll to the bottom of the page
def scroll_to_bottom(browser):
    browser.execute_script("window.scrollTo(0, document.body.scrollHeight);")
    time.sleep(3)  # Pause for page loading

# Function to extract employee data from the current page
def extract_employee_data(page_content):
    names = [name.text for name in page_content.find_all('span', class_='actor-name')]
    titles = [title.text.replace('\n', '') for title in page_content.find_all('p', class_='subline-level-1')]
    locations = [location.text.replace('\n', '') for location in page_content.find_all('p', class_='subline-level-2')]
    profiles = ['https://www.linkedin.com' + profile['href'] for profile in page_content.find_all('a', class_='search-result__result-link')][::2]
    return names, titles, locations, profiles

# Function to scrape LinkedIn search results
def scrape_linkedin(browser, search_url, max_employees):
    df = pd.DataFrame(columns=['name', 'title', 'location', 'profile'])
    current_url = 'url_placeholder'
    while True:
        if current_url.find('page=100') != -1:
            break
        previous_url = current_url
        current_url = browser.current_url
        if current_url == previous_url:
            break
        page = BeautifulSoup(browser.page_source, 'lxml')
        names, titles, locations, profiles = extract_employee_data(page)
        temp_df = pd.DataFrame({'name': names, 'title': titles, 'location': locations, 'profile': profiles})
        temp_df = temp_df[temp_df['name'] != 'LinkedIn Member']  # Exclude LinkedIn Member entries
        df = df.append(temp_df)
        if df.shape[0] >= max_employees:
            break
        next_button = browser.find_element_by_class_name('next')
        next_button.click()
        time.sleep(5)  # Pause for page loading
    df.reset_index(drop=True, inplace=True)
    return df

# Function to save DataFrame to CSV file
def save_to_csv(df):
    df.to_csv("output_search.csv", index=False)
