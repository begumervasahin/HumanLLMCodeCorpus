import tkinter as tk
from tkinter import messagebox, ttk
import requests
from bs4 import BeautifulSoup
import re

def fetch_movie_data():
    url = "http://www.imdb.com/chart/top"
    headers = {'Accept-Language': 'en-US,en;'}
    response = requests.get(url, headers=headers)
    html_content = response.content
    soup = BeautifulSoup(html_content, "html.parser")
    movie_names = soup.find_all("td", {"class": "titleColumn"})
    ratings = soup.find_all("td", {"class", "ratingColumn imdbRating"})
    all_movies = {}
    for name, rating in zip(movie_names, ratings):
        name_text = name.get_text(strip=True)
        rating_text = rating.get_text(strip=True)
        movie_details = re.search(r'(\d+)\.\s+(.*?)\s+\((\d{4})\)', name_text)
        if movie_details:
            all_movies[movie_details.group(2)] = [movie_details.group(3), rating_text]
    return all_movies

class MovieApp(tk.Tk):
    def __init__(self):
        super().__init__()
        self.title("IMDb Movies and Ratings Application")
        self.geometry("700x350")
        # self.iconbitmap('icon1.ico')  # Uncomment and set the path to your icon if you have one

        self.all_movies = fetch_movie_data()

        self.create_widgets()
        self.create_menu()

    def create_widgets(self):
        self.main_label = tk.Label(self, text="Welcome to IMDb Movies and Ratings Application!")
        self.main_label.pack(pady=20)

        self.search_label = tk.Label(self, text="Enter movie name to find:")
        self.search_entry = tk.Entry(self, width=30)
        self.search_button = tk.Button(self, text="Find Movie", command=self.find_movie)

        self.tree_view = ttk.Treeview(self, columns=("Year", "Rating"), show="headings")
        self.tree_view.heading("Year", text="Year")
        self.tree_view.heading("Rating", text="IMDb Rating")

        self.about_label = tk.Label(self, text="This application is developed for a course project.\n"
                                               "Contact: example@example.com\nVersion 1.0")

    def create_menu(self):
        menubar = tk.Menu(self)
        menubar.add_command(label="Get All Movies", command=self.display_all_movies)
        menubar.add_command(label="Find Movie by Name", command=self.display_movie_search)
        menubar.add_command(label="About", command=self.display_about)
        menubar.add_command(label="Exit", command=self.destroy)
        self.config(menu=menubar)

    def clear_screen(self):
        for widget in self.winfo_children():
            widget.pack_forget()

    def display_all_movies(self):
        self.clear_screen()
        self.main_label.pack(pady=20)
        for name, details in self.all_movies.items():
            self.tree_view.insert('', 'end', values=(details[0], details[1]))
        self.tree_view.pack(pady=20)

    def display_movie_search(self):
        self.clear_screen()
        self.search_label.pack()
        self.search_entry.pack()
        self.search_button.pack(pady=20)

    def find_movie(self):
        movie_name = self.search_entry.get()
        if movie_name in self.all_movies:
            year, rating = self.all_movies[movie_name]
            messagebox.showinfo("Found Successfully", f"Movie name: {movie_name}\nYear: {year}\nIMDb Rating: {rating}")
        else:
            messagebox.showerror("Error", "Movie not found.")

    def display_about(self):
        self.clear_screen()
        self.about_label.pack(pady=20)

if __name__ == "__main__":
    app = MovieApp()
    app.mainloop()
