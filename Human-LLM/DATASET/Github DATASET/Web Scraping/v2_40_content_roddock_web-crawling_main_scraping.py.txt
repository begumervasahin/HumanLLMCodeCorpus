import requests
from bs4 import BeautifulSoup

# Function to get the title of a web page
def getTitle(url):
    try:
        # Fetch the web page
        response = requests.get(url)
        # Parse HTML content
        soup = BeautifulSoup(response.text, 'html.parser')
        # Extract title
        title = soup.title.string
        return title
    except Exception as e:
        print("Error:", e)
        return None

# Function to get the body of a web page
def getBody(url):
    try:
        # Fetch the web page
        response = requests.get(url)
        # Parse HTML content
        soup = BeautifulSoup(response.text, 'html.parser')
        # Extract body
        body = soup.body
        return body
    except Exception as e:
        print("Error:", e)
        return None

# Function to get all text paragraphs from a web page
def getText(url):
    try:
        # Fetch the web page
        response = requests.get(url)
        # Parse HTML content
        soup = BeautifulSoup(response.text, 'html.parser')
        # Find all <p> tags (paragraphs)
        textList = soup.find_all('p')
        return textList
    except Exception as e:
        print("Error:", e)
        return None

# Function to get the head section of a web page
def getHead(url):
    try:
        # Fetch the web page
        response = requests.get(url)
        # Parse HTML content
        soup = BeautifulSoup(response.text, 'html.parser')
        # Extract head section
        head = soup.head
        return head
    except Exception as e:
        print("Error:", e)
        return None

# Function to get all links from a web page
def getLink(url):
    try:
        # Fetch the web page
        response = requests.get(url)
        # Parse HTML content
        soup = BeautifulSoup(response.text, 'html.parser')
        # Find all <a> tags (links)
        linkList = soup.find_all('a')
        return linkList
    except Exception as e:
        print("Error:", e)
        return None

# Function to get all images from a web page
def getImg(url):
    try:
        # Fetch the web page
        response = requests.get(url)
        # Parse HTML content
        soup = BeautifulSoup(response.text, 'html.parser')
        # Find all <img> tags (images)
        imgList = soup.find_all('img')
        return imgList
    except Exception as e:
        print("Error:", e)
        return None
