import requests
from bs4 import BeautifulSoup
from collections import deque
import datetime

# Set to keep track of visited URLs
visited = set()

# Queue to store URLs to be crawled
queue = deque([
    ("http://toscrape.com/", "", 0),
    ("http://www.fabpedigree.com/james/mathmen.htm", "", 0),
    ("http://scratchpads.eu/explore/sites-list", "", 0)
])

# Maximum depth to crawl
max_depth = 2

# Counter for generating file names
file_counter = 0

# Open index.txt file for writing
with open("index.txt", "w+") as urlfile:
    # Loop until the queue is empty
    while queue:
        # Dequeue URL to crawl
        base, path, depth = queue.popleft()

        # Check if the depth is within the limit
        if depth < max_depth:
            try:
                # Send HTTP GET request to fetch page content
                response = requests.get(base + path)

                # Check if the request was successful
                if response.status_code == 200:
                    # Parse HTML content using BeautifulSoup
                    soup = BeautifulSoup(response.text, "html.parser")

                    # Find all anchor tags with href attribute
                    for link in soup.find_all("a", href=True):
                        href = link["href"]

                        # Check if the URL has not been visited before
                        if href not in visited:
                            # Mark URL as visited
                            visited.add(href)

                            # Get current timestamp
                            timestamp = datetime.datetime.now().strftime("%m/%d/%Y %H:%M:%S")

                            # Construct line to write to index.txt
                            line = f"{file_counter}.html {timestamp} {href}\n"

                            # Write line to index.txt
                            urlfile.write(line)

                            # Increment file counter
                            file_counter += 1

                            # Enqueue new URLs for crawling
                            if href.startswith("http"):
                                queue.append((href, "", depth + 1))
                            else:
                                queue.append((base, href, depth + 1))
            except Exception as e:
                # Print error message if an exception occurs during crawling
                print(f"Error fetching {base + path}: {e}")

# Print message indicating crawling is completed
print("Crawling completed.")
