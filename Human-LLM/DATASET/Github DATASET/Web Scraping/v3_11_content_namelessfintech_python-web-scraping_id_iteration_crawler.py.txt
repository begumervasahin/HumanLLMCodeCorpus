import itertools
import urllib.request
from urllib.error import URLError, HTTPError, ContentTooShortError

def download(url, num_retries=2, user_agent='wswp', charset='utf-8'):
    """Download HTML content from a given URL."""
    print('Downloading:', url)

    # Set up request with user agent
    request = urllib.request.Request(url)
    request.add_header('User-agent', user_agent)

    try:
        # Open URL and read response
        with urllib.request.urlopen(request) as response:
            # Decode response content using character set
            charset = response.headers.get_content_charset() or charset
            html = response.read().decode(charset)

    # Handle various download errors
    except (URLError, HTTPError, ContentTooShortError) as e:
        print('Download error:', e.reason)
        html = None

        # Retry download if retries left and error is in 500 range
        if num_retries > 0 and hasattr(e, 'code') and 500 <= e.code < 600:
            return download(url, num_retries - 1)

    return html

def crawl_site(url, max_errors=5):
    """Crawl a website starting from a given URL."""
    num_errors = 0

    # Iterate over page numbers
    for page in itertools.count(1):
        page_url = f'{url}{page}'

        # Download HTML content of current page
        html = download(page_url)

        # Handle download errors
        if html is None:
            num_errors += 1
            if num_errors == max_errors:
                break
        else:
            num_errors = 0

# Starting the crawling process from a given URL
crawl_site('http://example.webscraping.com/view/-')
