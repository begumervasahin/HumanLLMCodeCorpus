import spacy
import pandas as pd
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer
import pickle
import os

# Load English model from spaCy
nlp = spacy.load('en_core_web_md')

def clean_my_text(song):
    """Filter punctuation, numbers, stop words, and return lemmatized words"""
    doc = nlp(song)
    clean_text = ''
    for word in doc:
        if (not word.is_stop) and (word.pos_ != 'PUNCT') and (word.pos_ != 'NUM'):
            word = word.lemma_
            clean_text += word + ' '
    return clean_text.strip()

def keep_english_for_spacy_nn(df):
    """Keep songs with English as the main language"""
    df = df[df['Main Language'] == 'en']
    df.drop(['English Translation Lyrics'], axis=1, inplace=True)
    return df

def apply_spacy_nn_to_DataFrame(df):
    """Apply reduction of words using clean_my_text Function to the lyrics column"""
    df['Text Lyrics'] = df['Text Lyrics'].apply(clean_my_text)
    return df

def save_transform_to_disk(cv, tf, folder_save):
    """Save CountVectorizer and TfidfTransformer to disk"""
    countvectorfile = os.path.join(folder_save, 'countvector.sav')
    pickle.dump(cv, open(countvectorfile, 'wb'))
    Tfidfile = os.path.join(folder_save, 'Tfidfile.sav')
    pickle.dump(tf, open(Tfidfile, 'wb'))

def prepare_training(df_read, folder_save):
    """Prepare data for training"""
    df_prep = keep_english_for_spacy_nn(df_read)
    df_prep = apply_spacy_nn_to_DataFrame(df_prep)
    
    # Count Vectorization
    cv = CountVectorizer()
    corpus_vec = cv.fit_transform(df_prep['Text Lyrics'])
    
    # TF-IDF Transformation
    tf = TfidfTransformer()
    transform_vec = tf.fit_transform(corpus_vec)
    
    # Save transformations to disk
    save_transform_to_disk(cv, tf, folder_save)
    
    # Convert transformed vectors to DataFrame
    df_word_vec = pd.DataFrame(transform_vec.toarray(), columns=cv.get_feature_names())
    
    X = df_word_vec
    y = df_prep['Name']
    
    return X, y
