import time
import pandas as pd
from selenium import webdriver
from selenium.webdriver.common.keys import Keys
from helpers import commands
from links import SCHOOL, CONST_EMAILS, BLOGS, SOCIAL_MEDIA

class Websites:
    def __init__(self, chromedriver_path="/Users/ramanshsharma/Downloads/chromedriver"):
        self.url = 'https://github.com/ramanshsharma2806'
        self.chromedriver_path = chromedriver_path
        self.chrome_options = webdriver.ChromeOptions()
        self.chrome_options.add_experimental_option("detach", True)
        self.file = pd.read_csv('secrets.csv')
        self.driver = None
        self.media_to_func = {
            'twitter': self.get_twitter,
            'instagram': self.get_instagram,
            'github': self.get_github,
            'linkedin': self.get_linkedin,
            'kaggle': self.get_kaggle,
            'mail': self.get_mail,
            'mail0': self.get_mail,
            'mail1': self.get_mail,
            'mail2': self.get_mail,
            'xkcd': self.get_xkcd,
            'blog': self.get_xkcd,
            'school': self.get_school,
            'sciencesurvey': self.get_school
        }

    def sign_in(self, url=None):
        if url:
            self.url = url
        media = link_to_media(self.url)
        self.driver = webdriver.Chrome(self.chromedriver_path, chrome_options=self.chrome_options)
        if media.startswith('mail'):
            self.media_to_func['mail'](int(media[-1]))
        elif media in ['school', 'sciencesurvey', 'xkcd', 'blog']:
            self.media_to_func[media](spec=media)
        else:
            self.media_to_func[media]()

    def get_school(self, spec=None):
        self.url = SCHOOL[0] if not spec or spec == 'school' else SCHOOL[1]
        self.driver.get(self.url)

    def get_linkedin(self):
        self.driver.get(self.url)
        time.sleep(2)
        username_field = self.driver.find_element_by_id('login-email')
        username_field.send_keys(self.file.iloc[4, 1])
        time.sleep(2)
        password_field = self.driver.find_element_by_id('login-password')
        time.sleep(2)
        password_field.send_keys(self.file.iloc[4, 2])
        time.sleep(2)
        password_field.send_keys(Keys.RETURN)

    def get_twitter(self):
        self.driver.get(self.url)
        time.sleep(2)
        login_button = self.driver.find_element_by_link_text('Log in')
        login_button.click()
        time.sleep(1)
        username_field = self.driver.find_element_by_class_name('email-input')
        username_field.send_keys(self.file.iloc[0, 1])
        time.sleep(2)
        password_field = self.driver.find_element_by_name('session[password]')
        password_field.send_keys(self.file.iloc[0, 2])
        time.sleep(2)
        password_field.send_keys(Keys.RETURN)

    def get_instagram(self):
        self.driver.get(self.url)
        time.sleep(2)
        login_link = self.driver.find_element_by_link_text('Log in')
        login_link.click()
        time.sleep(2)
        email_field = self.driver.find_element_by_class_name('_2hvTZ')
        email_field.send_keys(self.file.iloc[2, 1])
        time.sleep(2)
        password_field = self.driver.find_elements_by_class_name('_2hvTZ')[1]
        password_field.send_keys(self.file.iloc[2, 2])
        time.sleep(2)
        password_field.send_keys(Keys.RETURN)

    def get_github(self):
        self.driver.get(self.url)
        time.sleep(2)
        sign_in_button = self.driver.find_element_by_link_text('Sign in')
        sign_in_button.click()
        time.sleep(2)
        username_field = self.driver.find_element_by_id('login_field')
        time.sleep(2)
        username_field.send_keys(self.file.iloc[1, 1])
        time.sleep(2)
        password_field = self.driver.find_element_by_id('password')
        time.sleep(2)
        password_field.send_keys(self.file.iloc[1, 2])
        time.sleep(2)
        password_field.send_keys(Keys.RETURN)

    def get_kaggle(self):
        self.driver.get(self.url)
        time.sleep(2)
        login_button = self.driver.find_element_by_class_name('button--small')
        login_button.click()
        username_field = self.driver.find_element_by_id('username-input-text')
        username_field.send_keys(self.file.iloc[4, 1])
        time.sleep(2)
        password_field = self.driver.find_element_by_id('password-input-text')
        password_field.send_keys(self.file.iloc[4, 2])
        time.sleep(2)
        sign_in_button = self.driver.find_element_by_link_text('Sign in')
        sign_in_button.click()

    def get_mail(self, num=None):
        if num is None:
            for i in range(3):
                self.url = CONST_EMAILS[i]
                email, password = self.file.iloc[i + 5, 1], self.file.iloc[i + 5, 2]
                mail_action(self.driver, self.url, email, password)
        else:
            self.url = CONST_EMAILS[num]
            email = self.file.iloc[num + 5, 1]
            password = self.file.iloc[num + 5, 2]
            mail_action(self.driver, self.url, email, password)

    def get_xkcd(self, spec):
        self.url = BLOGS[0] if spec == 'xkcd' else BLOGS[1]
        self.driver.get(self.url)

def link_to_media(link):
    return [item[0] for item in commands.items() if link == item[1]][0]

def mail_action(driver, url, email, password):
    driver.get(url)
    time.sleep(2)
    email_input = driver.find_element_by_id('identifierId')
    email_input.send_keys(email)
    email_input.send_keys(Keys.RETURN)
    time.sleep(2)
    password_input = driver.find_element_by_class_name('whsOnd')
    password_input.send_keys(password)
    time.sleep(2)
    password_input.send_keys(Keys.RETURN)

if __name__ == '__main__':
    website = Websites()
    website.sign_in(SOCIAL_MEDIA[0])
