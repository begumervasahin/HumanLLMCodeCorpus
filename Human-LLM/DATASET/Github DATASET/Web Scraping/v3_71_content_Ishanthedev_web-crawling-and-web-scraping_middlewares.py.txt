import scrapy
from scrapy import signals

class QuoteTutorialSpiderMiddleware:
    """
    Middleware for processing various spider-level events and responses.
    """
    
    @classmethod
    def from_crawler(cls, crawler):
        """
        Initializes the middleware with the crawler instance.
        """
        instance = cls()
        crawler.signals.connect(instance.on_spider_opened, signal=signals.spider_opened)
        return instance

    def process_spider_input(self, response, spider):
        """
        Processes the input response for a spider.
        """
        # Currently not implemented, can be used for preprocessing responses.
        return None

    def process_spider_output(self, response, result, spider):
        """
        Processes the output results from the spider.
        """
        for item in result:
            yield item

    def process_spider_exception(self, response, exception, spider):
        """
        Handles exceptions that occur during the spider's processing.
        """
        # Exception handling can be implemented here.
        pass

    def process_start_requests(self, start_requests, spider):
        """
        Processes the starting requests of a spider.
        """
        for request in start_requests:
            yield request

    def on_spider_opened(self, spider):
        """
        Actions to perform when the spider opens.
        """
        spider.logger.info(f'Spider opened: {spider.name}')
