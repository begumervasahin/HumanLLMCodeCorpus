import sys
from tkinter import *
from tkinter import simpledialog

class PlotParam(simpledialog.Dialog):
    def body(self, master):
        self.title('Enter new plot parameters...')
        Label(master, text='Min X:').grid(row=0)
        Label(master, text='Max X:').grid(row=1)
        Label(master, text='Bin width:').grid(row=2)
        Label(master, text='Max Y:').grid(row=3)
        
        entry_min_x = Entry(master, justify=CENTER)
        entry_min_x.grid(row=0, column=1)
        entry_min_x.insert(END, '0')
        
        entry_max_x = Entry(master, justify=CENTER)
        entry_max_x.grid(row=1, column=1)
        entry_max_x.insert(END, '10')
        
        entry_bin_width = Entry(master, justify=CENTER)
        entry_bin_width.grid(row=2, column=1)
        entry_bin_width.insert(END, '1')
        
        entry_max_y = Entry(master, justify=CENTER)
        entry_max_y.grid(row=3, column=1)
        entry_max_y.insert(END, '1000')
        
        return entry_min_x
    
    def apply(self):
        self.x_min = int(self.entry_min_x.get())
        self.x_max = int(self.entry_max_x.get())
        self.bin_width = int(self.entry_bin_width.get())
        self.y_max = int(self.entry_max_y.get())

# Example usage:
if __name__ == "__main__":
    root = Tk()
    dialog = PlotParam(root)
    print("Min X:", dialog.x_min)
    print("Max X:", dialog.x_max)
    print("Bin width:", dialog.bin_width)
    print("Max Y:", dialog.y_max)
    root.mainloop()
