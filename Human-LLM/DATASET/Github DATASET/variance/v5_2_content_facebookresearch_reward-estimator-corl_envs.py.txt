# Function to create an environment with specific configurations
def make_env(env_id, seed, rank, log_dir):
    def _thunk():
        # Create the specified Gym environment
        env = gym.make(env_id)
        
        # Check if the environment is an Atari game
        is_atari = hasattr(gym.envs, 'atari') and isinstance(env.unwrapped, gym.envs.atari.atari_env.AtariEnv)
        
        # Preprocess Atari games
        if is_atari:
            env = make_atari(env_id)
        
        # Set seed for reproducibility
        env.seed(seed + rank)
        
        # Optionally log performance
        if log_dir is not None:
            env = bench.Monitor(env, os.path.join(log_dir, str(rank)))
        
        # Wrap Atari games for DeepMind style
        if is_atari:
            env = wrap_deepmind(env)
        
        # Adjust observation space for PyTorch compatibility
        if len(env.observation_space.shape) == 3 and env.observation_space.shape[2] in [1, 3]:
            env = WrapPyTorch(env)
        
        return env
    return _thunk

# Custom observation wrapper for PyTorch compatibility
class WrapPyTorch(gym.ObservationWrapper):
    def __init__(self, env=None):
        super(WrapPyTorch, self).__init__(env)
        obs_shape = self.observation_space.shape
        self.observation_space = Box(
            low=self.observation_space.low[0,0,0],
            high=self.observation_space.high[0,0,0],
            shape=[obs_shape[2], obs_shape[1], obs_shape[0]]
        )

    def observation(self, observation):
        # Transpose observation for PyTorch compatibility
        return observation.transpose(2, 0, 1)
