import argparse
import matplotlib.gridspec as gridspec
import matplotlib.pyplot as plt
import numpy as np
import pandas as pd

# Set initial debugging flag to False
debug = False

def main():
    # Parse command-line arguments
    args = parse_args()

    # Inform user about the process
    print("Plotting heritabilities from", args.infile)

    # Read data from input file
    data = pd.read_csv(args.infile, sep='\t')

    # Sort data columns by 'actual' values in descending order
    order = np.argsort(data.loc['actual', :])[::-1]
    data = data.iloc[:, order]

    # Create a figure for plotting
    fig = plt.figure(figsize=(5 + .25 * len(data.columns), 5))
    
    # Define grid for subplot arrangement
    grid = gridspec.GridSpec(nrows=100, ncols=100)
    
    # Add subplot to the figure
    ax = fig.add_subplot(grid[:80, :], title="Distributions of null heritabilities", xlabel="trait", ylabel="Heritability")

    # Identify which rows are permutations
    perms = data.index != "actual"

    # Prepare x-axis ticks and labels
    xticks, xlabels = list(), list()
    for x in range(len(data.columns)):
        trait = data.columns[x]
        xticks.append(x)
        # Plot violin plot for each trait
        ax.violinplot(data.loc[perms, trait], positions=[x])
        xlabels.append(trait.replace("trait_", ""))

    # Plot actual heritability values
    ax.scatter(xticks, data.loc['actual', :])

    # Set x-axis ticks and labels
    ax.set_xticks(xticks)
    ax.set_xticklabels(xlabels, rotation="vertical")

    # Save the figure to the output file
    fig.savefig(args.outfile, dpi=100)

def parse_args():
    # Create ArgumentParser object
    parser = argparse.ArgumentParser()

    # Define command-line arguments
    parser.add_argument("-i", "--infile")   # Input file
    parser.add_argument("-o", "--outfile")  # Output file
    parser.add_argument("--debug", default=False, action="store_true")  # Debugging flag

    # Parse the arguments
    args = parser.parse_args()

    # Update global debug flag
    global debug
    debug = args.debug

    return parser.parse_args()

# Execute main function if the script is run directly
if __name__ == '__main__':
    main()
