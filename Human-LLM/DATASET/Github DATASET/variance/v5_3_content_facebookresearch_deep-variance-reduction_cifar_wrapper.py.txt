import torchvision
from torch.utils.data import Dataset

class CIFAR10Wrapper(Dataset):
    """
    A wrapper for the CIFAR-10 dataset that caches transformed items to improve performance.
    """

    def __init__(self, root, train=True, download=False, transform=None):
        """
        Initializes the CIFAR10Wrapper.

        Args:
            root (str): Root directory where the dataset will be stored.
            train (bool, optional): If True, creates dataset from training set, otherwise from test set. Defaults to True.
            download (bool, optional): If True, downloads the dataset from the internet and puts it in root directory. Defaults to False.
            transform (callable, optional): A function/transform that takes in an PIL image and returns a transformed version. Defaults to None.
        """
        # Load the CIFAR-10 dataset
        self.dataset = torchvision.datasets.CIFAR10(root=root, train=train, download=download, transform=transform)

        # Initialize cache for transformed items
        self.transformed_cache = {}

        # Metrics for evaluating cache performance
        self.access_total = 0
        self.cache_hit = 0
        self.access_since_retransform = 0

    def __getitem__(self, index):
        """
        Retrieves an item from the dataset or cache.

        Args:
            index (int): Index of the item to be retrieved.

        Returns:
            Any: The requested dataset item.
        """
        # Check if the item is in cache
        if index in self.transformed_cache:
            item = self.transformed_cache[index]
            self.cache_hit += 1
        else:
            # If not in cache, get item from dataset and cache it
            item = self.dataset[index]
            self.transformed_cache[index] = item

        # Update access metrics
        self.access_total += 1
        self.access_since_retransform += 1

        return item

    def __len__(self):
        """
        Returns the total number of items in the dataset.
        """
        return len(self.dataset)

    def retransform(self):
        """
        Clears the transformation cache and resets metrics.
        This should be called when transformations are changed or periodically to free memory.
        """
        print(f"Total calls to retransform: {self.access_since_retransform}, Cache hits: {self.cache_hit}")
        self.transformed_cache = {}
        self.access_since_retransform = 0
        self.cache_hit = 0
