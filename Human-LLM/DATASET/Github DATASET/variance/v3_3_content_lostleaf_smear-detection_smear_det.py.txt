import numpy as np
import glob
import matplotlib.pyplot as plt
from collections import deque

# Constants for threshold values
THRESHOLD_LOW = 200
THRESHOLD_HIGH = 600

def bfs(variance, pos):
    # Define movement directions
    MOVEMENT_X = [1, 0, -1, 0]
    MOVEMENT_Y = [0, 1, 0, -1]
    
    # Initialize visited positions
    visited = set([(x, y) for x, y in pos])
    
    # Initialize queue for BFS
    que = deque(pos)
    
    # Initialize mask array
    mask = np.zeros_like(variance, dtype=np.uint8)
    mask[variance < THRESHOLD_LOW] = 255
    
    # Perform BFS
    while que:
        x, y = que.popleft()
        for dx, dy in zip(MOVEMENT_X, MOVEMENT_Y):
            nx, ny = x + dx, y + dy
            if variance[nx, ny] < THRESHOLD_HIGH and (nx, ny) not in visited:
                visited.add((nx, ny))
                mask[nx, ny] = 255
                que.append((nx, ny))
    return mask

def main():
    # Get file names
    file_names = glob.glob("*.npy")
    
    # Process each file
    for idx, name in enumerate(file_names):
        # Load variance data
        variance = np.load(name)
        
        # Get positions below THRESHOLD_LOW
        positions = np.argwhere(variance < THRESHOLD_LOW).tolist()
        
        # Perform BFS to create mask
        mask = bfs(variance, positions)
        
        # Plot the mask
        plt.subplot(231 + idx)
        plt.imshow(mask, cmap="gray")
        plt.title(name)
    
    # Show plots
    plt.show()

# Entry point
if __name__ == "__main__":
    main()
