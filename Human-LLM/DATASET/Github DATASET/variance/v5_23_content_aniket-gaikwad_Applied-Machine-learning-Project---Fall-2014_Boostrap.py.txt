import sys
import random

# Define the author of the script
__author__ = 'Aniket'

# Initialize a list to hold input data
inputData = []

# Set the value for sigma
SIGMA = 0.66

# Get the number of bootstraps from command line argument
NO_OF_BOOTSTRAPS = int(sys.argv[3])

# Function to perform bootstrapping
def perform_bootstrapping(train_set):
    # Initialize a list to store the bootstrap samples
    bootstrap_samples = []

    # Calculate the size of each bootstrap sample
    size_of_bootstrap = len(train_set) // NO_OF_BOOTSTRAPS

    print("Number of Instances:", len(train_set))
    print("One Bootstrap size:", size_of_bootstrap)

    # Generate bootstrap samples
    for _ in range(NO_OF_BOOTSTRAPS):
        bootstrap_sample = [random.choice(train_set) for _ in range(size_of_bootstrap)]
        bootstrap_samples.append(bootstrap_sample)

    # Print each bootstrap sample
    for i, sample in enumerate(bootstrap_samples):
        print("**********")
        print("Length of Sample", i + 1, ":", len(sample))

# Function to generate training and test samples
def generate_train_test_samples(input_data):
    # Initialize lists to store train and test sets
    train_set = []
    test_set = []

    # Iterate over each line in the input data
    for line in input_data:
        # Randomly assign each line to train or test set based on sigma
        if random.random() < SIGMA:
            train_set.append(line)
        else:
            test_set.append(line)

    # Print the generated train and test sets
    print("Train Set:")
    print(train_set)
    print("Test Set:")
    print(test_set)

    # Calculate the number of test examples
    num_test_examples = len(test_set)
    return train_set, test_set, num_test_examples

# Main function to execute the program
def main(input_data):
    # Generate train and test samples
    train_set, _, _ = generate_train_test_samples(input_data)

    # Perform bootstrapping on the train set
    perform_bootstrapping(train_set)

# Entry point of the script
if __name__ == "__main__":
    # Call the main function with the input data
    main(inputData)
