import sys
import random

# Define the author of the script
__author__ = 'Aniket'

# Initialize a list to hold input data
inputData = []

# Set the value for sigma
SIGMA = 0.66

# Get the number of bootstraps from command line argument
NO_OF_BOOTSTRAPS = int(sys.argv[3])

# Function to perform bootstrapping
def bootstrapping(trainSet):
    # Declare a global variable to store the bootstrap samples
    global bootstrap
    bootstrap = []

    # Print the total number of instances in the training set
    print("Number of Instances: %d" % (len(trainSet)))

    # Calculate the size of each bootstrap sample
    sizeOfBootstrap = int(len(trainSet) / NO_OF_BOOTSTRAPS)
    print("One Bootstrap size: %d " % (sizeOfBootstrap))

    # Generate bootstrap samples
    for i in range(NO_OF_BOOTSTRAPS):
        bootstrap_lst = []
        for j in range(sizeOfBootstrap):
            bootstrap_lst.append(random.choice(trainSet))
        bootstrap.append(bootstrap_lst)

    # Print each bootstrap sample
    for i in range(NO_OF_BOOTSTRAPS):
        print("**********")
        print("Length: %d" % (len(bootstrap[i])))

# Function to generate training and test samples
def generateTrainTestSample(inputData):
    # Declare global variables to store train and test sets
    global trainSet, testSet
    trainSet = []
    testSet = []

    # Iterate over each line in the input data
    for line in range(len(inputData)):
        # Randomly assign each line to train or test set based on sigma
        if random.random() < SIGMA:
            trainSet.append(inputData[line])
        else:
            testSet.append(inputData[line])

    # Print the generated train and test sets
    print("Train Set:")
    print(trainSet)
    print("Test Set:")
    print(testSet)

    # Calculate the number of test examples
    global noOfTestExamples
    noOfTestExamples = len(testSet)

# Main function to execute the program
def main(inputSet):
    # Declare global variables
    global trainSet, testSet, bootstrap

    # Generate train and test samples
    generateTrainTestSample(inputSet)

    # Perform bootstrapping on the train set
    bootstrapping(trainSet)

# Entry point of the script
if __name__ == "__main__":
    # Call the main function with the input data
    main(inputData)
