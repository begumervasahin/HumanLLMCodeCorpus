# Importing necessary libraries
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.linear_model import LinearRegression
from sklearn.metrics import mean_squared_error

# Reading the data
data = pd.read_csv('/Users/swaruptripathy/Desktop/Data Science/headbrain.csv')

# Extracting features and target variable
X = data['Head Size(cm^3)'].values
Y = data['Brain Weight(grams)'].values

# Calculating mean of X and Y
mean_x = np.mean(X)
mean_y = np.mean(Y)

# Total number of values
m = len(X)

# Calculating coefficients b0 and b1
numer = 0
denom = 0
for i in range(m):
    numer += (X[i] - mean_x) * (Y[i] - mean_y)
    denom += (X[i] - mean_x) ** 2
b1 = numer / denom
b0 = mean_y - (b1 * mean_x)

# Plotting the regression line
max_x = np.max(X) + 100
min_x = np.min(X) - 100
x = np.linspace(min_x, max_x, 1000)
y = b0 + b1 * x
plt.plot(x, y, color='blue', label='Regression Line')

# Plotting the data points
plt.scatter(X, Y, c='red', label='Data Points')

# Adding labels and legend
plt.xlabel('Head Size in cm3')
plt.ylabel('Brain Weight in grams')
plt.legend()

# Showing the plot
plt.show()

# Calculating Root Mean Squared Error (RMSE)
rmse = 0
for i in range(m):
    y_pred = b0 + b1 * X[i]
    rmse += (Y[i] - y_pred) ** 2
rmse = np.sqrt(rmse / m)
print("Root Mean Squared Error:", rmse)

# Calculating R-squared value
ss_t = 0
ss_r = 0
for i in range(m):
    y_pred = b0 + b1 * X[i]
    ss_t += (Y[i] - mean_y) ** 2
    ss_r += (Y[i] - y_pred) ** 2
r2 = 1 - (ss_r / ss_t)
print("R-squared Value:", r2)

# Using sklearn to perform linear regression
X = X.reshape((m, 1))  # Reshaping X
reg = LinearRegression()  # Creating a linear regression object
reg = reg.fit(X, Y)  # Fitting the model
Y_pred = reg.predict(X)  # Making predictions
mse = mean_squared_error(Y, Y_pred)  # Calculating Mean Squared Error (MSE)
rmse = np.sqrt(mse)  # Calculating RMSE
r2_score = reg.score(X, Y)  # Calculating R-squared value
print("Root Mean Squared Error (Sklearn):", np.sqrt(mse))
print("R-squared Value (Sklearn):", r2_score)
