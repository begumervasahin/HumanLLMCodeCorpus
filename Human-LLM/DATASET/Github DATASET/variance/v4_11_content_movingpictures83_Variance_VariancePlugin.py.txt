import sys

class VariancePlugin:
    def input(self, filename):
        # Store the filename
        self.myfile = filename
    
    def run(self):
        # Open the file for reading
        filestuff = open(self.myfile, 'r')
        # Read the first line to get the bacteria names
        firstline = filestuff.readline()
        # Split the line by comma to get individual bacteria names
        self.bacteria = firstline.split(',')
        # Remove any empty string elements if present
        if '\"\"' in self.bacteria:
            self.bacteria.remove('\"\"')
        # Calculate the number of bacteria
        self.n = len(self.bacteria)
        # Initialize an empty list to store adjacency matrix
        self.ADJ = [[] for _ in range(self.n)]
        # Read the rest of the lines to populate the adjacency matrix
        for line in filestuff:
            # Split each line by comma
            contents = line.split(',')
            # Iterate over the values (excluding the first one which is the bacteria name)
            for j in range(len(contents) - 1):
                # Convert the value to float and append to the appropriate list in the adjacency matrix
                value = float(contents[j + 1])
                self.ADJ[j].append(value)
    
    def output(self, filename):
        # Open the file for writing
        filestuff2 = open(filename, 'w')
        # Write the header
        filestuff2.write("Element\tVariance\n\n")
        # Initialize a list to store variances
        variances = []
        # Calculate variance for each bacteria
        for i in range(self.n):
            sum_values = sum(self.ADJ[i])
            variance = 0
            # If there are values present
            if len(self.ADJ[i]) != 0:
                # Calculate the average
                average = sum_values / len(self.ADJ[i])
                # Calculate the variance
                for j in range(len(self.ADJ[i])):
                    variance += (average - self.ADJ[i][j]) ** 2
                # Append the variance along with the bacteria name to the list
                variances.append((variance / len(self.ADJ[i]), self.bacteria[i]))
        # Sort the variances in descending order
        variances.sort(reverse=True)
        # Write the variances to the file
        for i in range(len(variances)):
            filestuff2.write(variances[i][1])
            filestuff2.write("\t")
            filestuff2.write(str(variances[i][0]))
            filestuff2.write("\n")
