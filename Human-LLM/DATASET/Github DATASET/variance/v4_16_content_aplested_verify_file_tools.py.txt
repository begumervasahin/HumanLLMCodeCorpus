import sys
import os

# Function to read lines from a file
def file_read(file_path):
    print('Reading ' + file_path + '...')
    with open(file_path, 'r') as file:
        lines = file.readlines()
    lines = [line.strip() for line in lines]  # Removing leading and trailing whitespaces
    del lines[0]  # Removing the first line
    del lines[-1]  # Removing the last line
    print(lines[0])
    return lines

# Function to write lines to a file
def file_write(file_path, data):
    with open(file_path, "w") as file:
        for line in data:
            joined_line = '\t'.join(line)
            file.write(joined_line + '\n')

# Function to add a prefix to a filename
def addFilenamePrefix(path, prefix=""):
    directory, filename = os.path.split(path)
    filename = prefix + filename
    return os.path.join(directory, filename)

# Function to convert lines of ASCII text into traces (lists of floats)
def lines_into_traces(lines):
    split_lines = [line.split('\t') for line in lines]
    traces = []
    num_of_traces = len(split_lines[0])
    for i in range(num_of_traces):
        traces.append([])
    for line in split_lines:
        for i in range(num_of_traces):
            traces[i].append(float(line[i]))
    return traces

# Function to convert traces (lists of floats) into lines of ASCII text
def traces_into_lines(traces):
    lines_of_output = len(traces[0])
    lines = [[] for _ in range(lines_of_output)]
    for point in range(lines_of_output):
        for trace in traces:
            lines[point].append(str(trace[point]))
    return lines
