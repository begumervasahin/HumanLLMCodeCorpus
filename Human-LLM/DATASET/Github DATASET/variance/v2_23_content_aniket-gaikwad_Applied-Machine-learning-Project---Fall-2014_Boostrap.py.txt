import sys
import random

# Initialize input data list
inputData = []

# Parameters
SIGMA = 0.66

# Take the number of bootstraps from command-line argument
NO_OF_BOOTSTRAPS = int(sys.argv[1])

# Function for bootstrapping
def bootstrapping(trainSet):
    bootstrap = []
    print("Number of Instances: %d" % len(trainSet))
    sizeOfBootstrap = len(trainSet) // NO_OF_BOOTSTRAPS
    print("Size of One Bootstrap: %d" % sizeOfBootstrap)
    
    for i in range(NO_OF_BOOTSTRAPS):
        bootstrap_lst = []
        for j in range(sizeOfBootstrap):
            bootstrap_lst.append(random.choice(trainSet))
        bootstrap.append(bootstrap_lst)
    
    for i in range(NO_OF_BOOTSTRAPS):
        print("**********")
        print("Length: %d" % len(bootstrap[i]))

# Function for generating train and test samples
def generateTrainTestSample(inputData):
    trainSet = []
    testSet = []
    
    for line in inputData:
        if random.random() < SIGMA:
            trainSet.append(line)
        else:
            testSet.append(line)
    
    print("Train Set:")
    print(trainSet)
    print("Test Set:")
    print(testSet)

# Main function
def main(inputSet):
    generateTrainTestSample(inputSet)
    bootstrapping(trainSet)

if __name__ == "__main__":
    # Assuming inputData is defined elsewhere in your code
    main(inputData)
