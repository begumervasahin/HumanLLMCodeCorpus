import sys

# Check Python version to import appropriate modules
if sys.version_info[0] < 3:
    from Tkinter import *
    import tkSimpleDialog
else:
    from tkinter import *
    from tkinter import simpledialog as tkSimpleDialog

# Author and date information
__author__ = "remis"
__date__ = "$27-May-2009 23:02:26$"

# Define a class for dialog window for entering plot parameters
class PlotParam(tkSimpleDialog.Dialog):
    # Method to create the body of the dialog window
    def body(self, master):
        # Set the title of the dialog window
        self.title('Enter new plot parameters...')
        
        # Labels for various parameters
        Label(master, text='Min X:').grid(row=0)
        Label(master, text='Max X:').grid(row=1)
        Label(master, text='Bin width:').grid(row=2)
        Label(master, text='Max Y:').grid(row=3)
        
        # Entry fields for inputting parameters
        self.e1 = Entry(master, justify=CENTER)
        self.e1.grid(row=0, column=1)
        self.e1.insert(END, '0')  # Default value for Min X
        self.e2 = Entry(master, justify=CENTER)
        self.e2.grid(row=1, column=1)
        self.e2.insert(END, '10')  # Default value for Max X
        self.e3 = Entry(master, justify=CENTER)
        self.e3.grid(row=2, column=1)
        self.e3.insert(END, '1')  # Default value for Bin width
        self.e4 = Entry(master, justify=CENTER)
        self.e4.grid(row=3, column=1)
        self.e4.insert(END, '1000')  # Default value for Max Y
        
        # Set focus on the first entry field
        return self.e1

    # Method to apply the entered parameters
    def apply(self):
        # Get and store the entered parameters
        self.xmin = int(self.e1.get())
        self.xmax = int(self.e2.get())
        self.dx = int(self.e3.get())
        self.ymax = int(self.e4.get())
