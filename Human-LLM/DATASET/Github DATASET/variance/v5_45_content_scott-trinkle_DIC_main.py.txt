# Import necessary modules
from dic.experiment import Experiment
from dic.plots import generate_plots

# Define experiment parameters
experiment_params = {
    'lens': 40,
    'weak_grad': False,
    'wavelength': 546,  # Renamed 'lamda' to 'wavelength' for clarity
    'approaches': ['A: 2x2', 'A: 2x3', 'B: 2x3', 'B: 2x4'],
    'k_value': 1e2,  # Renamed 'k' to 'k_value' for clarity
    'from_zero': True,  # Renamed 'fromZero' to 'from_zero' for consistency
    'save_data': False,  # Renamed 'save' to 'save_data' for clarity
    'filepath': None
}

# Initialize the experiment with given parameters
experiment = Experiment(**experiment_params)

# Generate data for the experiment
sigma_gamma, sigma_theta = experiment.generate_data(sample_size=100)  # Renamed variables for clarity

# Generate plots based on the experiment data
generate_plots(sigma_gamma, sigma_theta, experiment,
               polar=True,    # Plot in polar coordinates
               areaplot=True,  # Plot the "error area"
               snr=False,     # Signal-to-noise ratio option (lowercase for consistency)
               report=False,  # Whether to generate a report
               show=True)     # Whether to display the figure
