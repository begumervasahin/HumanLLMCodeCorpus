import numpy as np
import matplotlib.pyplot as plt
import time

# Parameters
T = 1
s0 = 100
v0 = 0.12
kappa = 2
theta = 0.15
sigma = 0.8
r = 0

# Function to simulate Heston model using Euler Discretization
def simulate_heseton(N):
    dt = T / N
    ST = s0
    VT = v0
    
    for i in range(N):
        ST += r * ST * dt + max(VT, 0) ** 0.5 * ST * dt ** 0.5 * np.random.randn()        
        VT += kappa * (theta - VT) * dt + sigma * max(VT, 0) ** 0.5 * dt ** 0.5 * np.random.randn()
    
    return [ST, VT]

# Strike prices, number of steps, and epsilon values
K = [100, 500, 1000, 5000, 10000]
nsteps = [int(100 ** 0.5), int(500 ** 0.5), int(1000 ** 0.5), int(5000 ** 0.5), int(10000 ** 0.5)]
epsilon = [70, 90, 100, 110, 130]

# Lists to store option price estimations and computation times
option_price_estimations = []
computation_times = []

# Estimate option prices
for strike in epsilon:
    estimations = []
    for num_steps in K:
        sum_estimator = 0
        for _ in range(num_steps):
            ST = simulate_heseton(int(num_steps ** 0.5))[0]
            sum_estimator += np.exp(-r * T) * max(ST - strike, 0)
        estimations.append(sum_estimator / num_steps)
    option_price_estimations.append(estimations)

# Measure computation times
for strike in epsilon:
    run_times = []
    for num_steps in K:
        sum_estimator = 0
        start_time = time.time()
        for _ in range(num_steps):
            ST = simulate_heseton(int(num_steps ** 0.5))[0]
            sum_estimator += np.exp(-r * T) * max(ST - strike, 0)
        end_time = time.time()
        run_times.append(end_time - start_time)
    computation_times.append(run_times)

# Correct option prices
correct_prices = [32.70326, 18.92499, 13.984957, 10.263762, 5.5888102]

print("Option Price Estimations:")
print(option_price_estimations)
print("\nComputation Times:")
print(computation_times)
print("\nCorrect Option Prices:")
print(correct_prices)
