# Define the list of games
games = [
    'Hopper-v2',
    'HalfCheetah-v2',
    'Walker2d-v2',
    'Reacher-v2',
]

# Define the list of seeds
seeds = [62, 43, 97, 104, 123, 1003, 554, 221, 11, 992]

# Define the list of noise levels
noise_levels = [0.0, 0.1, 0.2, 0.3, 0.4]

# Define the list of gamma values along with their configurations
gamma_configurations = [
    ([0.0, .99], "PPO+", False, False, False, False),
    ([.99], "PPO", False, False, False, False),
    ([0.0, .99], "PPO+RP", True, False, False, True)
]

# Generate all possible combinations of parameters
parameter_combinations = []
for seed in seeds:
    for game in games:
        for noise_level in noise_levels:
            for gamma_config in gamma_configurations:
                parameter_combinations.append((seed, game, *gamma_config, noise_level))

# Define a class to hold arguments
class Arguments:
    def __init__(self):
        pass

# Define a function to load parameters based on a given index
def load_parameters(args):
    # Unpack parameters based on the run index
    (args.seed, args.env_name, args.gamma, args.name, args.reward_predictor,
     args.reward_epsilon, args.use_s, args.use_s_a, args.use_s_a_sprime,
     args.noise_level) = parameter_combinations[args.run_index]
    
    # Construct log directory based on parameters
    args.log_dir = args.log_dir + args.env_name + '_' + str(args.seed) + '_' + args.name + 'UN' + str(args.reward_epsilon)
    args.save_dir = args.log_dir
    
    # Set additional algorithm parameters
    args.algo = 'ppo'
    args.use_gae = True
    args.log_interval = 1
    args.vis_interval = 1
    args.num_steps = 2048
    args.num_processes = 1
    args.entropy = 0.0
    args.lr = 3e-4
    args.value_loss_coef = 1
    args.ppo_epoch = 10
    args.rp_burn_in = 100
    args.lr_rp = args.lr
    args.num_stack = 1
    args.num_frames = 1000000
    args.reward_high = 1.0
    args.reward_low = -1.0

if __name__ == "__main__":
    args = Arguments()
    args.run_index = 0  # You can change this index to get different configurations
    args.log_dir = "./logs/"
    load_parameters(args)
    print("Loaded Parameters:")
    print(vars(args))
