# Importing necessary libraries
import matplotlib as mpl
mpl.use('Agg')  # Using Agg backend for matplotlib to save plots without displaying them
import matplotlib.pyplot as plt
plt.rcParams['image.cmap'] = 'gray'  # Setting default colormap for images to grayscale
import logging  # Importing logging module for logging messages
from pytorch import models  # Importing PyTorch models
from pytorch import dataset as datasets  # Importing PyTorch datasets
import torch  # Importing PyTorch library
import pytorch.experiment.utils as utils  # Importing utility functions from custom module

# Main script
if __name__ == "__main__":
    # Parsing model and dataset names using a utility function
    model_name, dataset_name = utils.parse_model_and_dataset("Experiment: accuracy of model for rotated vs unrotated dataset.")
else:
    # Default values if not executed as main script
    dataset_name = "cifar10"
    model_name = models.AllConvolutional.__name__

verbose = False  # Flag to control verbosity
use_cuda = torch.cuda.is_available()  # Checking if CUDA is available for GPU acceleration

# Getting dataset based on dataset name
dataset = datasets.get_dataset(dataset_name)

# Printing dataset summary if verbosity is enabled
if verbose:
    print(dataset.summary())

# Loading models for rotation experiment
from pytorch.experiment import rotation
model, rotated_model, scores, config = rotation.load_models(dataset, model_name, use_cuda)  

# Printing scores if verbosity is enabled
if verbose:
    rotation.print_scores(scores)

# Running variance experiment
from pytorch.experiment import variance
n_rotations = 16
conv_aggregation_function = "sum"

# Logging information about the experiment
logging.info(f"Calculating measure with aggregation function {conv_aggregation_function} for conv layers and {n_rotations} rotations...")

# Running variance experiment and obtaining results
results = variance.run_all(model, rotated_model, dataset, config, n_rotations, conv_aggregation_function, batch_size=64)

# Logging message indicating that results are being saved
logging.info("Saving results...")

# Saving results of the experiment
variance.save_results(model.name, dataset.name, results, conv_aggregation_function)
