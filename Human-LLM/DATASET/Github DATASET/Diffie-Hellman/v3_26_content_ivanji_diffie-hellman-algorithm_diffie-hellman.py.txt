# Define the shared base and prime numbers
shared_base = 47
shared_prime = 199

# Print the shared base and prime numbers
print("Shared base (g) is equal to %s and shared prime (p) is equal to %s" % (shared_base, shared_prime))
print("--------------------------")

# Define Alice's and Bob's secret keys
alice_secret = 6
bob_secret = 2

print("--------------------------")
print("Alice calculates: (g^a) mod p and sends the result (A) to Bob")
# Calculate A: (g^a) mod p
A = pow(shared_base, alice_secret, shared_prime)
print("--------------------------")

print("Bob calculates the same and sends the result (B) to Alice")
# Calculate B: (g^b) mod p
B = pow(shared_base, bob_secret, shared_prime)
print("--------------------------")

print("Alice now calculates using the received result (B) from Bob")
# Calculate Alice's shared modulo using Bob's result       
alice_modulo = pow(B, alice_secret, shared_prime)
print("--------------------------")

print("Bob now calculates using the received result (A) from Alice")
# Calculate Bob's shared modulo using Alice's result       
bob_modulo = pow(A, bob_secret, shared_prime)

# Print the shared modulos calculated by Alice and Bob     
print("Bob's calculated modulo:", bob_modulo)
print("Alice's calculated modulo:", alice_modulo)

print("--------------------------")
print("Shared Key is equal to %s." % bob_modulo)
