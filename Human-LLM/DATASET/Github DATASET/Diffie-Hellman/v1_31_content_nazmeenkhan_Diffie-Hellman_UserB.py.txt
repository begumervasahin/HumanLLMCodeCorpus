import socket
import random

c = socket.socket()
HOST = '127.0.0.1'
PORT = 5050
c.connect((HOST, PORT))

# Receive prime number q and primitive root alpha from the server
q = int(c.recv(1024))
alpha = int(c.recv(1024))

# Generate a random private key Xb
Xb = random.randint(1, q)

# Calculate Yb = alpha^Xb mod q
Yb = pow(alpha, Xb, q)
print("Yb:", Yb)

# Send Yb to the server
c.send(str(Yb).encode())

# Receive Ya from the server
Ya = int(c.recv(1024))

# Calculate the symmetric key Kb = Ya^Xb mod q
Kb = pow(Ya, Xb, q)
print("Symmetric Key:", Kb)

# Close the connection
c.close()
