LOCAL_IP = ""

class Handler(threading.Thread):
    """
    Thread class for receiving messages.
    """
    def __init__(self, local_host, local_port):
        threading.Thread.__init__(self, name="messenger_receiver")
        self.host = local_host
        self.port = local_port
        self.data_received = ""

    def listen(self):
        """
        Listen for incoming messages.
        """
        global FIRST_CONNECTION_FLAG, G, P
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.bind((self.host, self.port))
        sock.listen(10)
        while True:
            connection, _ = sock.accept()
            try:
                full_message = ""
                while True:
                    data = connection.recv(16)
                    full_message += data.decode("utf-8")
                    if not data:
                        if full_message.count(",") == 1:
                            self.data_received = full_message.split(",")
                            print("Data received:", self.data_received)
                            G = self.data_received[0]
                            P = self.data_received[1]
                            print("Transported P and G:", G, P)
                        break
            finally:
                connection.shutdown(2)
                connection.close()

    def run(self):
        """
        Run method for the thread.
        """
        self.listen()
