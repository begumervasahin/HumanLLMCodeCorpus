from random import randint
import math

# Function to check if a number is prime using the Miller-Rabin primality test
def is_prime(num, count):
    if num == 1:
        return False
    if count >= num:
        count = num - 1
    for x in range(count):
        a = randint(1, num - 1)
        if pow(a, num-1, num) != 1:
            return False
    return True

# Function to generate a big prime number using the Miller-Rabin primality test
def generate_big_prime(n):
    found_prime = False
    while not found_prime:
        p = randint(2**(n-1), 2**n)
        if is_prime(p, 1000):
            return p

# Generate two large prime numbers
prime1 = generate_big_prime(2**10)
print("Prime 1:")
print(prime1)
prime2 = generate_big_prime(2**10)
print("Prime 2:")

# Generate Alice and Bob's private keys
alice_private_key = randint(2**10, 2**15)

# Compute public keys for Alice and Bob
a = pow(prime2, alice_private_key, prime1)

# Compute shared secret keys for Alice and Bob
alice_key = pow(b, alice_private_key, prime1)
