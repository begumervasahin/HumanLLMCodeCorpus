import Seeded_Random_Generator

def generate_seed(a_private_key, b_private_key, a_peer_public_key, b_peer_public_key):
    # Extract key values from objects
    a_private_key_hex = extract_key_hex(a_private_key)
    b_private_key_hex = extract_key_hex(b_private_key)
    a_peer_public_key_hex = extract_key_hex(a_peer_public_key)
    b_peer_public_key_hex = extract_key_hex(b_peer_public_key)

    a_pkey = int(a_private_key_hex, 16)
    b_pkey = int(b_private_key_hex, 16)
    a_public_key = int(a_peer_public_key_hex, 16)
    b_public_key = int(b_peer_public_key_hex, 16)

    # Calculate intermediate values
    x1 = (a_public_key ^ a_pkey) % b_public_key
    x2 = (a_public_key ^ b_pkey) % b_public_key
    k1 = (x2 ^ a_pkey) % b_public_key
    k2 = (x1 ^ b_pkey) % b_public_key

    # Check if keys match and derive seed
    if k1 == k2:
        seed = k1 % 1111
        new_seed = Seeded_Random_Generator.reduce(seed)    
        print("Seed:", seed)
        print("Reduced Seed:", new_seed)
        Seeded_Random_Generator.random_generator(new_seed) 

def extract_key_hex(key_object):
    """Extract hexadecimal string representation of key object."""
    key_hex = str(key_object).replace("<cryptography.hazmat.backends.openssl.dh._DHPrivateKey object at 0x", "").replace(">", "")
    return key_hex
