import itertools

def generate_bit_combinations(num_bits):
    """
    Generate all possible combinations of bits for a given number of bits.

    Args:
        num_bits (int): The number of bits.

    Returns:
        list: A list of bit strings representing all possible combinations.
    """
    bit_tuples = itertools.product([0, 1], repeat=num_bits)
    bit_strings = [''.join(map(str, bits)) for bits in bit_tuples]
    return bit_strings

def crc(message, divisor):
    """
    Compute the cyclic redundancy check (CRC) for a message with a given divisor.

    Args:
        message (str): The message to calculate the CRC for.
        divisor (str): The divisor used for CRC calculation.

    Returns:
        str: The CRC value.
    """
    padding = '0' * (len(divisor) - 1)
    padded_message = message + padding
    padded_message = list(padded_message)
    divisor = list(divisor)
    for i in range(len(padded_message) - len(padding)):
        if padded_message[i] == '1':
            for j in range(len(divisor)):
                padded_message[i + j] = str(int(padded_message[i + j]) ^ int(divisor[j]))
    return ''.join(padded_message[-len(padding):])

def calculate_all_collisions(message, divisor, prefix_list=None):
    """
    Calculate all possible collisions for a given message and divisor.

    Args:
        message (str): The message for which to calculate the collisions.
        divisor (str): The divisor used for CRC calculation.
        prefix_list (list, optional): A list of prefixes to be appended to the message. 
                                      If None, all possible prefixes will be used.

    Returns:
        list: A list of all possible colliding messages.
    """
    final_crc = crc(message, divisor)
    all_combinations = generate_bit_combinations(len(divisor))
    collisions = []
    if prefix_list is None:
        prefix_list = generate_bit_combinations(len(message) - len(divisor))
    for prefix in prefix_list:
        for combination in all_combinations:
            current_message = prefix + combination
            if crc(current_message, divisor) == final_crc:
                collisions.append(current_message)
    return collisions

# Example usage
print(calculate_all_collisions("11010110", "10011", ["111"]))
