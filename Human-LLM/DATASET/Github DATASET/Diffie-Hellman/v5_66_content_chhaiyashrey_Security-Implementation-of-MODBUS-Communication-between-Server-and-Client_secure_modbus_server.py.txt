import logging
from socketserver import TCPServer
from collections import defaultdict
from umodbus import conf
from umodbus.server.tcp import RequestHandler, get_server
from umodbus.utils import log_to_stream

# Set up logging for debugging
log_to_stream(level=logging.DEBUG)

# Create a default data store to store Modbus data
data_store = defaultdict(int)

# Configure umodbus to use signed values for Modbus data
conf.SIGNED_VALUES = True

# Enable reusing the address for the TCP server
TCPServer.allow_reuse_address = True

# Create a Modbus TCP server instance
modbus_server = get_server(TCPServer, ('', 502), RequestHandler)

# Route for reading data from the Modbus data store
@app.route(slave_ids=[1], function_codes=[1, 2], addresses=list(range(0, 10)))
def read_data_store(slave_id, function_code, address):
    """Return the value stored at the specified address in the Modbus data store."""
    return data_store[address]

# Route for writing data to the Modbus data store
@app.route(slave_ids=[1], function_codes=[5, 15], addresses=list(range(0, 10)))
def write_data_store(slave_id, function_code, address, value):
    """Set the value at the specified address in the Modbus data store."""
    data_store[address] = value

if __name__ == '__main__':
    try:
        # Start serving Modbus requests indefinitely
        modbus_server.serve_forever()
    finally:
        # Clean up resources when the server is stopped
        modbus_server.shutdown()
        modbus_server.server_close()
