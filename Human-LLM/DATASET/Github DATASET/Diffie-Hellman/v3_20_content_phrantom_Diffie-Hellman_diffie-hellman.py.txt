from random import randint

# Diffie-Hellman key exchange system
# Authors: Whitfield Diffie, Martin Hellman
# Generally used as a means to agree on symmetric keys
# that will be used for session encryption
# (session key establishment).

print("\n")

# Choose a generator and a prime number
generator = 2
prime_number = getPrime(1024)

print("Generator:", generator)
print("Prime number:", prime_number)

# Generate random private keys for both parties
private_key_A = randint(prime_number // 2, prime_number - 1)
private_key_B = randint(prime_number // 2, prime_number - 1)

# Calculate public keys for both parties
public_key_A = pow(generator, private_key_A, prime_number) 
public_key_B = pow(generator, private_key_B, prime_number) 

# Calculate shared secrets
shared_secret_A = pow(public_key_B, private_key_A, prime_number)
shared_secret_B = pow(public_key_A, private_key_B, prime_number)

# Verify that both parties obtain the same shared secret   
assert shared_secret_A == shared_secret_B

print("\nPublic number of A:", public_key_A)
print("Public number of B:", public_key_B)

print("\nShared secret A:", shared_secret_A)
print("Shared secret B:", shared_secret_B)
