class PointAddition:
    def __init__(self, a, b):
        self.a = a
        self.b = b
        self.discriminant = -16 * (4 * a * a * a + 27 * b * b)
        if not self.is_smooth():
            raise Exception(f"The curve with parameters a={a} and b={b} is not smooth!")

    def is_smooth(self):
        """
        Check if the elliptic curve is smooth.
        """
        return self.discriminant != 0

    def test_point(self, x, y):
        """
        Check if the given point lies on the elliptic curve.
        """
        return y * y == x * x * x + self.a * x + self.b

    def print_equation(self):
        """
        Print the equation of the elliptic curve.
        """
        print(f"The elliptic curve equation is: y^2 = x^3 + {self.a}x + {self.b}")

    def __eq__(self, other):
        """
        Check if two elliptic curves are equal.
        """
        return (self.a, self.b) == (other.a, other.b)

    def sum_points(self, x1, y1, x2, y2, f):
        """
        Perform point addition operation on the elliptic curve.
        """
        E_y = self.egcd(x2 - x1, f)
        m = ((y2 - y1) * E_y) % f
        x3 = (m * m - x1 - x2) % f
        y3 = (m * (x1 - x3) - y1) % f
        return (x3, y3)

    def egcd(self, a, b):
        """
        Extended Euclidean algorithm for computing modular multiplicative inverse.
        """
        remainder = -1
        B = b
        A = a
        eq_set = []
        full_set = []
        mod_set = []

        while remainder != 1 and remainder != 0:
            remainder = b % a
            quotient = b // a
            eq_set = [remainder, b, a, -quotient]
            b = a
            a = remainder
            full_set.append(eq_set)

        for i in range(0, 4):
            mod_set.append(full_set[-1][i])
        mod_set.insert(2, 1)

        counter = 0
        for i in range(1, len(full_set)):
            if counter % 2 == 0:
                mod_set[2] = full_set[-1 * (i + 1)][3] * mod_set[4] + mod_set[2]        
                mod_set[3] = full_set[-1 * (i + 1)][1]
            elif counter % 2 != 0:
                mod_set[4] = full_set[-1 * (i + 1)][3] * mod_set[2] + mod_set[4]        
                mod_set[1] = full_set[-1 * (i + 1)][1]
            counter += 1

        return mod_set[2] % B if mod_set[3] == B else mod_set[4] % B
