def get_encoding():
    """Generate encoding and decoding dictionaries."""
    encoding_dict = {}
    decoding_dict = {}

    # Define encoding dictionary
    encoding_dict[' '] = 0
    decoding_dict[0] = ' '
    for i in range(65, 91):
        encoding_dict[chr(i)] = i - 64
        decoding_dict[i - 64] = chr(i)
    encoding_dict[','] = 27
    encoding_dict['.'] = 28
    encoding_dict['?'] = 29
    decoding_dict[27] = ','
    decoding_dict[28] = '.'
    decoding_dict[29] = '?'
    for i in range(48, 58):
        encoding_dict[chr(i)] = i - 18
        decoding_dict[i - 18] = chr(i)
    for i in range(97, 123):
        encoding_dict[chr(i)] = i - 57
        decoding_dict[i - 57] = chr(i)
    encoding_dict['!'] = 66
    decoding_dict[66] = '!'

    return encoding_dict, decoding_dict

def encrypt(string, key):
    """Encrypt the string using the given key."""
    encoding_dict, decoding_dict = get_encoding()
    keys = list(encoding_dict.keys())
    encrypted_string = ""
    for char in string:
        if char not in keys:
            return -1  # Invalid character
        else:
            now = encoding_dict[char]
            encrypted_char = (now + key) % 67
            encrypted_string += decoding_dict[encrypted_char]
    return encrypted_string

def decrypt(string, key):
    """Decrypt the string using the given key."""
    encoding_dict, decoding_dict = get_encoding()
    keys = list(encoding_dict.keys())
    decrypted_string = ""
    for char in string:
        if char not in keys:
            return -1  # Invalid character
        else:
            now = encoding_dict[char]
            decrypted_char = (now - key) % 67
            decrypted_string += decoding_dict[decrypted_char % 67]
    return decrypted_string

# Example usage
if __name__ == "__main__":
    message = "Hello, World!"
    key = 5
    encrypted_message = encrypt(message, key)
    print("Encrypted message:", encrypted_message)
    decrypted_message = decrypt(encrypted_message, key)
    print("Decrypted message:", decrypted_message)
