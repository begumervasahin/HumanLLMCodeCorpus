def generate_fibonacci_list(limit=100):
    """
    Generate a list of Fibonacci numbers.

    Args:
        limit (int): The number of Fibonacci numbers to generate. Default is 100.

    Returns:
        list: A list containing Fibonacci numbers.
    """
    fibonacci_list = [1, 1]  # Start with the first two Fibonacci numbers
    for _ in range(limit - 2):  # Generate the next Fibonacci numbers
        next_fib = fibonacci_list[-1] + fibonacci_list[-2]  # Calculate the next Fibonacci number
        fibonacci_list.append(next_fib)  # Append the next Fibonacci number to the list
    return fibonacci_list

def print_fibonacci_sequence(fibonacci_list):
    """
    Print the Fibonacci sequence.

    Args:
        fibonacci_list (list): A list containing Fibonacci numbers.
    """
    for i, fib_num in enumerate(fibonacci_list):
        if i == 0:
            print(f"{fib_num} + 0 = {fib_num}")
        elif i == 1:
            print(f"0 + {fib_num} = {fib_num}")
        else:
            prev_fib = fibonacci_list[i - 1]
            prev_prev_fib = fibonacci_list[i - 2]
            print(f"{prev_prev_fib} + {prev_fib} = {fib_num}")

def main():
    """
    Main function to generate and print the Fibonacci sequence.
    """
    fibonacci_list = generate_fibonacci_list()  # Generate the Fibonacci list
    print_fibonacci_sequence(fibonacci_list)  # Print the Fibonacci sequence

if __name__ == "__main__":
    main()
