def calculate_fibonacci_and_suffix(n):
    """
    Calculate the Fibonacci number and determine the ordinal suffix based on the input number.
    
    Parameters:
    - n (int): The input number for which Fibonacci number and suffix are to be calculated.

    Returns:
    - (int): The calculated Fibonacci number.
    - (str): The ordinal suffix corresponding to the input number.
    """
    current = 1
    previous = 1

    # Calculate Fibonacci number
    for _ in range(n - 1):
        total = current + previous
        previous = current
        current = total

    # Determine ordinal suffix
    suffix = 'th'
    str_n = str(n)
    if str_n[-2:] not in ('11', '12', '13'):
        if str_n[-1] == '1':
            suffix = 'st'
        elif str_n[-1] == '2':
            suffix = 'nd'
        elif str_n[-1] == '3':
            suffix = 'rd'

    return current, suffix

if __name__ == "__main__":
    # Get user input for the Fibonacci sequence number
    user_input = int(input("Enter the number of the Fibonacci sequence you want displayed: "))

    # Calculate Fibonacci number and determine ordinal suffix
    fib_number, suffix = calculate_fibonacci_and_suffix(user_input)

    # Display the result
    print(f"The {user_input}{suffix} number of the Fibonacci sequence is {fib_number:,d}.")
