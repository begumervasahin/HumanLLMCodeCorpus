import unittest
from fibonacci import Fibonacci
import json

class TestFibonacci(unittest.TestCase):
    def setUp(self):
        # Load configuration data from config.json
        with open('config.json') as file:
            config_data = json.load(file)
        # Initialize Fibonacci object using configuration data
        self.fibonacci = Fibonacci(config_data['redis_host'], config_data['redis_db'])

    def test_generate_fibonacci_sequence(self):
        # Generate Fibonacci numbers for the specified range
        self.fibonacci.numbers(9, 13)
        # Get the generated Fibonacci sequence
        generated_sequence = self.fibonacci.control(9, 13)
        # Expected Fibonacci sequence for the specified range
        expected_sequence = [
            "<div>[9] == b'34'</div>",
            "<div>[10] == b'55'</div>",
            "<div>[11] == b'89'</div>",
            "<div>[12] == b'144'</div>"
        ]
        # Assert that the generated sequence matches the expected sequence
        self.assertEqual(generated_sequence, expected_sequence)

    def test_initial_fibonacci_sequence(self):
        # Get the initial Fibonacci sequence for the specified range
        initial_sequence = self.fibonacci.control(1, 3)
        # Expected initial Fibonacci sequence for the specified range
        expected_initial_sequence = [
            "<div>[1] == b'1'</div>",
            "<div>[2] == b'1'</div>"
        ]
        # Assert that the initial sequence matches the expected initial sequence
        self.assertEqual(initial_sequence, expected_initial_sequence)

if __name__ == '__main__':
    unittest.main()
