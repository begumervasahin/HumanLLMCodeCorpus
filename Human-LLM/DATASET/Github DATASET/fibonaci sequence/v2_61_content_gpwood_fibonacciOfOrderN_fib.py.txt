class Fibonacci:
    def __init__(self, fib_type=2, stop=10):
        # Initializing variables
        self.steps = 0
        self.ratio = 0
        self.fib_type = fib_type  # Fibonacci type: default is 2
        self.stop = stop  # Number of steps to stop at
        self.previous = [1, 1]  # List to store previous Fibonacci numbers

    def calculate_fibonacci(self):
        # Check if reached the desired number of steps
        if self.stop == self.steps:
            # Print the ratio of Fibonacci numbers
            print(f"Ratio {self.fib_type}: {self.ratio}")
            return
        else:
            # Calculate the sum of the last two Fibonacci numbers
            my_sum = sum(self.previous)
            # Calculate the ratio of the new Fibonacci number to the previous one
            self.ratio = my_sum / self.previous[-1]
            # Increment the step counter
            self.steps += 1
            # Check if there are enough previous Fibonacci numbers to calculate the new one
            if len(self.previous) >= self.fib_type:
                # Slice the list to get the required number of previous Fibonacci numbers
                new_array = self.previous[slice(len(self.previous) + 1 - self.fib_type, len(self.previous))]
                # Append the sum of the previous numbers to get the new Fibonacci number
                new_array.append(my_sum)
                # Update the list of previous Fibonacci numbers
                self.previous = new_array
            else:
                # If not enough previous Fibonacci numbers, simply append the sum
                self.previous.append(my_sum)
            # Recursively call the function to calculate the next Fibonacci number
            self.calculate_fibonacci()

if __name__ == "__main__":
    # Iterate over a range of numbers (0 to 54)
    for i in range(55):
        # Create an instance of Fibonacci class with different Fibonacci types
        # and stopping points
        my_fib = Fibonacci(i, 54)
        # Calculate Fibonacci numbers and print their ratios
        my_fib.calculate_fibonacci()
