import unittest
import http_server
import json
import requests

class TestHttpServer(unittest.TestCase):
    def setUp(self):
        # Load configuration from config.json
        with open('config.json') as file:
            self.conf = json.load(file)

    def test_fibonacci_from_a_to_b(self):
        # Test Fibonacci endpoint from a to b
        response = requests.get('http://localhost:{}/fibonacci?from=9&to=13'.format(self.conf['http_server_port']))
        control_sequence = response.content
        etalon_sequence = [b"<div>[9] == 34</div>", b"<div>[10] == 55</div>", b"<div>[11] == 89</div>", b"<div>[12] == 144</div>"]
        self.assertEqual(control_sequence, b"".join(etalon_sequence))

    def test_get_sequence_from_a_to_b(self):
        # Test Get Sequence endpoint from a to b
        response = requests.get('http://localhost:{}/get-sequence?from=9&to=13'.format(self.conf['http_server_port']))
        control_sequence = response.content
        etalon_sequence = [b"<div>[9] == 34</div>", b"<div>[10] == 55</div>", b"<div>[11] == 89</div>", b"<div>[12] == 144</div>"]
        self.assertEqual(control_sequence, b"".join(etalon_sequence))

if __name__ == '__main__':
    unittest.main()
