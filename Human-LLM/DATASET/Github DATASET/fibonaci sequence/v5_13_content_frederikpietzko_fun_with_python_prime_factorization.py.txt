from find_prime_numbers import sieve_of_eratosthenes
from gdc_euclidian_algorithm import modern_euclic_recursive as gcd

def naive_factorization(n):
    # Find prime numbers up to n//2 + 1 using the sieve of Eratosthenes
    primes = sieve_of_eratosthenes(n // 2 + 1)
    factors = []
    # Iterate over the primes and check for factors
    for i, is_prime in enumerate(primes[2:], start=2):
        if is_prime and n % i == 0:
            factors.append(i)
    return factors

def g(x, n):
    # Define the function g(x, n) = (x^2 + 1) mod n
    return (x ** 2 + 1) % n

def pollard_rho(n):
    # Initialize variables for Pollard's rho algorithm
    x = y = 2
    d = 1
    # Implement Pollard's rho algorithm to find a non-trivial divisor of n
    while d == 1:
        x = g(x, n)
        y = g(g(y, n), n)
        d = gcd(abs(x - y), n)
    # Return the divisor found
    if d == n:
        return False  # Indicate failure if no non-trivial divisor is found
    return d

        algorithm_choice = input("Choose the approach: Naive (n) or Pollard Rho (p)? ").lower()
