from Memoria import Cerebro

class Memorization:
    def __init__(self):
        self.memory = Cerebro(10000000, 0.7)  # Initialize memory with specified limit and cleaning threshold

    def calc_fibonacci(self, n):
        if n == 0:  # Base case for Fibonacci sequence
            return 0
        if n == 1:  # Base case for Fibonacci sequence
            return 1

        str_n1 = str(n - 1)  # Convert n-1 to a string for memoization
        str_n2 = str(n - 2)  # Convert n-2 to a string for memoization

        # Retrieve Fibonacci numbers from memory if available
        fib_n1 = self.memory.get_lembranca(str_n1)
        fib_n2 = self.memory.get_lembranca(str_n2)

        # Calculate Fibonacci numbers recursively if not in memory
        if fib_n1 == -1:
            fib_n1 = self.calc_fibonacci(n - 1)
            self.memory.nova_memoria(str_n1, fib_n1)

        if fib_n2 == -1:
            fib_n2 = self.calc_fibonacci(n - 2)
            self.memory.nova_memoria(str_n2, fib_n2)

        return fib_n1 + fib_n2  # Return the sum of Fibonacci numbers

    def get_fibonacci_function(self):
        return lambda n: self.calc_fibonacci(n)  # Return a lambda function for calculating Fibonacci numbers

    def __str__(self):
        return "Recursive Fibonacci with Memoization"

if __name__ == "__main__":
    memorization = Memorization()  # Create an instance of the Memorization class

    n = int(input("Enter a number: "))  # Input the number for which Fibonacci number is to be calculated

    result = memorization.calc_fibonacci(n)  # Calculate the Fibonacci number
    print("Result:", result)  # Print the result
