from flask import Flask

app = Flask(__name__)

# Global variable to keep track of the current Fibonacci number index
n = 0

# Function to calculate the nth Fibonacci number recursively
def fibo(n: int) -> int:
    if n < 1:
        return 0
    elif n == 1:
        return 1
    else:
        return fibo(n - 1) + fibo(n - 2)

# Route for the home page
@app.route("/")
def home():
    return (
        "Available Routes:<br/>"
        "/previous<br/>"
        "/current<br/>"
        "/next<br/>"
    )

# Route for the current Fibonacci number
@app.route("/current")
def current_fib():
    global n
    return str(fibo(n))

# Route for the next Fibonacci number
@app.route("/next")
def next_fib():
    global n
    n += 1
    return str(fibo(n))

# Route for the previous Fibonacci number
@app.route("/previous")
def prev_fib():
    global n
    n -= 1
    return str(fibo(n))

if __name__ == '__main__':
    app.run()
