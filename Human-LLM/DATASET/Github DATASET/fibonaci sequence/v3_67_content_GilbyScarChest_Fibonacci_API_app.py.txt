def fibo(n: int) -> int:
    """
    Calculates the nth Fibonacci number recursively.

    Args:
        n (int): Index of the Fibonacci number to calculate.

    Returns:
        int: The calculated Fibonacci number.
    """
    if n < 1:
        return 0
    elif n == 1:
        return 1
    else:
        return fibo(n - 1) + fibo(n - 2)

current_index = 0  # Global variable to keep track of the current Fibonacci number index

def home():
    """
    Displays the available routes.

    Returns:
        str: HTML-formatted string listing available routes.
    """
    return (
        f"Available Routes:<br/>"
        f"/previous<br/>"
        f"/current<br/>"
        f"/next<br/>"
    )

def current_fib():
    """
    Returns the current Fibonacci number.

    Returns:
        str: The current Fibonacci number as a string.
    """
    global current_index
    return str(fibo(current_index))

def next_fib():
    """
    Moves to the next Fibonacci number and returns it.

    Returns:
        str: The next Fibonacci number as a string.
    """
    global current_index
    current_index += 1
    return str(fibo(current_index))

def prev_fib():
    """
    Moves to the previous Fibonacci number and returns it.

    Returns:
        str: The previous Fibonacci number as a string.
    """
    global current_index
    current_index -= 1
    return str(fibo(current_index))

def handle_request(endpoint):
    """
    Handles user requests based on the provided endpoint.

    Args:
        endpoint (str): The endpoint requested by the user.

    Returns:
        str: The response to the user's request.
    """
    if endpoint == "/":
        return home()
    elif endpoint == "/current":
        return current_fib()
    elif endpoint == "/next":
        return next_fib()
    elif endpoint == "/previous":
        return prev_fib()
    else:
        return "Invalid endpoint"

if __name__ == '__main__':
    # Continuously prompt the user for input and handle requests
    while True:
        user_input = input("Enter endpoint (current/next/previous): ")
        print(handle_request(user_input))
