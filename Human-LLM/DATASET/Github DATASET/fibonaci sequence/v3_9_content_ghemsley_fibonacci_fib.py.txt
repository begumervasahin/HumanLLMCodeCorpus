import argparse
from math import sqrt

def get_fibonacci_index():
    """
    Prompt the user to enter the index of the Fibonacci sequence.

    Returns:
        int: The index of the Fibonacci sequence entered by the user.
    """
    while True:
        try:
            fib_index = abs(int(input("Please enter the index of the Fibonacci sequence: ")))
            return fib_index
        except ValueError:
            print("Invalid input. Please enter a positive integer.")

def calculate_fibonacci(n):
    """
    Calculate the Fibonacci number at the given index.

    Args:
        n (int): The index of the Fibonacci sequence.

    Returns:
        int: The Fibonacci number at the given index.
    """
    phi = (1 + sqrt(5)) / 2
    return int((phi**n - (-1/phi)**n) / sqrt(5))

def main():
    """
    Parse command-line arguments and calculate Fibonacci number.
    """
    parser = argparse.ArgumentParser(description="Calculate the Fibonacci sequence value at a given index.")
    parser.add_argument("-i", "--index", type=int, help="Index of the Fibonacci sequence")
    args = parser.parse_args()

    if args.index is not None:
        fib_index = abs(args.index)
    else:
        fib_index = get_fibonacci_index()

    fib_value = calculate_fibonacci(fib_index)
    print(f"Fibonacci sequence value at index {fib_index}: {fib_value}")

if __name__ == "__main__":
    main()
