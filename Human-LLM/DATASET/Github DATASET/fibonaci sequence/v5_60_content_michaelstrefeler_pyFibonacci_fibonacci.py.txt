from turtle import *

def get_valid_input():
    user_input = input('Choose the number of Fibonacci sequence elements (min 5): ')
    while not user_input.isdecimal() or int(user_input) < 5:
        user_input = input('Please enter a number greater than or equal to 5: ')
    return int(user_input)

def generate_fibonacci_sequence(amount):
    fibonacci_sequence = [0, 1]
    while len(fibonacci_sequence) < amount:
        next_number = fibonacci_sequence[-1] + fibonacci_sequence[-2]
        fibonacci_sequence.append(next_number)
    return fibonacci_sequence

def print_fibonacci_sequence(sequence):
    for index, number in enumerate(sequence):
        print(f'Fibonacci {index}: {number}')

def draw_fibonacci_sequence(sequence):
    penup()
    goto(135.00, -20.00)
    pendown()
    first_one = True
    for fib in sequence:
        if fib == 0:
            continue
        elif fib == 1:
            if first_one:
                forward(12.5)
                write(fib)
                forward(12.5)
                for _ in range(3):
                    left(90)
                    forward(25)
                first_one = False
            else:
                for _ in range(2):
                    forward(25)
                forward(12.5)
                write(fib)
                forward(12.5)
                for _ in range(3):
                    left(90)
                    forward(25)
        else:
            for _ in range(2):
                forward(12.5 * fib)
                left(90)
            forward(12.5)
            write(fib)
            forward(12.5)
            left(90)

def draw_fibonacci_circles(sequence):
    penup()
    goto(135.00, 5.00)
    pendown()
    color('red')
    shape('turtle')
    pensize(2)
    setheading(180)
    for i in range(len(sequence)):
        if i < 9:
            circle(12.5 * sequence[i], 90)
        else:
            break

def main():
    amount = get_valid_input()
    fibonacci_sequence = generate_fibonacci_sequence(amount)
    print_fibonacci_sequence(fibonacci_sequence)
    draw_fibonacci_sequence(fibonacci_sequence)
    draw_fibonacci_circles(fibonacci_sequence)
    done()

if __name__ == "__main__":
    main()
