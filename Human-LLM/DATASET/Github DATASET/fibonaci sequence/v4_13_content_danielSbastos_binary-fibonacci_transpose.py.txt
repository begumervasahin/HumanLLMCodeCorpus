import numpy as np
import matplotlib.pyplot as plt

# Function to generate Fibonacci numbers iteratively
def fibonacci_iterative(n):
    a, b = 0, 1
    while n > 0:
        a, b = b, a + b
        n -= 1
    return a

# Function to format a number into an 80-bit binary representation
def format_binary(number):
    return format(number, '080b')

# Number of Fibonacci numbers to generate
n = 100

# Generate Fibonacci numbers and convert them to binary
fibonacci_list = [fibonacci_iterative(i) for i in range(n + 1)]
binary_list = [format_binary(num) for num in fibonacci_list]

# Split each binary number into a list of digits
binary_split_list = [[int(bit) for bit in binary] for binary in binary_list]

# Create a numpy array from the split binary lists
binary_array = np.array(binary_split_list)

# Transpose the binary array
transposed_binary_array = np.transpose(binary_array)

# Plot the transposed binary array
plt.imshow(transposed_binary_array, interpolation='nearest')
plt.show()
