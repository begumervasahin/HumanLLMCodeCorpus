def classic_euclid_recursive(a, b):
    # Base case: if b is 0, return a; if a is 0, return b
    if b == 0:
        return a
    if a == 0:
        return b
    # Recursive case: subtract smaller from larger until they're equal
    return classic_euclid_recursive(a - min(a, b), b - min(a, b))

def modern_euclid_recursive(a, b):
    # Base case: if b is 0, return a
    if b == 0:
        return a
    # Recursive case: call the function with b and the remainder of a divided by b      
    return modern_euclid_recursive(b, a % b)

if __name__ == "__main__":
    while True:
        # Prompt the user to choose between classic and modern Euclid algorithm
        alg = input("Classic (c) or modern Euclid (e) algorithm? ")
        # Validate user input
        if alg.lower() not in ['c', 'e']:
            print("Invalid input. Please choose 'c' or 'e'.")
            continue
        while True:
            # Prompt the user to enter a number or exit
            n = input("Enter 'exit' to choose another algorithm. Otherwise, enter a number: ")
            if n.lower() == 'exit':
                break
            elif n.isnumeric():
                # Convert input to integers
                a = int(n)
                b = int(input("Enter the second number: "))
                # Calculate GCD using the chosen algorithm
                gcd = classic_euclid_recursive(a, b) if alg == "c" else modern_euclid_recursive(a, b)
                # Display the result
                print(f"GCD of {a} and {b} is: {gcd}")
            else:
                print("Invalid input. Please enter a number or 'exit'.")
