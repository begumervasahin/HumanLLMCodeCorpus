import os
import datetime

def fibonacci_recursive(n):
    """Calculate the nth Fibonacci number using a recursive method."""
    if n <= 2:
        return 1
    else:
        return fibonacci_recursive(n - 1) + fibonacci_recursive(n - 2)

def get_integer_input(prompt):
    while True:
        try:
            value = int(input(prompt))
            return value
        except ValueError:
            print('\nYou must enter an integer.\n')

def n_times_fibonacci(n):
    """Calculate Fibonacci sequence n times."""
    num = 1
    next_num = 1
    print(num)

    for _ in range(n - 1):
        print(next_num)
        num, next_num = next_num, num + next_num

    print('\nS    - Save to a file')
    print('else - Go to the main menu')
    choice = input('==> ')

    return choice.lower() == 's'

def fibonacci_up_to(maxnum):
    """Calculate Fibonacci sequence up to a certain number."""
    num = 1
    next_num = 1
    itera = 0
    print(num)

    while next_num <= maxnum:
        print(next_num)
        num, next_num = next_num, num + next_num
        itera += 1

    print('\nS    - Show more information')
    print('else - Go to the main menu')
    choice = input('==> ')

    if choice.lower() == 's':
        if num == maxnum:
            print("The number", maxnum, "is in the Fibonacci sequence")
        print("Number of iterations:       ", itera)
        print("Your number:                ", maxnum)
        print("Next number in the sequence:", next_num)
        print("Difference to that number:   ", (next_num - maxnum))
        print("\n\n")

def save_to_file(n):
    name = input('Name of the file: ').replace(" ", "") + '.txt'
    with open(name, "w") as file:
        conc = "Number of iterations: " + str(n) + '\n'
        file.write(conc)
        num = 1
        next_num = 1

        for _ in range(n):
            str_num = str(num) + '\n'
            file.write(str_num)
            num, next_num = next_num, num + next_num

    return name

def get_greeting():
    hour = datetime.datetime.now().hour
    if hour >= 20:
        return 'night!'
    elif hour >= 12:
        return 'afternoon!'
    elif hour >= 5:
        return 'day!'
    else:
        return 'night!'

def clear_screen():
    os.system('cls' if os.name == 'nt' else 'clear')

print(' __________________________')
print('|                          |')
print('|** Fibonacci calculator **|')
print('|__________________________|\n\n')

while True:
    print('1 - Calculate Fibonacci sequence n times')
    print('2 - Calculate Fibonacci sequence up to a certain number')
    print('T - Terminate')

    choice = input('==> ').upper()
    clear_screen()

    if choice == '1':
        n = get_integer_input('Number of times: ')
        if n > 0:
            if n_times_fibonacci(n):
                name = save_to_file(n)
                clear_screen()
                print('\nFile saved as:', name, '\n\n')
    elif choice == '2':
        maxnum = get_integer_input('Calculate up to the number: ')
        if maxnum >= 1:
            fibonacci_up_to(maxnum)
    elif choice == 'T':
        greeting = get_greeting()
        print(f'\nThank you, have a good {greeting}')
        break
    else:
        print('\nInvalid option\n')
