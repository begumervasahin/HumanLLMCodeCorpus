# Importing the necessary function from the difflib module
from difflib import ndiff

# Defining two lists of strings
text1 = 'one\ntwo\nthree\n'.splitlines(keepends=True)
text2 = 'ore\ntree\nemu\n'.splitlines(keepends=True)

# Using the ndiff function to find the differences between the two lists
diff = ndiff(text1, text2)

# Joining the differences and printing them
print(''.join(diff), end="")
