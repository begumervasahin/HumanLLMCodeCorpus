# Importing SequenceMatcher from difflib
from difflib import SequenceMatcher

# Defining two strings to compare
string1 = "private Thread currentThread;"
string2 = "private volatile Thread currentThread;"

# Creating a SequenceMatcher object with a custom lambda function to consider spaces as delimiters
matcher = SequenceMatcher(lambda x: x == " ", string1, string2)

# Printing the similarity ratio between the two strings
print(round(matcher.ratio(), 3))

# Iterating over matching blocks and printing details
for block in matcher.get_matching_blocks():
    print("In both strings, a[%d] and b[%d] match for %d elements" % block)

# Iterating over operations and printing details
for opcode in matcher.get_opcodes():
    print("%6s for elements a[%d:%d] and b[%d:%d]" % opcode)
