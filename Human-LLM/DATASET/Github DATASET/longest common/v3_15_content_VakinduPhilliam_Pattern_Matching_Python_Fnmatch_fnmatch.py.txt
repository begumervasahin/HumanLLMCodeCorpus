import fnmatch
import os

# Define the directory to search for text files
directory_to_search = '.'

# Iterate over each file in the directory
for file_name in os.listdir(directory_to_search):
    # Check if the file matches the '*.txt' pattern
    if fnmatch.fnmatch(file_name, '*.txt'):
        # Print a message indicating the discovery of a text file
        print("Found a text file:", file_name)
