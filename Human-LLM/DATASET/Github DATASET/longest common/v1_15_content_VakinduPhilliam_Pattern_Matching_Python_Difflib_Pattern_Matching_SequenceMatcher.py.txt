from difflib import SequenceMatcher

# Define the strings to be compared
string1 = "private Thread currentThread;"
string2 = "private volatile Thread currentThread;"

# Create a SequenceMatcher object
s = SequenceMatcher(lambda x: x == " ", string1, string2)

# Print the ratio of similarity
print(round(s.ratio(), 3))

# Print matching blocks
for block in s.get_matching_blocks():
    print("a[%d] and b[%d] match for %d elements" % block)

# Print the operations to transform string1 into string2
for opcode in s.get_opcodes():
    print("%6s a[%d:%d] b[%d:%d]" % opcode)
