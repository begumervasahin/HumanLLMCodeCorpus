import glob  # Import the glob module for file searching

# Search for GIF files in the current directory
gif_files = glob.glob('*.gif')
print("GIF files found in this folder:")
print(gif_files)

# Search for files starting with 'c' in the current directory
c_files = glob.glob('.c*')
print("\nFiles starting with 'c' found in this folder:")
print(c_files)
