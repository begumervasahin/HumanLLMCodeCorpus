# Import the glob module
import glob

# Find files in the current directory that start with a single digit followed by any character
files_with_single_digit = glob.glob('./[0-9].*')

# Find files in the current directory with a .gif extension
gif_files_in_current_dir = glob.glob('*.gif')

# Find files in the current directory with a single-character filename and a .gif extension
single_char_gif_files = glob.glob('?.gif')

# Find all .txt files in the current directory and its subdirectories
all_txt_files_recursive = glob.glob('**/*.txt', recursive=True)

# Find all directories in the current directory and its subdirectories
all_directories_recursive = glob.glob('./**/', recursive=True)
