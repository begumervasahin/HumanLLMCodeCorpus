import glob

# Find files with names starting with a single digit in the current directory
files_with_single_digit = glob.glob('./[0-9].*')

# Find all GIF files in the current directory
gif_files_in_current_dir = glob.glob('*.gif')

# Find GIF files with single-character filenames in the current directory
single_char_gif_files = glob.glob('?.gif')

# Find all TXT files recursively in the current directory and its subdirectories
all_txt_files_recursive = glob.glob('**/*.txt', recursive=True)

# Find all directories recursively in the current directory and its subdirectories
all_directories_recursive = glob.glob('./**/', recursive=True)
