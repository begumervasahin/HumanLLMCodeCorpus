# Import the necessary modules for working with files and patterns
import fnmatch
import os

# Define the directory we want to search for files in
directory_to_search = '.'

# Look through all the files in the directory
for filename in os.listdir(directory_to_search):
    # Check if the filename matches the pattern '*.txt'
    if fnmatch.fnmatch(filename, '*.txt'):
        # If it does, print the filename
        print("Found a text file:", filename)
