# Define a function that checks if a character is either a space or a tab
def is_space_or_tab(x):
    return x in " \t"

# Create a SequenceMatcher object without any special function for comparison
s = SequenceMatcher(None, " abcd", "abcd abcd")
# Find the longest matching subsequence between two strings within specified ranges
longest_match = s.find_longest_match(0, 5, 0, 9)

# Create a SequenceMatcher object with a custom function for comparison
s = SequenceMatcher(is_space_or_tab, " abcd", "abcd abcd")
# Find the longest matching subsequence between two strings within specified ranges
longest_match_custom = s.find_longest_match(0, 5, 0, 9)

# Create a SequenceMatcher object without any special function for comparison
s = SequenceMatcher(None, "abxcd", "abcd")
# Get the matching blocks between two strings
matching_blocks = s.get_matching_blocks()

# Define two strings
a = "qabxcd"
b = "abycdf"
# Create a SequenceMatcher object between the two strings
s = SequenceMatcher(None, a, b)
# Iterate over the operations between the two strings
for tag, i1, i2, j1, j2 in s.get_opcodes():
    print('{:7}   a[{}:{}] --> b[{}:{}] {!r:>8} --> {!r}'.format(        
        tag, i1, i2, j1, j2, a[i1:i2], b[j1:j2]))
