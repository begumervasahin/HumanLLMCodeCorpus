import fnmatch
import re

# Define the file pattern we want to match
file_pattern = '*.txt'

# Translate the file pattern into a regular expression pattern
regex_pattern = fnmatch.translate(file_pattern)

# Compile the regular expression pattern into a regex object
regex_object = re.compile(regex_pattern)

# Check if the filename "foobar.txt" matches the regex pattern
is_match = regex_object.match('foobar.txt')
