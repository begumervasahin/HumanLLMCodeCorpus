# Importing the dircmp function from the filecmp module
from filecmp import dircmp

# Defining a function to print the differing files between two directories
def print_diff_files(dcmp):
    # Loop through the differing files
    for name in dcmp.diff_files:
        # Print the name of the differing file and the directories it's found in
        print(f"Differing file '{name}' found in '{dcmp.left}' and '{dcmp.right}'")
    
    # Recursively call the function for subdirectories
    for sub_dcmp in dcmp.subdirs.values():
        print_diff_files(sub_dcmp)

# Comparing two directories
dcmp = dircmp('dir1', 'dir2')

# Calling the function to print the differing files
print_diff_files(dcmp)
