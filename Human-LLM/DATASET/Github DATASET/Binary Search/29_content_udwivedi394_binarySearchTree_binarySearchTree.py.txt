class Node:
	def __init__(self,data):
		self.data = data
		self.left = None
		self.right = None

	def addNode(self,data):
		if self.root == None:
			self.root = Node(data)
			return
		temp = self.root
		while temp:
			prevNode = temp
			if data < temp.data:
				temp = temp.left
			elif data >= temp.data:
				temp = temp.right
		if data < prevNode.data:
			prevNode.left = Node(data)
		else:
			prevNode.right = Node(data)
		return
