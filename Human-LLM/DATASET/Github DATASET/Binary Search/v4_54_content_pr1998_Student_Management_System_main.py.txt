import time
import bst  # Assuming this is a custom binary search tree module

def main():
    # Input the number of students
    num_students = int(input("Enter the number of students: "))

    # Create an instance of the Binary Search Tree
    binary_tree = bst.BST()

    # Input details for each student and insert them into the BST
    for i in range(num_students):
        print(f"\nENTER THE DETAILS OF STUDENT {i + 1}:")
        roll_num = int(input("Enter the roll number: "))
        name = input("Enter name: ")
        address = input("Enter the address: ")
        mobile_num = int(input("Enter the mobile number: "))
        course = input("Enter the course: ")
        
        print('Enter the marks in three subjects:')
        m1 = int(input("First subject: "))
        m2 = int(input("Second subject: "))
        m3 = int(input("Third subject: "))
        total_marks = m1 + m2 + m3
        average_marks = total_marks / 3
        
        print("Total Marks: ", total_marks)
        print("Average Marks: ", average_marks)

        # Insert student details into the BST
        binary_tree.insert(roll_num)

    print("\nBST TRAVERSAL\n")
    binary_tree.inorder()

    # Ask user if they want to delete keys in the tree
    delete_choice = input("\nDo you want to delete any keys in the tree? (Enter 'y' or 'Y'): ")
    if delete_choice.lower() == 'y':
        while delete_choice.lower() == 'y':
            delete_key = int(input("\nEnter the number to be deleted: "))
            binary_tree.remove(delete_key)
            delete_choice = input("Enter 'y' or 'Y' to continue deleting: ")

        print("\nAFTER THE DELETION :\n")
        binary_tree.inorder()

    # Ask user if they want to search for keys in the tree
    search_choice = input("\nDo you want to search any keys in the tree? (Enter 'y' or 'Y'): ")
    while search_choice.lower() == 'y':
        search_key = int(input("\nEnter the key to be searched: "))
        binary_tree.search(search_key)
        search_choice = input("Enter 'y' or 'Y' to continue searching: ")

    print("\n*************************END OF BST OPERATIONS*******************************\n")

if __name__ == '__main__':
    main()
