class BinarySearchTree:
    def __init__(self):
        self._size = 0  # Initialize the size of the BST to zero
        self._root = None  # Initialize the root of the BST to None

    class _BSTNode:
        def __init__(self, key, value):
            self.key = key
            self.value = value
            self.left = None
            self.right = None

    def add(self, key, value):
        # Add a new node with the given key and value to the BST
        z = self._BSTNode(key, value)
        y = None
        x = self._root
        while x is not None:
            y = x
            if key < x.key:
                x = x.left
            else:
                x = x.right
        if y is None:
            self._root = z  # If the BST is empty, set the new node as the root
        elif key < y.key:
            y.left = z  # Add the new node as the left child of y
        else:
            y.right = z  # Add the new node as the right child of y
        self._size += 1  # Increment the size of the BST

    def smallest(self):
        # Find and return the smallest key in the BST
        if self._root is not None:
            current = self._root
            while current.left is not None:
                current = current.left
            return current.key

    def largest(self):
        # Find and return the largest key in the BST
        if self._root is not None:
            current = self._root
            while current.right is not None:
                current = current.right
            return current.key

    def search(self, key):
        # Search for a key in the BST and return True if found, False otherwise
        found = []
        self._search(self._root, key, found)
        return len(found) > 0

    def _search(self, subtree, key, found):
        # Recursive helper function to search for a key in the BST
        if subtree:
            if key == subtree.key:
                found.append(1)
            elif key < subtree.key:
                self._search(subtree.left, key, found)
            elif key > subtree.key:
                self._search(subtree.right, key, found)

    def is_empty(self):
        # Check if the BST is empty
        return self._size == 0

    def size(self):
        # Return the size of the BST
        return self._size

    def inorder_walk(self):
        # Perform an inorder traversal of the BST and return the keys in sorted order
        nodes = []
        self._inorder_walk(self._root, nodes)
        return nodes

    def _inorder_walk(self, subtree, nodes):
        # Recursive helper function for inorder traversal
        if subtree:
            self._inorder_walk(subtree.left, nodes)
            nodes.append(subtree.key)
            self._inorder_walk(subtree.right, nodes)

    def preorder_walk(self):
        # Perform a preorder traversal of the BST and return the keys
        nodes = []
        self._preorder_walk(self._root, nodes)
        return nodes

    def _preorder_walk(self, subtree, nodes):
        # Recursive helper function for preorder traversal
        if subtree:
            nodes.append(subtree.key)
            self._preorder_walk(subtree.left, nodes)
            self._preorder_walk(subtree.right, nodes)

    def postorder_walk(self):
        # Perform a postorder traversal of the BST and return the keys
        nodes = []
        self._postorder_walk(self._root, nodes)
        return nodes

    def _postorder_walk(self, subtree, nodes):
        # Recursive helper function for postorder traversal
        if subtree:
            self._postorder_walk(subtree.left, nodes)
            self._postorder_walk(subtree.right, nodes)
            nodes.append(subtree.key)
