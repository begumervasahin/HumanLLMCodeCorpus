class Solution:
    def search(self, nums, target):
        # Function to find the break point where the array is rotated
        def find_break_point(start, end, nums):
            mid = start + (end - start) // 2
            if len(nums) < 3:
                return start  # If the array has only 2 or fewer elements, return the start index
            if nums[mid] < nums[mid - 1] and nums[mid] < nums[mid + 1]:
                return mid  # Found the break point where the array is rotated
            elif nums[mid] > nums[mid - 1] and nums[mid] > nums[mid + 1]:
                return mid  # Found the break point where the array is rotated
            elif nums[mid] > nums[len(nums) - 1]:
                return find_break_point(mid, len(nums), nums)  # Search in the right half
            elif nums[mid] < nums[0]:
                return find_break_point(start, mid, nums)  # Search in the left half
            else:
                return -1  # No rotation found

        # Binary search to find the target element
        def binary_search(start, end, target, nums):
            mid = start + (end - start) // 2
            if target == nums[mid]:
                return mid  # Target found
            if end - start < 3:
                if target == nums[start]:
                    return start  # Target found at the start index
                elif target == nums[end]:
                    return end  # Target found at the end index
                else:
                    return -1  # Target not found
            elif target > nums[mid]:
                return binary_search(mid, end, target, nums)  # Search in the right half
            else:
                return binary_search(start, mid, target, nums)  # Search in the left half

        break_point = find_break_point(0, len(nums) - 1, nums)
        if break_point == -1:
            return binary_search(0, len(nums) - 1, target, nums)  # No rotation, perform binary search
        else:
            left_search = binary_search(0, break_point - 1, target, nums)
            if left_search != -1:
                return left_search  # Target found in the left part of the array
            else:
                return binary_search(break_point, len(nums) - 1, target, nums)  # Search in the right part

# Example usage
if __name__ == "__main__":
    sol = Solution()
    nums = [4, 5, 6, 7, 0, 1, 2]
    target = 0
    result = sol.search(nums, target)
    print("Index of the target element in the rotated sorted array:", result)
