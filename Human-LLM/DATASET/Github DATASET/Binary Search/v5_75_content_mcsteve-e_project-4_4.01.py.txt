class BinarySearchTree:
    class Node:
        def __init__(self, payload):
            self.payload = payload
            self.left = None
            self.right = None

    def __init__(self):
        self.root = None

    def pretty_print(self):
        """Prints the binary search tree in a pretty format."""
        self._pretty_print_aux(self.root, 0)

    def _pretty_print_aux(self, current_node, indent_level):
        """Helper method for pretty printing."""
        if current_node.right:
            self._pretty_print_aux(current_node.right, indent_level + 1)
        print("    " * indent_level, current_node.payload)
        print()
        if current_node.left:
            self._pretty_print_aux(current_node.left, indent_level + 1)

    def find(self, target):
        """Find a node with the given payload."""
        return self._find_aux(self.root, target)

    def _find_aux(self, current_node, target):
        """Helper method for finding a node."""
        if current_node is None:
            return None
        if current_node.payload == target:
            return current_node
        elif target < current_node.payload:
            return self._find_aux(current_node.left, target)
        else:
            return self._find_aux(current_node.right, target)

    def attach_in_order(self, new_value):
        """Attach a new node with the given payload in order."""
        if self.root is None:
            self.root = self.Node(new_value)
        else:
            self._attach_in_order_aux(self.root, new_value)

    def _attach_in_order_aux(self, current_node, new_value):
        """Helper method for attaching a new node in order."""
        if new_value < current_node.payload:
            if current_node.left is None:
                current_node.left = self.Node(new_value)
            else:
                self._attach_in_order_aux(current_node.left, new_value)
        elif new_value > current_node.payload:
            if current_node.right is None:
                current_node.right = self.Node(new_value)
            else:
                self._attach_in_order_aux(current_node.right, new_value)

    def flip(self):
        """Flip the binary search tree."""
        self._flip_aux(self.root)

    def _flip_aux(self, current_node):
        """Helper method for flipping the binary search tree."""
        if current_node is None:
            return None
        current_node.left, current_node.right = current_node.right, current_node.left
        self._flip_aux(current_node.left)
        self._flip_aux(current_node.right)

def main():
    # Create a binary search tree
    tree = BinarySearchTree()
    tree.attach_in_order("man")
    tree.attach_in_order("dog")
    tree.attach_in_order("zebra")
    tree.attach_in_order("ape")
    tree.attach_in_order("elephant")
    tree.attach_in_order("yak")
    tree.attach_in_order("zorse")
    tree.attach_in_order("fly")

    # Print the binary search tree in a pretty format
    tree.pretty_print()

if __name__ == "__main__":
    main()
