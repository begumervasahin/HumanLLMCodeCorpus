class Node:
    def __init__(self, value):
        self.value = value
        self.left_child = None
        self.right_child = None

    def insert(self, data):
        """
        Inserts a value into the binary search tree.
        
        Args:
            data (int): The value to insert.
            
        Returns:
            bool: True if the value is successfully inserted, False otherwise.
        """
        if self.value == data:
            return False
        elif self.value > data:
            if self.left_child:
                return self.left_child.insert(data)
            else:
                self.left_child = Node(data)
                return True
        else:
            if self.right_child:
                return self.right_child.insert(data)
            else:
                self.right_child = Node(data)
                return True

    def find(self, data):
        """
        Finds if a value exists in the binary search tree.
        
        Args:
            data (int): The value to find.
            
        Returns:
            bool: True if the value is found, False otherwise.
        """
        if self.value == data:
            return True
        elif self.value > data:
            if self.left_child:
                return self.left_child.find(data)
            else:
                return False
        else:
            if self.right_child:
                return self.right_child.find(data)
            else:
                return False

    def preorder(self):
        """Performs a preorder traversal of the binary search tree."""
        if self:
            print(str(self.value))
            if self.left_child:
                self.left_child.preorder()
            if self.right_child:
                self.right_child.preorder()

    def inorder(self):
        """Performs an inorder traversal of the binary search tree."""
        if self:
            if self.left_child:
                self.left_child.inorder()
            print(str(self.value))
            if self.right_child:
                self.right_child.inorder()

    def postorder(self):
        """Performs a postorder traversal of the binary search tree."""
        if self:
            if self.left_child:
                self.left_child.postorder()
            if self.right_child:
                self.right_child.postorder()
            print(str(self.value))

class Tree:
    def __init__(self):
        self.root = None

    def insert(self, data):
        """Inserts a value into the binary search tree."""
        if self.root:
            return self.root.insert(data)
        else:
            self.root = Node(data)
            return True

    def find(self, data):
        """Finds if a value exists in the binary search tree."""
        if self.root:
            return self.root.find(data)
        else:
            return False

    def preorder(self):
        """Performs a preorder traversal of the binary search tree."""
        print("Preorder:")
        if self.root:
            self.root.preorder()

    def inorder(self):
        """Performs an inorder traversal of the binary search tree."""
        print("Inorder:")
        if self.root:
            self.root.inorder()

    def postorder(self):
        """Performs a postorder traversal of the binary search tree."""
        print("Postorder:")
        if self.root:
            self.root.postorder()
