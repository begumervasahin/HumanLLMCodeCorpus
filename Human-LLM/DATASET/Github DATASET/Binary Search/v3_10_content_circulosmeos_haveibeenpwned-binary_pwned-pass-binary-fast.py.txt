import os
import sys
import hashlib
import binascii
from re import match
from platform import system

# Constants
WINDOWS_ENCODING = 'cp1252' if system() == 'Windows' else 'utf-8'
HASH_LENGTH = 20  # Length of SHA1 hash in bytes
PWNED_PASSWORDS_FILES = ['pwned-passwords-ordered-2.0.bin']
DETECT_HASH = True  # Set to True to automatically detect SHA1 hash input
VERBOSE = False  # Set to True for verbose output

# Function to get the size of a file
def get_file_size(filename):
    """Returns the size of a file."""
    return os.path.getsize(filename)

# Function to search for a password in the list of hashed passwords
def search_for_password(password):
    """Searches for a password in the list of hashed passwords."""
    # Decode password based on the system encoding
    try:
        password = password.decode(WINDOWS_ENCODING).encode('utf-8')
    except UnicodeDecodeError:
        password = password.encode('utf-8')

    # Calculate the SHA1 hash of the password
    pass_hash = hashlib.sha1(password).hexdigest().upper()

    # Automatically detect if the input is a SHA1 hash
    if DETECT_HASH and len(password) == HASH_LENGTH * 2 and match(r'^[a-fA-F0-9]+$', password.decode('utf-8')):
        pass_hash = password.decode('utf-8').upper()

    # Display verbose output
    if VERBOSE:
        print("\nSearching for SHA1 hash:", pass_hash)

    # Iterate over each password file
    for password_file in PWNED_PASSWORDS_FILES:
        if VERBOSE:
            print("Searching in:", password_file)

        try:
            filesize = get_file_size(password_file)
        except FileNotFoundError:
            if VERBOSE:
                print("File not found:", password_file)
            return 1

        # Calculate the number of hashes in the file
        number_of_hashes = filesize // HASH_LENGTH

        # Display verbose output
        if VERBOSE:
            print(f"File size is: {filesize} bytes, hash length is {HASH_LENGTH} bytes, number of hashes is {number_of_hashes}")

        # Binary search for the password hash in the file
        pos_from = 0
        pos_to = number_of_hashes

        with open(password_file, "rb") as file:
            while True:
                current_pos = int((pos_to + pos_from) / 2)
                file.seek(current_pos * HASH_LENGTH)
                pwned_hash = binascii.hexlify(bytearray(file.read(HASH_LENGTH))).upper().decode('utf-8')

                # Display verbose output
                if VERBOSE:
                    print("Go to", current_pos, "hash position")

                # Check if the hash matches the password hash
                if pwned_hash == pass_hash:
                    if VERBOSE:
                        print("Found:", password.decode('utf-8'), "as", pwned_hash)
                    return 0

                # Break the loop if the search range is small
                if abs(pos_to - pos_from) < 1:
                    break

                # Update the search range
                if pwned_hash < pass_hash:
                    pos_from = current_pos
                    if abs(pos_to - pos_from) <= 1:
                        pos_from = pos_to
                else:
                    pos_to = current_pos
                    if abs(pos_to - pos_from) <= 1:
                        pos_to = pos_from

    # Display verbose output
    if VERBOSE:
        print("Not found, all clear!")

    return 1

# Main function
def main():
    """Main function to handle command-line arguments."""
    if len(sys.argv) == 2 and len(sys.argv[1]) > 0:
        exit(search_for_password(sys.argv[1]))
    else:
        print("\tUse:\n\t" + os.path.basename(__file__) + " [password|SHA1]")

if __name__ == "__main__":
    main()
