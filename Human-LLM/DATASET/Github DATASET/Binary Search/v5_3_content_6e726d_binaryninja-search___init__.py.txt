from binaryninja import *

def instruction_text_token_to_string(instruction_token_list):
    """
    Convert a list of InstructionTextToken objects into a string representing the instruction.
    """
    return " ".join(token.text.strip() for token in instruction_token_list if isinstance(token, InstructionTextToken))

def process_immediate(immediate):
    """
    Convert a string representing an immediate value into an integer.
    """
    if not isinstance(immediate, str):
        raise TypeError('Immediate must be a string object.')
    return int(immediate, 16 if immediate.startswith("0x") else 10)

def lookup_for_immediate(bv, immediate):
    """
    Search for instructions containing the specified immediate value in all functions.
    """
    results = []
    for function_item in bv.functions:
        for block in function_item:
            for instruction in block:
                for token in instruction:
                    if token.type == InstructionTextTokenType.PossibleAddressToken and token.value == immediate:
                        instruction_str = instruction_text_token_to_string(instruction.tokens)
                        results.append((block.start, function_item.name, instruction_str))
    return results

def show_results_report(bv, value, results):
    """
    Show an HTML report with the search results.
    """
    table_header = "\n\t\t\t<tr>\n\t\t\t\t<th width=\"150\">Address</th>\n\t\t\t\t<th width=\"150\">Function</th>\n\t\t\t\t<th>Instruction</th>\n\t\t\t</tr>"
    rows = "".join(f"\n\t\t\t<tr>\n\t\t\t\t<td><pre>0x{address:016X}</pre></td>\n\t\t\t\t<td><pre>{function_name}</pre></td>\n\t\t\t\t<td><pre>{instruction_str}</pre></td>\n\t\t\t</tr>" for address, function_name, instruction_str in results)
    html = f"<!DOCTYPE html>\n<html>\n\t<body>\n\t\t<table>{table_header}{rows}\n\t\t</table>\n\t</body>\n</html>"
    bv.show_html_report(f"Search immediate - 0x{value:X}", html)

def do_stuff(bv):
    immediate_str = get_text_line_input('Value to search', 'Search Immediate')
    immediate = process_immediate(immediate_str)
    results = lookup_for_immediate(bv, immediate)
    show_results_report(bv, immediate, results)

name = "Search Immediate"
description = "Search for a specific value in the instruction operands."
PluginCommand.register(name, description, do_stuff)
