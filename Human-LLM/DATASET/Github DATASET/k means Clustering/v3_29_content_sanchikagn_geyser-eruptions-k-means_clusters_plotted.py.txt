import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans

# Step 1: Read data
data_frame = pd.read_csv('C:/Users/Admin/PycharmProjects/GeyserEruption/resources/faithful_geyser.csv')

# Step 2: Standardize the data
scaler = StandardScaler()
X_std = scaler.fit_transform(data_frame)

# Step 3: Perform KMeans clustering
kmeans = KMeans(n_clusters=2, max_iter=100)
kmeans.fit(X_std)

# Step 4: Get cluster centroids
centroids = kmeans.cluster_centers_

# Step 5: Plotting
plt.figure(figsize=(8, 8))

# Plot points for each cluster
plt.scatter(X_std[kmeans.labels_ == 0, 0], X_std[kmeans.labels_ == 0, 1], c='green', label='Cluster 1')
plt.scatter(X_std[kmeans.labels_ == 1, 0], X_std[kmeans.labels_ == 1, 1], c='blue', label='Cluster 2')

# Plot centroids
plt.scatter(centroids[:, 0], centroids[:, 1], marker='X', s=100, c='red', label='Centroid')

# Plot settings
plt.legend()
plt.xlim([-2, 2])
plt.ylim([-2, 2])
plt.xlabel('Eruption time (minutes)', fontsize=16)
plt.ylabel('Waiting time to next eruption (minutes)', fontsize=16)
plt.title('Clustered Data from Old Faithful Geyser, \n Yellowstone, USA', fontsize=20, fontweight='bold')
plt.gca().set_aspect('equal', adjustable='box')

# Step 6: Save plot
plt.savefig('plot-clusters.png')

# Step 7: Show plot
plt.show()
