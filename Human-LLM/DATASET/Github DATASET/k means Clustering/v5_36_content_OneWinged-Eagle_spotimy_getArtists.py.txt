# Importing necessary libraries
import spotipy
import spotipy.util as util
import config
import json
from time import perf_counter
from sys import stderr

# Attempting to retrieve token
try:
    token = util.prompt_for_user_token(config.username, config.scope,
                                       config.client_id, config.client_secret,
                                       config.redirect_uri)
    # Creating a Spotify object with the obtained token
    sp = spotipy.Spotify(token)
except:
    # Printing an error message if token retrieval fails
    print(f"Token is not accessible for {config.username}", file=stderr)

# Function to retrieve followed artists
def retrieve_artists() -> list:
    artists = []  # List to store retrieved artists
    last_artist_id = None  # Variable to track the last artist ID   
    # Looping until all followed artists are retrieved
    while True:
        # Retrieving followed artists in batches of 50
        items = sp.current_user_followed_artists(50, last_artist_id)["artists"]["items"]
        # Checking if no more artists are left
        if not items:
            break
        # Updating last artist ID for the next batch
        last_artist_id = items[-1]["id"]
        # Extending the list of artists with the retrieved batch  
        artists.extend(items)
    return artists

# Main function to retrieve and save artists' data
def main():
    print("Start retrieving artists...")
    start_time = perf_counter()  # Recording start time
    artists = retrieve_artists()  # Retrieving artists
    elapsed_time = perf_counter() - start_time  # Calculating elapsed time  
    # Printing the number of retrieved artists and elapsed time       
    print(f"Retrieved {len(artists)} artists in {elapsed_time:.2f}s.")     

    # Writing retrieved artists to a JSON file
    with open("json/artists.json", "w") as outfile:
        json.dump(artists, outfile)

# Entry point of the program
if __name__ == "__main__":
    main()
