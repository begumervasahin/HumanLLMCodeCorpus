import random
import numpy as np
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
from sklearn.datasets.samples_generator import make_blobs

# Setting seed for reproducibility
np.random.seed(0)

# Generating synthetic data
X, y = make_blobs(n_samples=5000, centers=[[4,4], [-2, -1], [2, -3], [1, 1]], cluster_std=0.9)

# Plotting the generated data
plt.scatter(X[:, 0], X[:, 1], marker='.')

# Initializing KMeans clustering
k_means = KMeans(init="k-means++", n_clusters=4, n_init=12)
k_means.fit(X)

# Extracting cluster labels and cluster centers
k_means_labels = k_means.labels_
k_means_cluster_centers = k_means.cluster_centers_

# Generating colors for clusters
colors = plt.cm.Spectral(np.linspace(0, 1, len(set(k_means_labels))))

# Creating a figure for plotting
fig = plt.figure(figsize=(6, 4))
ax = fig.add_subplot(1, 1, 1)

# Plotting each cluster
for k, col in zip(range(len([[4, 4], [-2, -1], [2, -3], [1, 1]])), colors):
    # Members of the cluster
    my_members = (k_means_labels == k)
    cluster_center = k_means_cluster_centers[k]
    
    # Plotting data points in the cluster
    ax.plot(X[my_members, 0], X[my_members, 1], 'w', markerfacecolor=col, marker='.')
    
    # Plotting cluster center
    ax.plot(cluster_center[0], cluster_center[1], 'o', markerfacecolor=col, markeredgecolor='k', markersize=6)

# Setting title and removing axis ticks
ax.set_title('KMeans')
ax.set_xticks(())
ax.set_yticks(())

# Displaying the plot
plt.show()
