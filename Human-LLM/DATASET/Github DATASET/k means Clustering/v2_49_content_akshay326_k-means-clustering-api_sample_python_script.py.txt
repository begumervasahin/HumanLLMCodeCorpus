from PIL import Image
from sklearn.metrics.pairwise import cosine_similarity
from img_to_vec import Img2Vec
import numpy as np
import http.client
import os
import csv
import base64

# Dictionary to store image vectors
image_vectors = {}

def read_from_csv():
    # Read image vectors from CSV files in the './csv' directory
    for filename in os.listdir('./csv'):
        with open(os.path.join('./csv', filename), 'r') as csvfile:
            csvreader = csv.reader(csvfile)
            for row in csvreader:
                # Convert string representation of vector to numpy array and store in dictionary
                image_vectors[row[1]] = np.fromstring(row[2][1:-1], dtype=np.float_, sep=' ')

def add_to_dict(image_rel_path):
    # Create an instance of Img2Vec for extracting image vectors
    img2vec = Img2Vec()
    # Decode relative path of image
    filename = os.fsdecode(image_rel_path)
    print("Processing image: %s" % filename)
    # Open image using PIL
    img = Image.open(os.path.join('.', filename))
    # Extract vector representation of image
    vec = img2vec.get_vec(img)
    # Store image vector in dictionary
    image_vectors[filename] = vec

def search_offline():
    # Prompt user to enter the relative path of the image to search
    image_rel_path = str(input("Enter the relative path of the image to search:\n"))
    # Add image to dictionary of image vectors
    add_to_dict(image_rel_path)
    # Read image vectors from CSV files
    read_from_csv()
    try:
        # Calculate cosine similarity between the input image and other images
        similarities = {}
        for key in list(image_vectors.keys()):
            if key == image_rel_path:
                continue
            similarities[key] = cosine_similarity(image_vectors[image_rel_path].reshape((1, -1)), image_vectors[key].reshape((1, -1)))[0][0]
        # Sort images based on similarity and print top 5 results
        sorted_similarities = sorted(similarities.items(), key=lambda x: x[1], reverse=True)
        for image, similarity in sorted_similarities[:5]:
            print("Similarity:", similarity, "Image:", image)
    except KeyError as e:
        print('Error: Image not found - %s' % e)
    except Exception as e:
        print('Error:', e)

def search_online():
    # Prompt user to enter the relative path of the image to search
    image_rel_path = str(input("Enter the relative path of the image to search:\n"))
    # Open and read image file in binary mode
    with open(image_rel_path, "rb") as image_file:
        # Encode image file content as base64
        encoded_string = base64.b64encode(image_file.read())
    # Create HTTP connection to the online service
    conn = http.client.HTTPConnection("beard-app.herokuapp.com")
    # Prepare payload and headers
    payload = encoded_string.decode('utf-8')
    headers = {
        'Content-Type': "text/plain"
    }
    # Send HTTP request to the online service
    conn.request("PUT", "image_clustering", payload, headers)
    # Get HTTP response
    res = conn.getresponse()
    # Read and print response data
    data = res.read()
    print(data.decode("utf-8"))

if __name__ == '__main__':
    search_online()
