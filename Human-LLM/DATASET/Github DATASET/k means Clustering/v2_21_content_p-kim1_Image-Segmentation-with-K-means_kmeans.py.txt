import numpy as np

class KMeans:
    """Implementation of the k-Means algorithm"""
    
    def __init__(self, k, data):
        # Initialize KMeans object with the number of clusters (k) and input data
        self.nData = np.shape(data)[0]  # Number of data points
        self.nDim = np.shape(data)[1]   # Dimensionality of the data
        self.k = k                      # Number of clusters
    
    def kmeans_train(self, data, max_iterations=10):
        # Train the KMeans model using the input data
        minima = data.min(axis=0)       # Minimum values along each dimension
        maxima = data.max(axis=0)       # Maximum values along each dimension
        
        # Initialize cluster centers randomly within the data range
        self.centres = np.random.rand(self.k, self.nDim) * (maxima - minima) + minima
        
        # Random initialization for oldCentres
        old_centres = np.random.rand(self.k, self.nDim) * (maxima - minima) + minima
        
        # Iterate until convergence or reaching maximum iterations
        iteration_count = 0
        while np.sum(np.sum(old_centres - self.centres)) != 0 and iteration_count < max_iterations:
            old_centres = self.centres.copy()  # Store the current centers for convergence check
            iteration_count += 1  # Increment iteration counter
            
            # Compute distances between data points and cluster centers
            distances = np.ones((self.k, self.nData)) * np.sum((data - self.centres[0, :]) ** 2, axis=1)
            for j in range(1, self.k):
                distances[j, :] = np.sum((data - self.centres[j, :]) ** 2, axis=1)
            
            # Assign data points to the nearest cluster
            cluster = np.argmin(distances, axis=0)
            
            # Update cluster centers based on assigned data points
            for j in range(self.k):
                this_cluster = (cluster == j)
                if np.sum(this_cluster) > 0:
                    self.centres[j, :] = np.sum(data[this_cluster, :], axis=0) / np.sum(this_cluster)
        
        # Return the final cluster centers
        return self.centres
    
    def kmeans_fwd(self, data):
        # Predict clusters for new data points using the trained model
        n_data = np.shape(data)[0]  # Number of new data points
        
        # Compute distances between new data points and cluster centers
        distances = np.ones((self.k, n_data)) * np.sum((data - self.centres[0, :]) ** 2, axis=1)
        for j in range(1, self.k):
            distances[j, :] = np.sum((data - self.centres[j, :]) ** 2, axis=1)
        
        # Assign new data points to the nearest cluster
        cluster = np.argmin(distances, axis=0)
        
        # Return the predicted clusters
        return cluster

# Example usage:
if __name__ == "__main__":
    # Generate sample data
    np.random.seed(42)
    data = np.random.rand(100, 2)
    
    # Initialize and train kmeans
    kmeans = KMeans(k=3, data=data)
    centers = kmeans.kmeans_train(data)
    print("Final cluster centers:")
    print(centers)
    
    # Predict clusters for new data
    new_data = np.random.rand(10, 2)
    clusters = kmeans.kmeans_fwd(new_data)
    print("Clusters for new data:")
    print(clusters)
