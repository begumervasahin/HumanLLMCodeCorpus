import random
import numpy as np
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans

# Set the style for plotting
plt.style.use('ggplot')

# Function to generate a random integer within a specified range  
def get_random_int(min_value=1, max_value=20):
    return random.randint(min_value, max_value)

# Function to plot the data points and centroids
def plot_data(features, labels, centroids):
    # Define colors for different clusters
    colors = ["g.", "r.", "c.", "y."]

    # Plot each data point with its assigned color
    for i, feature in enumerate(features):
        plt.plot(feature[0], feature[1], colors[labels[i]], markersize=10)

    # Plot centroids with 'x' marker
    plt.scatter(centroids[:, 0], centroids[:, 1], marker="x", s=150, linewidths=5, zorder=10)

    # Display the plot
    plt.show()

# Generate sample data points
features = np.array([[get_random_int(), get_random_int()] for _ in range(6)])       

# Create a KMeans classifier with 2 clusters
classifier = KMeans(n_clusters=2)

# Fit the classifier to the data
classifier.fit(features)

# Get the centroids and labels
centroids = classifier.cluster_centers_
labels = classifier.labels_

# Plot the data points and centroids
plot_data(features, labels, centroids)
