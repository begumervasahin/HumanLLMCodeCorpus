import csv
import random

class DataGenerator:
    def __init__(self, dimensions, num_rows, lower_bound, upper_bound, output_file):
        # Initialize the DataGenerator object with provided parameters
        self.dimensions = dimensions
        self.num_rows = num_rows
        self.lower_bound = lower_bound
        self.upper_bound = upper_bound
        self.output_file = output_file

    def write_data(self):
        # Write randomly generated data points to a CSV file
        with open(self.output_file, 'w', newline='\n') as csvfile:
            # Create a CSV writer object
            writer = csv.writer(csvfile, delimiter=',')
            # Generate and write data points for the specified number of rows
            for row in range(self.num_rows):
                # Generate a data point with random values within specified bounds for each dimension
                data_point = [random.uniform(self.lower_bound, self.upper_bound) for _ in range(self.dimensions)]
                # Write the data point to the CSV file
                writer.writerow(data_point)

if __name__ == "__main__":
    # Generate random parameters
    dimensions = random.randint(2, 5)
    num_rows = random.randint(10000, 100000)
    lower_bound = -10
    upper_bound = 10
    output_file = "data.csv"
    # Create a DataGenerator object with the generated parameters
    data_generator = DataGenerator(dimensions, num_rows, lower_bound, upper_bound, output_file)
    # Generate and write data to the CSV file
    data_generator.write_data()
