# Function to calculate word frequency in English sentences
def word_freq(sents_list: list):
    word_freq = {}
    f_word = []

    # Iterating through each sentence in the list
    for sentence in sents_list:
        sentence = str(sentence)

        # Removing a specific phrase from sentences if present
        if "(Read more about how we rate cars.)" in sentence:
            sentence = sentence.replace("(Read more about how we rate cars.)", "")

        sentence = sentence.lower()

        # Removing punctuation from the sentence
        punctuation = '[,，。.!?:"@'
        sentence = str(re.sub(punctuation, '', sentence))

        # Iterating through each word in the sentence
        for word in sentence.split(" "):
            if word in f_word:
                word_freq[word] += 1
            else:
                word_freq[word] = 1
                f_word.append(word)
    return word_freq

# Function to calculate word frequency in English sentences from the database
def en_freq():
    sql = "select data from RAW_DATA"
    cursor.execute(sql)
    result = list(cursor.fetchall())
    cursor.close()
    conn.close()

    # Calculating word frequency using the word_freq function
    word_freq_dict = word_freq(result)

    # Sorting the word frequency dictionary
    word_freq_sort = sorted(word_freq_dict.items(), key=lambda x: x[1], reverse=True)
    index = 1

    # Writing the sorted word frequency to a text file
    with open('./word_freq_en_v2.txt', 'w') as f:
        for (key, value) in word_freq_sort:
            line = str(index) + " : " + key + " : " + str(value) + "\n"
            index += 1
            f.write(line)
