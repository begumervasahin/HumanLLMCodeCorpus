# Import necessary libraries
import numpy as np
import matplotlib.pyplot as plt
from sklearn import manifold
from scipy.io import loadmat
import seaborn as sns

# Set seaborn style and context for plots
sns.set(style='white', context='notebook', rc={'figure.figsize': (14, 10)})

# Load data from the file 'Parekh_all2.mat'
annots = loadmat('Parekh_all2.mat')

# Extract relevant data
Xn = annots['Xn']
IDX = annots['IDX_new'].flatten()  # Flatten to 1D array
target = np.array([1, 2, 3, 4])  # Unused, consider removing
perplexities = [5, 30, 50, 100]  # Unused, consider removing

# Initialize t-SNE with parameters
tsne = manifold.TSNE(n_components=2, init='random', random_state=0, perplexity=30)

# Perform t-SNE dimensionality reduction
Y = tsne.fit_transform(Xn)

# Scatter plot the t-SNE results
plt.scatter(Y[:, 0], Y[:, 1], c=[sns.color_palette()[x] for x in IDX])
plt.gca().set_aspect('equal', 'datalim')
plt.title('t-SNE projection of sFe-all model', fontsize=24)
plt.savefig('tsne-sFe-all.pdf')
plt.show()
