import pandas as pd

# Read ratings data
r_cols = ['user_id', 'movie_id', 'rating']
ratings = pd.read_csv('u.data', sep='\t', names=r_cols, usecols=range(3))

# Read movies data
m_cols = ['movie_id', 'title']
movies = pd.read_csv('u.item', sep='|', names=m_cols, usecols=range(2))

# Merge movies and ratings data
ratings = pd.merge(movies, ratings)

# Create user-ratings pivot table
userRatings = ratings.pivot_table(index=['user_id'], columns=['title'], values='rating')

# Calculate correlation matrix
corrMatrix = userRatings.corr(method='pearson', min_periods=100)

# Define user ID
userId = 2

# Get user ratings
myRatings = userRatings.loc[userId].dropna()

# Find similar candidates
simCandidates = pd.Series()
for i in range(0, len(myRatings.index)):
    sims = corrMatrix[myRatings.index[i]].dropna()
    sims = sims.map(lambda x: x * myRatings[i])
    simCandidates = simCandidates.append(sims)

# Sort similar candidates
simCandidates.sort_values(inplace=True, ascending=False)

# Group and sum similar candidates
simCandidates = simCandidates.groupby(simCandidates.index).sum()

# Sort again
simCandidates.sort_values(inplace=True, ascending=False)

# Filter out already rated movies
filteredSims = simCandidates.drop(myRatings.index, errors='ignore')

# Print and save filtered similarities
print(filteredSims.head())
filteredSims.to_csv("filteredSims.csv")
