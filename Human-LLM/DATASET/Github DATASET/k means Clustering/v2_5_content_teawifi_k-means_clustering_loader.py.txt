import os
import re
import pandas as pd

# Define the text encoding to use
ENCODING = 'utf-8'  

# Function to load data from text files in a directory
def load_data(path_to_dir):
    # Initialize an empty list to store the data
    data = []
    
    # Get a list of all files in the directory that end with '.txt'
    files = [file for file in os.listdir(path_to_dir)
             if os.path.isfile(os.path.join(path_to_dir, file)) and file.endswith(".txt")]
    
    # Iterate through each file
    for file in files:
        # Open the file for reading
        with open(os.path.join(path_to_dir, file), "r", encoding=ENCODING) as f:
            # Read the file content and split it into lines
            content = f.read().split('\n')
            # Extend the data list with the content of the file
            data.extend(content)
    
    # Initialize an empty list to store filtered data
    filtered_data = []
    
    # Iterate through each item in the data list
    for item in data:
        # Filter the item and append it to the filtered data list
        filtered_data.append(filter_data(item))
    
    # Return the filtered data
    return filtered_data

# Function to filter data based on a regex pattern
def filter_data(data):
    # Define a regex pattern to match numbers
    pattern = "[0-9]+"
    # Find all matches of the pattern in the data
    strings = re.findall(pattern, data)
    # Convert the matched strings to floats and take the first three elements
    numbers = [float(i) for i in strings]
    return numbers[:3]

# Function to save data to a CSV file
def save_to_csv(data, file_path):
    # Create a DataFrame from the data
    df = pd.DataFrame(data)
    # Write the DataFrame to a CSV file without index and with specified header
    df.to_csv(file_path, index=False, encoding=ENCODING, header=['x', 'y', 'z'])

# Main block of code
if __name__ == "__main__":
    # Example usage:
    
    # Directory containing the text files
    data_directory = "/path/to/directory"
    # Output file path
    output_file = "output.csv"

    # Load data from the specified directory
    data = load_data(data_directory)

    # Save the loaded data to a CSV file
    save_to_csv(data, output_file)
