import numpy as np
from itertools import combinations
from math import acos, sqrt, pi, cos, sin

# Projections along dimensions: (d+1)->(d+1)^2 and back

def compute_projection(A):
    """
    Compute the projection of matrix A onto its column space.

    Args:
    - A: Input matrix

    Returns:
    - Projection of A
    """
    interim = np.matmul(A.transpose(), A)
    return interim.flatten()

def decompose_vector(q, d):
    """
    Decompose a vector q into its corresponding coefficients.

    Args:
    - q: Input vector
    - d: Dimension

    Returns:
    - Decomposed vector x
    """
    xxT = np.outer(q, q)
    xxT = _get_closest_symmetric_matrix(xxT)
    x = -xxT[:, d - 1]
    return x

def compose_vector(x):
    """
    Compose a vector x.

    Args:
    - x: Input vector

    Returns:
    - Composed vector
    """
    interim = np.outer(x, x)
    return interim.flatten()

# Cartesian -> Spherical

def cartesian_to_spherical(x):
    """
    Convert Cartesian coordinates to Spherical coordinates.

    Args:
    - x: Input vector in Cartesian coordinates

    Returns:
    - Spherical coordinates
    """
    d = len(x)
    phi = np.zeros((d - 1))
    sum_squares = pow(x[-1], 2) + pow(x[-2], 2)
    last_angle = acos(x[-2] / sqrt(sum_squares))
    phi[-1] = last_angle if x[-1] >= 0 else 2 * pi - last_angle
    for k in range(d - 3, -1, -1):
        sum_squares += pow(x[k], 2)
        phi[k] = acos(x[k] / sqrt(sum_squares))
    return phi

# Spherical -> Cartesian

def spherical_to_cartesian(phi):
    """
    Convert Spherical coordinates to Cartesian coordinates.

    Args:
    - phi: Input vector in Spherical coordinates

    Returns:
    - Cartesian coordinates
    """
    dim = len(phi)
    x = np.zeros(dim + 1)
    sins = 1
    for i in range(dim):
        x[i] = sins * cos(phi[i])
        sins *= sin(phi[i])
    x[-1] = sins
    return x

def compute_alt_projection(A, j):
    """
    Compute the alternative projection of matrix A onto its column space.

    Args:
    - A: Input matrix
    - j: Parameter

    Returns:
    - Projection of A
    """
    d = len(A[0])
    p = np.array([])
    p_1 = np.multiply(A, A).reshape(1, -1)
    p = np.append(p, p_1)
    index = range(d + 1)
    comb = list(combinations(index, 2))
    for i in range(d - j):
        p_2 = 2 * np.multiply(A[i, [x[0] for x in comb]], A[i, [x[1] for x in comb]])
        p = np.append(p, p_2)
    return p

def decompose_alt_vector(q, d, j):
    """
    Compute the alternative decomposition of a vector q into its corresponding coefficients.

    Args:
    - q: Input vector
    - d: Dimension
    - j: Parameter

    Returns:
    - Decomposed vector x
    """
    x_ = np.sqrt(np.abs(q[:d]))
    signs = q[d * (d - j):d * (d - j) + d - 1]
    x = np.zeros(d)
    x[0] = np.absolute(x_[0])
    for i in range(1, len(x_)):
        x[i] = -np.absolute(x_[i]) if signs[i - 1] < 0 else np.absolute(x_[i])
    return x

# Helper function to get the closest symmetric matrix

def _get_closest_symmetric_matrix(A):
    """
    Get the closest symmetric matrix to the input matrix.

    Args:
    - A: Input matrix

    Returns:
    - Closest symmetric matrix
    """
    return (A + A.T) / 2
