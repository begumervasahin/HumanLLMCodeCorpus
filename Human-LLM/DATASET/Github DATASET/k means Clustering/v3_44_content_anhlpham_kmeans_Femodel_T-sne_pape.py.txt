# Importing necessary libraries
import numpy as np
import matplotlib.pyplot as plt
from sklearn.manifold import TSNE
from scipy.io import loadmat
import seaborn as sns

def load_data(file_path):
    """Load data from the given file path."""
    data = loadmat(file_path)
    Xn = data['Xn']
    IDX_new = data['IDX_new']
    IDX = np.reshape(IDX_new, (35885))
    return Xn, IDX

def visualize_tsne(X, labels, title, save_path=None):
    """Visualize t-SNE embedding of the data."""
    tsne = TSNE(n_components=2, init='random', random_state=0)
    Y = tsne.fit_transform(X)

    sns.set(style='white', context='notebook', rc={'figure.figsize': (14, 10)})
    plt.scatter(Y[:, 0], Y[:, 1], c=[sns.color_palette()[x] for x in labels])
    plt.gca().set_aspect('equal', 'datalim')  # Fix aspect ratio
    plt.title(title, fontsize=24)

    if save_path:
        plt.savefig(save_path)
    plt.show()

def main():
    """Main function."""
    # Load data
    file_path = 'Parekh_all2.mat'
    Xn, IDX = load_data(file_path)

    # Define target classes
    target_classes = np.array([1, 2, 3, 4])

    # Visualize t-SNE embedding
    title = 't-SNE Projection of sFe-all Model'
    save_path = 'tsne-sFe-all.pdf'
    visualize_tsne(Xn, IDX, title, save_path)

if __name__ == "__main__":
    main()
