# Import necessary libraries
from pandas import DataFrame
import matplotlib.pyplot as plt
from sklearn.cluster import KMeans
import tkinter as tk
from matplotlib.backends.backend_tkagg import FigureCanvasTkAgg   

# Define the data points
data = {'x': [25,34,22,27,33,33,31,22,35,34,67,54,57,43,50,57,59,52,65,47,49,48,35,33,44,45,38,43,51,46],
        'y': [79,51,53,78,59,74,73,57,69,75,51,32,40,47,53,36,35,58,59,50,25,20,14,12,20,5,29,27,8,7]
       }

# Convert the data into a DataFrame
df = DataFrame(data, columns=['x','y'])

# Perform KMeans clustering with 3 clusters
kmeans = KMeans(n_clusters=3).fit(df)

# Get the centroids of the clusters
centroids = kmeans.cluster_centers_

# Create a Tkinter window
root = tk.Tk()

# Create a canvas to display elements
canvas1 = tk.Canvas(root, width=100, height=100)
canvas1.pack()

# Display centroids label
label1 = tk.Label(root, text=centroids, justify='center')       
canvas1.create_window(70, 50, window=label1)

# Create a matplotlib figure
figure1 = plt.Figure(figsize=(5, 4), dpi=100)
ax1 = figure1.add_subplot(111)

# Plot data points and centroids
ax1.scatter(df['x'], df['y'], c=kmeans.labels_.astype(float), s=50, alpha=0.5)
ax1.scatter(centroids[:, 0], centroids[:, 1], c='red', s=50)      

# Embed the matplotlib figure into Tkinter
scatter1 = FigureCanvasTkAgg(figure1, root)
scatter1.get_tk_widget().pack(side=tk.LEFT, fill=tk.BOTH)

# Run the Tkinter event loop
root.mainloop()
