import numpy as np
from itertools import combinations
from math import acos, sqrt, pow, pi, cos, sin

# Projections along dimensions: (d+1)->(d+1)^2 and back

def pComp(A):
    """
    Computes the projection of matrix A onto its column space.

    Args:
    - A: Input matrix

    Returns:
    - Projection of A
    """
    interim = np.matmul(A.transpose(), A)
    return np.reshape(interim, -1)

def qDecomp(q, d):
    """
    Computes the decomposition of a vector q into its corresponding coefficients.

    Args:
    - q: Input vector
    - d: Dimension

    Returns:
    - Decomposed vector x
    """
    xxT = np.outer(q, q)
    xxT = getClosestSymMatrix(xxT)
    x = -xxT[:, d - 1]
    return x

def qComp(x):
    """
    Computes the composition of a vector x.

    Args:
    - x: Input vector

    Returns:
    - Composed vector
    """
    interim = np.outer(x, x)
    return np.reshape(interim, -1)

# Cartesian -> Spherical

def cartesianToSpherical(x):
    """
    Converts Cartesian coordinates to Spherical coordinates.

    Args:
    - x: Input vector in Cartesian coordinates

    Returns:
    - Spherical coordinates
    """
    d = len(x)
    phi = np.zeros((d-1))
    d_ = len(phi)
    sum_squares = pow(x[d_], 2) + pow(x[d_-1], 2)
    last_angle = acos(x[d_-1] / sqrt(sum_squares))
    if x[d_] >= 0:
        phi[d_-1] = last_angle
    else:
        phi[d_-1] = 2 * pi - last_angle
    for k in range(d_-2, -1, -1):
        sum_squares = sum_squares + pow(x[k], 2)
        phi[k] = acos(x[k] / sqrt(sum_squares))
    return phi

# Spherical -> Cartesian

def sphericalToCartesian(phi):
    """
    Converts Spherical coordinates to Cartesian coordinates.

    Args:
    - phi: Input vector in Spherical coordinates

    Returns:
    - Cartesian coordinates
    """
    dim = len(phi)
    x = np.zeros(dim + 1)
    sins = 1
    for i in range(0, dim):
        x[i] = sins * cos(phi[i])
        sins *= sin(phi[i])
    x[dim] = sins
    return x

# P-Q: Appendix Version

def pCompAlt(A, j):
    """
    Computes the alternative projection of matrix A onto its column space.

    Args:
    - A: Input matrix
    - j: Parameter

    Returns:
    - Projection of A
    """
    d = len(A[0, :])
    p = np.array([])
    p_1 = np.reshape(np.multiply(A, A), (1, A.shape[0] * A.shape[1]))
    p = np.append(p, p_1)
    index = range(d + 1)
    comb = list(combinations(index, 2))
    for i in range(d - j):
        p_2 = 2 * np.multiply(A[i, [x[0] for x in comb]], A[i, [x[1] for x in comb]])
        p = np.append(p, p_2)
    return p

def qDecompAlt(q, d, j):
    """
    Computes the alternative decomposition of a vector q into its corresponding coefficients.

    Args:
    - q: Input vector
    - d: Dimension
    - j: Parameter

    Returns:
    - Decomposed vector x
    """
    x_ = np.sqrt(np.abs(q[0:d]))
    signs = q[d * (d - j):d * (d - j) + d - 1]
    x = np.zeros(d)
    x[0] = np.absolute(x_[0])
    for i in range(1, len(x_)):
        if signs[i - 1] < 0:
            x[i] = -np.absolute(x_[i])
        else:
            x[i] = np.absolute(x_[i])
    return x

# Helper function to get the closest symmetric matrix

def getClosestSymMatrix(A):
    """
    Gets the closest symmetric matrix to the input matrix.

    Args:
    - A: Input matrix

    Returns:
    - Closest symmetric matrix
    """
    return (A + A.T) / 2
