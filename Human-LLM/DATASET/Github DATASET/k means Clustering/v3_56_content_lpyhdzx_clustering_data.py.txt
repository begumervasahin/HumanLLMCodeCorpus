import pandas as pd

class DataLoader:
    """
    Load data from a CSV file based on specified targets.

    Args:
        target (list): List of target values to filter the data.
        path (str): Path to the CSV file containing the data.

    Returns:
        DataFrame: Data filtered based on the specified targets.
    """
    def __init__(self, target=None, path=None):
        self.target = target
        self.path = path
    
    def load_data(self):
        """
        Load data from the CSV file and filter it based on the specified targets.

        Returns:
            DataFrame: Filtered data.
        """
        dataset = pd.read_csv(self.path)
        df = pd.DataFrame(dataset)
        target_values = self.target
        filtered_data = df.iloc[:, 1:][df['V1'].isin(target_values)]
        return filtered_data

class NewData(DataLoader):
    """
    Load data from a CSV file, filter it based on specified targets,
    and apply thresholding to the filtered data.

    Args:
        target (list): List of target values to filter the data.
        path (str): Path to the CSV file containing the data.

    Returns:
        DataFrame: Thresholded data.
    """
    def load_data(self):
        """
        Load data from the CSV file, filter it based on the specified targets,
        and apply thresholding to the filtered data.

        Returns:
            DataFrame: Thresholded data.
        """
        dataset = pd.read_csv(self.path)
        df = pd.DataFrame(dataset)
        target_values = self.target
        filtered_data = df.iloc[:, 1:][df['V1'].isin(target_values)]
        filtered_data[filtered_data < 0.8] = -1
        return filtered_data
