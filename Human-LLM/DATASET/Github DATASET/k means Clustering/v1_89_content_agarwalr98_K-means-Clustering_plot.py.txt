import numpy as np
import matplotlib.pyplot as plt
from matplotlib.lines import Line2D

print("========== K MEANS CLUSTERING ===========\n")

# Define colors for different classes
colors = ['red', 'blue', 'green']

# Count total data points
TotalData = 0
with open("Data/NonLinear/group03.txt") as myfile:
    next(myfile)
    for i in myfile:
        TotalData += 1

# Read data from file
index = 0
Data = np.zeros(shape=(TotalData, 2), dtype=float)
with open("Data/NonLinear/group03.txt") as myfile:
    next(myfile)
    for i in myfile:
        i = i.split()
        Data[index][0] = float(i[0])
        Data[index][1] = float(i[1])
        index += 1

# Plot data points with different colors based on index ranges
index = 0
for i in Data:
    if index < 300:
        plt.plot(float(i[0]), float(i[1]), "o", color=colors[0])
    elif 300 <= index < 800:
        plt.plot(float(i[0]), float(i[1]), "o", color=colors[1])
    else:
        plt.plot(float(i[0]), float(i[1]), "o", color=colors[2])
    index += 1

# Create legend for different classes
points = [Line2D([0], [0], color=c, marker='o', linestyle='None') for c in colors]
labels = ['Class 1', 'Class 2', 'Class 3']
plt.legend(points, labels)

# Show plot
plt.show()
