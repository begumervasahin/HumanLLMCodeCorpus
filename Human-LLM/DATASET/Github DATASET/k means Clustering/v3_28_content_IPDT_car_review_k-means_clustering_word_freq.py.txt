import pymysql

# Function to execute a query and fetch data from the database
def fetch_data_from_db(query):
    try:
        conn = connect_to_database()
        with conn.cursor() as cursor:
            cursor.execute(query)
            result = cursor.fetchall()
        return result
    finally:
        conn.close()

# Function to clean and split a sentence into words
def clean_and_split_sentence(sentence):
    # Convert the sentence to lowercase
    sentence = sentence.lower()
    # Define a regular expression pattern to remove punctuation
    punctuation_pattern = r'[,\，。.!?:"@]'
    # Remove punctuation from the sentence
    sentence = re.sub(punctuation_pattern, '', sentence)
    # Split the sentence into words and return as a list
    return sentence.split()

# Function to calculate word frequency from a list of sentences
def calculate_word_frequency(sents_list):
    word_freq = {}  # Dictionary to store word frequencies
    for sentence in sents_list:
        # Remove specific phrase from the sentence if present
        phrase_to_remove = "(Read more about how we rate cars.)"
        if phrase_to_remove in sentence:
            sentence = sentence.replace(phrase_to_remove, "")
        # Split the cleaned sentence into words
        for word in clean_and_split_sentence(sentence):
            # Update word frequency count in the dictionary
            word_freq[word] = word_freq.get(word, 0) + 1
    return word_freq

# Function to write word frequencies to a file
def write_word_frequencies_to_file(word_freq, filepath):
    # Sort the word frequencies in descending order of frequency
    sorted_word_freq = sorted(word_freq.items(), key=lambda x: x[1], reverse=True)
    # Write the sorted word frequencies to the specified file
    with open(filepath, 'w', encoding='utf-8') as file:
        for index, (word, frequency) in enumerate(sorted_word_freq, start=1):
            file.write(f"{index} : {word} : {frequency}\n")

# Function to handle English word frequency calculation
def calculate_and_write_en_word_frequency():
    # SQL query to fetch English data from the database
    sql_query = "SELECT data FROM RAW_DATA"
    # Fetch data from the database
    fetched_data = fetch_data_from_db(sql_query)
    # Extract sentences from the fetched data
    sentences_list = [row[0] for row in fetched_data]
    # Calculate word frequency from the extracted sentences
    word_frequency = calculate_word_frequency(sentences_list)
    # Write word frequencies to a file
    write_word_frequencies_to_file(word_frequency, './word_freq_en_v2.txt')
