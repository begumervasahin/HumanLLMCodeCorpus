import tkinter as tk

class CalculatorApp:
    def __init__(self, master):
        self.master = master
        self.master.title("Calculator")
        self.master.geometry("500x500")
        self.master.configure(background='powder blue')
        self.create_widgets()

    def create_widgets(self):
        tk.Label(self.master, text="Please Select a method from below", bg='powder blue', font=("Arial",20,'bold')).pack()

        self.create_button("Check Prime", self.check_prime_number)
        self.create_button("Radians to Degrees", self.radian_to_degree)
        self.create_button("Degrees to Radian", self.degree_to_radian)
        self.create_button("Compute Arc Length", self.compute_arc_length)
        self.create_button("Area of Sector", self.area_of_sector)
        self.create_button("Unknown", self.unknown)

    def create_button(self, text, command):
        button = tk.Button(self.master, text=text, font=('Arial',20,'bold'), command=command)
        button.pack()

    def check_prime_number(self):
        self.open_window("Check Prime Number")
        entry_value = tk.StringVar()
        entry_box = tk.Entry(self.sub_window, textvariable=entry_value, font=('Arial', 20, 'bold'))
        entry_box.pack()
        self.create_calc_button("Calculate", lambda: self.calculate_prime(entry_box))

    def calculate_prime(self, entry_box):
        number = int(entry_box.get())
        result_label = tk.Label(self.sub_window, font=('Arial',20,'bold'))
        result_label.pack()
        if number <= 1:
            result_label.config(text=f'{number} is not prime')
        else:
            for i in range(2, number):
                if number % i == 0:
                    result_label.config(text=f'{number} is not prime')
                    break
            else:
                result_label.config(text=f'{number} is prime')

    def radian_to_degree(self):
        self.open_window('Radian to Degree')
        self.create_calc_entry("Enter Radian Value", self.convert_to_degree)

    def convert_to_degree(self):
        radian_value = float(self.entry_value.get())
        degree = radian_value * 180 / 3.14
        self.result_label.config(text=f'degrees = {degree}')

    def degree_to_radian(self):
        self.open_window('Degree to Radian')
        self.create_calc_entry("Enter Degree Value", self.convert_to_radian)

    def convert_to_radian(self):
        degree = float(self.entry_value.get())
        radian = degree * 3.14 / 180
        self.result_label.config(text=f'radian = {radian}')

    def compute_arc_length(self):
        self.open_window('Compute Arc Length of an Angle')
        self.create_calc_entry("Enter Diameter", self.compute_arc_length_inner, "Please Enter Angle")

    def compute_arc_length_inner(self):
        diameter = float(self.entry_value.get())
        angle = float(self.entry_value2.get())
        pi = 22/7
        if angle >= 360:
            self.result_label.config(text="Angle not possible")
        else:
            arc_length_value = (pi * diameter) * (angle / 360)
            self.result_label.config(text=f"Arc Length is: {arc_length_value}")

    def area_of_sector(self):
        self.open_window('Compute Area Of Sector')
        self.create_calc_entry("Enter Radius", self.compute_area_of_sector_inner, "Please Enter Angle")

    def compute_area_of_sector_inner(self):
        radius = float(self.entry_value.get())
        angle = float(self.entry_value2.get())
        area = (1/2) * radius**2 * angle
        self.result_label.config(text=f"Area of Sector: {area}")

    def unknown(self):
        self.open_window('Unknown')

    def open_window(self, title):
        self.sub_window = tk.Toplevel(self.master)
        self.sub_window.title(title)
        self.sub_window.geometry("400x400")
        self.sub_window.configure(background="yellow")

    def create_calc_entry(self, label_text, command, label2_text=None):
        tk.Label(self.sub_window, text=label_text, bg='yellow', font=('Arial',15,'bold')).pack()
        self.entry_value = tk.StringVar()
        tk.Entry(self.sub_window, textvariable=self.entry_value, font=('Arial',15,'bold')).pack()
        if label2_text:
            tk.Label(self.sub_window, text=label2_text, bg='yellow', font=('Arial',15,'bold')).pack()
            self.entry_value2 = tk.StringVar()
            tk.Entry(self.sub_window, textvariable=self.entry_value2, font=('Arial',15,'bold')).pack()
        self.create_calc_button("Calculate", command)

    def create_calc_button(self, text, command):
        btn = tk.Button(self.sub_window, text=text, command=command, font=('Arial',20,'bold'))
        btn.pack()
        self.result_label = tk.Label(self.sub_window, text="", font=('Arial',20,'bold'))
        self.result_label.pack()

def main():
    root = tk.Tk()
    app = CalculatorApp(root)
    root.mainloop()

if __name__ == "__main__":
    main()
