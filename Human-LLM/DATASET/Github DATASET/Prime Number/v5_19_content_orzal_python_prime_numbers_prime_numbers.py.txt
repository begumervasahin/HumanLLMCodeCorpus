def print_prime_numbers():
    """
    Prints prime numbers from 2 to 99 and their factors if they are not prime.

    Returns:
        None
    """
    # Iterate through numbers from 2 to 99
    for n in range(2, 100):
        # Initialize a flag to track if n is prime
        is_prime = True
        
        # Iterate through numbers from 2 to n-1
        for x in range(2, n):
            # Check if n is divisible by x
            if n % x == 0:
                # If divisible, print the factors and set is_prime flag to False
                print(n, 'equals', x, '*', n // x)
                is_prime = False
                break
        
        # If no factors found, print that n is a prime number
        if is_prime:
            print(n, 'is a prime number')

# Example usage:
print_prime_numbers()
