def prime_number_generator():
    """
    Generates prime numbers from 2 to n.

    Returns:
        list: A list of prime numbers from 2 to n.
    """
    print("Welcome to the prime number generator.")
    print("This program will print all prime numbers from 2 to n.")

    n = int(input("Enter the value of n: "))

    # Create a list to mark numbers as prime or not
    primes = [True for _ in range(n + 1)]

    # Start checking from 2
    p = 2

    # Iterate until p*p is less than or equal to n
    while p * p <= n:
        # If p is prime
        if primes[p]:
            # Mark multiples of p as not prime
            for multiple in range(p * 2, n + 1, p):
                primes[multiple] = False
        p += 1

    # Generate list of prime numbers from 2 to n
    prime_numbers = [x for x in range(2, n) if primes[x]]
    return prime_numbers

# Example usage:
prime_numbers = prime_number_generator()
print("Prime numbers from 2 to", n, "are:")
for prime_number in prime_numbers:
    print(prime_number)
