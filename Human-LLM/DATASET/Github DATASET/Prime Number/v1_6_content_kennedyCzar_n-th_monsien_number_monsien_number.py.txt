import random

def nth_prime(n):
    """
    Finds the nth prime number.
    """
    number = 10
    p = 2
    while p < n:
        if all(number % i != 0 for i in range(2, number)):
            p += 1
        number += 1
    return number - 1

def is_prime(num):
    """
    Checks if a number is prime.
    """
    return all(num % i != 0 for i in range(2, num))

def find_monisen(n):
    """
    Finds the n-th Monisen number.
    """
    prime_count = 0
    number = 10
    while prime_count < n:
        if is_prime(number):
            m = 2 ** number - 1
            if is_prime(m):
                prime_count += 1
        number += 1
    return m

def generate_monisen_file(n):
    """
    Generates a text file containing the n-th Monisen number.
    """
    monisen = find_monisen(n)
    with open(f'monisen_{n}.txt', 'w') as file:
        file.write(str(monisen))

def foo(num, base):
    """
    Converts a number to a different base recursively.
    """
    if num >= base:
        foo(num // base, base)
    print(num % base, end=' ')
