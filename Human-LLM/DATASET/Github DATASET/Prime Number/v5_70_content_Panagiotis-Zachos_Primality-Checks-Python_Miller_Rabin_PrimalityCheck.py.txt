import math
import random
import time

def power_of_two(n):
    """
    Decomposes a number into the highest power of 2 that divides it
    and the remaining odd number.

    Args:
    - n (int): The number to be decomposed.

    Returns:
    - tuple: A tuple containing the highest power of 2 that divides n
             and the remaining odd number.
    """
    power = 0
    odd_number = n
    while odd_number % 2 == 0:
        power += 1
        odd_number //= 2
    assert 2**power * odd_number == n
    return power, odd_number

def miller_rabin_test(n, k):
    """
    Performs the Miller-Rabin primality test on a number.

    Args:
    - n (int): The number to be tested for primality.
    - k (int): The number of iterations for the Miller-Rabin test.

    Returns:
    - bool: True if n is likely prime, False otherwise.
    """
    if n <= 3:
        return False
    power, odd_number = power_of_two(n - 1)
    for _ in range(k):
        a = random.randint(2, n - 2)
        x = pow(a, odd_number, n)
        if x == 1 or x == n - 1:
            continue
        composite = False
        for _ in range(power - 1):
            x = pow(x, 2, n)
            if x == n - 1:
                composite = True
                break
        if not composite:
            return False
    return True

def is_prime(n):
    """
    Performs a simple primality check on a number.

    Args:
    - n (int): The number to be tested for primality.

    Returns:
    - bool: True if n is prime, False otherwise.
    """
    if n <= 1:
        return False
    elif n <= 3:
        return True
    elif n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True

n = 200000015717
