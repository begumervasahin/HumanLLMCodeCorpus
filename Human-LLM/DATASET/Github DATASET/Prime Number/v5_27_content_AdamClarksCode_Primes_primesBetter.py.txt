def sieve_of_eratosthenes(limit):
    """
    Returns a list of prime numbers up to the specified limit using the Sieve of Eratosthenes algorithm.
    """
    sieve = [True] * (limit + 1)
    sieve[0], sieve[1] = False, False  # 0 and 1 are not primes
    for i in range(2, int(limit ** 0.5) + 1):
        if sieve[i]:
            for j in range(i * i, limit + 1, i):
                sieve[j] = False
    return [num for num, is_prime in enumerate(sieve) if is_prime]

def optimized_sieve_of_eratosthenes(limit):
    """
    Returns a list of prime numbers up to the specified limit using an optimized Sieve of Eratosthenes algorithm.
    """
    sieve = [True] * (limit // 2)
    sieve[0] = False  # 1 is not prime
    for i in range(3, int(limit ** 0.5) + 1, 2):
        if sieve[i // 2]:
            for j in range(i * i // 2, limit // 2, i):
                sieve[j] = False
    return [2] + [2 * i + 1 for i, is_prime in enumerate(sieve) if is_prime]

if __name__ == "__main__":
    # Define the upper limit
    limit = 2000000000

    # Get the list of primes using the optimized sieve_of_eratosthenes function
    list_of_primes = optimized_sieve_of_eratosthenes(limit)

    # Print the number of primes found within the specified limit
    print(f"There are {len(list_of_primes)} primes from 0 to {limit}")
