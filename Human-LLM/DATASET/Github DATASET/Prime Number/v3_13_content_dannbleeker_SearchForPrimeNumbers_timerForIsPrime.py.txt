import timeit
import isPrimeSearch

def search_for_primes(start_number, end_number, prime_search_function, loop_type):
    """Search for primes using different methods and measure execution time."""
    print(f"Searching from {start_number:,} to {end_number:,} (interval: {end_number - start_number:,})")
    # Loop over different prime search functions
    for current_prime_search in prime_search_function:
        # Define the test code
        test_code = f'''
import isPrimeSearch
prime_counter = 0
{loop_type}
    if isPrimeSearch.checkForPrime{current_prime_search}(current_number):
        prime_counter += 1
'''
        # Measure the time to run the test code
        time_to_run = round(timeit.timeit(stmt=test_code, number=1), 4)
        print(f"CheckForPrime{current_prime_search} ({loop_type}): {time_to_run}")

# Define the range of numbers to search for primes
start_number = 0
end_number = 100_000_000

# Define whether to use a log file
use_log_file = True

# Open the log file if needed
if use_log_file:
    log_file = open('log.txt', 'a')

# Define the prime search functions and loop types to test
prime_search_function = range(6, 7)  # Example: range(6, 7) for isPrimeSearch.checkForPrime6
loop_types = ["while", "for"]  # Loop types to test

# Search for primes using different methods and measure execution time
for loop_type in loop_types:
    search_for_primes(start_number, end_number, prime_search_function, loop_type)

# Close the log file if it was opened
if use_log_file:
    log_file.close()
