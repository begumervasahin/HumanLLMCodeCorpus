# Initialize variables
n = 2
primes = [1]  # List to store prime numbers, initialized with 1
count = 0

# Loop through numbers from 2 to 3001
while n <= 3001:
    i = 0
    m = primes[i]

    # Check divisibility of n by each prime in the list
    while m < n:
        p = n % m
        if p == 0:  # If n is divisible by a prime, it's not a prime number
            count += 1
            break
        i += 1
        if i + 1 > len(primes):
            break
        m = primes[i]

    # If count is 0, n is a prime number
    if count == 0:
        print(n, "is prime")
        primes.append(n)  # Add n to the list of primes

    # Increment n by 2 if it's not 2, otherwise increment by 1
    if n == 2:
        n += 1
    else:
        n += 2

    count = 0  # Reset count for the next iteration
