# Prompt user to enter the end of the range
n = int(input('Enter the end of the range: '))

# Initialize empty lists
a = []
b = 0

# Iterate through a range of numbers to find square roots
for i in range(1, 1000000000000000000):
    b = (24 * i + 1) ** 0.5
    # Check if the square root is an integer and less than the specified end of the range
    if b % 1 == 0 and int(b) < n:
        a.append(int(b))
    # Stop loop if the square root exceeds the specified end of the range
    elif int(b) > n:
        break

# Remove elements from the list if they are divisible by numbers from 2 to 9
for i in range(2, 10):
    for z in a:
        if z % i == 0:
            a.remove(z)

# Square the remaining elements in the list
z = [y ** 2 for y in a]

# Remove elements from the list if they match the squares
for i in z:
    for hh in a:
        if hh == i:
            a.remove(hh)

# Add prime numbers 2, 3, 5, and 7 to the list
aaa = [2, 3, 5, 7]
a = a + aaa

# Print the sorted list of prime numbers
print(sorted(a))
