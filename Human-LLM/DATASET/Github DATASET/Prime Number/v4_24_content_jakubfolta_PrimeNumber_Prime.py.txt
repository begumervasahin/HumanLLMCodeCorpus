def is_prime(digit):
    """Check if a given number is prime."""
    if digit < 2:
        return False
    elif digit == 2:
        return True
    for number in range(2, digit):
        if digit % number == 0:
            return False
            break
    else:
        return True

print(is_prime(21))

def check_if_prime_number(number):
    """Check if a given number is prime."""
    if number < 2:
        return False
    else:
        for x in range(2, number - 1):
            if number % x == 0:
                return False
        else:
            return True

print(check_if_prime_number(15))
