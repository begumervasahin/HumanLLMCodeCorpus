def get_prime():
    """
    Accepts input from the user and validates if it's an integer.
    """
    while True:
        try:
            a_number = input("Please enter a number: ")
            proper_number = int(a_number)
            return proper_number
        except ValueError:
            print("The value entered is not an integer")

def is_prime(number):
    """
    Checks if a number is prime.
    """
    if number == 1:
        return False
    elif number == 2:
        return True
    else:
        for element in range(2, number // 2 + 1):
            if number % element == 0:
                return False
        return True

def get_range():
    """
    Gets a range of numbers from the user and finds prime numbers within that range.
    """
    print("Provide a range of numbers to find out which ones are prime.")
    first_number = get_prime()
    second_number = get_prime()

def range_of_prime(num1, num2):
    """
    Finds prime numbers within a given range.
    """
    list_of_primes = []
    for element in range(num1, num2 + 1):
        if is_prime(element):
            list_of_primes.append(element)
    return list_of_primes

def next_hundred_primes(number):
    """
    Finds the next hundred prime numbers after a given number.
    """
    list_of_primes = []
    prime_start = number + 1
    while len(list_of_primes) < 100:
        if is_prime(prime_start):
            list_of_primes.append(prime_start)
        prime_start += 1
    return list_of_primes
