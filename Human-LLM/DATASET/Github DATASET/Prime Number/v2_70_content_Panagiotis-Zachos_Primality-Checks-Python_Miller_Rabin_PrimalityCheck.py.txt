import math
import random
import time

def power_of_two(n):
    """
    Returns the power of two for the given number.
    """
    r = 0
    d = n
    while d % 2 == 0:
        r += 1
        d >>= 1
    assert 2 ** r * d == n
    return r, d

def miller_rabin_test(n, k):
    """
    Performs the Miller-Rabin primality test.
    Returns True if n is very likely prime, False otherwise.
    """
    if n <= 3:
        return False
    r, d = power_of_two(n - 1)
    for _ in range(k):
        r_int = random.randint(2, n - 2)
        x = pow(r_int, d, n)
        if x == 1 or x == n - 1:
            continue
        flag = 0
        for _ in range(r - 1):
            x = pow(x, 2, n)
            if x == n - 1:
                flag = 1
                break
        if flag == 1:
            continue
        return False
    return True

def is_prime_simple(n):
    """
    Simple primality check function.
    """
    if n <= 1:
        return False
    elif n <= 3:
        return True
    elif n % 2 == 0 or n % 3 == 0:
        return False
    i = 5
    while i * i <= n:
        if n % i == 0 or n % (i + 2) == 0:
            return False
        i += 6
    return True
