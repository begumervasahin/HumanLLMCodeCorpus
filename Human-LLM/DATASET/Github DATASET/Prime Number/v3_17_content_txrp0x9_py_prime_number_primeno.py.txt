def sieve_of_eratosthenes(n):
    """Generate prime numbers up to n using the Sieve of Eratosthenes algorithm."""
    primes = [True] * (n + 1)  # Initialize a list to store whether each number is prime or not

    # Sieve of Eratosthenes algorithm to find prime numbers
    p = 2
    while p * p <= n:
        if primes[p]:
            # Mark multiples of prime numbers as non-prime
            for i in range(p * 2, n + 1, p):
                primes[i] = False
        p += 1

    return primes

def print_primes(n, primes):
    """Print all prime numbers up to n."""
    print(f"Prime numbers from 2 to {n}:")
    for x in range(2, n):
        if primes[x]:
            print(x)

if __name__ == "__main__":
    print("Welcome to the prime number generator!")
    print("This program will print all prime numbers from 2 to n.")

    # Input
    n = int(input("Enter the value of n: "))

    # Generate prime numbers using Sieve of Eratosthenes algorithm
    primes = sieve_of_eratosthenes(n)

    # Print all prime numbers from 2 to n
    print_primes(n, primes)
