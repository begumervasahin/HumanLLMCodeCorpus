def generate_prime_numbers(n):
    """Generate prime numbers up to n using the Sieve of Eratosthenes algorithm."""
    if n is not None:
        print("Starting prime number generation...")
        if n > 0:
            if isinstance(n, int):
                if n == 0:
                    return [0]
                elif n == 2:
                    return [2]
                elif n < 2:
                    return []
                
                # Initialize sieve of Eratosthenes
                sieve = list(range(3, n + 1, 2))
                mroot = int(n ** 0.5)
                half = (n + 1) // 2 - 1
                i = 0
                m = 3
                
                # Sieve algorithm
                while m <= mroot:
                    if sieve[i]:
                        j = (m * m - 3) // 2
                        sieve[j] = 0
                        while j < half:
                            sieve[j] = 0
                            j += m
                    i += 1
                    m = 2 * i + 3
                
                # Return the list of prime numbers
                return [2] + [x for x in sieve if x]
            
            # Print error message if n is not an integer
            print("Error: The input is not an integer. Please enter a whole number.")
            return "Error: The input is not an integer. Please enter a whole number."
        
        # Print error message if n is not positive
        print("Error: Please enter a positive integer.")
        return "Error: Please enter a positive integer."
    
    # Print error message if n is None
    print("Error: Please enter a positive integer as the limit.")
    return "Error: Please enter a positive integer as the limit."

# Example usage
n = 20
print(generate_prime_numbers(n))
