def generate_prime_numbers(n):
    """
    Generates prime numbers up to n.

    Parameters:
        n (int): The upper limit to generate prime numbers.

    Returns:
        list: A list of prime numbers up to the limit.
    """
    prime_numbers = []
    for num in range(2, n + 1):
        is_prime = True
        for divisor in range(2, int(num ** 0.5) + 1):
            if num % divisor == 0:
                is_prime = False
                break
        if is_prime:
            prime_numbers.append(num)
    return prime_numbers

# Example usage
prime_numbers_up_to_10 = generate_prime_numbers(10)
print(prime_numbers_up_to_10)
