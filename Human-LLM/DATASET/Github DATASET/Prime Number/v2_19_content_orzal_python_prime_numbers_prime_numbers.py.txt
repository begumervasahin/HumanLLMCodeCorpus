def is_prime(num):
    """Check if a number is prime."""
    if num < 2:
        return False
    for i in range(2, int(num ** 0.5) + 1):
        if num % i == 0:
            return False
    return True

# Loop through numbers from 2 to 99 to check for primality
for n in range(2, 100):
    if is_prime(n):
        # Print if the number is prime
        print(f"{n} is a prime number")
    else:
        # If not prime, find and print its factors
        for x in range(2, n):
            if n % x == 0:
                print(f"{n} equals {x} * {n // x}")
                break
