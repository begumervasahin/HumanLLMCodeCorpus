import math

def mersenne_number(p):
    """Calculate the pth Mersenne number."""
    return (2 ** p) - 1

def is_prime(number):
    """Check if a number is prime."""
    if number <= 1:
        return False
    for i in range(2, int(math.sqrt(number)) + 1):
        if number % i == 0:
            return False
    return True

def get_primes(start, end):
    """Generate a list of prime numbers within the given range."""
    prime_list = []
    for num in range(start, end):
        if is_prime(num):
            prime_list.append(num)
    return prime_list

def lucas_lehmer(p):
    """Perform the Lucas-Lehmer test for primality."""
    lucas_lehmer_sequence = [4] * (p - 1)
    for i in range(1, p - 1):
        lucas_lehmer_sequence[i] = (((lucas_lehmer_sequence[i - 1]) ** 2) - 2) % ((2 ** p) - 1)
    return lucas_lehmer_sequence[p - 2] == 0

def sieve(n):
    """Implement the Sieve of Eratosthenes algorithm to find prime numbers."""
    bool_list = [False, False] + [True] * (n - 1)  # Initialize a list of booleans representing prime numbers
    p = 2
    while p is not None:
        for i in range(2 * p, n + 1, p):
            bool_list[i] = False  # Mark multiples of p as non-prime
        p = next((i for i in range(p + 1, n + 1) if bool_list[i]), None)  # Find the next prime
    return [i for i, prime in enumerate(bool_list) if prime]  # Extract prime numbers from the boolean list
