# Prompt the user to enter an integer to check for primality
user_input = int(input("Please enter an integer you want to check: "))

# Initialize a flag to track primality
is_prime = True
divisor = 2

# Iterate through potential divisors up to the square root of the input number
while divisor < user_input ** 0.5:
    if user_input % divisor == 0:
        is_prime = False
        break
    divisor += 1

# Display the result based on the primality flag
if is_prime:
    print(str(user_input) + " is a prime number")
else:
    print(str(user_input) + " is not a prime number")
