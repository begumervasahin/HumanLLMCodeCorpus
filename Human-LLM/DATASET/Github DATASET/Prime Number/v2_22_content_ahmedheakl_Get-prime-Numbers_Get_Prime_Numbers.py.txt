# Prompt the user to enter the end of the range
end_range = int(input('Enter the end of the range: '))

# Initialize lists to store prime numbers and temporary variables
prime_numbers = []
temp = 0

# Iterate through a range of numbers to find prime numbers
for i in range(1, 1000000000000000000):
    temp = (24 * i + 1) ** 0.5
    if temp % 1 == 0 and int(temp) < end_range:
        prime_numbers.append(int(temp))
    elif int(temp) > end_range:
        break

# Remove non-prime numbers from the list
for i in range(2, 10):
    for num in prime_numbers:
        if num % i == 0:
            prime_numbers.remove(num)

# Square the prime numbers and remove them from the list if present
squared_primes = [num ** 2 for num in prime_numbers]
for num_squared in squared_primes:
    for prime in prime_numbers:
        if prime == num_squared:
            prime_numbers.remove(prime)

# Add initial prime numbers [2, 3, 5, 7] to the list
initial_primes = [2, 3, 5, 7]
prime_numbers += initial_primes

# Sort and print the list of prime numbers
sorted_primes = sorted(prime_numbers)
print(sorted_primes)
