import numpy as np
import csv

def is_prime(x, prime_list):
    """
    Check if a number is prime.
    
    Args:
        x (int): The number to check for primality.
        prime_list (list): List of prime numbers less than or equal to sqrt(x).

    Returns:
        bool: True if x is prime, False otherwise.
    """
    for prime_number in prime_list:
        if prime_number <= np.sqrt(x):
            if x % prime_number == 0:
                return False
        else:
            break
    return True

def generate_primes(target):
    """
    Generate prime numbers up to the target.
    
    Args:
        target (int): The upper limit to generate prime numbers.

    Returns:
        list: List of prime numbers up to the target.
    """
    primes = []
    for number in range(2, target + 1):
        if is_prime(number, primes):
            primes.append(number)
    return primes

def save_primes_to_csv(primes, target):
    """
    Save prime numbers to a CSV file.
    
    Args:
        primes (list): List of prime numbers.
        target (int): The upper limit used for generating primes.
    """
    filename = f'prime_{target}_({len(primes)}).csv'
    with open(filename, 'w', newline='') as csvfile:
        writer = csv.writer(csvfile, delimiter=',')
        writer.writerow(primes)

if __name__ == "__main__":
    target = 100
    primes = generate_primes(target)
    save_primes_to_csv(primes, target)
    print("Target:", target)
    print("Number of primes:", len(primes))
