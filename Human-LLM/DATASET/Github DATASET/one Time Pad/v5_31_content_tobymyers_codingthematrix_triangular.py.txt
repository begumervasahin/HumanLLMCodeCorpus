from vec import Vec
from vecutil import zero_vec

def triangular_solve_n(row_vectors, b):
    """
    Solves an upper-triangular linear system.

    Args:
        row_vectors: A nonempty list of Vecs representing the rows of the upper-triangular matrix.
        b: An n-element list or a Vec whose domain is {0, 1, ..., n-1} representing the right-hand side of the equations.

    Returns:
        A Vec x that is the unique solution to the linear system.
    """
    domain = row_vectors[0].D
    n = len(domain)
    assert domain == set(range(n))
    solution = zero_vec(domain)
    for j in reversed(range(n)):
        solution[j] = (b[j] - row_vectors[j] * solution) / row_vectors[j][j]
    return solution

def triangular_solve(row_vectors, label_list, b):
    """
    Solves an upper-triangular linear system with respect to a given label_list.

    Args:
        row_vectors: A nonempty list of Vecs representing the rows of the upper-triangular matrix.
        label_list: A list consisting of all the elements of D, where D is the domain of each of the vectors in row_vectors.
        b: An n-element list or a Vec over domain {0, 1, ..., n-1} representing the right-hand side of the equations.

    Returns:
        A Vec x that is the unique solution to the linear system.
    """
    domain = row_vectors[0].D
    solution = zero_vec(domain)
    for j in reversed(range(len(domain))):
        c = label_list[j]
        row = row_vectors[j]
        solution[c] = (b[j] - solution * row) / row[c]
    return solution
