def hex_to_dec(hex_string):
    """
    Converts a hexadecimal string to a list of decimal integers.

    Parameters:
        hex_string (str): Hexadecimal string containing bytes.

    Returns:
        list: List of decimal integers representing the hexadecimal values.
    """
    num_bytes = len(hex_string) // 2
    dec_list = []

    for i in range(num_bytes):
        digit1 = hex_digit_to_dec(hex_string[2 * i])
        digit2 = hex_digit_to_dec(hex_string[2 * i + 1])
        dec_list.append(16 * digit1 + digit2)

    return dec_list

def dec_to_hex(dec_list):
    """
    Converts a list of integers to a list of their 2-digit hexadecimal representations.

    Parameters:
        dec_list (list): List of integers.

    Returns:
        list: List of 2-digit hexadecimal strings.
    """
    return [format(n, '02X') for n in dec_list]

def hex_digit_to_dec(hex_val):
    """ Converts a hexadecimal character to its decimal equivalent. """
    if ord(hex_val) < 65:  # Check if the character represents a digit
        value = int(hex_val)
    else:
        value = ord(hex_val.upper()) - 55
    return value

def bytes_to_chars(byte_list):
    """ Converts a list of bytes to a list of characters. """
    return [chr(byte) for byte in byte_list]

def xor_characters(char_list):
    """ Computes the XOR of all characters in the input list and returns the result as a character. """
    xor_value = 0
    for char in char_list:
        xor_value ^= ord(char)
    return chr(xor_value)

# Example usage
hex_message = "48656C6C6F20576F726C64"  # Hex representation of "Hello World"
decimal_list = hex_to_dec(hex_message)
print("Decimal List:", decimal_list)

hex_representation = dec_to_hex(decimal_list)
print("Hexadecimal Representation:", hex_representation)

character_list = bytes_to_chars(decimal_list)
print("Character List:", character_list)

xor_result = xor_characters(character_list)
print("XOR Result:", xor_result)
