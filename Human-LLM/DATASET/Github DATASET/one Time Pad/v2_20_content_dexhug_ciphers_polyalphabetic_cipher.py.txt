import string

# Function to encrypt a message using the Polyalphabetic Cipher
def encrypt(message, shift):
    encryption = ""  # Initialize an empty string to store the encrypted message
    counter = 0  # Initialize a counter to keep track of the shift positions
    for char in message:
        if char.isalpha():  # Check if the character is a letter
            # Calculate the shifted index based on the character and the corresponding shift value
            shifted_index = (string.ascii_uppercase.index(char) + shift[counter]) % 26
            # Get the encrypted character from the shifted index
            encrypted_char = string.ascii_uppercase[shifted_index]
            encryption += encrypted_char  # Append the encrypted character to the encryption string
            counter = (counter + 1) % len(shift)  # Increment the counter and loop back if necessary
        else:
            encryption += char  # Non-alphabetic characters remain unchanged
    return encryption

# Function to decrypt an encrypted message using the Polyalphabetic Cipher
def decrypt(encryption, shift):
    # Calculate the decryption shift for each character in the shift list
    dec_shift = [(26 - s) % 26 for s in shift]
    # Call the encrypt function with the decryption shift to decrypt the message
    return encrypt(encryption, dec_shift)

# Main function
def main():
    print("Welcome to the Polyalphabetic Cipher.\n")
    shift_word = "encrypt"  # The shift word used to determine the shifting pattern
    message = "Thanks for taking a look at my polyalphabetic cipher!"  # The message to be encrypted
    
    # Clean shift word and message: convert to uppercase and remove non-alphabetic characters
    shift_word = ''.join(char.upper() for char in shift_word if char.isalpha())
    message = ''.join(char.upper() for char in message if char.isalpha())
    
    # Calculate the shift for each character in the shift word
    shift = [(ord(char) - ord('A') + 1) for char in shift_word]

    print("Shifting the input by this list:", shift)
    print("Message to be encrypted:\n", message)

    # Encrypt the message using the calculated shift
    encryption = encrypt(message, shift)
    print("\nYour encrypted message is:\n", encryption)

    # Decrypt the encrypted message using the same shift
    decryption = decrypt(encryption, shift)
    print("Your decrypted message is:\n", decryption)

if __name__ == "__main__":
    main()
