def tuple_sum(A, B):
    """
    Sums up corresponding elements of two lists of tuples.

    Args:
        A: A list of tuples.
        B: A list of tuples.

    Returns:
        A list of pairs (x, y) where each pair is the sum of corresponding elements from tuples in A and B.
    """
    return [(a[0] + b[0], a[1] + b[1]) for a, b in zip(A, B)]

def inv_dict(d):
    """
    Computes the inverse of a dictionary representing an invertible function.

    Args:
        d: A dictionary representing an invertible function f.

    Returns:
        A dictionary representing the inverse of f, where keys are the values of d and values are the keys of d.
    """
    return {v: k for k, v in d.items()}

def row(p, n):
    """
    Generates a list of n elements where each element is p + i.

    Args:
        p: A number.
        n: A number.

    Returns:
        An n-element list where each element is p + i.
    """
    return [p + i for i in range(n)]
