import sys
import numpy as np

def convert_to_numeric(plain_text):
    num_array = np.ones(len(plain_text))
    for i, char in enumerate(plain_text):
        num_array[i] = ord(char)
    num_array = num_array.astype(int)
    sep_array = np.ones(3 * len(num_array))
    for i, num in enumerate(num_array):
        if num_digits(num) == 3:
            sep_array[3*i] = num // 100 % 10
            sep_array[3*i + 1] = num // 10 % 10
            sep_array[3*i + 2] = num % 10
        elif num_digits(num) == 2:
            sep_array[3*i] = 0
            sep_array[3*i + 1] = num // 10 % 10
            sep_array[3*i + 2] = num % 10
        else:
            sep_array[3*i] = 0
            sep_array[3*i + 1] = 0
            sep_array[3*i + 2] = num % 10
    sep_array = sep_array.astype(int)
    return sep_array

def encrypt_message(converted_array, pad):
    length = len(converted_array)
    final_array = [(converted_array[i] + pad[i]) % 10 for i in range(length)]
    return final_array

def decrypt_message(encrypted_array, pad):
    length = len(encrypted_array)
    decrypted = []
    for i in range(0, length, 3):
        value = ((encrypted_array[i] - pad[i] + 10) % 10) * 100
        value += ((encrypted_array[i + 1] - pad[i + 1] + 10) % 10) * 10
        value += ((encrypted_array[i + 2] - pad[i + 2] + 10) % 10)
        decrypted.append(value)
    return decrypted

def num_digits(x):
    if x == 0:
        return 1
    if x < 0:
        x *= -1
    counter = 0
    while x != 0:
        x //= 10
        counter += 1
    return counter

def main():
    if len(sys.argv) != 4:
        sys.exit("Error: incorrect number of arguments")
    try:
        with open(sys.argv[2]) as message_file:
            plain_text = message_file.read()
    except IOError:
        sys.exit("Error: cannot open message file")
    try:
        with open(sys.argv[3]) as pad_file:
            pad_list = pad_file.read()
    except IOError:
        sys.exit("Error: cannot open pad file")

    pad_array = [int(char) for char in pad_list if char.isdigit()]

    if sys.argv[1] == "encrypt":
        num_array = convert_to_numeric(plain_text)
        if len(pad_array) < len(num_array):
            sys.exit("Error: pad is shorter than numeric version of message")
        final = encrypt_message(num_array, pad_array)
        file_name = "encrypted_" + sys.argv[2]
        final_message = "".join(map(str, final)) + "\n"
        with open(file_name, "w+") as outfile:
            outfile.write(final_message)
    elif sys.argv[1] == "decrypt":
        en_message = [int(char) for char in plain_text if char.isdigit()]
        if len(pad_array) < len(en_message):
            sys.exit("Error: pad is shorter than numeric version of message")
        decrypted = decrypt_message(en_message, pad_array)
        file_name = "decrypted_" + sys.argv[2]
        final_message = "".join(map(chr, decrypted))
        with open(file_name, "w+") as outfile:
            outfile.write(final_message)
    else:
        print("Error: incorrect command, must be \"encrypt\" or \"decrypt\"")

if __name__ == "__main__":
    main()
