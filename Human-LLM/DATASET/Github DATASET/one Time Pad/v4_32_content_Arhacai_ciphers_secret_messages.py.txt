import os

def clear_screen():
    """Clears the screen to prepare it for displaying the menu."""
    os.system('cls' if os.name == 'nt' else 'clear')

def show_menu():
    """Displays the main menu of the Secret Messages program."""
    clear_screen()
    print(
        "Welcome to the Secret Messages project for the Treehouse Techdegree.\n\n"
        "Available ciphers:\n\n"
        "- Caesar\n"
        "- Atbash\n"
        "- Affine\n"
        "- Keyword\n\n"
        "Type 'QUIT' to exit Secret Messages.\n\n"
    )

def get_cipher_or_quit():
    """Prompts the user to choose a cipher or quit the program."""
    available_ciphers = ['Caesar', 'Atbash', 'Affine', 'Keyword']
    module = __import__('ciphers')
    while True:
        cipher = input("Which cipher would you like to use? ").lower().capitalize()
        if cipher == 'Quit':
            return False
        if cipher in available_ciphers:
            pad = get_pad()
            blocks = in_blocks()
            return getattr(module, cipher)(pad, blocks)
        print("Sorry, that's not a valid cipher.\n")

def get_message():
    """Prompts the user to enter a message."""
    message = input("Enter the message: ")
    return message

def get_option():
    """Prompts the user to choose whether to encode or decode the message."""
    while True:
        try:
            option = input("Do you want to encode or decode? ").lower()
            if option == 'encode' or option == 'decode':
                return option
            else:
                raise ValueError
        except ValueError:
            print("Sorry, please enter a valid option.\n")

def get_pad():
    """Allows the user to specify a one-time pad for encryption."""
    while True:
        pad = input("Enter a pad number (leave blank for no pad): ")
        if pad == '':
            return pad
        else:
            try:
                return str(int(pad))
            except ValueError:
                print("Sorry, please enter a valid numeric pad key.\n")

def in_blocks():
    """Allows the user to choose whether to encrypt the message in blocks of five characters."""
    option = input("Encrypt in blocks of 5? (y/N): ").lower()
    return option == 'y'

def show_message(cipher, option, message):
    """Encrypts or decrypts the message using the chosen cipher and displays the result."""
    result = getattr(cipher, option)(message)
    print("\nYour {}d message is: {}".format(option, result))

def cipher_again():
    """Prompts the user to encrypt or decrypt another message or exit the program."""
    start = input("\nDo you want to encrypt or decrypt another message? (Y/n): ")
    if start.lower() != 'n':
        run()
    else:
        print("Thanks for using the Secret Messages project. Goodbye!")

def run():
    """Runs the main program."""
    show_menu()
    cipher = get_cipher_or_quit()
    if cipher:
        message = get_message()
        option = get_option()
        show_message(cipher, option, message)
        cipher_again()
    else:
        print("Thanks for using the Secret Messages project. Goodbye!")

if __name__ == '__main__':
    run()
