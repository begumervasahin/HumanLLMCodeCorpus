from scipy.sparse import csc_matrix
import numpy as np
import math
import heapq as hp
import sys
from collections import Counter

def centroid(indices, matrix):
    """
    Calculate the centroid vector for a set of document indices.
    """
    centroid_sum = sum(matrix[i] for i in indices)
    return centroid_sum / len(indices)

def read_input(file_path):
    """
    Read input file and return necessary data.
    """
    with open(file_path, 'r') as file:
        file_content = file.readlines()
    num_documents = int(file_content[0])
    num_words = int(file_content[1])
    data = [list(map(int, line.strip().split())) for line in file_content[3:]]
    return num_documents, num_words, data

def tf_idf_transform(data, num_documents):
    """
    Perform TF-IDF transformation.
    """
    col_indices = [word_idx - 1 for _, word_idx, _ in data]
    values = [value * math.log(float(num_documents + 1) / (Counter(col_indices)[word_idx] + 1), 2)
              for _, word_idx, value in data]
    return col_indices, values

def create_sparse_matrix(num_documents, num_words, row_indices, col_indices, values):
    """
    Create a sparse matrix using CSC format.
    """
    sparse_matrix = csc_matrix((values, (row_indices, col_indices)), shape=(num_documents, num_words))
    norms = np.asarray(np.sqrt(sparse_matrix.power(2).sum(axis=1)))
    normalized_matrix = csc_matrix(sparse_matrix / norms)
    return normalized_matrix

def initialize_similarity_heap(document_vectors):
    """
    Initialize a heap to store pairwise similarities.
    """
    similarity_heap = []
    for i in range(len(document_vectors)):
        for j in range(i + 1, len(document_vectors)):
            temp_matrix = document_vectors[i].multiply(document_vectors[j])
            denominator = np.sqrt((document_vectors[i].power(2).sum())) * np.sqrt((document_vectors[j].power(2).sum()))
            similarity = (temp_matrix.sum()) / denominator
            hp.heappush(similarity_heap, (1 - similarity, i, j))
    return similarity_heap

def hierarchical_clustering(similarity_heap, num_documents, cluster_size, document_vectors):
    """
    Perform hierarchical clustering until the desired cluster size is reached.
    """
    clustered_documents = {}
    iterations = 0

    while iterations < num_documents - cluster_size:
        similarity_tuple = hp.heappop(similarity_heap)
        if similarity_tuple[1] in clustered_documents or similarity_tuple[2] in clustered_documents:
            continue
        elif isinstance(similarity_tuple[1], int) and isinstance(similarity_tuple[2], int):
            clustered_documents[similarity_tuple[1]] = None
            clustered_documents[similarity_tuple[2]] = None
            current_cluster = (similarity_tuple[1], similarity_tuple[2])
            centroid_vector = centroid(current_cluster, document_vectors)
            del document_vectors[similarity_tuple[1]]
            del document_vectors[similarity_tuple[2]]
        else:
            clustered_documents[similarity_tuple[1]] = None
            clustered_documents[similarity_tuple[2]] = None
            if isinstance(similarity_tuple[1], int):
                current_cluster = similarity_tuple[2] + (similarity_tuple[1],)
            elif isinstance(similarity_tuple[2], int):
                current_cluster = similarity_tuple[1] + (similarity_tuple[2],)
            else:
                current_cluster = similarity_tuple[1] + similarity_tuple[2]
            del document_vectors[similarity_tuple[1]]
            del document_vectors[similarity_tuple[2]]
            centroid_vector = centroid(current_cluster, document_vectors)

        for i in document_vectors:
            if i not in clustered_documents:
                temp_matrix = centroid_vector.multiply(document_vectors[i])
                denominator = np.sqrt((document_vectors[i].power(2).sum())) * np.sqrt((centroid_vector.power(2).sum()))
                similarity = (temp_matrix.sum()) / denominator
                hp.heappush(similarity_heap, (1 - similarity, current_cluster, i))

        document_vectors[current_cluster] = centroid_vector
        iterations += 1

    return clustered_documents

def print_clusters(clustered_documents):
    """
    Print the clusters.
    """
    for cluster in clustered_documents:
        if isinstance(cluster, int):
            print(cluster + 1)
        else:
            print(','.join(str(doc_index + 1) for doc_index in sorted(cluster)))

def main():
    # Read input file and cluster size from command-line arguments
    file_path = sys.argv[1]
    cluster_size = int(sys.argv[2])

    # Read input file
    num_documents, num_words, data = read_input(file_path)

    # Perform TF-IDF transformation
    row_indices, values = tf_idf_transform(data, num_documents)

    # Create sparse matrix
    document_vectors = create_sparse_matrix(num_documents, num_words, row_indices, col_indices, values)

    # Initialize similarity heap
    similarity_heap = initialize_similarity_heap(document_vectors)

    # Perform hierarchical clustering
    clustered_documents = hierarchical_clustering(similarity_heap, num_documents, cluster_size, document_vectors)

    # Print the clusters
    print_clusters(clustered_documents)

if __name__ == "__main__":
    main()
