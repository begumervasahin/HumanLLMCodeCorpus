import networkx as nx
import community as c
import matplotlib.pyplot as plt
import sys
import copy

def remove_edges(G):
    # Calculate edge betweenness centrality
    b = nx.edge_betweenness_centrality(G)
    # Find the maximum betweenness value
    max_betweenness = max(b.values())
    remove = []
    # Identify edges with maximum betweenness and remove them
    for k, v in b.items():
        if v == max_betweenness:
            remove.append(k)
    G.remove_edges_from(remove)
    # Identify connected subgraphs after edge removal
    graphs = list(nx.connected_component_subgraphs(G))
    d = {}
    counter = 0
    # Assign community labels to nodes based on connected subgraphs
    for graph in graphs:
        counter += 1
        for node in graph:
            d[node] = counter
    # Calculate modularity of the current graph
    if G.number_of_edges() == 0:
        return [list(nx.connected_component_subgraphs(G)), 0, G]
    modularity = c.modularity(d, G)
    return [list(nx.connected_component_subgraphs(G)), modularity, G]

if __name__ == "__main__":
    # Check if the correct number of command-line arguments is provided
    if len(sys.argv) != 3:
        print("Usage: detect_communities <inputfile> <outputfile>")
        print("Inputfile : Contains representation of the graph")
        print("Outputfile : This file store community visualization")
        sys.exit(-1)

    # Initialize a list to store detected communities
    result_communities = []

    # Read the graph from the input file
    G = nx.read_edgelist(sys.argv[1])
    # Create a copy of the original graph for visualization
    copy_graph = copy.deepcopy(G)

    # Initialize a dictionary to store initial community assignments
    d = {node: 0 for node in G}
    # Calculate the initial modularity of the graph
    initial_modularity = c.modularity(d, G)
    result_communities.append([d, initial_modularity, G])

    # Iteratively remove edges and detect communities
    while G.number_of_edges() > 0:
        subgraphs = remove_edges(G)
        result_communities.append(subgraphs)
        G = subgraphs[-1]

    # Output the detected communities
    for step in result_communities:
        if step[1] > initial_modularity:
            ng = step[0]
            result = []
            modularity = step[1]
            for graph in step[0]:
                result.append(sorted([int(vertex) for vertex in graph]))
            for community in result:
                print(community)

    # Assign community labels to nodes
    d = {}
    counter = 0
    for graph in ng:
        for node in graph:
            d[node] = counter
        counter += 1

    # Visualize the graph with detected communities
    pos = nx.spring_layout(copy_graph)
    colors = ["violet", "black", "orange", "cyan", "red", "blue", "green", "yellow", "indigo", "pink"]
    for i in range(len(ng)):
        graph = ng[i]
        nlist = [node for node in graph]
        nx.draw_networkx_nodes(copy_graph, pos, nodelist=nlist, node_color=colors[i % 10], node_size=500, alpha=0.8)
    nx.draw_networkx_edges(copy_graph, pos)
    nx.draw_networkx_labels(copy_graph, pos, font_size=10)
    plt.axis('off')

    # Save the visualization to the output file and display it
    plt.savefig(sys.argv[2])
    plt.show()
