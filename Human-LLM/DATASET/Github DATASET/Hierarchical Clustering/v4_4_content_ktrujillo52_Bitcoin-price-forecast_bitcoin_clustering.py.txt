import os
import csv
import collections
import pandas as pd
from matplotlib import pyplot as plt
from matplotlib.dates import datestr2num
from scipy.cluster.hierarchy import dendrogram, linkage

# Define the path to PCA CSV files
pca_csv_path = "/home/kevin/Desktop/Bitcoin/simulations/MCHMC/24hour/5_16_17/pca_csv/"  

class DataProcessor:
    """
    Class to process data from CSV files in a given directory and generate a hierarchical clustering dendrogram.
    """

    def __init__(self, pca_csv_path):
        self.pca_csv_path = pca_csv_path
        self.directories = []
        self.imports = {}
        self.exports = {}
        self.timeseries = None
        self.Z = None

    def get_data(self):
        """Get the PCA CSV path."""
        return self.pca_csv_path

    def change_directory(self):
        """Change directory to the PCA CSV path."""
        os.chdir(self.pca_csv_path)

    def read_csv_files(self):
        """Read CSV files in the directory and append data."""
        self.directories = os.listdir(self.pca_csv_path)
        for directory in self.directories:
            with open(directory, "r") as file:
                csv_reader = csv.reader(file)
                for row in csv_reader:
                    try:
                        self.imports[directory].append(float(row[1]))
                        self.exports[directory].append(row[0])
                    except Exception as e:
                        print(f"Error reading CSV file: {e}")

    def get_countries(self):
        """Get the list of countries."""
        return self.directories

    def create_timeseries(self):
        """Create timeseries DataFrame from imports data."""
        self.timeseries = pd.DataFrame.from_dict(self.imports)

    def sort_exports(self):
        """Sort exports data."""
        self.exports = collections.OrderedDict(sorted(self.exports.items(), key=lambda t: t[1]))

    def prepare_axes(self):
        """Prepare axes for plotting."""
        self.sort_exports()
        import_list = datestr2num(list(self.exports.values()))
        self.values = zip(import_list, list(self.exports.keys()))

    def perform_clustering(self):
        """Perform hierarchical clustering."""
        correlation_matrix = self.timeseries.corr(method='pearson')
        self.Z = linkage(self.timeseries, 'average')

    def plot_dendrogram(self):
        """Plot the hierarchical clustering dendrogram."""
        plt.figure()
        plt.title('Hierarchical Clustering Dendrogram')
        plt.xlabel('Sample Index')
        plt.ylabel('Distance')
        country_labels = list(self.exports.keys())
        dendrogram(self.Z, leaf_rotation=90., leaf_font_size=8., labels=country_labels)
        plt.show()

# Create an instance of DataProcessor and perform operations
data_processor = DataProcessor(pca_csv_path)
data_processor.change_directory()
data_processor.read_csv_files()
data_processor.create_timeseries()
data_processor.prepare_axes()
data_processor.perform_clustering()
data_processor.plot_dendrogram()
