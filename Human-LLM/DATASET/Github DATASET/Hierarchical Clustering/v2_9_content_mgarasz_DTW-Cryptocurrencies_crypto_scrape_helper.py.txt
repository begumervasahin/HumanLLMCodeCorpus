import pandas as pd

# Function to retrieve cryptocurrency data from Poloniex API
def coin_lookup(coin):
    # URL for Bitcoin data to be used as base currency
    btc_url = 'https://poloniex.com/public?command=returnChartData&currencyPair=USDT_BTC&start=1439020800&end=1511202699&period=14400'
    # Construct URL for specific cryptocurrency
    currency_pair = 'BTC_' + coin
    url = f'https://poloniex.com/public?command=returnChartData&currencyPair={currency_pair}&start=1439020800&end=1511202699&period=14400'
    # Read cryptocurrency data from API
    crypto_data = pd.read_json(url)
    return crypto_data
