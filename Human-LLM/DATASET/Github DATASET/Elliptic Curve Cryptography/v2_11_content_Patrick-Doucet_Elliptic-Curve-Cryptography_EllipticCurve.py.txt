import matplotlib.pyplot as plt
import math

class EllipticCurve:
    def __init__(self, p, a, b, G=None, n=None, h=None):
        """
        Initialize an elliptic curve object with the parameters p, a, b, G, n, and h.

        :param p: Prime modulus for arithmetic operations
        :param a: Coefficient 'a' in the curve equation
        :param b: Coefficient 'b' in the curve equation
        :param G: Base point (optional)
        :param n: Order of the base point G (optional)
        :param h: Cofactor (optional)
        """
        self.p = p
        self.a = a
        self.b = b
        self.G = G
        self.n = n
        self.h = h

    def plot(self, n_points):
        """
        Plot the elliptic curve by generating n_points points on the curve.

        :param n_points: Number of points to plot on the curve
        """
        x_values = []
        y_values = []
        for i in range(n_points):
            x_values.append(i)
            y_values.append(math.sqrt(i**3 + self.a*i + self.b) % self.p)
        plt.plot(x_values, y_values)
        plt.title(f'Elliptic Curve: y^2 = x^3 + {self.a}x + {self.b} (mod {self.p})')
        plt.xlabel('x')
        plt.ylabel('y')
        plt.grid(True)
        plt.show()

class EllipticCurvePoint(EllipticCurve):
    def __init__(self, x, y, curve):
        """
        Initialize a point on the elliptic curve with coordinates (x, y) and associated curve.

        :param x: x-coordinate of the point
        :param y: y-coordinate of the point
        :param curve: EllipticCurve object to which the point belongs
        """
        super().__init__(curve.p, curve.a, curve.b, curve.G, curve.n, curve.h)
        self.x = x % self.p
        self.y = y % self.p

    def __add__(self, other):
        """
        Add two points on the elliptic curve.

        :param other: Another EllipticCurvePoint object
        :return: Result of the addition operation as a new EllipticCurvePoint object
        """
        if self.x == other.x and self.y == other.y:
            m = (3 * (self.x**2) + self.a) * self.inv_mod(2*self.y)
        else:
            m = (other.y - self.y) * self.inv_mod(other.x - self.x)
        x3 = (m**2 - self.x - other.x) % self.p
        y3 = (m * (self.x - x3) - self.y) % self.p
        return EllipticCurvePoint(x3, y3, self)

    def __eq__(self, other):
        """
        Check if two EllipticCurvePoint objects are equal.

        :param other: Another EllipticCurvePoint object
        :return: True if equal, False otherwise
        """
        return self.x == other.x and self.y == other.y and self.curve == other.curve

    def __ne__(self, other):
        """
        Check if two EllipticCurvePoint objects are not equal.

        :param other: Another EllipticCurvePoint object
        :return: True if not equal, False otherwise
        """
        return not self == other

    def inv_mod(self, x):
        """
        Compute the modular inverse of x modulo p.

        :param x: Input value
        :return: Modular inverse of x modulo p
        """
        return pow(x, -1, self.p) if x != 0 else 0

    def multiply(self, scalar):
        """
        Multiply the point by a scalar value using the double-and-add algorithm.

        :param scalar: Scalar value for multiplication
        :return: Result of the multiplication as a new EllipticCurvePoint object
        """
        result = EllipticCurvePoint(0, 0, self)
        binary_scalar = bin(scalar)[2:]
        for bit in binary_scalar:
            result = result + result
            if bit == '1':
                result = result + self
        return result

# Example usage:
curve = EllipticCurve(23, 0, 7)
point1 = EllipticCurvePoint(1, 3, curve)
point2 = EllipticCurvePoint(18, 20, curve)

# Plot the elliptic curve
curve.plot(20)

# Perform arithmetic operations on points
print(point1 + point2)
print(point1 == point2)
print(point1.multiply(5))
