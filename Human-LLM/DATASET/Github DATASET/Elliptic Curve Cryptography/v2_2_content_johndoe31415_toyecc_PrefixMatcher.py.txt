class PrefixMatcher:
    def __init__(self, options):
        """
        Initialize PrefixMatcher with a list of options.

        Args:
            options (list): List of options to match against.
        """
        self._options = options

    def matchunique(self, value):
        """
        Match a unique option based on the provided value.

        Args:
            value (str): The value to match.

        Returns:
            str: The unique matched option.

        Raises:
            Exception: If no option matches the value or if the value is ambiguous.
        """
        matches = self.match(value)
        if len(matches) == 0:
            raise Exception("'%s' did not match any options." % value)
        elif len(matches) > 1:
            raise Exception("'%s' is ambiguous. Please clarify further. Available options: %s" % (value, ", ".join(sorted(matches))))
        return matches[0]

    def match(self, value):
        """
        Match options based on the provided value.

        Args:
            value (str): The value to match.

        Returns:
            list: A list of matched options.
        """
        return [option for option in self._options if option.startswith(value)]
