def new_ascii(ch):
    """Calculate the new ASCII value of a character by adding 100."""
    return ord(ch) + 100

def old_ascii(ascii_val):
    """Calculate the old ASCII value by subtracting 100 from the given ASCII value."""  
    return int(ascii_val) - 100

def encode(msg):
    """Encode a message by converting each character to its new ASCII value."""
    encoded_message = ''
    for char in msg:
        encoded_message += str(new_ascii(char))
    return encoded_message

def decode(new_ascii_string):
    """Decode a string of new ASCII values back to the original message."""
    decoded_message = ''
    i = 0
    while i < len(new_ascii_string):
        chunk = new_ascii_string[i:i+3]
        decoded_message += chr(old_ascii(chunk))
        i += 3
    return decoded_message

if __name__ == "__main__":
    # Example usage
    message = "Hello, World!"
    encoded_message = encode(message)
    print("Encoded:", encoded_message)
    decoded_message = decode(encoded_message)
    print("Decoded:", decoded_message)
