def encode_string_to_points(string, curve, chunk_length):

# Function to split a string into chunks of specified length
def split_string_into_chunks(string, length):
    return [string[i:i+length] for i in range(0, len(string), length)]

# Function to decode points on the elliptic curve back into a string
def decode_points_to_string(points):
    decoded = []
    for point in points:
        decoded.append(decode_point(point))
    return ''.join(decoded)

# Function to decode a point on the elliptic curve back into a character
def decode_point(point):
    n = (point.x - point.x % 1000) // 1000
    chars = []
    while n != 0:
        chars.append(chr(n % 256))
        n //= 256
    return ''.join(chars[::-1])  # Reverse the characters

# Function to encrypt a string using elliptic curve cryptography
def encrypt_string(string, key):
    curve, p = key.curve, key.curve.p
    chunk_length = int(math.log(p // (1000 - 1), 256))
    encoded_points = encode_string_to_points(string, curve, chunk_length)
    return encrypt_points(encoded_points, key)

# Function to encrypt points using elliptic curve cryptography
def encrypt_points(points, key):
    curve, p = key.curve, key.curve.p
    l = randint(1, p - 1)
    res = [curve.multiply(l, key.P)]
    lQ = curve.multiply(l, key.Q)
    for point in points:
        res.append(curve.add_points(lQ, point))
    return res

# Function to decrypt points using elliptic curve cryptography
def decrypt_points(points, key, k):
    curve = key.curve
    neg_kC1 = curve.multiply(k, points[0]).negate()
    decrypted = []
    for point in points[1:]:
        ans = curve.add_points(point, neg_kC1)
        decrypted.append(ans)
    return decrypted
