import zipfile

def crack_zip_password(zip_file_path, dictionary_file_path):
    try:
        # Open the ZIP file
        with zipfile.ZipFile(zip_file_path) as zip_file:
            # Open and read the dictionary file
            with open(dictionary_file_path, 'r') as dictionary_file:
                passwords = [line.strip() for line in dictionary_file]

            # Iterate through each password and attempt to crack the ZIP file
            for password in passwords:
                try:
                    # Attempt to extract the ZIP file using the password
                    zip_file.extractall(pwd=password.encode())
                    print('Password cracked:', password)
                    return
                except Exception:
                    # Password attempt failed, continue to the next password
                    pass

            # If no password is cracked
            print('Password not found')

    except zipfile.BadZipfile:
        print("ERROR: Invalid ZIP file")
        return

if __name__ == "__main__":
    zip_file_path = 'test_zip.zip'
    dictionary_file_path = 'english.txt'
    crack_zip_password(zip_file_path, dictionary_file_path)
