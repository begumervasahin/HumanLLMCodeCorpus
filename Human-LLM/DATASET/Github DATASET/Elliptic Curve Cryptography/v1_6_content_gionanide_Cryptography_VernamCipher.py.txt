import binascii

def text_to_bits(text):
    """
    Convert text to binary representation.
    """
    return bin(int(binascii.hexlify(text.encode()), 16))[2:]

def bits_to_text(bits):
    """
    Convert binary representation to text.
    """
    n = int(bits, 2)
    return binascii.unhexlify('%x' % n)

def xor(m, k):
    """
    Perform XOR operation between two binary strings.
    """
    result = []
    for i, j in zip(m, k):
        if i.isalpha() and j.isalpha():
            result.append(i)
        else:
            result.append(str(int(i) ^ int(j)))
    return "".join(result)

def encrypt(plaintext, key):
    """
    Encrypt plaintext using the key.
    """
    plaintext_bits = text_to_bits(plaintext)
    key_bits = text_to_bits(key)
    return xor(plaintext_bits, key_bits)

def decrypt(ciphertext, key):
    """
    Decrypt ciphertext using the key.
    """
    key_bits = text_to_bits(key)
    decrypted_bits = xor(ciphertext, key_bits)
    return bits_to_text(decrypted_bits)
