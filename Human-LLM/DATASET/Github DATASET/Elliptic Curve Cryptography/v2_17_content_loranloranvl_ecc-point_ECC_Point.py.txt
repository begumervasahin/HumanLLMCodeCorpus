BASE = 23
infinite_point = (BASE, BASE)

# Generate points that satisfy the elliptic curve equation
F_BASE = [(x, y) for x in range(BASE) for y in range(BASE) if (x ** 3 + x * 2 + 7) % BASE == y * y % BASE]

def get_reciprocal(x):
    """
    Calculate the reciprocal of x modulo BASE.

    Args:
        x (int): The value for which the reciprocal needs to be calculated.

    Returns:
        int: The reciprocal of x modulo BASE.

    Raises:
        Exception: If the reciprocal is not found.
    """
    for i in range(BASE):
        if i * x % BASE == 1:
            return i
    raise Exception('Reciprocal not found')

class ECC_Point:
    def __init__(self, x_init, y_init):
        """
        Initialize an ECC Point.

        Args:
            x_init (int): The x-coordinate of the point.
            y_init (int): The y-coordinate of the point.

        Raises:
            Exception: If the point is not on the elliptic curve.
        """
        self.x = x_init
        self.y = y_init
        if self.val() not in F_BASE and self.val() != infinite_point:
            raise Exception('Invalid initialization')

    def val(self):
        """
        Get the coordinates of the point.

        Returns:
            tuple: A tuple containing the x and y coordinates of the point.
        """
        return (self.x, self.y)

    def show_val(self):
        """
        Print the coordinates of the point.
        """
        print(self.val())

    def __add__(self, another):
        """
        Add two ECC points.

        Args:
            another (ECC_Point): The other point to be added.

        Returns:
            ECC_Point: The result of adding two ECC points.
        """
        if self.val() == infinite_point:
            return another
        if another.val() == infinite_point:
            return self

        x1, y1 = self.x, self.y
        x2, y2 = another.x, another.y

        if x1 == x2 and y1 == -y2:
            return ECC_Point(BASE, BASE)

        if self == another:
            lamda = (3 * x1 * x1 + 2) * get_reciprocal(2 * y2) % BASE
        else:
            lamda = (y2 - y1) * get_reciprocal(x2 - x1) % BASE  

        x3 = (lamda * lamda - x1 - x2) % BASE
        y3 = (lamda * (x1 - x3) - y1) % BASE

        return ECC_Point(x3, y3)

    def __mul__(self, number):
        """
        Multiply an ECC point by an integer.

        Args:
            number (int): The number by which the point is to be multiplied.

        Returns:
            ECC_Point: The result of multiplying the point by the given number.
        """
        result = self
        for i in range(number - 1):
            result = result + self
        return result

    # Create ECC points
    point_A = ECC_Point(5, 7)
    point_B = ECC_Point(12, 19)
    # Add the points
    result = point_A + point_B
    # Print the result
    print("Result of point addition:", result.val())
