import numpy as np
import matplotlib.pyplot as plt

# Define the data
n_groups = 4
ecc_time = (0.001567, 0.003833, 0.006130, 0.007508)
hes_time = (0.001109, 0.001981, 0.005481, 0.005547)

# Create the plot
fig, ax = plt.subplots()
index = np.arange(n_groups)
bar_width = 0.35
opacity = 0.8

# Plot ECC and Hessian curve times
ecc_bar = ax.bar(index, ecc_time, bar_width, alpha=opacity, color='white', edgecolor='black', hatch="*", label='ECC')
hes_bar = ax.bar(index + bar_width, hes_time, bar_width, alpha=opacity, color='white', edgecolor='black', hatch="//", label='Hessian')

# Add labels and title
ax.set_ylabel('Time taken (seconds)')
ax.set_xlabel('File size (bits)')
ax.set_title('ECC vs Hessian Curve')
ax.set_xticks(index + bar_width)
ax.set_xticklabels((1, 8, 10, 12))
ax.legend()

# Show plot
plt.show()
