def decrypt_message(ciphertext):
    decrypted_messages = []
    
    # Try all possible shifts
    for shift in range(1, 26):
        decrypted_text = ""
        
        # Decrypt each character
        for char in ciphertext:
            if 'A' <= char <= 'Z':
                decrypted_char = chr(((ord(char) - shift - ord('A')) % 26) + ord('A'))
            else:
                decrypted_char = char
            decrypted_text += decrypted_char
        
        decrypted_messages.append(decrypted_text)
    
    return decrypted_messages

def main():
    ciphertext = 'YMJJDJXHFSRNXQJFIYMJXRNQJHFSQNJGZYYMJXMTJXYJQQYMJYWZYM'
    print('Ciphertext:', ciphertext)
    decrypted_messages = decrypt_message(ciphertext)
    
    # Print all decrypted messages
    for i, message in enumerate(decrypted_messages, start=1):
        print(f'Decryption {i}: {message}')

if __name__ == "__main__":
    main()
