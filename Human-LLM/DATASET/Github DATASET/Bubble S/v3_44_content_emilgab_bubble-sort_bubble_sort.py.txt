import re
import sys

def process_data(file_name):
    '''
    Reads the file, processes its content, and returns a list of words.
    Each word is converted to lowercase and stripped of punctuation.
    '''
    words = []
    try:
        with open(file_name, mode="r") as file:
            for word in file.read().split():
                word = re.sub("\W", "", word).lower()
                words.append(word)
    except FileNotFoundError:
        print("File not found")
    return words

def bubble_sort_by_length_and_lexico(words):
    '''
    Sorts a list of words by length using bubble sort.
    If words have equal length, they are sorted lexicographically.
    '''
    for i in range(len(words) - 1, 0, -1):
        for j in range(i):
            if len(words[j]) > len(words[j + 1]):
                words[j], words[j + 1] = words[j + 1], words[j]
            elif len(words[j]) == len(words[j + 1]) and words[j] > words[j + 1]:
                words[j], words[j + 1] = words[j + 1], words[j]
    return words

if __name__ == "__main__":
    # Get file name from command line argument or user input
    file_name = sys.argv[1] if len(sys.argv) == 2 else input("What is the name of the file? ")

    # Process the data, sort, and print the words
    words = process_data(file_name)
    sorted_words = bubble_sort_by_length_and_lexico(words)
    for word in sorted_words:
        print(word)
