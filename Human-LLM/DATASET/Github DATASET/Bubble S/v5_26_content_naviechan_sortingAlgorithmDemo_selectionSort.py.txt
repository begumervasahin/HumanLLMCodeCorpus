import math

class SelectionSort:
    def __init__(self, lst, trace_mode):
        """
        Initialize the SelectionSort object.

        Args:
        lst (list): The list to be sorted.
        trace_mode (bool): Whether to enable trace mode for step-by-step visualization.
        """
        self.lst = lst
        self.trace_mode = trace_mode

    def sort(self):
        """
        Sorts the list using the selection sort algorithm.

        Returns:
        list: The sorted list.
        """
        lst = self.lst
        for start_idx in range(len(lst)):
            min_val = lst[start_idx]
            min_idx = start_idx
            for i in range(start_idx + 1, len(lst)):
                if lst[i] < min_val:
                    min_val = lst[i]
                    min_idx = i
            lst[start_idx], lst[min_idx] = lst[min_idx], lst[start_idx]
            if self.trace_mode:
                self.display_list(start_idx, min_idx)
        return lst

    def display_list(self, i, j):
        """
        Displays the list in a readable format with '*' indicating the current and minimum index.

        Args:
        i (int): Current index.
        j (int): Minimum index.
        """
        lst = [str(elm) + '*' if idx in (i, j) else str(elm) for idx, elm in enumerate(self.lst)]
        output = '|' + ' | '.join(lst) + ' |'
        print(output)
        input("Press Enter to continue...")
