import random
from time import time

# Generate a list of unique random numbers
max_range = 10001
unsorted_list = random.sample(range(1, max_range), max_range - 1)

def bubble_sort(input_list):
    # Flag to indicate whether the list is sorted
    is_sorted = False
    # Continue sorting until the list is sorted
    while not is_sorted:
        is_sorted = True
        # Iterate through the list
        for i in range(len(input_list) - 1):
            # If two adjacent elements are in the wrong order, swap them
            if input_list[i] > input_list[i + 1]:
                input_list[i], input_list[i + 1] = input_list[i + 1], input_list[i]
                is_sorted = False
    return input_list

# Create a copy of the unsorted list for each sorting algorithm
unsorted_list_cpy = unsorted_list[:]

# Measure the time taken for bubble sort
start = time()
bubble_sort(unsorted_list_cpy)
print("Bubble sort algorithm ends in " + str(time() - start) + " seconds to sort a list of " + str(max_range - 1)
      + " items.")

# Create a copy of the unsorted list for the system sort algorithm
unsorted_list_cpy = unsorted_list[:]

# Measure the time taken for the system sort (Timsort)
start = time()
unsorted_list_cpy.sort()
print("System Python sort() (Timsort) algorithm ends in " + str(time() - start) + " seconds to sort a list of "
      + str(max_range - 1) + " items.")
