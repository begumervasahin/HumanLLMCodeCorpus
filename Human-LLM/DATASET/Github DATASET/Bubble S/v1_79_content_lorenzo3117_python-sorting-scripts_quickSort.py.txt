import time

comparisons = 0
swaps = 0

def quickSort(arr):
    start = time.time()
    sort(arr)
    end = time.time()
    print("QUICK SORT\nComparisons: " + str(comparisons) + "\nSwaps: " + str(swaps))
    print("Time elapsed: ", str(end - start) + "\n")

def sort(array):
    """Sort the array by using quicksort."""
    global comparisons
    global swaps
    less = []
    equal = []
    greater = []
    if len(array) > 1:
        pivot = array[len(array)-1]
        for x in array:
            if x < pivot:
                comparisons += 1
                less.append(x)
            elif x == pivot:
                comparisons += 1
                equal.append(x)
            elif x > pivot:
                comparisons += 1
                greater.append(x)
        swaps += 1
        return sort(less)+equal+sort(greater)
    else:
        return array

# Example usage
arr = [12, 11, 13, 5, 6, 7]
print("Unsorted array:", arr)
quickSort(arr)
print("Sorted array:", arr)
