def bubble_sort(lst):
    """
    Sorts a list using the Bubble Sort algorithm.

    Args:
        lst (list): The list to be sorted.

    Returns:
        list: The sorted list.
    """
    new_list = lst[:]  # Create a copy of the original list to avoid modifying it
    is_sorted = False
    while not is_sorted:
        n_swaps = 0
        for i in range(len(new_list) - 1):
            if new_list[i] > new_list[i + 1]:
                new_list[i], new_list[i + 1] = new_list[i + 1], new_list[i]  # Swap elements
                n_swaps += 1
        if n_swaps == 0:
            is_sorted = True
    return new_list

def main():
    # Test the bubble_sort function
    lst = [4, 2, 7, 1, 9, 5, 3]
    sorted_lst = bubble_sort(lst)
    print("Original list:", lst)
    print("Sorted list:", sorted_lst)

if __name__ == "__main__":
    main()
