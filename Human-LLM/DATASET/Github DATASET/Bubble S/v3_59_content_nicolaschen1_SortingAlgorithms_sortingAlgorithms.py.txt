def insertion_sort(lst):
    """Sorts the input list using the insertion sort algorithm."""
    for i in range(1, len(lst)):
        current_element = lst[i]
        j = i
        while j > 0 and lst[j - 1] > current_element:
            lst[j] = lst[j - 1]
            j -= 1
        lst[j] = current_element
    return lst

def bubble_sort(lst):
    """Sorts the input list using the bubble sort algorithm."""
    permutation = True
    passage = 0
    while permutation:
        permutation = False
        passage += 1
        for index in range(0, len(lst) - passage):
            if lst[index] > lst[index + 1]:
                permutation = True
                lst[index], lst[index + 1] = lst[index + 1], lst[index]
    return lst

def shaker_sort(lst):
    """Sorts the input list using the shaker sort algorithm."""
    permutation = True
    direction, element = 1, 0
    start, end = 0, len(lst) - 2
    while permutation:
        permutation = False
        while (element < end and direction == 1) or (element > start and direction == -1):
            if lst[element] > lst[element + 1]:
                permutation = True
                lst[element], lst[element + 1] = lst[element + 1], lst[element]
            element += direction
        if direction == 1:
            end -= 1
        else:
            start += 1
        direction = -direction
    return lst

def gnome_sort(lst):
    """Sorts the input list using the gnome sort algorithm."""
    i_backward, i_forward, size = 1, 2, len(lst)
    while i_backward < size:
        if lst[i_backward - 1] <= lst[i_backward]:
            i_backward, i_forward = i_forward, i_forward + 1
        else:
            lst[i_backward - 1], lst[i_backward] = lst[i_backward], lst[i_backward - 1]
            i_backward -= 1
            if i_backward == 0:
                i_backward, i_forward = i_forward, i_forward + 1
    return lst

def merge(left, right):
    """Merges two sorted lists into one sorted list."""
    merged = []
    left_index, right_index = 0, 0
    while left_index < len(left) and right_index < len(right):
        if left[left_index] <= right[right_index]:
            merged.append(left[left_index])
            left_index += 1
        else:
            merged.append(right[right_index])
            right_index += 1
    if left_index < len(left):
        merged.extend(left[left_index:])
    if right_index < len(right):
        merged.extend(right[right_index:])
    return merged

def merge_sort(lst):
    """Sorts the input list using the merge sort algorithm."""
    if len(lst) <= 1:
        return lst
    mid = len(lst) // 2
    left_half = lst[:mid]
    right_half = lst[mid:]
    left_half = merge_sort(left_half)
    right_half = merge_sort(right_half)
    return merge(left_half, right_half)

def selection_sort(lst):
    """Sorts the input list using the selection sort algorithm."""
    size = len(lst)
    for index in range(0, size):
        min_index = index
        for j in range(index + 1, size):
            if lst[j] < lst[min_index]:
                min_index = j
        if min_index != index:
            lst[index], lst[min_index] = lst[min_index], lst[index]
    return lst
