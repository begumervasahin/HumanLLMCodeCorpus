import random
import datetime

sample_list = []

def populate_list(x, length=100):
    for _ in range(length):
        random_num = random.randint(0, 10000)
        x.append(random_num)
    print('\nThis is your Sample List')
    print(sample_list)

def bubble_sort(a):
    startTime = datetime.datetime.now()
    n = len(a)
    for i in range(n - 1):
        for j in range(n - i - 1):
            if a[j] > a[j + 1]:
                a[j], a[j + 1] = a[j + 1], a[j]
    print('\nThis is your Bubble Sort')
    print(a)
    endTime = datetime.datetime.now()
    calcdTime = endTime - startTime
    print('This operation took ' + str(calcdTime) + ' seconds to complete')

def selection_sort(a):
    startTime = datetime.datetime.now()
    n = len(a)
    for i in range(n):
        min_index = i
        for j in range(i + 1, n):
            if a[j] < a[min_index]:
                min_index = j
        if min_index != i:
            a[i], a[min_index] = a[min_index], a[i]
    print('\nThis is your Selection Sort')
    print(a)
    endTime = datetime.datetime.now()
    calcdTime = endTime - startTime
    print('This operation took ' + str(calcdTime) + ' seconds to complete')

if __name__ == "__main__":
    populate_list(sample_list)

    bubble_sorted_list = sample_list.copy()
    bubble_sort(bubble_sorted_list)

    selection_sorted_list = sample_list.copy()
    selection_sort(selection_sorted_list)
