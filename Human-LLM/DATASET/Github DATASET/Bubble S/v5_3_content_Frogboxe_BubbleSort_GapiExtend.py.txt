# Function to return the type based on the given text
def type_of(text):
    types = {"int": int, "str": str, "flt": float, "Non": Pass}
    return types.get(text)

# File Loading Functions

# Function to check if the root exists
def root_check(root):
    return path.exists(root)

# Function to load dictionary data from a file
def load_dictionary(root, types):
    if not root_check(root):
        raise FileNotFound
    ret = dict()
    with open(root, "r") as f:
        for line in f.readlines():
            line = line.strip()
            if line == "ENDKEK":
                return ret
            key, val = line.split(":")
            key, val = types[0](key), types[1](val)
            ret[key] = val
    return ret

# Function to save dictionary data to a file
def save_dictionary(item, root):
    with open(root, "w") as f:
        for key, val in item.items():
            f.write(f"{key}:{val}\n")
        f.write("ENDKEK")

# Function to load set data from a file
def load_set(root):
    if not root_check(root):
        raise FileNotFound
    ret = []
    with open(root, "r") as f:
        curr = dict()
        for line in f.readlines():
            line = line.strip()
            if line == "ENDOBJ":
                ret.append(curr)
                curr = dict()
            elif line == "ENDKEK":
                return ret
            else:
                cls, attr, val = (type_of(line.split(".")[0]),
                                  line.split(":")[0].split(".")[1],
                                  line.split(":")[1])
                curr[attr] = cls(val)

# Function to save set data to a file
def save_set(root, items):
    with open(root, "w") as f:
        for item in items:
            for key, val in item.items():
                f.write(f"{type(val).__name__[0:3]}.{key}:{val}\n")
            f.write("ENDOBJ\n")
        f.write("ENDKEK\n")

# Function to load list data from a file
def load_list(root, _type):
    if not root_check(root):
        raise FileNotFound
    ret = []
    with open(root, "r") as f:
        for line in f.readlines():
            line = line.strip()
            if line == "ENDKEK":
                return ret
            ret.append(_type(line))

# Function to save list data to a file
def save_list(item, root):
    with open(root, "w") as f:
        for obj in item:
            f.write(f"{obj}\n")
        f.write("ENDKEK")

# Function to load string data from a file
def load_string(root):
    if not root_check(root):
        raise FileNotFound
    with open(root, "r") as f:
        ret = "".join(f.readlines())
    return ret

# Function to save string data to a file
def save_string(item, root):
    with open(root, "w") as f:
        for letter in item:
            f.write(letter)

# Function to exclude items from a set based on a map ID
def exclude(_set, mapID, index=0):
    tmp = set()
    for item in _set:
        if item[index] == mapID:
            tmp.add(item)
    _set -= tmp
    return _set

# Function to generate ranges
def generate_range(x, y=None, z=False):
    if y is None:
        if isinstance(x, int):
            yield from range(x)
        elif isinstance(x, dict):
            yield from x.items()
        else:
            for n, item in enumerate(x):
                yield item, n
    elif not z:
        for ix in range(x):
            for iy in range(y):
                yield ix, iy
    elif z:
        for ix in range(x):
            for iy in range(y):
                yield ix, iy, (ix * x) + iy

# Key Mapping Class
class KeyMapping:
    @staticmethod
    def define_key_map():
        for char in generate_range(26):
            setattr(KeyMapping, chr(char + 97).upper(), char + 97)
        for char in generate_range(12):
            setattr(KeyMapping, f"F{char + 1}", char + 282)
    @staticmethod
    def populate(pop):
        KeyMapping.MAP = pop
        for key, val in generate_range(pop):
            setattr(KeyMapping, val, key)
    @staticmethod
    def is_number(kKey):
        kKey -= 48
        return 0 <= kKey <= 9
    @staticmethod
    def number(kKey):
        return kKey - 48

# Colour Mapping Functions
def get_color(col):
    col = int(col)
    r = col // (256 ** 3)
    col -= r * (256 ** 3)
    g = col // (256 ** 2)
    col -= g * (256 ** 2)
    b = col // 256
    col -= b * 256
    a = col
    return (r, g, b, a)

def color_to_int(col):
    _int = 0
    for i in range(4):
        _int += col[i] * (256 ** (3 - i))
    return _int

# Colour Mapping Class
class ColourMapping:
    @staticmethod
    def populate(pop):
        for key, val in generate_range(pop):
            setattr(ColourMapping, key, val)
