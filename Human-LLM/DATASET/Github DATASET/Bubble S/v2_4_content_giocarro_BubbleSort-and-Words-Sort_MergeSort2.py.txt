# Given list
a = [0, 9, 1, 4, 6, 7, 2, 1, 8, 7, 7, 4, 3, 5, 6, 1, 7, 8, 0, 3]
l = len(a)  # Length of the list

# Find the largest power of 2 that divides the length of the list
x = 0
li = l
while li % 2 == 0:
    li = li / 2
    x = x + 1
p = 2 ** x  # Smallest power of 2 greater than or equal to the length of the list
tamis = l / p  # Size of each subinterval
inters = l / tamis  # Number of intervals

print('Number of intervals =', inters, '\nInterval size =', tamis, '\nLength of list =', l, '\nList:', a)

s = 0
g = 1

# Iterating over the intervals
for h in range(s, l, int(tamis)):
    # Iterating over the elements within each interval
    for k in range(s, g * li - 1):
        b = []  # Temporary list to store sorted elements
        n = 0  # Index to track the end of the previous subinterval
        m = 0  
        # Comparing elements within the subinterval
        for i in range(s, g * li - 1):
            for j in range(s, g * li):
                if i >= n:
                    if i < j:
                        if a[i] <= a[j]:
                            b.append(a[i])
                            n = j
                            if j == li - 1:
                                b.append(a[j])
                            else:
                                break
                        else:
                            b.append(a[j])
                            if j == li - 1:
                                m = m + 1
                                b.append(a[i])
                                break
                else:
                    break
            if m > 0:
                break
        a = list(b)  # Update the original list with sorted elements
    g = g + 1
    s = s + li
