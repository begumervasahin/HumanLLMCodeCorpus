# Data
names = ["", "", "ROSE", "ALGERNON", "CLARENCE", "VIOLET", "CECIL", "CUTHBERT"]
genders = ["", "", "FEMALE", "MALE", "MALE", "FEMALE", "MALE", "MALE"]
links = [0, 0]

def initialise():
    """
    Initialize the links list based on genders.
    """
    for index in range(2, len(names)):
        if genders[index] == "FEMALE":
            que = 0
        elif genders[index] == "MALE":
            que = 1
        links[index] = links[que]
        links[que] = index

def ripple_sort():
    """
    Sort the names list using ripple sort algorithm.
    """
    for index in range(2, len(names)):
        swapped = False
        current = links[0]
        for index1 in range(2, len(names) - index):
            previous = current
            current = links[current]
            following = links[current]
            if names[current] >= names[following]:
                swapped = True
                # Swap the links
                temp = links[previous]
                links[previous] = links[current]
                links[current] = links[following]
                links[following] = temp
                current = following
        if not swapped:
            return

# Test the functions
initialise()
ripple_sort()
sorted_names = [names[index] for index in links if index != 0]
print("Sorted names:", sorted_names)
