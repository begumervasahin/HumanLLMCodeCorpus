# File Loading Functions

# Function to check if the root exists
def RootCheck(root):
    return path.exists(root)

# Function to load dictionary data from a file
def LoadDict(root, types):
    if not RootCheck(root):
        return FileNotFound
    ret = dict()
    with open(root, "r") as f:
        for line in f.readlines():
            line = line.replace("\n", "")
            if line == "ENDKEK":
                return ret
            key, val = line.split(":")
            key, val = types[0](key), types[1](val)
            ret[key] = val
    return ret

# Function to save dictionary data to a file
def SaveDict(item, root):
    with open(root, "w") as f:
        for key, val in item.items():
            f.write("{}:{}\n".format(key, val))
        f.write("ENDKEK")

# Function to load set data from a file
def LoadSet(root):
    if not RootCheck(root):
        return FileNotFound
    ret = list()
    with open(root, "r") as f:
        curr = dict()
        for line in f.readlines():
            line = line.replace("\n", "")
            if line == "ENDOBJ":
                ret.append(curr)
                curr = dict()
            elif line == "ENDKEK":
                return ret
            else:
                cls, attr, val = (TypeOf(line.split(".")[0]),
                                  line.split(":")[0].split(".")[1],
                                  line.split(":")[1])
                curr[attr] = cls(val)

# Function to save set data to a file
def SaveSet(root, items):
    with open(root, "w") as f:
        for item in items:
            for key, val in item.items():
                f.write("{}.{}:{}\n".format(type(val).__name__[0:3], key, val))
            f.write("ENDOBJ\n")
        f.write("ENDKEK\n")

# Function to load list data from a file
def LoadList(root, _type):
    if not RootCheck(root):
        return FileNotFound
    ret = list()
    with open(root, "r") as f:
        for line in f.readlines():
            line = line.replace("\n", "")
            if line == "ENDKEK":
                return ret
            ret.append(_type(line))

# Function to load string data from a file
def LoadString(root):
    if not RootCheck(root):
        return FileNotFound
    with open(root, "r") as f:
        ret = "\n".join(f.readlines())
    return ret

# Function to exclude items from a set based on a map ID
def Exclude(_set, mapID, index=0):
    tmp = set()
    for item in _set:
        if item[index] == mapID:
            tmp.add(item)
    for item in tmp:
        _set.remove(item)
    return _set

# Function to generate ranges
def Range(x, y=None, z=False):
    if y == None:
        if isinstance(x, int):
            for i in range(x):
                yield i
        elif isinstance(x, dict):
            for key, val in zip(x, x.values()):
                yield key, val
        else:
            for item, n in zip(x, range(len(x))):
                yield item, n
    elif not z:
        for ix in range(x):
            for iy in range(y):
                yield ix, iy
    elif z:
        for ix in range(x):
            for iy in range(y):
                yield ix, iy, (ix*x) + iy

# Key Mapping Class
class K:
    @staticmethod
    def DefineKeyMap():
        for char in Range(26):
            setattr(K, chr(char+97).upper(), char+97)
        for char in Range(12):
            setattr(K, "F"+str(char+1), char+282)
    @staticmethod
    def Populate(pop):
        K.MAP = pop
        for key, val in Range(pop):
            setattr(K, val, key)
    @staticmethod
    def IsNumber(kKey):
        kKey -= 48
        if kKey >= 0 and kKey <= 9:
            return True
        return False
    @staticmethod
    def Number(kKey):
        return kKey - 48

# Colour Mapping Functions
def Col(col):
    col = int(col)
    r = col//(256**3)
    col -= r*(256**3)
    g = col//(256**2)
    col -= g*(256**2)
    b = col//(256)
    col -= b*(256)
    a = col
    return (r, g, b, a)

def ColToInt(col):
    _int = 0
    for i in range(4):
        _int += col[i]*(256**(3-i))
    return _int
