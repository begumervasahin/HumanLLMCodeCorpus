import io
import requests
import pandas as pd
from dateutil import parser, rrule
from datetime import datetime, time, date
import time

def get_temperature_data(station, day, month, year):
    """
    Function to retrieve minute-level weather data for a single Wunderground PWS station.
    Args:
        station (string): Station code from the Wunderground website
        day (int): Day of the month for which data is requested
        month (int): Month for which data is requested
        year (int): Year for which data is requested
    Returns:
        Pandas DataFrame with weather data for the specified station and date.
    """
    # Construct URL for fetching data
    url = "http://www.wunderground.com/weatherstation/WXDailyHistory.asp?ID={station}&day={day}&month={month}&year={year}&graphspan=day&format=1"
    full_url = url.format(station=station, day=day, month=month, year=year)
    
    # Fetch data from the URL
    response = requests.get(full_url)
    data = response.text
    
    # Preprocess data to remove HTML tags
    data = data.replace("<br>", "")
    
    try:
        # Read data into DataFrame, extracting only Time and TemperatureF columns
        dataframe = pd.read_csv(io.StringIO(data), usecols=["Time", "TemperatureF"], index_col=False)
    except:
        print("Issue with date: {}-{}-{} for station {}".format(day, month, year, station))
        return None
    
    return dataframe

def main():
    # Define start and end dates for data retrieval
    start_date = "2010-07-30"
    end_date = "2018-07-30"
    start = parser.parse(start_date)
    end = parser.parse(end_date)
    
    # Generate list of dates within the specified range
    dates = list(rrule.rrule(rrule.DAILY, dtstart=start, until=end))
    
    # Specify the station(s) to retrieve data for
    stations = ["KCASANFR49"]
    
    # Backoff time in case of connection errors
    backoff_time = 10
    
    # Dictionary to store retrieved data
    data = {}
    
    # Iterate over each station
    for station in stations:
        print("Working on {}".format(station))
        data[station] = []
        
        # Iterate over each date
        for date in dates:
            if date.day % 10 == 0:
                print("Working on date: {} for station {}".format(date, station))
            
            done = False
            while not done:
                try:
                    # Retrieve temperature data for the current date and station
                    weather_data = get_temperature_data(station, date.day, date.month, date.year)
                    done = True
                except ConnectionError:
                    print("Got connection error on {}".format(date))
                    print("Will retry in {} seconds".format(backoff_time))
                    time.sleep(backoff_time)
            
            # Append retrieved data to the station's entry in the dictionary
            data[station].append(weather_data)
        
        # Concatenate data frames for each date and station, then save to CSV
        pd.concat(data[station]).to_csv("temp_data_raw.csv".format(station))

if __name__ == "__main__":
    main()
