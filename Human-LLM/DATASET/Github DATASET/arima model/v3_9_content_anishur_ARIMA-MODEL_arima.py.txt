import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from statsmodels.tsa.stattools import adfuller, acf, pacf
from statsmodels.tsa.arima.model import ARIMA
from statsmodels.tsa.seasonal import seasonal_decompose

# Load the dataset
data_path = 'C:/Users/PATH/DataSet/Milano_WeatherPhenomena/mi_meteo_2001.csv'
df = pd.read_csv(data_path, names=['id', 'Day', 'Temperature'])

# Convert 'Day' column to datetime and set as index
df['Day'] = pd.to_datetime(df['Day'], infer_datetime_format=True)
df.set_index('Day', inplace=True)

# Resample data to daily frequency and compute daily mean temperature
daily_mean_temperature = df.resample('24H').mean()

# Plot the original data
plt.xlabel('Date')
plt.ylabel('Temperature (Celsius)')
plt.plot(daily_mean_temperature)
plt.show()

# Calculate rolling statistics
rolling_mean = daily_mean_temperature.rolling(window=2).mean()
rolling_std = daily_mean_temperature.rolling(window=2).std()

# Plot rolling statistics
plt.plot(daily_mean_temperature, color='blue', label='Original')
plt.plot(rolling_mean, color='red', label='Rolling Mean')
plt.plot(rolling_std, color='black', label='Rolling Std')
plt.legend(loc='best')
plt.title('Rolling Mean & Standard Deviation')
plt.show()

# Test stationarity using Dickey-Fuller test
def test_stationarity(timeseries):
    rolling_mean = timeseries.rolling(window=2).mean()
    rolling_std = timeseries.rolling(window=2).std()

    plt.plot(timeseries, color='blue', label='Original')
    plt.plot(rolling_mean, color='red', label='Rolling Mean')
    plt.plot(rolling_std, color='black', label='Rolling Std')
    plt.legend(loc='best')
    plt.title('Rolling Mean & Standard Deviation')
    plt.show()

    print('Results of Dickey-Fuller Test:')
    dftest = adfuller(timeseries['Temperature'], autolag='AIC')
    dfoutput = pd.Series(dftest[0:4], index=['Test Statistic', 'p-value', 'Lags Used', 'Observations Used'])
    for key, value in dftest[4].items():
        dfoutput['Critical Value (%s)' % key] = value
    print(dfoutput)

test_stationarity(daily_mean_temperature)

# Log transformation
log_transformed = np.log(daily_mean_temperature)

# Moving average
moving_avg = log_transformed.rolling(window=2).mean()

# Differencing
differenced = log_transformed - moving_avg
differenced.dropna(inplace=True)

# Stationarity check after differencing
test_stationarity(differenced)

# Decompose into trend, seasonal, and residual components
decomposition = seasonal_decompose(log_transformed)
trend = decomposition.trend
seasonal = decomposition.seasonal
residual = decomposition.resid

# Plot decomposed components
plt.subplot(411)
plt.plot(log_transformed, label='Original')
plt.legend(loc='best')
plt.subplot(412)
plt.plot(trend, label='Trend')
plt.legend(loc='best')
plt.subplot(413)
plt.plot(seasonal, label='Seasonal')
plt.legend(loc='best')
plt.subplot(414)
plt.plot(residual, label='Residuals')
plt.legend(loc='best')
plt.tight_layout()
plt.show()

# Autocorrelation and partial autocorrelation
lag_acf = acf(differenced, nlags=20)
lag_pacf = pacf(differenced, nlags=20, method='ols')

# Plot ACF and PACF
plt.subplot(121)
plt.plot(lag_acf)
plt.axhline(y=0, linestyle='--', color='gray')
plt.axhline(y=-1.96/np.sqrt(len(differenced)), linestyle='--', color='gray')
plt.axhline(y=1.96/np.sqrt(len(differenced)), linestyle='--', color='gray')
plt.title('Autocorrelation Function')

plt.subplot(122)
plt.plot(lag_pacf)
plt.axhline(y=0, linestyle='--', color='gray')
plt.axhline(y=-1.96/np.sqrt(len(differenced)), linestyle='--', color='gray')
plt.axhline(y=1.96/np.sqrt(len(differenced)), linestyle='--', color='gray')
plt.title('Partial Autocorrelation Function')
plt.tight_layout()
plt.show()
