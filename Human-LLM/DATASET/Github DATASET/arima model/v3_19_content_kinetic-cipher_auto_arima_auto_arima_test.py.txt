import numpy as np
import matplotlib.pyplot as plt
from statsmodels.tsa.arima.model import ARIMA
import argparse

class AutoArimaModel:
    def __init__(self, start_p=1, max_p=3, start_q=1, max_q=3):
        self.start_p = start_p
        self.max_p = max_p
        self.start_q = start_q
        self.max_q = max_q
        self.model = None

    def fit(self, data):
        order = self._get_best_model_params(data)
        self.model = ARIMA(data, order=order)
        self.model_fit = self.model.fit()

    def _get_best_model_params(self, data):
        return self.model.auto_arima(data, start_p=self.start_p, max_p=self.max_p,
                                     start_q=self.start_q, max_q=self.max_q).order

    def forecast(self, num_steps, conf_int=False):
        return self.model_fit.forecast(steps=num_steps, alpha=0.05) if conf_int else self.model_fit.forecast(steps=num_steps)

def get_command_line_args():
    parser = argparse.ArgumentParser()
    parser.add_argument("-f", "--forecast", help="Perform forecasts and plot results", action="store_true")
    args = parser.parse_args()
    return args

def generate_noisy_sine_data(t_start=0, t_end=60, t_delta=0.1, noise_mean=0, noise_std=1, A=1, a=0.35):
    time = np.arange(t_start, t_end, t_delta)
    y = A * np.sin(time)
    noise = np.random.normal(noise_mean, noise_std, len(y))
    yn = y + a * noise
    return time, yn

def plot_forecast(time_axis, actual_data, forecast_values, confidence_interval=None, title='Data'):
    plt.plot(time_axis[:len(actual_data)], actual_data, 'b', label='Actual')
    plt.plot(time_axis[len(actual_data):], forecast_values, 'r', label='Forecast')
    if confidence_interval is not None:
        plt.fill_between(time_axis[len(actual_data):],
                         confidence_interval[:, 0], confidence_interval[:, 1],
                         alpha=0.3, color='b', label='Confidence Interval')
    plt.grid()
    plt.title(title)
    plt.legend()
    plt.show()

args = get_command_line_args()
time, yn = generate_noisy_sine_data()

arima_model = AutoArimaModel(start_p=1, max_p=3, start_q=1, max_q=3)
arima_model.fit(yn)
model_params = arima_model._get_best_model_params(yn)
print("Best model parameters:", model_params)

if args.forecast:
    forecast_steps = 100
    forecast_values, confidence_interval = arima_model.forecast(num_steps=forecast_steps, conf_int=True)
    time_axis = np.arange(len(yn) + len(forecast_values))
    plot_forecast(time_axis, yn, forecast_values, confidence_interval, title='Noisy Sine Forecast')
