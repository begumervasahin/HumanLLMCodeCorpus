import os
import time
import datetime
import numpy as np
import pandas as pd
import json
from os import listdir
from os.path import isfile, join
from objdict import ObjDict
import pyflux as pf
import matplotlib.pyplot as plt
from statsmodels.tsa.stattools import adfuller, acf, pacf
from statsmodels.tsa.arima_model import ARIMA

def fit_models(mypath='', js=None):
    # Change directory if a path is provided
    if mypath != '':
        os.chdir(mypath)
        lista_archivos = [f for f in listdir(mypath) if isfile(join(mypath, f))]
        lista_dat = []
        for dat in lista_archivos:
            with open(dat) as json_data:
                lista_dat.append(json.load(json_data))
        fechas, valores = [], []
        for j in lista_dat[0]:
            fechas.append(j['fecha'])
            valores.append(j['valor'])
    elif js != None:
        fechas, valores = [], []
        for i in js:
            fechas.append(i['fecha'])
            valores.append(i['valor'])

    # Data preprocessing
    fechas_list = [fechas[x:x+1] for x in range(0, len(fechas), 1)]
    fechas_format = [datetime.datetime.fromtimestamp(date/1000.0).strftime('%Y-%m-%d-%H') for date in fechas]
    ano, mes, dia, hora = [], [], [], []
    for date in fechas_format:
        fecha = date.split('-')
        ano.append(int(fecha[0]))
        mes.append(int(fecha[1]))
        dia.append(int(fecha[2]))
        hora.append(int(fecha[3]))
    dia_semana = [time.strptime(time.ctime(date/1000)).tm_wday + 1 for date in fechas]
    fechas_pandas = pd.to_datetime(fechas_format)
    dframe = pd.Series(valores, index=fechas_pandas)

    # Removing outliers and zero values
    dframe = dframe[~((dframe-dframe.mean()).abs() > 3*dframe.std())]
    dframe = dframe[(dframe != 0)]

    # Train-test split
    features_train = dframe[0:int(len(dframe)*0.9)]
    features_test = dframe[int(len(dframe)*0.9)+1:len(dframe)]

    # Stationarity test
    def test_stationarity(timeseries, plot=False):
        rolmean = pd.Series(timeseries).rolling(window=12).mean()
        rolstd = pd.Series(timeseries).rolling(window=12).std()
        dftest = adfuller(timeseries, autolag='AIC')
        dfoutput = pd.Series(dftest[0:4], index=['Test Statistic', 'p-value', 'Lags Used', 'Number of Observations Used'])
        if plot:
            plt.plot(timeseries, color='blue', label='Original')
            plt.plot(rolmean, color='red', label='Rolling Mean')
            plt.plot(rolstd, color='black', label='Rolling Std')
            plt.legend(loc='best')
            plt.title('Rolling Mean & Standard Deviation')
            plt.show()
            print('Results of Dickey-Fuller Test:')
            print(dfoutput)
        else:
            return dfoutput

    # ARIMA modeling
    def fit_arima_model(timeseries):
        p_values = range(0, 6)
        d_values = range(0, 3)
        q_values = range(0, 6)
        best_score, best_cfg = float("inf"), None
        for p in p_values:
            for d in d_values:
                for q in q_values:
                    order = (p, d, q)
                    try:
                        model = ARIMA(timeseries, order=order)
                        model_fit = model.fit(disp=0)
                        error = np.sqrt(np.mean((model_fit.fittedvalues - timeseries[1:])**2))
                        if error < best_score:
                            best_score, best_cfg = error, order
                    except:
                        continue
        return best_cfg

    # Test stationarity and fit ARIMA model
    test_stationarity(features_train, plot=True)
    order = fit_arima_model(features_train)
    model = ARIMA(features_train, order=order)
    model_fit = model.fit(disp=0)
    predictions = model_fit.forecast(steps=len(features_test))[0]

    # Evaluate model performance
    error = np.sqrt(np.mean((predictions - features_test)**2))
    accuracy = 100 - error.mean()

    # Prepare results
    data = []
    for i in range(len(predictions)):
        entry = ObjDict()
        entry.fecha = str(features_test.index[i])
        entry.prediccion = predictions[i]
        entry.error = error[i]
        data.append(entry)

    return json.dumps(data), error.mean(), accuracy
