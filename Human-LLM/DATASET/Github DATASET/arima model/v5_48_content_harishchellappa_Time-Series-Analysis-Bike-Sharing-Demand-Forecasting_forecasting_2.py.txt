import pandas as pd
from fbprophet import Prophet
from sklearn.metrics import mean_squared_error
from math import sqrt

# Load data from an Excel file
df = pd.read_excel('newtable.xlsx', sheet_name='newtable')

# Convert 'start_time' and 'end_time' columns to datetime format
df['start_time'] = pd.to_datetime(df['start_time'], format="%d/%m/%Y %I:%M:%S %p")
df['end_time'] = pd.to_datetime(df['end_time'], format="%d/%m/%Y %I:%M:%S %p")

# Aggregate data by start time and station
df_pivot = pd.pivot_table(df[['start_station', 'start_time']], aggfunc='count', index=df['start_time'].dt.date, columns=['start_station'], fill_value=0)

# Remove specific stations and adjust index to datetime
stations_to_remove = [
    3021, 3053, 3055, 3059, 3060, 3061, 3079, 3080,
    4108, 4138, 4142, 4143, 4144, 4146, 4147, 4148,
    4149, 4150, 4151, 4152, 4153, 4154, 4155, 4156,
    4157, 4158, 4159, 4160, 4162, 4163, 4165, 4166,
    4167, 4169, 4170, 4174, 4176, 4177, 4180, 4181,
    4183, 4194, 4244, 4276
]
df_pivot.drop(columns=stations_to_remove, inplace=True)
df_pivot.index = pd.to_datetime(df_pivot.index)

# Prepare data for Prophet model
def prepare_prophet_input(df):
    return pd.DataFrame({'ds': df.index, 'y': df.values})

# Train and test Prophet model
def train_test_prophet(df_pivot, test_samples=-184):
    removal_dates = ['2016-08-14', '2018-10-16', '2017-03-26', '2017-05-11', '2017-08-13', '2017-10-08',
                     '2017-12-10', '2018-04-22', '2018-06-24', '2018-09-30', '2018-12-02']

    df_train = df_pivot.loc[~df_pivot.index.isin(removal_dates)].iloc[:test_samples]
    df_test = df_pivot.iloc[test_samples:]

    return prepare_prophet_input(df_train), prepare_prophet_input(df_test)

# Iterate over each station's data
df_predicted = pd.DataFrame()
mse_train = []
mse_test = []

for station in df_pivot.columns:
    train_data, test_data = train_test_prophet(df_pivot)

    model = Prophet(yearly_seasonality=True, weekly_seasonality=False, daily_seasonality=False)
    model.fit(train_data)

    future = model.make_future_dataframe(periods=len(test_data) + 90 + 3, freq='D')
    forecast = model.predict(future)

    df_predicted[station] = forecast['yhat']

    y_actual_test = test_data['y']
    y_predicted_test = df_predicted[station].tail(184)
    mse_test.append(sqrt(mean_squared_error(y_actual_test, y_predicted_test)))

    y_actual_train = train_data['y']
    y_predicted_train = df_predicted[station].iloc[:-277]
    mse_train.append(sqrt(mean_squared_error(y_actual_train, y_predicted_train)))

# Create dataframe for actual data and save it with predictions and RMSE in an Excel file
df_actual = df_pivot.reset_index(drop=True)
mse_train_df = pd.DataFrame(mse_train)
mse_test_df = pd.DataFrame(mse_test)

with pd.ExcelWriter('outputY.xlsx') as writer:
    df_actual.to_excel(writer, sheet_name='actual(Y)')
    df_predicted.to_excel(writer, sheet_name='predicted(Y)')
    mse_train_df.to_excel(writer, sheet_name='mse_trn(Y)')
    mse_test_df.to_excel(writer, sheet_name='mse_tst(Y)')
