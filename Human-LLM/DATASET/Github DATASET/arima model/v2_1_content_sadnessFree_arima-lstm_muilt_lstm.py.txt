import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from sklearn.preprocessing import MinMaxScaler
from sklearn.metrics import mean_absolute_error
from keras.models import Sequential
from keras.layers import Dense, LSTM

# Define mean absolute percentage error function
def mean_absolute_percentage_error(y_true, y_pred):
    return np.mean(np.abs((y_pred - y_true) / y_true)) * 100

# Convert time series data to supervised learning format
def series_to_supervised(data, n_in=1, n_out=1, dropnan=True):
    n_vars = 1 if isinstance(data, list) else data.shape[1]
    df = pd.DataFrame(data)
    columns, names = [], []
    for i in range(n_in, 0, -1):
        columns.append(df.shift(i))
        names += [('var%d(t-%d)' % (j+1, i)) for j in range(n_vars)]
    for i in range(0, n_out):
        columns.append(df.shift(-i))
        if i == 0:
            names += [('var%d(t)' % (j+1)) for j in range(n_vars)]
        else:
            names += [('var%d(t+%d)' % (j+1, i)) for j in range(n_vars)]
    aggregated = pd.concat(columns, axis=1)
    aggregated.columns = names
    if dropnan:
        aggregated.dropna(inplace=True)
    return aggregated

# Load dataset
dataset = pd.read_csv('your_dataset.csv')

# Set parameters
n_lag = 1
n_features = 1

# Convert dataset to supervised learning format
reframed = series_to_supervised(dataset, n_lag, 1)
reframed.drop(reframed.columns[-n_features:], axis=1, inplace=True)

# Split into train and test sets
values = reframed.values
train_size = int(len(values) * 0.7)
train, test = values[:train_size, :], values[train_size:, :]

# Split into input and output variables
train_X, train_y = train[:, :-1], train[:, -1]
test_X, test_y = test[:, :-1], test[:, -1]

# Reshape input to be 3D [samples, timesteps, features]
train_X = train_X.reshape((train_X.shape[0], 1, train_X.shape[1]))
test_X = test_X.reshape((test_X.shape[0], 1, test_X.shape[1]))

# Define and compile the LSTM model
model = Sequential()
model.add(LSTM(50, input_shape=(train_X.shape[1], train_X.shape[2])))
model.add(Dense(1))
model.compile(loss='mae', optimizer='adam')

# Fit the model
history = model.fit(train_X, train_y, epochs=50, batch_size=72, validation_data=(test_X, test_y), verbose=2, shuffle=False)

# Make predictions
yhat = model.predict(test_X)

# Calculate mean absolute percentage error
mape = mean_absolute_percentage_error(test_y, yhat)
print('Test MAPE: %.3f' % mape)

# Plot training & validation loss values
plt.plot(history.history['loss'], label='Train')
plt.plot(history.history['val_loss'], label='Test')
plt.title('Model loss')
plt.ylabel('Loss')
plt.xlabel('Epoch')
plt.legend()
plt.show()
