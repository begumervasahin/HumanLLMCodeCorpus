import random
import copy
import numpy as np
from itertools import combinations
from cromossome import Cromossome
from train_rbfn import train
from math import log

class GeneticAlgorithm:
    def __init__(self, training, test):
        self.training = training
        self.test = test
        self.pop = []
        self.min_size = 2
        self.max_size = training["INPUT"].size - 1
        self.training_input = np.asarray([[x.tolist()] for x in training["INPUT"].values])
        self.training_output = np.asarray([[x.tolist()] for x in training["OUTPUT"].values])

    def initialize_population(self, size):
        for _ in range(size):
            individual = Cromossome()
            individual.string = random.sample(range(0, len(self.training["INPUT"])), random.randint(self.min_size, self.max_size))
            self.pop.append(individual)

    def decode(self, chromosome):
        centers = []
        for center in chromosome:
            centers.append([self.training["INPUT"].values[center].tolist()])        
        return np.asarray(centers)

    def evaluate(self, chromosome):
        net = train(self.training_input, self.training_output, self.decode(chromosome.string))
        chromosome.fitness = [net.aic(self.training), net.aic(self.test)]

    def evolve(self, generations):
        generation = 1
        for _ in range(generations):
            print("Generation ", generation)
            generation += 1
            for individual in self.pop:
                self.evaluate(individual)
            self.rank_all()
            new_population = []
            while len(new_population) < len(self.pop):
                parents = [self.select_individual() for _ in range(2)]
                offsprings = self.crossover(parents) if random.random() <= 0.90 else [copy.deepcopy(i) for i in parents]
                for offspring in offsprings:
                    self.mutation(offspring)
                    self.deletion(offspring)
                    self.addition(offspring)
                    if offspring.string != parents[0].string and offspring.string != parents[1].string:
                        self.evaluate(offspring)
                        parents.append(offspring)
                selected = []
                while len(selected) < 2:
                    for individual in parents:
                        individual.is_optimal = True
                    self.compare(parents)
                    selected.extend([individual for individual in parents if individual.is_optimal][:2 - len(selected)])
                    parents = [e for e in parents if not e.is_optimal]
                new_population.extend(selected)
            self.pop = new_population
        for individual in self.pop:
            individual.is_optimal = True
        self.compare(self.pop)
        best_config = [individual for individual in self.pop if individual.is_optimal][0]
        print(self.decode(best_config.string))
        return train(self.training_input, self.training_output, self.decode(best_config.string))

    def compare(self, population):
        for a, b in combinations(population, 2):
            if a.partially_less_than(b):
                b.is_optimal = False
            elif b.partially_less_than(a):
                a.is_optimal = False

    def rank_all(self):
        new_pop = []
        current_rank = 0
        while self.pop:
            for individual in self.pop:
                individual.is_optimal = True
            self.compare(self.pop)
            for individual in self.pop:
                if individual.is_optimal:
                    individual.rank = len(self.pop) - current_rank
                    new_pop.append(individual)
            self.pop = [individual for individual in self.pop if not individual.is_optimal]
            current_rank += 1
        self.pop = new_pop

    def select_individual(self):
        max_sum = sum(individual.rank for individual in self.pop)
        pick = random.uniform(0, max_sum)
        current = 0
        for individual in self.pop:
            current += individual.rank
            if current > pick:
                return individual

    def crossover(self, parents):
        if len(parents[0].string) <= len(parents[-1].string):
            smaller, bigger = [copy.deepcopy(parent) for parent in parents]
        else:
            bigger, smaller = [copy.deepcopy(parent) for parent in parents]
        common = set(smaller.string).intersection(bigger.string)
        diff = list(set(smaller.string) - common)
        if diff:
            quantity = random.randint(1, len(diff))
            first_sample = [diff[i] for i in sorted(random.sample(range(len(diff)), quantity))]
            diff = list(set(bigger.string) - common)
            if diff:
                second_sample = [diff[i] for i in sorted(random.sample(range(len(diff)), quantity))]
                findex = [smaller.string.index(item) for item in first_sample]      
                sindex = [bigger.string.index(item) for item in second_sample]      
                for i, j in zip(findex, sindex):
                    smaller.string[i], bigger.string[j] = bigger.string[j], smaller.string[i]
        return [smaller, bigger]

    def mutation(self, chromosome):
        for locus in range(len(chromosome.string)):
            if random.random() <= 0.01:
                chromosome.string[locus] = random.sample(set(range(0, len(self.training["INPUT"]))) - set(chromosome.string), 1)[0]

    def deletion(self, chromosome):
        if random.random() <= 0.01:
            starting_from = random.randint(0, len(chromosome.string) - 1)
            upto = starting_from + random.randint(1, len(chromosome.string) - starting_from)
            if (upto - starting_from) == len(chromosome.string):
                upto -= 1
            chromosome.string = [gene for index, gene in enumerate(chromosome.string) if index not in range(starting_from, upto)]

    def addition(self, chromosome):
        if random.random() <= 0.01 and len(chromosome.string) < self.max_size:      
            quantity = random.randint(1, self.max_size - len(chromosome.string))    
            chromosome.string.extend(random.sample(set(range(0, len(self.training["INPUT"]))) - set(chromosome.string), quantity)))
