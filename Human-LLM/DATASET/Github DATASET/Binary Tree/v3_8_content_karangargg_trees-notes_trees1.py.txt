class BinaryTree:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None

def build_tree():
    root_data = int(input("Enter root data (-1 to exit): "))
    if root_data == -1:
        return None
    root = BinaryTree(root_data)
    print("Enter left child of", root.data)
    root.left = build_tree()
    print("Enter right child of", root.data)
    root.right = build_tree()
    return root

def print_tree(root):
    if root is None:
        return
    print(root.data, end=":")
    if root.left:
        print(root.left.data, end=",")
    else:
        print("-1", end=",")
    if root.right:
        print(root.right.data, end="")
    else:
        print("-1", end="")
    print()
    print_tree(root.left)
    print_tree(root.right)

root = build_tree()
print("Binary Tree:")
print_tree(root)

def count_nodes(root):
    if root is None:
        return 0
    return count_nodes(root.left) + count_nodes(root.right) + 1

print("Number of nodes in the tree:", count_nodes(root))

def preorder_traversal(root):
    if root:
        print(root.data, end=" ")
        preorder_traversal(root.left)
        preorder_traversal(root.right)

print("Preorder Traversal:")
preorder_traversal(root)

def inorder_traversal(root):
    if root:
        inorder_traversal(root.left)
        print(root.data, end=" ")
        inorder_traversal(root.right)

print("\nInorder Traversal:")
inorder_traversal(root)

def postorder_traversal(root):
    if root:
        postorder_traversal(root.left)
        postorder_traversal(root.right)
        print(root.data, end=" ")

print("\nPostorder Traversal:")
postorder_traversal(root)

def height(root):
    if root is None:
        return 0
    return max(height(root.left), height(root.right)) + 1

print("\nHeight of the tree:", height(root))

def print_at_depth_k(root, k):
    if root is None:
        return
    if k == 0:
        print(root.data)
    print_at_depth_k(root.left, k - 1)
    print_at_depth_k(root.right, k - 1)

print("Elements at depth 2:")
print_at_depth_k(root, 2)

def replace_with_depth_k(root, depth):
    if root is None:
        return
    root.data = depth
    replace_with_depth_k(root.left, depth + 1)
    replace_with_depth_k(root.right, depth + 1)

replace_with_depth_k(root, 0)
print("Replaced tree:")
print_tree(root)

def remove_leaves(root):
    if root is None:
        return None
    if root.left is None and root.right is None:
        return None
    root.left = remove_leaves(root.left)
    root.right = remove_leaves(root.right)
    return root

print("Tree after removing leaves:")
print_tree(remove_leaves(root))

def mirror_tree(root):
    if root is None:
        return None
    root.left, root.right = mirror_tree(root.right), mirror_tree(root.left)
    return root

root = mirror_tree(root)
print("Mirrored Tree:")
print_tree(root)

class BinaryTreeNode:
    def __init__(self, data):
        self.data = data
        self.left = None
        self.right = None

def take_input_level_wise():
    q = queue.Queue()
    root_data = int(input("Enter root data (-1 to exit): "))
    if root_data == -1:
        return None
    root = BinaryTreeNode(root_data)
    q.put(root)
    while not q.empty():
        current = q.get()
        left_data = int(input("Enter left child of {}: ".format(current.data)))
        if left_data != -1:
            left_child = BinaryTreeNode(left_data)
            current.left = left_child
            q.put(left_child)
        right_data = int(input("Enter right child of {}: ".format(current.data)))
        if right_data != -1:
            right_child = BinaryTreeNode(right_data)
            current.right = right_child
            q.put(right_child)
    return root

def print_level_wise(root):
    if root is None:
        return
    q = queue.Queue()
    q.put(root)
    while not q.empty():
        current = q.get()
        print(current.data, end=":")
        if current.left:
            print(current.left.data, end=",")
            q.put(current.left)
        if current.right:
            print(current.right.data, end="")
            q.put(current.right)
        print()

root = take_input_level_wise()
print("Level-wise Tree:")
print_level_wise(root)

def construct_tree(postorder, inorder):
    if not postorder or not inorder:
        return None
    root_data = postorder.pop()
    root = BinaryTreeNode(root_data)
    idx = inorder.index(root_data)
    root.right = construct_tree(postorder, inorder[idx+1:])
    root.left = construct_tree(postorder, inorder[:idx])
    return root

def diameter_of_tree(root):
    if root is None:
        return 0

    def dfs(node):
        if node is None:
            return 0, 0
        left_diameter, left_height = dfs(node.left)
        right_diameter, right_height = dfs(node.right)
        return max(left_diameter, right_diameter, left_height + right_height), max(left_height, right_height) + 1

    return dfs(root)[0]

print("Diameter of the tree:", diameter_of_tree(root))
