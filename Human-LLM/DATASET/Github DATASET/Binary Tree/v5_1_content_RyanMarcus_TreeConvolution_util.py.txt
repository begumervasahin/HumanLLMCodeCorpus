import numpy as np
import torch

class TreeConvolutionError(Exception):
    pass

def _is_leaf(node, left_child, right_child):
    has_left = left_child(node) is not None
    has_right = right_child(node) is not None
    if has_left != has_right:
        raise TreeConvolutionError("All nodes must have both a left and a right child or no children")
    return not has_left

def _flatten_tree(root, transformer, left_child, right_child):
    if not callable(transformer):
        raise TreeConvolutionError("Transformer must be a function mapping a tree node to a vector")
    if not callable(left_child) or not callable(right_child):
        raise TreeConvolutionError("left_child and right_child must be functions mapping a tree node to its child, or None")

    flattened_nodes = []

    def recurse(node):
        if _is_leaf(node, left_child, right_child):
            flattened_nodes.append(transformer(node))
            return
        flattened_nodes.append(transformer(node))
        recurse(left_child(node))
        recurse(right_child(node))

    recurse(root)

    try:
        flattened_nodes = [np.zeros(flattened_nodes[0].shape)] + flattened_nodes
    except:
        raise TreeConvolutionError("Output of transformer must have a .shape (e.g., numpy array)")

    return np.array(flattened_nodes)

def _preorder_indexes(root, left_child, right_child, idx=1):
    if not callable(left_child) or not callable(right_child):
        raise TreeConvolutionError("left_child and right_child must be functions mapping a tree node to its child, or None")

    if _is_leaf(root, left_child, right_child):
        return idx

    def rightmost(tree):
        if isinstance(tree, tuple):
            return rightmost(tree[2])
        return tree

    left_subtree = _preorder_indexes(left_child(root), left_child, right_child, idx=idx+1)
    max_index_in_left = rightmost(left_subtree)
    right_subtree = _preorder_indexes(right_child(root), left_child, right_child, idx=max_index_in_left + 1)

    return (idx, left_subtree, right_subtree)

def _tree_convolution_indexes(root, left_child, right_child):
    if not callable(left_child) or not callable(right_child):
        raise TreeConvolutionError("left_child and right_child must be functions mapping a tree node to its child, or None")

    index_tree = _preorder_indexes(root, left_child, right_child)

    def recurse(root):
        if isinstance(root, tuple):
            my_id = root[0]
            left_id = root[1][0] if isinstance(root[1], tuple) else root[1]
            right_id = root[2][0] if isinstance(root[2], tuple) else root[2]
            yield [my_id, left_id, right_id]
            yield from recurse(root[1])
            yield from recurse(root[2])
        else:
            yield [root, 0, 0]

    return np.array(list(recurse(index_tree))).flatten().reshape(-1, 1)

def _pad_and_combine_vectors(vectors):
    assert len(vectors) >= 1
    assert len(vectors[0].shape) == 2

    for vec in vectors:
        if vec.dtype == np.dtype("object"):
            raise TreeConvolutionError("Transformer outputs could not be unified into an array. Are they all the same size?")

    second_dim = vectors[0].shape[1]

    for vec in vectors[1:]:
        assert vec.shape[1] == second_dim

    max_first_dim = max(arr.shape[0] for arr in vectors)
    padded_vectors = []

    for arr in vectors:
        padded = np.zeros((max_first_dim, second_dim))
        padded[0:arr.shape[0]] = arr
        padded_vectors.append(padded)

    return np.array(padded_vectors)

def prepare_trees_for_convolution(trees, transformer, left_child, right_child):
    flat_trees = [_flatten_tree(tree, transformer, left_child, right_child) for tree in trees]
    flat_trees = _pad_and_combine_vectors(flat_trees)
    flat_trees = torch.Tensor(flat_trees)
    flat_trees = flat_trees.transpose(1, 2)
    indexes = [_tree_convolution_indexes(tree, left_child, right_child) for tree in trees]
    indexes = _pad_and_combine_vectors(indexes)
    indexes = torch.Tensor(indexes).long()
    return (flat_trees, indexes)
