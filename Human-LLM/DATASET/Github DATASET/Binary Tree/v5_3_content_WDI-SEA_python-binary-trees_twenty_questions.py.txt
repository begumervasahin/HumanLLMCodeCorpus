import shelve

class QuestionNode:
    """
    Node class representing a question or an answer in the question tree.
    """

    def __init__(self, data):
        """
        Initialize a node with the given data.
        """
        self.data = data
        self.left = None
        self.right = None

class QuestionTree:
    """
    Class representing the question tree for the 20 Questions game.
    """

    def __init__(self):
        """
        Initialize the question tree with a default root node.
        """
        self.root = QuestionNode("hippo")

    def play_game(self):
        """
        Start playing the 20 Questions game.
        """
        print("Welcome to 20 Questions.")
        self.root = self._play_game_helper(self.root)

    def _play_game_helper(self, node):
        """
        Helper function to recursively play the game.
        """
        if node.left is None and node.right is None:
            print("You're thinking of:", node.data)
            response = input("Is this correct? (y/n) ")
            if response == "y":
                print("I win!!")
                return node
            else:
                return self._add_answer(node)
        else:
            print(node.data)
            response = input("Is this correct? (y/n) ")
            if response == "y":
                node.right = self._play_game_helper(node.right)
            elif response == "n":
                node.left = self._play_game_helper(node.left)
            else:
                self._play_game_helper(node)
            return node

    def _add_answer(self, old_node):
        """
        Add a new answer and question to the tree based on user input.
        """
        your_thing = input("Enter the thing you were thinking of: ")
        question = input("Enter a disambiguating question: ")
        yes_or_no = input("Is the answer 'yes' for your thing? (y/n) ")
        your_thing = QuestionNode(your_thing)
        new_question = QuestionNode(question)

        if yes_or_no == "y":
            new_question.right = your_thing
            new_question.left = old_node
        elif yes_or_no == "n":
            new_question.right = old_node
            new_question.left = your_thing
        else:
            print("Invalid input. Please enter 'y' or 'n'.")
            return self._add_answer(old_node)

        return new_question

def main():
    """
    Main function to run the 20 Questions game.
    """
    savefile = shelve.open("twenty_questions.save")

    if "tree" in savefile:
        twenty_questions = savefile["tree"]
    else:
        twenty_questions = QuestionTree()

    twenty_questions.play_game()
    keep_playing = True

    while keep_playing:
        savefile["tree"] = twenty_questions
        response = input("Do you want to play again? (y/n) ")
        if response == "y":
            twenty_questions.play_game()
        elif response == "n":
            print("Thanks for playing! Goodbye.")
            keep_playing = False
        else:
            print("Invalid input. Please enter 'y' or 'n'.")

    savefile.close()

if __name__ == "__main__":
    main()
