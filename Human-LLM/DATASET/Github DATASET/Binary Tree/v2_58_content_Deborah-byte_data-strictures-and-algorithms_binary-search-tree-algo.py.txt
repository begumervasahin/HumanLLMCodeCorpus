class TreeNode:
    def __init__(self, data):
        self.left = None
        self.right = None
        self.data = data

    def insert(self, data):
        """Insert a new node with given data into the tree."""
        if self.data:
            if data < self.data:
                if self.left is None:
                    self.left = TreeNode(data)
                else:
                    self.left.insert(data)
            elif data > self.data:
                if self.right is None:
                    self.right = TreeNode(data)
                else:
                    self.right.insert(data)
        else:
            self.data = data

    def find_value(self, lookup_value):
        """Search for a value in the tree."""
        if lookup_value < self.data:
            if self.left is None:
                return str(lookup_value) + " Not Found"
            return self.left.find_value(lookup_value)
        elif lookup_value > self.data:
            if self.right is None:
                return str(lookup_value) + " Not Found"
            return self.right.find_value(lookup_value)
        else:
            return str(self.data) + ' is found'

    def print_tree(self):
        """Print the tree in sorted order."""
        if self.left:
            self.left.print_tree()
        print(self.data)
        if self.right:
            self.right.print_tree()

# Creating the root node and inserting values
root = TreeNode(14)
root.insert(6)
root.insert(18)
root.insert(3)

# Testing find_value method
print(root.find_value(7))   # Output: 7 Not Found
print(root.find_value(14))  # Output: 14 is found

# Printing the tree in sorted order
root.print_tree()
