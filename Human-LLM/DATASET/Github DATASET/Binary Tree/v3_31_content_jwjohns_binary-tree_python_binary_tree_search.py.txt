import datetime
from random import randint

class Node:
    def __init__(self, value):
        """Initialize a node with a given value."""
        self.value = value
        self.left_child = None
        self.right_child = None

    def insert(self, data):
        """Insert data into the binary search tree rooted at this node."""
        if self.value == data:
            return False
        elif self.value > data:
            if self.left_child:
                return self.left_child.insert(data)
            else:
                self.left_child = Node(data)
                return True
        else:
            if self.right_child:
                return self.right_child.insert(data)
            else:
                self.right_child = Node(data)
                return True

    def find(self, data):
        """Check if the given data is present in the binary search tree rooted at this node."""
        if self.value == data:
            return True
        elif self.value > data and self.left_child:
            return self.left_child.find(data)
        elif self.value < data and self.right_child:
            return self.right_child.find(data)
        else:
            return False

    def preorder(self):
        """Traverse the binary search tree rooted at this node in preorder."""
        if self:
            print(str(self.value))
            if self.left_child:
                self.left_child.preorder()
            if self.right_child:
                self.right_child.preorder()

    def postorder(self):
        """Traverse the binary search tree rooted at this node in postorder."""
        if self:
            if self.left_child:
                self.left_child.postorder()
            if self.right_child:
                self.right_child.postorder()
            print(str(self.value))

    def inorder(self):
        """Traverse the binary search tree rooted at this node in inorder."""
        if self:
            if self.left_child:
                self.left_child.inorder()
            print(str(self.value))
            if self.right_child:
                self.right_child.inorder()

class BinaryTree:
    def __init__(self):
        """Initialize an empty binary search tree."""
        self.root = None

    def insert(self, data):
        """Insert data into the binary search tree."""
        if self.root:
            return self.root.insert(data)
        else:
            self.root = Node(data)
            return True

    def find(self, data):
        """Check if the given data is present in the binary search tree."""
        if self.root:
            return self.root.find(data)
        else:
            return False

    def preorder(self):
        """Print the elements of the binary search tree in preorder."""
        print("PreOrder")
        self.root.preorder()

    def postorder(self):
        """Print the elements of the binary search tree in postorder."""
        print("PostOrder")
        self.root.postorder()

    def inorder(self):
        """Print the elements of the binary search tree in inorder."""
        print("InOrder")
        self.root.inorder()

# Example usage
bst = BinaryTree()
n = 10000
start = datetime.datetime.now()
for i in range(n):
    bst.insert(randint(0, 20000))
finish = datetime.datetime.now()
print((finish - start).microseconds)

start = datetime.datetime.now()
bst.inorder()
finish = datetime.datetime.now()
print((finish - start).microseconds % 1000)
