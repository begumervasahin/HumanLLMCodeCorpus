# Import necessary libraries
import serial
import time
import json
from Crypto.Cipher import AES

# Define the AES encryption key
encryption_key = 'abcdefghijklmnop'

# Create an AES cipher object with ECB mode
cipher = AES.new(encryption_key, AES.MODE_ECB)

# Configure the serial port
serial_port = '/dev/ttyS0'
baud_rate = 115200
serial_connection = serial.Serial(port=serial_port, baudrate=baud_rate)

# Define the JSON data to send
json_data = {"oi": "abcdef"}

# Print the name of the serial port
print("Serial port name:", serial_connection.name)

# Continuous loop to send encrypted JSON messages
while True:
    try:
        # Convert JSON data to a string
        json_string = json.dumps(json_data)
        
        # Encrypt the message using AES
        encrypted_message = cipher.encrypt(json_string.encode())
        
        # Write the encrypted message to the serial port
        serial_connection.write(encrypted_message)
        
        # Wait for 1 second before sending the next message
        time.sleep(1)
    
    except KeyboardInterrupt:
        # Close the serial connection if Ctrl+C is pressed
        serial_connection.close()
        print("Serial connection closed. Exiting.")
        break
    
    except Exception as e:
        # Handle any other exceptions
        print("An error occurred:", str(e))
