import struct

# Function to perform a left rotation on a 32-bit integer
def left_rotate(i, n):
    return ((i << n) & 0xffffffff) | (i >> (32 - n))

# MD4 round functions
def F(x, y, z):
    return (x & y) | (~x & z)

def G(x, y, z):
    return (x & y) | (x & z) | (y & z)

def H(x, y, z):
    return x ^ y ^ z

class MD4(object):
    def __init__(self):
        # Initial state variables
        self.remainder = ""
        self.count = 0
        self.h = [
            0x67452301,
            0xefcdab89,
            0x98badcfe,
            0x10325476
        ]

    # Process a 512-bit chunk of the input message
    def _add_chunk(self, chunk):
        # Increment the message length in bits
        self.count += 1

        # Unpack the chunk into 16 32-bit words
        X = list(struct.unpack("<16I", chunk) + (None,) * (80 - 16))

        # Make a copy of the initial hash values
        h = [x for x in self.h]

        # Round constants
        s = ((3, 7, 11, 19), (3, 5, 9, 13), (3, 9, 11, 15))
        k = ((0, 1, 2, 3), (0, 4, 8, 12), (0, 2, 1, 3))
        t = (0x00000000, 0x5a827999, 0x6ed9eba1)

        # Perform the 3 rounds of processing
        for i in range(3):
            round_shifts = s[i]
            round_indices = k[i]
            round_constant = t[i]

            for r in range(16):
                idx = (16 - r) % 4
                h[idx] = left_rotate((h[idx] + globals()[chr(72 + i)](h[(idx + 1) % 4], h[(idx + 2) % 4], h[(idx + 3) % 4]) + X[round_indices[r]] + round_constant) % 2 ** 32, round_shifts[r % 4])

        # Update the hash values
        for i, v in enumerate(h):
            self.h[i] = (v + self.h[i]) % 2 ** 32

    # Update the hash with additional message data
    def update(self, data):
        # Append the new data to any remaining data from previous updates
        message = self.remainder + data

        # Compute the length of the message in bytes
        remainder_length = len(message) % 64

        # Determine the number of padding bytes needed to reach a multiple of 512 bits
        padding_length = (55 - remainder_length) % 64

        # Pad the message with a single '1' bit followed by zeros, and the length of the original message in bits
        padded_message = "\x80" + "\x00" * padding_length + struct.pack("<Q", len(message) * 8)

        # Process each 512-bit chunk of the message
        for chunk_start in range(0, len(message) - remainder_length, 64):
            self._add_chunk(message[chunk_start:chunk_start + 64])

        # Store any remaining data for the next update
        self.remainder = message[-remainder_length:]

        # Return self for method chaining
        return self

    # Finalize the hash computation and return the digest
    def finish(self):
        # Final padding and processing of remaining data
        self.update("\x80" + "\x00" * ((55 - len(self.remainder)) % 64) + struct.pack("<Q", len(self.remainder) * 8))

        # Pack the hash values into bytes and reset the hash state
        digest = struct.pack("<4I", *self.h)
        self.__init__()

        # Return the hexadecimal representation of the digest
        return digest.encode("hex")

# Example usage:
md4 = MD4()
md4.update("Hello, world!")
print(md4.finish())
