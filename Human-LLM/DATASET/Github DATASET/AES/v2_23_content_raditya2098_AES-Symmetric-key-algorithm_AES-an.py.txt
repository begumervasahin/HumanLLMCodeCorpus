def sbox_substitute(input_string):
    """
    Perform S-box substitution for each character in the input string.
    
    Args:
    - input_string: The string to be substituted.
    
    Returns:
    - The substituted string after applying the AES S-box.
    """
    # Initialize an empty list to hold substituted hexadecimal values
    substituted_hex_values = []

    # Iterate over each character in the input string
    for char in input_string:
        # Convert the character to its ASCII value and then to a hexadecimal string
        char_hex = hex(ord(char))
        
        # Extract the hexadecimal value (without the '0x' prefix)
        hex_value = char_hex[2:]
        
        # Convert the hex value to an integer, use it to look up the S-box, and then convert back to hex
        substituted_hex = hex(Sbox[int(hex_value, 16)])
        
        # Append the substituted hex value to the list
        substituted_hex_values.append(substituted_hex)

    # Convert the list of substituted hex values back into a string
    # For each substituted value, convert from hex to integer, then to a character, and join them together
    substituted_string = ''.join(chr(int(hex_val, 16)) for hex_val in substituted_hex_values)

    return substituted_string
