import sys

def read_input(filename):
    """
    Read input from a file and extract flags and outputs from each line.
    Returns a tuple of lists: (flags, outputs).
    """
    flags = []
    outputs = []

    # Open the file and read its contents
    with open(filename, 'r') as file:
        lines = file.read().splitlines()

    # Iterate over each line in the file
    for line in lines:
        # Find the position of ':' to separate flag and output
        delimiter_index = line.find(":")

        # Extract flag and output from the line
        flag = line[:delimiter_index]
        output = line[delimiter_index + 1:]

        flags.append(flag)
        outputs.append(output)

    return flags, outputs

def main():
    # Check if the input file is provided
    if len(sys.argv) < 2:
        print("Usage: python script.py <input_file>")
        return

    # Get the filename from command-line arguments
    filename = sys.argv[1]

    # Read input from the file
    flags, outputs = read_input(filename)

    # Print the list of outputs
    print(outputs)

if __name__ == "__main__":
    main()
