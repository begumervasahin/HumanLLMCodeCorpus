from Crypto.Cipher import AES
from Crypto.Util.Padding import pad, unpad
from Crypto.Random import get_random_bytes
import base64

class AESCipher:
    def __init__(self, key):
        self.key = key.encode('utf-8')  # Ensure key is in bytes

    def encrypt(self, message):
        message = message.encode('utf-8')  # Ensure message is in bytes
        cipher = AES.new(self.key, AES.MODE_CBC)
        ct_bytes = cipher.encrypt(pad(message, AES.block_size))
        iv = base64.b64encode(cipher.iv).decode('utf-8')
        ct = base64.b64encode(ct_bytes).decode('utf-8')
        return iv + ct

    def decrypt(self, encrypted_message):
        iv = base64.b64decode(encrypted_message[:24])  # Extract iv
        ct = base64.b64decode(encrypted_message[24:])  # Extract the actual encrypted message
        cipher = AES.new(self.key, AES.MODE_CBC, iv=iv)
        pt = unpad(cipher.decrypt(ct), AES.block_size)
        return pt.decode('utf-8')

# Usage
key = "Thats my Kung Fu"  # AES requires a key of 16 (AES-128), 24 (AES-192), or 32 (AES-256) bytes
if len(key) not in [16, 24, 32]:
    raise ValueError("Key must be 16, 24, or 32 bytes long")

message = "Two One Nine Two"
cipher = AESCipher(key)

# Since AES key needs to be either 16, 24, or 32 bytes long, we'll adjust the provided key.
# For demonstration, let's just pad the key if it's not the correct length (not secure, just for demo).
correct_length_key = pad(key.encode(), 16)[:16]  # Adjust key to 16 bytes for AES-128 (for demo purposes)
cipher.key = correct_length_key

enc = cipher.encrypt(message)
print("Message:\t", message)
print("Encrypted:\t", enc)

# For demonstration, showing how to decrypt as well
dec = cipher.decrypt(enc)
print("Decrypted:\t", dec)
