import sys
from hashlib import sha256
from Crypto import Random
from Crypto.Cipher import AES

# Define the block size for AES encryption
BLOCK_SIZE = 16

def pad(text):
    """
    Pads an input string to ensure its length is a multiple of BLOCK_SIZE.
    """
    padding = BLOCK_SIZE - len(text) % BLOCK_SIZE
    return text + chr(padding) * padding

def unpad(text):
    """
    Removes padding from the input string.
    """
    return text[:-ord(text[-1])]

class AESCipher:
    """
    A class to handle AES encryption and decryption.
    """
    def __init__(self, key):
        self.key = sha256(key.encode()).digest()

    def encrypt(self, plaintext):
        """
        Encrypts plaintext.
        """
        plaintext = pad(plaintext)
        iv = Random.new().read(AES.block_size)
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        return iv + cipher.encrypt(plaintext.encode())

    def decrypt(self, ciphertext):
        """
        Decrypts ciphertext.
        """
        iv = ciphertext[:BLOCK_SIZE]
        cipher = AES.new(self.key, AES.MODE_CBC, iv)
        return unpad(cipher.decrypt(ciphertext[BLOCK_SIZE:])).decode()

def print_usage_and_exit():
    """
    Prints the script usage and exits.
    """
    print("Usage: script.py <encrypt|decrypt> <filename> <key>")
    sys.exit(1)

def main():
    if len(sys.argv) != 4:
        print_usage_and_exit()

    option, filename, key = sys.argv[1:4]
    if option not in ["encrypt", "decrypt"]:
        print_usage_and_exit()

    try:
        with open(filename, "rb") as f:
            data = f.read()
    except IOError as e:
        print(f"Failed to read '{filename}': {e}")
        sys.exit(1)

    cipher = AESCipher(key)

    if option == "encrypt":
        processed_data = cipher.encrypt(data.decode())
        output_filename = f"{filename}.crypt"
    else:  # decrypt
        processed_data = cipher.decrypt(data).encode()
        output_filename = filename.rsplit('.', 1)[0]  # Remove the .crypt extension more reliably

    try:
        with open(output_filename, "wb") as f:
            f.write(processed_data)
    except IOError as e:
        print(f"Failed to write '{output_filename}': {e}")
        sys.exit(1)

if __name__ == "__main__":
    main()
