import torch
import torch.nn as nn
import torchvision.transforms as transforms
from PIL import Image

# Define the basic block for the encoder and decoder
class BasicBlock(nn.Module):
    def __init__(self, channels=128, stride=1, padding=1):
        super(BasicBlock, self).__init__()
        self.conv1 = nn.Conv2d(channels, channels, kernel_size=3, stride=stride, padding=padding)
        self.bn1 = nn.BatchNorm2d(channels)
        self.prelu1 = nn.PReLU()
        self.conv2 = nn.Conv2d(channels, channels, kernel_size=3, stride=stride, padding=padding)
        self.bn2 = nn.BatchNorm2d(channels)
        self.prelu2 = nn.PReLU()

    def forward(self, x):
        identity = x
        x = self.prelu1(self.bn1(self.conv1(x)))
        x = self.bn2(self.conv2(x)) + identity
        return self.prelu2(x)

# Pre-Encoder Network
class PreEncoder(nn.Module):
    def __init__(self):
        super(PreEncoder, self).__init__()
        self.input_pass = nn.Sequential(
            nn.Conv2d(3, 32, kernel_size=3, stride=1, padding=1),
            nn.BatchNorm2d(32),
            nn.PReLU(),
            nn.Conv2d(32, 32, kernel_size=3, stride=1, padding=1),
            nn.BatchNorm2d(32),
            nn.PReLU(),
            nn.Conv2d(32, 64, kernel_size=3, stride=1, padding=1),
            nn.BatchNorm2d(64),
            nn.PReLU()
        )

    def forward(self, x):
        return self.input_pass(x)

# Encoder Network
class Encoder(nn.Module):
    def __init__(self, block):
        super(Encoder, self).__init__()
        self.block = block
        self.input_conv = nn.Sequential(
            nn.Conv2d(64, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.PReLU(),
            nn.Conv2d(128, 128, kernel_size=3, padding=1),
            nn.BatchNorm2d(128),
            nn.PReLU(),
        )
        self.layer1 = self._make_layers(1)
        self.layer2 = self._make_layers(1)
        self.layer3 = self._make_layers(1)
        self.downsample_conv1 = nn.Sequential(
            nn.Conv2d(128, 128, kernel_size=5, stride=2),
            nn.BatchNorm2d(128),
            nn.PReLU()
        )
        self.downsample_conv2 = nn.Sequential(
            nn.Conv2d(128, 128, kernel_size=5, stride=2),
            nn.BatchNorm2d(128),
            nn.PReLU()
        )
        self.downsample_conv3 = nn.Sequential(
            nn.Conv2d(128, 128, kernel_size=5, stride=2),
            nn.BatchNorm2d(128),
            nn.PReLU()
        )

    def _make_layers(self, num_layers):
        layers = []
        for _ in range(num_layers):
            layers.append(self.block())
        return nn.Sequential(*layers)

    def forward(self, x):
        x = self.input_conv(x)
        x = self.layer1(x)
        x = self.downsample_conv1(x)
        x = self.layer2(x)
        x = self.downsample_conv2(x)
        x = self.layer3(x)
        x = self.downsample_conv3(x)
        return x

# Decoder Network
class Decoder(nn.Module):
    def __init__(self, block):
        super(Decoder, self).__init__()
        self.block = block
        self.layer1 = self._make_layers(1)
        self.layer2 = self._make_layers(1)
        self.layer3 = self._make_layers(1)
        self.trans_conv1 = nn.Sequential(
            nn.ConvTranspose2d(128, 128, kernel_size=4, stride=2),
            nn.BatchNorm2d(128),
            nn.PReLU()
        )
        self.trans_conv2 = nn.Sequential(
            nn.ConvTranspose2d(128, 128, kernel_size=4, stride=2),
            nn.BatchNorm2d(128),
            nn.PReLU()
        )
        self.trans_conv3 = nn.Sequential(
            nn.ConvTranspose2d(128, 128, kernel_size=4, stride=2),
            nn.BatchNorm2d(128),
            nn.PReLU()
        )
        self.trans_conv4 = nn.Sequential(
            nn.ConvTranspose2d(128, 128, kernel_size=5, stride=2),
            nn.BatchNorm2d(128),
            nn.PReLU()
        )
        self.trans_conv5 = nn.Sequential(
            nn.ConvTranspose2d(128, 128, kernel_size=5, stride=2),
            nn.BatchNorm2d(128),
            nn.PReLU()
        )
        self.output_conv = nn.Sequential(
            nn.Conv2d(128, 64, kernel_size=2),
            nn.BatchNorm2d(64),
            nn.PReLU(),
            nn.Conv2d(64, 32, kernel_size=3, padding=1),
            nn.BatchNorm2d(32),
            nn.PReLU(),
            nn.Conv2d(32, 3, kernel_size=3, padding=1),
        )

    def _make_layers(self, num_layers):
        layers = []
        for _ in range(num_layers):
            layers.append(self.block())
        return nn.Sequential(*layers)

    def forward(self, x):
        x = self.trans_conv1(x)
        x = self.layer1(x)
        x = self.trans_conv2(x)
        x = self.layer2(x)
        x = self.trans_conv3(x)
        x = self.layer3(x)
        x = self.trans_conv4(x)
        x = self.trans_conv5(x)
        x = self.output_conv(x)
        return x

# Discriminator Network
class Discriminator(nn.Module):
    def __init__(self, block):
        super(Discriminator, self).__init__()
        self.block = block
        self.layer1 = self._make_layers(1)
        self.layer2 = self._make_layers(1)
        self.layer3 = self._make_layers(1)
        self.main_conv1 = nn.Sequential(
            nn.Conv2d(3, 32, kernel_size=3, stride=2),
            nn.BatchNorm2d(32),
            nn.PReLU()
        )
        self.main_conv2 = nn.Sequential(
            nn.Conv2d(32, 128, kernel_size=3, stride=2),
            nn.BatchNorm2d(128),
            nn.PReLU()
        )
        self.main_conv3 = nn.Sequential(
            nn.Conv2d(128, 128, kernel_size=5, stride=2, padding=2),
            nn.BatchNorm2d(128),
            nn.PReLU()
        )
        self.main_conv4 = nn.Sequential(
            nn.Conv2d(128, 128, kernel_size=3, stride=2),
            nn.BatchNorm2d(128),
            nn.PReLU()
        )
        self.main_conv5 = nn.Sequential(
            nn.Conv2d(128, 64, kernel_size=4, stride=2),
            nn.BatchNorm2d(64),
            nn.PReLU()
        )
        self.main_conv6 = nn.Sequential(
            nn.Conv2d(64, 32, kernel_size=3, stride=2),
            nn.BatchNorm2d(32),
            nn.PReLU()
        )
        self.main_conv7 = nn.Sequential(
            nn.Conv2d(32, 16, kernel_size=3, stride=1),
            nn.BatchNorm2d(16),
            nn.PReLU()
        )
        self.main_conv8 = nn.Sequential(
            nn.Linear(256, 128),
            nn.BatchNorm1d(128),
            nn.PReLU(),
            nn.Linear(128, 10),
            nn.BatchNorm1d(10),
            nn.PReLU(),
            nn.Linear(10, 2),
        )

    def _make_layers(self, num_layers):
        layers = []
        for _ in range(num_layers):
            layers.append(self.block())
        return nn.Sequential(*layers)

    def forward(self, x):
        x = self.main_conv1(x)
        x = self.main_conv2(x)
        x = self.layer1(x)
        x = self.main_conv3(x)
        x = self.layer2(x)
        x = self.main_conv4(x)
        x = self.layer3(x)
        x = self.main_conv5(x)
        x = self.main_conv6(x)
        x = self.main_conv7(x)
        x = x.view(x.size(0), -1)
        x = self.main_conv8(x)
        return x
