import collections

class SimpleGraph:
    def __init__(self):
        self.edges = {}
    
    def neighbors(self, node):
        """Return the neighbors of the given node."""
        return self.edges.get(node, [])

    def print_graph(self):
        """Print the graph's nodes and their corresponding neighbors."""
        for node, neighbors in self.edges.items():
            print(node, neighbors)

class Stack:
    def __init__(self):
        self.elements = collections.deque()
    
    def empty(self):
        """Check if the stack is empty."""
        return len(self.elements) == 0
    
    def push(self, x):
        """Push an item onto the stack."""
        self.elements.append(x)
    
    def pop(self):
        """Pop the top item from the stack."""
        return self.elements.pop()

def depth_first_search(graph, start_node):
    """Perform depth-first search starting from the given node."""
    frontier = Stack()
    frontier.push(start_node)
    visited = {start_node: True}

    while not frontier.empty():
        current_node = frontier.pop()
        print("Visiting", current_node)
        for neighbor_node in graph.neighbors(current_node):
            if neighbor_node not in visited:
                frontier.push(neighbor_node)
                visited[neighbor_node] = True

# Create an instance of SimpleGraph
example_graph = SimpleGraph()
example_graph.edges = {
    'A': ['B'],
    'B': ['A', 'C', 'D'],
    'C': ['A'],
    'D': ['E', 'A'],
    'E': ['B']
}

# Perform depth-first search starting from node 'A'
depth_first_search(example_graph, 'A')
