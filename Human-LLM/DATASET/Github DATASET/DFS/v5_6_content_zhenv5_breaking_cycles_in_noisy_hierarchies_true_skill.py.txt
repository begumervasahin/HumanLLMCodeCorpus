from trueskill import Rating, rate_1vs1
import networkx as nx
import random
from datetime import datetime
from measures import measure_pairs_agreement
import argparse

def compute_trueskill(pairs, players):
    """
    Compute Trueskill ratings for players based on matches.

    Args:
        pairs (list): List of player pairs representing matches.
        players (dict): Dictionary containing players and their ratings.

    Returns:
        dict: Updated players dictionary with Trueskill ratings.
    """
    # Initialize players if not already initialized
    if not players:
        for u, v in pairs:
            if u not in players:
                players[u] = Rating()
            if v not in players:
                players[v] = Rating()

    # Shuffle pairs for randomness
    random.shuffle(pairs)

    # Update players' ratings based on matches
    for u, v in pairs:
        players[v], players[u] = rate_1vs1(players[v], players[u])

    return players

def get_players_score(players, n_sigma):
    """
    Calculate relative scores for players based on their Trueskill ratings.

    Args:
        players (dict): Dictionary containing players and their Trueskill ratings.
        n_sigma (int): Number of standard deviations for calculating relative scores.

    Returns:
        dict: Relative scores for each player.
    """
    # Calculate relative scores for players
    relative_score = {}
    for k, v in players.items():
        relative_score[k] = players[k].mu - n_sigma * players[k].sigma
    return relative_score

def trueskill_ratings(pairs, iter_times=15, n_sigma=3, threshold=0.85):
    """
    Compute Trueskill ratings for players based on multiple iterations.

    Args:
        pairs (list): List of player pairs representing matches.
        iter_times (int): Number of iterations to compute Trueskill ratings.
        n_sigma (int): Number of standard deviations for calculating relative scores.
        threshold (float): Threshold for accuracy to stop iterations.

    Returns:
        dict: Relative scores for each player.
    """
    start_time = datetime.now()
    players = {}

    # Iterate to compute Trueskill ratings
    for i in range(iter_times):
        players = compute_trueskill(pairs, players)
        relative_scores = get_players_score(players, n_sigma=n_sigma)
        accu = measure_pairs_agreement(pairs, relative_scores)
        if accu >= threshold:
            return relative_scores

    end_time = datetime.now()
    time_used = end_time - start_time
    print("Time used in computing Trueskill: %0.4f s, iteration time is: %i" % (time_used.seconds, (i + 1)))
    return relative_scores

def graphbased_trueskill(g, iter_times=15, n_sigma=3, threshold=0.95):
    """
    Compute graph-based Trueskill ratings for players.

    Args:
        g (networkx.DiGraph): Directed graph representing player matches.
        iter_times (int): Number of iterations to compute Trueskill ratings.
        n_sigma (int): Number of standard deviations for calculating relative scores.
        threshold (float): Threshold for accuracy to stop iterations.

    Returns:
        dict: Relative scores for each player.
    """
    from s_c_c import scc_nodes_edges
    
    # Compute Trueskill ratings
    relative_scores = trueskill_ratings(list(g.edges()), iter_times=iter_times, n_sigma=n_sigma, threshold=threshold)

    # Split edges into strongly connected and non-strongly connected components
    scc_nodes, scc_edges, nonscc_nodes, nonscc_edges = scc_nodes_edges(g)

    # Print and return results
    print("---- Strongly Connected Components ----")
    scc_accu = measure_pairs_agreement(scc_edges, relative_scores)
    print("---- Non-Strongly Connected Components ----")
    nonscc_accu = measure_pairs_agreement(nonscc_edges, relative_scores)
    print("SCC Accuracy: %0.4f, Non-SCC Accuracy: %0.4f" % (scc_accu, nonscc_accu))
    
    return relative_scores

def main(edges_file_name="/home/sunjiank/Dropbox/Data/cit-Patents/cit-Patents.txt"):    
    """
    Main function to compute Trueskill ratings for a given graph.

    Args:
        edges_file_name (str): File path of the graph edges list.

    Returns:
        None
    """
    # Read the graph from the edges file
    g = nx.read_edgelist(edges_file_name, create_using=nx.DiGraph(), nodetype=int)      

    # Compute graph-based Trueskill ratings
    graphbased_trueskill(g)

if __name__ == "__main__":
    # Parse command-line arguments
    parser = argparse.ArgumentParser()
    parser.add_argument("-g", "--graph", type=str, default="", help="Graph edges list file")
    args = parser.parse_args()
    edges_file_name = args.graph

    # Execute main function with the specified edges file
    main(edges_file_name)
