import sys
import itertools
import subprocess
import numpy as np
from datetime import datetime
from os import path, environ, mkdir
import tabulate_results

# Global Configuration
get_results_only = False
switch_gpus = False
n_parallel_threads = 5

# Define Hyperparameters and Values
args = {
    'hyper_params': ['dataset', 'lr', 'l2', 'drop_in', 'drop_out', 'wce'],
    'dataset': ['cora', 'citeseer', 'wiki', 'cora_multi', 'mlgene'],
    'lr': [1e-3],
    'l2': [1e-5],
    'drop_in': [0.25],
    'drop_out': [0.5],
    'wce': [0]
}

# Generate a timestamp with custom suffix
custom_suffix = '_subGraph_5_'
now = datetime.now()
args['timestamp'] = f"{now.month}|{now.day}|{now.hour}:{now.minute}:{now.second}{custom_suffix}"

# Swap 'dataset' with the first hyper-parameter
dataset_index = args['hyper_params'].index('dataset')
args['hyper_params'][0], args['hyper_params'][dataset_index] = args['hyper_params'][dataset_index], args['hyper_params'][0]

# Save Arguments to a File
args_path = 'args'
if not path.exists(args_path):
    mkdir(args_path)
np.save(path.join(args_path, args['timestamp']), args)

# Create Directory for Stdout Dumps if it Doesn't Exist
stdout_dump_path = 'stdout_dumps'
if not path.exists(stdout_dump_path):
    mkdir(stdout_dump_path)

# Generate Combinations of Hyperparameter Values
param_values = [args[param] for param in args['hyper_params']]
combinations = list(itertools.product(*param_values))
n_combinations = len(combinations)
print('Total number of experiments:', n_combinations)

# Initialize Lists for Process IDs and File Handles
pids = [None] * n_combinations
file_handles = [None] * n_combinations
last_process = False

# Iterate over Each Combination of Hyperparameter Values
for i, setting in enumerate(combinations):
    command = "python __main__.py "
    folder_suffix = args['timestamp']

    # Construct Command Line Arguments Based on Hyperparameter Values
    for name, value in zip(args['hyper_params'], setting):
        command += f"--{name} {value} "
        if name != 'dataset':
            folder_suffix += f"_{value}"
    command += f"--folder_suffix {folder_suffix}"
    print(f"{i+1}/{n_combinations} {command}")

    # Set CUDA Environment Variables for GPU Switching if Specified
    cuda_device = "1" if switch_gpus and i % 2 == 0 else "0"
    env = dict(environ, CUDA_DEVICE_ORDER="PCI_BUS_ID", CUDA_VISIBLE_DEVICES=cuda_device)

    # Execute the Command with Subprocess
    output_file = path.join(stdout_dump_path, folder_suffix)
    with open(output_file, 'w') as file_handles[i]:
        pids[i] = subprocess.Popen(command.split(), env=env, stdout=file_handles[i])

    # Wait for Processes to Finish in Batches
    if i == n_combinations - 1:
        last_process = True
    if ((i + 1) % n_parallel_threads == 0 and i >= n_parallel_threads - 1) or last_process:
        if last_process and not ((i + 1) % n_parallel_threads) == 0:
            n_parallel_threads = (i + 1) % n_parallel_threads
        start_time = datetime.now()
        print('Waiting for processes to finish...')
        for t in range(n_parallel_threads - 1, -1, -1):
            pids[i - t].wait()
        end_time = datetime.now()
        print('Batch execution time:', end_time - start_time)

# Write Results if Not in get_results_only Mode
if not get_results_only:
    tabulate_results.write_results(args)
else:
    tabulate_results.write_results(args)
    print("Tabulation completed.")
