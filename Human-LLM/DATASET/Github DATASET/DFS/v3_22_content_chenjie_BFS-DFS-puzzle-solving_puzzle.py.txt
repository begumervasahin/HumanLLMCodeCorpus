from abc import ABC, abstractmethod

class Puzzle(ABC):
    """Abstract base class representing a full-information puzzle."""

    @abstractmethod
    def fail_fast(self):
        """Check if Puzzle self can never be extended to a solution."""
        return False

    @abstractmethod
    def is_solved(self):
        """Check if Puzzle self is solved."""
        raise NotImplementedError

    @abstractmethod
    def extensions(self):
        """Generate legal extensions of Puzzle self."""
        raise NotImplementedError

class NumberPuzzle(Puzzle):
    """Concrete subclass of Puzzle representing a number arrangement puzzle."""

    def __init__(self, numbers):
        """Initialize the NumberPuzzle with a list of numbers."""
        self.numbers = numbers

    def fail_fast(self):
        """Check if the NumberPuzzle can't be extended to a solution."""
        return False

    def is_solved(self):
        """Check if the NumberPuzzle is solved."""
        return sorted(self.numbers) == list(range(1, len(self.numbers) + 1))

    def extensions(self):
        """Generate legal extensions of the NumberPuzzle."""
        for i in range(len(self.numbers)):
            for j in range(i + 1, len(self.numbers)):
                new_numbers = self.numbers[:]
                new_numbers[i], new_numbers[j] = new_numbers[j], new_numbers[i]
                yield NumberPuzzle(new_numbers)

# Usage example
if __name__ == "__main__":
    initial_puzzle = NumberPuzzle([3, 1, 2])
    print("Initial puzzle:", initial_puzzle.numbers)
    print("Is solved?", initial_puzzle.is_solved())
    print("Extensions:", [p.numbers for p in initial_puzzle.extensions()])
