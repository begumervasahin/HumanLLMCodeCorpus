# Default arguments for lineup generation
DEFAULT_ARGS = {
    'duo': 'n',             # Allow duo type positions
    's': 'n',               # Undetermined argument
    'w': 5,                 # Undetermined argument
    'i': 1,                 # Undetermined argument
    'league': 'NBA',        # Fantasy league type
    'limit': 'n',           # Limitation flag
    'lp': 0,                # Undetermined argument
    'mp': 500,              # Maximum players to consider
    'ms': 100000,           # Maximum salary limit
    'sp': 1000,             # Salary limit
    'banned': [],           # Banned players list
    'historical_date': None,# Historical date for data
    'game': 'fanduel',      # Fantasy game type
    'po': 0,                # Undetermined argument
    'pids': 'data/pid-file-nba-fanduel.csv',             # Player IDs file
    'salary_file': 'data/current-nba-salaries-fanduel.csv',    # Salary data file
    'projection_file': 'data/current-nba-projections-fanduel.csv',  # Player projections file
    'home': None,           # Home team filter
    'v_avg': 100,           # Undetermined argument
    'source': 'nba_rotogrinders',   # Data source
    'historical': None,     # Historical data flag
    'season': None,         # Season filter
    'dtype': None,          # Data type filter
    'flex_position': None,  # Flex position filter
    'locked': [],           # Locked players list
    'teams': None,          # Team filter
    'po_location': None,    # Undetermined argument
    'min_avg': None,        # Minimum average score
}

def is_duplicate(roster, roster_list):
    """
    Checks if a roster is a duplicate of any roster in the list.
    """
    roster_players = sorted([p.name for p in roster.players])
    return any(sorted([p.name for p in r.players]) == roster_players for r in roster_list)

def generate_lineups(lineups, exposure, min_avg):
    """
    Generates multiple fantasy lineups with specified constraints.
    """
    DEFAULT_ARGS['min_avg'] = min_avg
    roster_list, player_list = [], []
    max_exposure = lineups * exposure
    exposure_counter = None

    for _ in range(lineups):
        args = DEFAULT_ARGS.copy()

        if exposure_counter:
            # Exclude players exceeding maximum exposure
            args['banned'] = [name for name, freq in exposure_counter.items() if freq > max_exposure]

        roster = optimizer_run('NBA', Namespace(**args), [])

        if is_duplicate(roster, roster_list):
            # Regenerate roster if it's a duplicate
            while is_duplicate(roster, roster_list):
                args['banned'].append(random.choice(roster.players).name)
                roster = optimizer_run('NBA', Namespace(**args), [])

        roster_list.append(roster)
        player_list += [p.name for p in roster.players]
        exposure_counter = Counter(player_list)

    unique_lineups = len(set([str(r.players) for r in roster_list]))

    if unique_lineups != lineups:
        raise Exception('Duplication error in logic. Expected {} and got {} lineups'.format(lineups, unique_lineups))

    # Calculate and display lineup statistics
    scores = [r.projected() for r in roster_list]
    print('Generated {} lineups.'.format(lineups))
    print('Maximum score: {}'.format(np.max(scores)))
    print('Minimum score: {}'.format(np.min(scores)))
    print('Average score: {}'.format(np.average(scores)))
    print('Median score: {}'.format(np.median(scores)))

if __name__ == '__main__':
    lineups = 3
    exposure = 0.5
    min_avg = 10
    generate_lineups(lineups, exposure, min_avg)
