def bfs_shortest_path(graph, start, goal):
    # Initialize explored nodes and queue with the start node
    explored = set()
    queue = [[start]]

    # Check if the start node is the goal
    if start == goal:
        return "The start node is the goal."

    # Perform BFS until the queue is empty
    while queue:
        # Get the first path from the queue
        path = queue.pop(0)
        node = path[-1]

        # Check if the node has not been explored
        if node not in explored:
            # Get the neighbours of the current node
            neighbours = graph[node]
            for neighbour in neighbours:
                # Create a new path by appending the neighbour to the current path
                new_path = path + [neighbour]
                queue.append(new_path)
                # Check if the neighbour is the goal node
                if neighbour == goal:
                    return new_path
            # Mark the current node as explored
            explored.add(node)

    return "Sorry, the selected node does not exist in the graph."

# Define the graph
graph = {
    'A': {'B'},
    'B': {'A', 'C'},
    'C': {'B', 'D', 'H', 'I'},
    'D': {'C', 'E', 'F', 'H'},
    'E': {'D'},
    'F': {'D', 'G'},
    'G': {'F', 'H'},
    'H': {'C', 'D', 'G', 'L'},
    'I': {'C', 'J', 'K'},
    'J': {'I'},
    'K': {'I', 'L'},
    'L': {'H', 'K'}
}

# Get input from the user
start_node = input("Enter the start node: ")
goal_node = input("Enter the goal node: ")

# Call the function and print the result
print(bfs_shortest_path(graph, start_node, goal_node))
