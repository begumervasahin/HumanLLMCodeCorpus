from collections import defaultdict

class Graph:
    def __init__(self, count):
        # Initialize the graph as a defaultdict with lists as values
        self.graph = defaultdict(list)
        # Store the total count of nodes in the graph
        self.count = count

    def add_edge(self, u, v):
        # Add an edge from node u to node v
        self.graph[u].append(v)

    def DFS(self, start, goal):
        # Initialize a list to track visited nodes
        visited = [False] * self.count
        # Initialize a stack for DFS traversal, start with the start node
        stack = [start]
        # Mark the start node as visited
        visited[start] = True

        # Perform Depth-First Search
        while stack:
            # Pop a node from the stack
            current_node = stack.pop()
            # Print the current node
            print(current_node, end=" ")

            # Check if the current node is the goal
            if current_node == goal:
                print("Goal found.")
                return True

            # Explore neighbors of the current node
            for neighbor in self.graph[current_node]:
                # If neighbor is not visited, add it to the stack and mark as visited
                if not visited[neighbor]:
                    stack.append(neighbor)
                    visited[neighbor] = True

        # If goal not found, print message
        print("Search failed.")
        return False

# Read input from file
with open('input.txt', 'r') as inputFile:
    # Read the total count of nodes, start node, and goal node
    count = int(inputFile.readline())
    start = int(inputFile.readline())
    goal = int(inputFile.readline())

    # Create an instance of the Graph class
    graph = Graph(count)

    # Add edges to the graph
    for line in inputFile.readlines():
        u, v = map(int, line.split())
        graph.add_edge(u, v)

# Perform Depth-First Search
graph.DFS(start, goal)
