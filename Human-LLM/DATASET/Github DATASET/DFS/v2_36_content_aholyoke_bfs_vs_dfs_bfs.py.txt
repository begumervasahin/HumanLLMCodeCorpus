from collections import deque

class Queue(deque):
    """A queue implementation using deque."""
    def enqueue(self, items):
        """Enqueue items into the queue."""
        self.extend(items)

    def dequeue(self):
        """Dequeue an item from the queue."""
        return self.popleft()

class Stack(deque):
    """A stack implementation using deque."""
    def push(self, items):
        """Push items onto the stack."""
        self.extend(items)

    def pop(self):
        """Pop an item from the stack."""
        return self.pop()

def search(structure, root):
    """Traverse a tree using the given structure."""
    while True:
        yield root
        structure.enqueue(root.children)
        if not structure:
            break
        root = structure.dequeue()

def bfs(tree):
    """Perform breadth-first search traversal of a tree."""
    return search(Queue(), tree)

def dfs(tree):
    """Perform depth-first search traversal of a tree."""
    return search(Stack(), tree)

if __name__ == "__main__":
    class Node:
        """Node class representing a node in a tree."""
        def __init__(self, value, *children):
            self.value = value
            self.children = list(children)

        def __repr__(self):
            return f"Node({self.value})"

    # Define the tree structure
    tree = Node(
        0,
        Node(1, Node(3), Node(4, Node(6), Node(7), Node(8))),
        Node(2, Node(5))
    )

    # Perform BFS and DFS traversals
    bfs_result = [node.value for node in bfs(tree)]
    dfs_result = [node.value for node in dfs(tree)]

    # Print results
    print("BFS traversal:", bfs_result)
    print("DFS traversal:", dfs_result)
