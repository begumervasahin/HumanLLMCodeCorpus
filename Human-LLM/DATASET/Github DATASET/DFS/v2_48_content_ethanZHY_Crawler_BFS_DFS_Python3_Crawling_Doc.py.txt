import time
import requests
from bs4 import BeautifulSoup
from collections import deque

seed_url = "https://en.wikipedia.org/wiki/Sustainable_energy"
result_urls = []
queue = deque([])
title_list = []

def BFS_crawler(seed_url):
    depth = 1
    queue.append(seed_url)
    
    # Continue crawling until the queue is empty, 1000 URLs are collected, or depth reaches 6
    while queue and len(result_urls) < 1000 and depth < 6:
        size = len(queue)
        
        # Process URLs at the current depth level
        for _ in range(size):
            current_url = queue.popleft()
            time.sleep(1)
            
            # Check if the URL has not been visited before
            if not is_visited_url(current_url):
                source_code = requests.get(current_url).text
                soup = BeautifulSoup(source_code, "html.parser")
                content = soup.find('div', {'id': 'mw-content-text'})
                
                # Extract links from the content
                if content:
                    for item in content.find_all('a', {'title': True, 'class': False}):
                        href = item.get('href', '')
                        
                        # Filter out unwanted URLs
                        if href and not href.startswith('/wiki/Category:') \
                            and not href.startswith('/wiki/File:') \
                            and not href.startswith('/wiki/Template:') \
                            and not href.startswith('/wiki/Book:') \
                            and not href.startswith('/wiki/Portal:') \
                            and not href.startswith('/wiki/Help:') \
                            and not href.startswith('/wiki/Template_talk:') \
                            and not href.startswith('/wiki/Talk:'):
                            
                            url = "https://en.wikipedia.org" + href
                            
                            # Add the URL to the result_urls list and the queue
                            if len(result_urls) < 1000 and url not in result_urls:  
                                result_urls.append(url)
                                queue.append(url)
                                print(url)
                                print(len(result_urls))
                            
                            # If 1000 URLs are collected, write them to a file and return
                            if len(result_urls) == 1000:
                                output_urls(result_urls)
                                return
        depth += 1

def output_urls(urls):
    file_loc = 'Task_2_A.txt'
    with open(file_loc, "w") as f:
        for url in urls:
            f.write(url + '\n')
        f.write('\nNum of Urls: ' + str(len(urls)) + '\n')

def is_visited_url(arg_url):
    time.sleep(1)
    source_code = requests.get(arg_url).text
    soup = BeautifulSoup(source_code, "html.parser")
    title = soup.find('h1', {'id': 'firstHeading'})
    
    # Check if the title is already in the title_list
    if title in title_list:
        return True
    else:
        title_list.append(title)
        return False

if __name__ == "__main__":
    BFS_crawler(seed_url)
