from typing import Set

class Cell:
    def __init__(self, row: int, col: int, grid: int, value: int, is_immutable: bool = False):
        """
        Initialize a Cell object.

        Args:
            row (int): Row index of the cell.
            col (int): Column index of the cell.
            grid (int): Grid index of the cell.
            value (int): Value of the cell.
            is_immutable (bool, optional): Whether the cell is immutable or not. Defaults to False.
        """
        self.row = row
        self.col = col
        self.grid = grid
        self._value = value
        self.is_immutable = is_immutable

        # Initialize possibilities based on the initial value of the cell
        if self.value == 0:
            self._possibilities = set(range(1, 10))
        else:
            self._possibilities = set()

    @property
    def is_empty(self) -> bool:
        """Check if the cell is empty."""
        return self.value == 0

    @property
    def value(self) -> int:
        """Get the value of the cell."""
        return self._value

    @value.setter
    def value(self, value: int):
        """
        Set the value of the cell.

        Args:
            value (int): New value to set.

        Note:
            Value can be set only if the cell is not immutable and the value is in the range [0, 9].
        """
        if value in range(0, 10) and not self.is_immutable:
            self._value = value

    @property
    def possibilities(self) -> Set[int]:
        """Get the possibilities for the cell."""
        return self._possibilities

    @possibilities.setter
    def possibilities(self, possibilities: Set[int]):
        """
        Set the possibilities for the cell.

        Args:
            possibilities (Set[int]): Set of possible values for the cell.
        """
        self._possibilities = possibilities

    def __repr__(self) -> str:
        """Return a string representation of the cell."""
        if self.is_empty:
            return '.'
        else:
            return str(self.value)
