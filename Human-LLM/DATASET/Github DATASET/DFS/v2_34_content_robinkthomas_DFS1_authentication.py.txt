import re
import base64
import shelve
from http.server import BaseHTTPRequestHandler, HTTPServer

class MyRequestHandler(BaseHTTPRequestHandler):
    def do_GET(self):
        self.send_unauthorized_response()

    def do_POST(self):
        auth_header = self.headers.get('Authorization')
        if auth_header:
            auth_credentials = self.parse_basic_auth_header(auth_header)
            if auth_credentials:
                username, password = auth_credentials
                if self.authenticate_user(username, password):
                    self.send_authenticated_response()
                    return
        self.send_unauthorized_response()

    def send_unauthorized_response(self):
        self.send_response(401)
        self.send_header('WWW-Authenticate', 'Basic realm="Auth example"')
        self.end_headers()
        self.wfile.write(b'Not Authorized!')

    def send_authenticated_response(self):
        self.send_response(200)
        self.end_headers()
        self.wfile.write(b'Authorized!')

    def parse_basic_auth_header(self, auth_header):
        auth_scheme, auth_b64 = auth_header.split()
        if auth_scheme.lower() == 'basic':
            auth_decoded = base64.b64decode(auth_b64).decode()
            return auth_decoded.split(':')
        return None

    def authenticate_user(self, username, password):
        try:
            with shelve.open("login.dat") as shelve1:
                return username in shelve1 and shelve1[username] == password
        except Exception as e:
            print("Error:", e)
            return False

def run(server_class=HTTPServer, handler_class=MyRequestHandler, port=8083):
    server_address = ('', port)
    httpd = server_class(server_address, handler_class)
    print(f"Starting server on port {port}...")
    httpd.serve_forever()

if __name__ == '__main__':
    run()
