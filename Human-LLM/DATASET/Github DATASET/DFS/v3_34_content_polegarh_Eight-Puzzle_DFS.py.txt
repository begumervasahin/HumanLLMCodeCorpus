from collections import deque

class PuzzleSolver:
    def __init__(self, initial_state):
        self.goal_state = [1, 2, 3, 8, 0, 4, 7, 6, 5]
        self.initial_state = PuzzleState(initial_state)
        self.is_solved = False

    def build_path_actions(self, end_state):
        actions = []
        state = end_state
        while state.parent:
            actions.append(state.action)
            state = state.parent
        actions.pop()
        actions.reverse()
        actions.append("")
        return actions

    def build_path_states(self, end_state):
        states = []
        state = end_state
        while state.parent:
            states.append(state.arr)
            state = state.parent
        states.reverse()
        return states

    def solve(self):
        self._solve_dfs(self.initial_state, [], 0)

    def _solve_dfs(self, state, visited, count):
        if self.is_solved:
            return
        if state.arr == self.goal_state:
            actions = self.build_path_actions(state)
            states = self.build_path_states(state)
            print("Solution path and actions:")
            for i, state in enumerate(states):
                print(f"{state}   {actions[i]}")
            print(f"Depth = {len(states)}")
            print(f"States explored: {count}")
            self.is_solved = True
            return visited
        if state.arr not in visited:
            visited.append(state.arr)
            index_of_zero = state.arr.index(0)
            for move in ["Up", "Down", "Right", "Left"]:
                new_state = state.make_move(index_of_zero, move)
                if new_state is not None:
                    self._solve_dfs(new_state, visited, count + 1)

class PuzzleState:
    def __init__(self, arr, parent=None, action=""):
        self.arr = arr
        self.parent = parent
        self.goal_state = [1, 2, 3, 8, 0, 4, 7, 6, 5]
        self.action = action

    def make_move(self, zero_index, move):
        moves = {"Up": -3, "Down": 3, "Right": 1, "Left": -1}
        new_index = zero_index + moves[move]
        if 0 <= new_index < len(self.arr):
            new_arr = self.arr[:]
            new_arr[zero_index], new_arr[new_index] = new_arr[new_index], new_arr[zero_index]
            return PuzzleState(new_arr, self, move)
        return None

print("Solving the puzzle using Depth-First Search:")
solver = PuzzleSolver([2, 8, 1, 0, 4, 3, 7, 6, 5])
solver.solve()
