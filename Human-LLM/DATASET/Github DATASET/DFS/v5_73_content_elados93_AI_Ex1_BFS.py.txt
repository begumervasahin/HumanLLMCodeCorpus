from TilePuzzleLogic import backtracking

class BFS_Search:
    """
    Breadth-First Search (BFS) algorithm for searching a puzzle graph.
    """

    def __init__(self, logic):
        """
        Initialize the BFS search algorithm with the given puzzle logic.

        Args:
            logic: An instance of the puzzle logic.
        """
        self.logic = logic

    def search(self, initial_state):
        """
        Search for a path to the goal state starting from the initial state.

        Args:
            initial_state: The initial state of the puzzle.

        Returns:
            A tuple containing the path of operators, the number of states visited, and 0.

        Raises:
            Exception: If the puzzle is not solvable.
        """
        open_list = [initial_state]
        states_visited = 0

        while open_list:
            current_state = open_list.pop(0)
            states_visited += 1

            if self.logic.is_goal_state(current_state):
                path = backtracking(current_state)
                return path, str(states_visited), '0'

            children = self.logic.next_states(current_state)
            open_list.extend(children)

        raise Exception('Puzzle is not solvable!')
