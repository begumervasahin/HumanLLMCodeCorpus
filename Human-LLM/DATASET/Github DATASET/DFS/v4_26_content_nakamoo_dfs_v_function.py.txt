import chainer
import chainer.functions as F
import chainer.links as L

class VFunction(object):
    pass

class FCVFunction(chainer.ChainList, VFunction):
    def __init__(self, n_input_channels, n_hidden_layers=0,
                 n_hidden_channels=None):
        # Initialize the parent class
        super(FCVFunction, self).__init__()

        # Define attributes
        self.n_input_channels = n_input_channels
        self.n_hidden_layers = n_hidden_layers
        self.n_hidden_channels = n_hidden_channels

        # Create layers
        layers = []

        if n_hidden_layers > 0:
            # Add hidden layers
            layers.append(L.Linear(n_input_channels, n_hidden_channels))
            for i in range(n_hidden_layers - 1):
                layers.append(L.Linear(n_hidden_channels, n_hidden_channels))
            layers.append(L.Linear(n_hidden_channels, 1))
        else:
            # If no hidden layers, add a single linear layer
            layers.append(L.Linear(n_input_channels, 1))

        # Add layers to the chain list
        self.layers = layers

    def __call__(self, state):
        h = state
        # Apply ReLU activation for all hidden layers
        for layer in self.layers[:-1]:
            h = F.relu(layer(h))
        # Apply the final layer without activation
        h = self.layers[-1](h)
        return h
