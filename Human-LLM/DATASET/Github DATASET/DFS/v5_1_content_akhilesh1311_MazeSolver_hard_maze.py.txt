import random
import copy

def breadth_first_search(maze, is_path=False):
    queue = [(0, 0)]
    visited = set(queue)
    parent = {}
    path = []
    
    while queue:
        cur_i, cur_j = queue.pop(0)
        
        if maze[cur_i][cur_j] == 'G':
            ans = 0
            while (cur_i, cur_j) in parent:
                path.append((cur_i, cur_j))
                cur_i, cur_j = parent[(cur_i, cur_j)]
                ans += 1
            
            if is_path:
                for p, q in path:
                    if maze[p][q] in ['O', 'X']:
                        maze[p][q] = ' '
                display_maze(maze)
                return
            else:
                return ans
        
        for di, dj in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
            next_i, next_j = cur_i + di, cur_j + dj
            if 0 <= next_i < len(maze) and 0 <= next_j < len(maze[0]) and maze[next_i][next_j] != 'X' and (next_i, next_j) not in visited:
                queue.append((next_i, next_j))
                visited.add((next_i, next_j))
                parent[(next_i, next_j)] = (cur_i, cur_j)

    return -1

def find_best_maze(maze_list):
    evaluations = []

    for index, maze in enumerate(maze_list):
        nodes_expanded = breadth_first_search(maze)
        evaluations.append((nodes_expanded, random.random(), index))

    evaluations.sort(reverse=True)
    return evaluations[0][-1]

def display_maze(maze):
    for row in maze:
        print(*row, sep=' ')
    print()

if __name__ == '__main__':
    dimension = 15
    obstacle_probability = 0.3
    iterations = 100

    maze = [['X' if random.random() < obstacle_probability else 'O' for _ in range(dimension)] for _ in range(dimension)]
    maze[0][0] = 'S'
    maze[dimension - 1][dimension - 1] = 'G'

    print("Initial maze:")
    display_maze(maze)

    print("Shortest path in initial maze:")
    breadth_first_search(maze, True)

    for i in range(dimension):
        for j in range(dimension):
            if maze[i][j] == ' ':
                maze[i][j] = 'O'

    x, y = random.randint(0, dimension - 1), random.randint(0, dimension - 1)

    for _ in range(iterations):
        maze_list = [maze]

        for di, dj in [(1, 0), (-1, 0), (0, 1), (0, -1)]:
            next_i, next_j = x + di, y + dj
            if 0 <= next_i < dimension and 0 <= next_j < dimension and maze[next_i][next_j] in ['O', 'X']:
                new_maze = copy.deepcopy(maze)
                new_maze[next_i][next_j] = 'O' if new_maze[next_i][next_j] == 'X' else 'X'
                maze_list.append(new_maze)

        index = find_best_maze(maze_list)
        maze = maze_list[index]

        if index:
            x, y = x + [-1, 1, 0, 0][index - 1], y + [0, 0, -1, 1][index - 1]
        else:
            x, y = random.randint(0, dimension - 1), random.randint(0, dimension - 1)

    print("Final maze after hill climbing:")
    display_maze(maze)

    print("Shortest path in final maze after hill climbing:")
    breadth_first_search(maze, True)
