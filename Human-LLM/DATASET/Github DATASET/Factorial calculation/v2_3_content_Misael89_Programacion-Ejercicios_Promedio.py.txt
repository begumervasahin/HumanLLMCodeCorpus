import time
import os

def get_integer_input(prompt):
    """
    Prompt the user for an integer input and validate it.
    """
    while True:
        try:
            value = int(input(prompt))
            return value
        except ValueError:
            print('Por favor, introduzca solo números.')
            print('Vuelva a intentarlo.')
            time.sleep(1)
            os.system('cls' if os.name == 'nt' else 'clear')

def main():
    """
    Main function to execute the program.
    """
    while True:
        numeroElementos = get_integer_input("Introduzca el número de elementos: ")
        listaElementos = [None] * numeroElementos
        break

    for i in range(numeroElementos):
        while True:
            try:
                elementos = get_integer_input('Elemento {}: '.format(i + 1))
                listaElementos[i] = elementos
                break
            except ValueError:
                print('Por favor, introduzca solo números.')
                print('Vuelva a intentarlo.')
                time.sleep(1)
                os.system('cls' if os.name == 'nt' else 'clear')

    total = sum(listaElementos)
    print('El promedio es:', total / numeroElementos)

if __name__ == "__main__":
    main()
