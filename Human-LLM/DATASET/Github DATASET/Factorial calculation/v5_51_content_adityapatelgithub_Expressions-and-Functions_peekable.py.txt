class Peekable:
    """An iterator with the ability to examine a value without advancement"""

    def __init__(self, iterator):
        """Initialize Peekable with an iterator."""
        self._iterator = iter(iterator)
        self._peeked = None

    def __iter__(self):
        """Return the iterator."""
        return self

    def __next__(self):
        """Return the next element of the data; no advancement if already peeked."""
        if self._peeked is None:
            self._peeked = next(self._iterator)
        peeked_value = self._peeked
        self._peeked = None
        return peeked_value

    def peek(self):
        """Peek at the next element of the data."""
        if self._peeked is None:
            self._peeked = next(self._iterator)
        return self._peeked

def peek(x):
    """Peek at the next element of a Peekable object."""
    return x.peek()

if __name__ == "__main__":
    # Example usage
    i = Peekable([1, 2, 3, 4, 5])
    print(peek(i))  # Peek at the next element
    print(peek(i))  # Peek again without advancing
    print(next(i))  # Advance to the next element
    print(next(i))  # Advance to the next element
    print(next(i))  # Advance to the next element
    print(peek(i))  # Peek at the next element without advancing
    print(next(i))  # Advance to the next element
    print(list(Peekable([1, 2, 3, 4, 5])))  # Consume the iterator and collect all elements
