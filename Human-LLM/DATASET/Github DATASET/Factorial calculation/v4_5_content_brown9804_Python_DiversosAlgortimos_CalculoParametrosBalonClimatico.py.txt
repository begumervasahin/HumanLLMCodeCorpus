def calculate_velocity(t):
    """
    Calculate the velocity of a weather balloon at a given time.

    Args:
        t (float): Time in hours.

    Returns:
        float: Velocity of the weather balloon at the given time.
    """
    vel = -0.48 * (t ** 3) + 36 * (t ** 2) - 760 * t + 4100
    return vel

def calculate_altitude(t):
    """
    Calculate the altitude of a weather balloon at a given time.

    Args:
        t (float): Time in hours.

    Returns:
        float: Altitude of the weather balloon at the given time.
    """
    alt = -0.12 * t ** 4 + 12 * t ** 3 - 380 * t ** 2 + 4100 * t + 220
    return alt

def convert_velocity(t):
    """
    Convert velocity from meters per hour to meters per second.

    Args:
        t (float): Velocity in meters per hour.

    Returns:
        float: Velocity in meters per second.
    """
    v = t / 3600
    return v

def process(initial_time, final_time, increment):
    """
    Process the velocity and altitude calculations for the weather balloon.

    Args:
        initial_time (int): Initial time for calculations.
        final_time (int): Final time for calculations.
        increment (int): Increment of hours.

    Returns:
        None
    """
    max_altitude = 0
    hour_of_max_altitude = 0

    for hour in range(initial_time, final_time + 1, increment):
        vel = calculate_velocity(hour)
        alt = calculate_altitude(hour)
        vel_converted = convert_velocity(vel)

        if alt > max_altitude:
            max_altitude = alt
            hour_of_max_altitude = hour

        print("{0:2}h   {1:8.2f}m       {2:3.2f}m/s".format(hour, alt, vel_converted))

    print("The maximum altitude was reached at", hour_of_max_altitude, "hours.")
    print("The maximum altitude reached was", max_altitude, "meters.")

# Get user input for initial time, final time, and increment
initial_time = int(input("Enter the initial time to calculate velocity and altitude for the weather balloon: "))
final_time = int(input("Enter the final time: "))
increment = int(input("Enter the increment in hours: "))

# Validate the input range
if 0 <= initial_time < 48 and initial_time < final_time <= 48:
    process(initial_time, final_time, increment)
else:
    print("Error: Initial time should be greater than or equal to 0 and less than 48.")
    print("Final time should be greater than initial time and less than or equal to 48.")
