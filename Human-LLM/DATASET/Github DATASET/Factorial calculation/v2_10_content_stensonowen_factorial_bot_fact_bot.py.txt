import praw
import sqlite3
import re
import mpmath
import atoi

# Regular expression pattern to match numerical values possibly formatted with commas
fact_ptn = re.compile("\d{1,3}(,?\d{3})*!")

def login():
    # Load Reddit app information from an external module or file
    import info  # Assuming info.py contains your Reddit app credentials

    # Create a Reddit instance
    reddit = praw.Reddit(user_agent=info.app_ua)
    
    # Set OAuth app information
    reddit.set_oauth_app_info(client_id=info.app_id, client_secret=info.app_secret, redirect_uri=info.app_uri)
    
    # Refresh access information
    reddit.refresh_access_information(refresh_token=info.app_refresh)
    
    return reddit

def shorten(x):
    # Calculate the order of magnitude of a number
    order_of_magnitude = mpmath.floor(mpmath.log(x) / mpmath.log(10)) - 1
    
    # Shorten the number and return both the shortened number and its order of magnitude
    return (x / mpmath.power(10, order_of_magnitude), order_of_magnitude)

def stirling(x):
    # Calculate the Stirling approximation of a number
    n = mpmath.mpf(x)
    coeff = mpmath.sqrt(2 * mpmath.pi * n)
    expon = mpmath.power(n / mpmath.e, n)
    return coeff * expon

def findall(comment):
    # Find all numerical values in a comment
    numbers = []
    matches = fact_ptn.findall(comment)
    for match in matches:
        # Extract numerical values and append them to the list
        result = atoi.extract(match)
        if result is not None:
            numbers.append(result)
    return numbers

def get_comments(reddit, cursor):
    # Retrieve comments from the 'all' subreddit
    all_comments = reddit.subreddit('all').comments()
    
    # Iterate over each comment
    for comment in all_comments:
        created_utc = comment.created_utc
        
        # Extract numerical values from the comment body
        numbers = findall(comment.body)
        
        # If numerical values are found, insert them into the database
        if numbers:
            for number in numbers:
                cursor.execute('INSERT INTO history (id) VALUES (?)', (number,))
    
    # Commit changes to the database
    cursor.connection.commit()

def main():
    # Connect to SQLite database
    db_connection = sqlite3.connect('history.db')
    cursor = db_connection.cursor()
    
    # Create history table if it doesn't exist
    cursor.execute('CREATE TABLE IF NOT EXISTS history(id TEXT)')
    
    # Log in to Reddit
    reddit_instance = login()
    
    # Retrieve comments and store relevant data in the database
    get_comments(reddit_instance, cursor)
    
    # Close database connection
    cursor.close()
    db_connection.close()

if __name__ == "__main__":
    main()
