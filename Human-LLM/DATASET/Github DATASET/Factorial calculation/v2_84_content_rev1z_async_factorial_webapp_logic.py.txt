import asyncio
import math
import sys

async def calculate_factorial(n):
    """
    Calculate factorial of a number.
    """
    return math.factorial(n)

async def produce_factorials(queue: asyncio.Queue, start: int = 0, finish: int = None):
    """
    Produce factorial numbers and put them into the queue.
    """
    async def __produce(n):
        item = await calculate_factorial(n)
        await queue.put(item)

    n = start if start else 0
    if finish is None:
        while True:
            await __produce(n)
            n += 1
    else:
        while n != finish:
            await __produce(n)
            n += 1
    await queue.put(None)

async def consume_factorials(queue: asyncio.Queue):
    """
    Consume factorial numbers from the queue and print them.
    """
    while True:
        item = await queue.get()
        if not item:
            break
        sys.stdout.write(f"Current value: {item}\n")
        await asyncio.sleep(0.2)

async def main():
    queue = asyncio.Queue(maxsize=1)
    producer_coro = produce_factorials(queue)
    consumer_coro = consume_factorials(queue)
    await asyncio.gather(producer_coro, consumer_coro)

if __name__ == "__main__":
    asyncio.run(main())
