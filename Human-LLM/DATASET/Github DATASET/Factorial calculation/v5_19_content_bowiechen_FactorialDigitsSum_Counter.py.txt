class Counter:
    def __init__(self, count, initial_value=0, zeros=True):
        """
        Initialize a Counter object.

        Args:
        - count (int): Number of digits in the counter.
        - initial_value (int, optional): Initial value of the counter (default is 0).
        - zeros (bool, optional): Whether to pad with zeros if initial value has fewer digits (default is True).
        """
        self.count = count
        self.digits = []
        self.overflow = False

        # Initialize digits based on initial_value and zeros flag
        if initial_value != 0:
            self._initialize_from_value(initial_value)
        else:
            self._initialize_with_zeros(zeros)

    def _initialize_from_value(self, initial_value):
        """
        Initialize digits from the given initial value.

        Args:
        - initial_value (int): Initial value to set the counter.
        """
        self.digits = [int(digit) for digit in str(initial_value)]
        self._pad_with_zeros()

    def _initialize_with_zeros(self, zeros):
        """
        Initialize digits with zeros or ones.

        Args:
        - zeros (bool): If True, initialize with zeros; if False, initialize with ones. 
        """
        self.digits = [0 if zeros else 1] * self.count

    def _pad_with_zeros(self):
        """
        Pad the digits list with zeros if necessary.
        """
        while len(self.digits) < self.count:
            self.digits.insert(0, 0)

    def increment(self):
        """
        Increment the counter by one.
        """
        self.digits[-1] += 1

        # Handle carry-over
        for i in range(self.count - 1, -1, -1):
            if self.digits[i] > 9:
                if i == 0:
                    self.overflow = True
                self.digits[i] = 0
                if i - 1 >= 0:
                    self.digits[i - 1] += 1

    def get_value(self):
        """
        Get the value of the counter as an integer.
        """
        return int(''.join(map(str, self.digits)))

    def get_list(self):
        """
        Get the digits of the counter as a list.
        """
        return self.digits
