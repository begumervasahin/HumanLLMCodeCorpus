class LinkedList:
    class Node:
        """
        Represents a node in the linked list.
        """
        __slots__ = "_value", "_next"

        def __init__(self, value, next_node):
            self._value = value
            self._next = next_node

    def __init__(self):
        """
        Initializes an empty linked list.
        """
        self._tail = None
        self._head = None
        self._length = 0

    def __iter__(self):
        """
        Iterates over the elements in the linked list.
        """
        current = self._head
        while current is not None:
            yield current._value
            current = current._next

    def __len__(self):
        """
        Returns the number of elements in the linked list.
        """
        return self._length

    def __str__(self):
        """
        Returns a string representation of the linked list.
        """
        return "".join(list(iter(self)))

    def push(self, value):
        """
        Adds a new node with the given value to the beginning of the linked list.
        """
        new_node = self.Node(value, None)
        if self._head is None:
            self._head = self._tail = new_node
        else:
            new_node._next = self._head
            self._head = new_node
        self._length += 1

    def pop(self):
        """
        Removes and returns the value of the node at the beginning of the linked list.
        """
        value = self._head._value
        self._head = self._head._next
        self._length -= 1
        return value

    def top(self):
        """
        Returns the value of the node at the end of the linked list.
        """
        return self._tail._value

    def is_empty(self):
        """
        Checks if the linked list is empty.
        """
        return self._length == 0
