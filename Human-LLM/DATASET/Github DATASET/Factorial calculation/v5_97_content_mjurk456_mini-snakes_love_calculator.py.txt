def count_matching_chars(source, example):
    """
    Count the number of characters from 'example' that appear in 'source'.
    """
    return sum(1 for char in source if char in example)

love_points = 0
vowels = "aeiuo"
consonants = "qrtpsdfghjklzxcvbnmy"

# Input two names and validate the input
while True:
    try:
        name1, name2 = input("Input two names divided by comma (e.g. 'Jan, Joanna'): ").split(",")
        name1 = name1.lower().strip()
        name2 = name2.lower().strip()
        break
    except ValueError:
        print("Invalid input. Please provide two names separated by a comma.")

# Calculate love points based on name characteristics
first_letters_match = name1[0] == name2[0]
both_start_with_vowels = name1[0] in vowels and name2[0] in vowels
both_start_with_consonants = name1[0] in consonants and name2[0] in consonants

if first_letters_match:
    love_points += 20
if both_start_with_vowels:
    love_points += 10
if both_start_with_consonants:
    love_points += 5

vowel_match = count_matching_chars(name1, vowels) == count_matching_chars(name2, vowels)
consonant_match = count_matching_chars(name1, consonants) == count_matching_chars(name2, consonants)

if vowel_match:
    love_points += 12
if consonant_match:
    love_points += 12
if any(char in name1 and char in name2 for char in 'love'):
    love_points += 7

# Print the result
print(f"Your love points are {love_points}")
print("REMEMBER: Your relationship depends only on you and your partner, not on the stupid love calculator!")
