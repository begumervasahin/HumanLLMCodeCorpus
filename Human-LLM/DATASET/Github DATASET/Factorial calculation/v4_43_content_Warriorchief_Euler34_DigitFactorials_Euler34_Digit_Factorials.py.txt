"""
Digit Factorials
Problem 34
Find the sum of all numbers which are equal to the sum of the factorial of their digits.
Note: Numbers like 1! = 1 and 2! = 2 are not considered as they are not sums.
"""

def factorial(x):
    """
    Calculate the factorial of a given number 'x'.
    
    Args:
        x (int): The number to calculate the factorial of.
    
    Returns:
        int: The factorial of 'x'.
    """
    fac = 1
    for i in range(1, x + 1):
        fac *= i
    return fac

def separate_into_chars(num):
    """
    Separate a number into its individual digits and return as a list of characters.
    
    Args:
        num (int): The number to be separated into digits.
    
    Returns:
        list: A list containing the individual digits of the number.
    """
    return list(str(num))

def is_special_number(num):
    """
    Check if a number is a special number.
    
    A special number is equal to the sum of the factorial of its digits.
    
    Args:
        num (int): The number to be checked.
    
    Returns:
        bool: True if the number is a special number, False otherwise.
    """
    chars = separate_into_chars(num)
    total = sum(factorial(int(char)) for char in chars)
    return total == num

def find_all_special_numbers(max_num):
    """
    Find all special numbers less than a given maximum number.
    
    Args:
        max_num (int): The maximum number to check.
    
    Returns:
        list: A list of special numbers found.
    """
    special_numbers = []
    i = 3
    while i < max_num:
        if i % 100000 == 0:
            print("Trying i as", i, "so there's another hundred thousand tested")
        if is_special_number(i):
            special_numbers.append(i)
        i += 1
    print("Special numbers:", special_numbers)
    print("Count:", len(special_numbers))
    return special_numbers

if __name__ == "__main__":
    max_number = 100000000
    find_all_special_numbers(max_number)
