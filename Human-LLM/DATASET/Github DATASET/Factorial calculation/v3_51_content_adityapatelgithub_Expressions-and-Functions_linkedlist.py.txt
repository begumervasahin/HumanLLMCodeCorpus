class LinkedList:
    class _Node:
        """A private class representing a node in the linked list."""
        def __init__(self, value, next_node=None):
            self.value = value
            self.next = next_node

    def __init__(self):
        """Initializes an empty linked list."""
        self._head = None
        self._tail = None
        self._length = 0

    def __iter__(self):
        """Allows iteration over the elements of the linked list."""
        current = self._head
        while current:
            yield current.value
            current = current.next

    def __len__(self):
        """Returns the number of elements in the linked list."""
        return self._length

    def __str__(self):
        """Provides a string representation of the linked list."""
        return "->".join(str(value) for value in self)

    def append(self, value):
        """
        Adds a new node with the specified value to the end of the linked list.
        """
        new_node = self._Node(value)
        if not self._head:  # List is empty
            self._head = self._tail = new_node
        else:
            self._tail.next = new_node
            self._tail = new_node
        self._length += 1

    def prepend(self, value):
        """
        Adds a new node with the specified value to the beginning of the linked list.
        """
        new_node = self._Node(value, self._head)
        self._head = new_node
        if not self._tail:  # List was empty
            self._tail = new_node
        self._length += 1

    def pop_first(self):
        """
        Removes and returns the value of the first node in the linked list.
        Throws an exception if the list is empty.
        """
        if not self._head:
            raise IndexError("pop from empty list")
        value = self._head.value
        self._head = self._head.next
        if not self._head:  # List became empty
            self._tail = None
        self._length -= 1
        return value

    def first(self):
        """Returns the value of the first node in the linked list."""
        if not self._head:
            raise IndexError("list is empty")
        return self._head.value

    def last(self):
        """Returns the value of the last node in the linked list."""
        if not self._tail:
            raise IndexError("list is empty")
        return self._tail.value

    def is_empty(self):
        """Checks if the linked list is empty."""
        return self._length == 0
