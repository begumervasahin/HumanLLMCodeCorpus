import db
import profile
import itertools
import time

def wsieve(N):
    # Initialize the state list with -1 for indices 0 and 1, and 0 for the rest
    state = [-1, -1] + [0] * (N - 1)
    primes = [2]  # Initialize the list of prime numbers with 2
    prime_i = 0  # Index of the current prime number
    lcm = 1  # Initialize the least common multiple to 1
    p = 2  # Current prime number

    while p ** 2 < N:  # Iterate until the square of the current prime is less than N
        # Extend primes list with numbers marked as prime in the state list
        primes.extend(i for i in range(state.index(0, primes[-1] + 1), p ** 2) if state[i] == 0)

        if lcm < p ** 2:
            # Generate a chain of numbers to mark as composite
            U = itertools.chain([1], [i for i in range(p, min(lcm, N // p) + 1) if state[i] == 0])
        else:
            U = itertools.chain([1], primes[prime_i:], (i for i in range(p ** 2, min(lcm, N // p) + 1) if state[i] == 0))

        next_lcm = p * lcm  # Calculate the next least common multiple
        for sortaprime in U:
            for i in range(p * sortaprime, N + 1, next_lcm):
                state[i] = p  # Mark multiples as composite

        lcm = next_lcm  # Update the least common multiple
        prime_i += 1
        p = primes[prime_i]  # Move to the next prime number

    return primes

def go():
    N = int(10 ** 6)
    t0 = time.time()
    r = wsieve(N)
    t1 = time.time()
    print("for %d took %0.4f" % (N, t1 - t0))

    # Ensure the prime numbers are stored in the database
    db.ensure(r)

go()
