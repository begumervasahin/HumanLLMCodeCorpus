# Define factor lists
first_factor_list = ["factor list 1 item 1", "factor list 1 item 2", "factor list 1 item 3"]
second_factor_list = ["factor list 2 item 1", "factor list 2 item 2", "factor list 2 item 3", "factor list 2 item 4"]
third_factor_list = ["factor list 3 item 1", "factor list 3 item 2"]
fourth_factor_list = ["factor list 4 item 1", "factor list 4 item 2", "factor list 4 item 3", "factor list 4 item 4", "factor list 4 item 5"]
fifth_factor_list = ["factor list 5 item 1", "factor list 5 item 2", "factor list 5 item 3"]

# Initialize question number
question_number = 1

# Generate combinations and print questions
for first_factor in first_factor_list:
    for second_factor in second_factor_list:
        for third_factor in third_factor_list:
            for fourth_factor in fourth_factor_list:
                for fifth_factor in fifth_factor_list:
                    print("[[Block]]")
                    print(f"{question_number}. You are {first_factor}. You see {second_factor}. When you, {third_factor} reacts {fourth_factor} in order to {fifth_factor}.")
                    print("\nThis situation is acceptable to me:")
                    print("\nStrongly Agree.")
                    print("Agree.")
                    print("Neither Agree nor Disagree.")
                    print("Disagree.")
                    print("Strongly Disagree.\n")
                    question_number += 1
