# Import necessary libraries
from nltk.tokenize import RegexpTokenizer
from nltk.stem.porter import PorterStemmer
from stop_words import get_stop_words
from gensim import corpora, models
import gensim

# Text preprocessing steps: Tokenization, Stopping, Stemming
# Tokenization: Convert documents into atomic elements using regular expressions
tokenizer = RegexpTokenizer(r'\w+')

# Stopping: Remove meaningless words
en_stop = get_stop_words('en')

# Stemming: Merge words with equivalent meanings
p_stemmer = PorterStemmer()

# Sample documents for testing
doc_a = "Broccoli is good to eat. My brother likes to eat good broccoli, but not my mother."
doc_b = "My mother spends a lot of time driving my brother around to baseball practice."
doc_c = "Some health experts suggest that driving may cause increased tension and blood pressure."
doc_d = "I often feel pressure to perform well at school, but my mother never seems to drive my brother to do better."
doc_e = "Health professionals say that broccoli is good for your health."

# Compile sample documents into a list
doc_set = [doc_a, doc_b, doc_c, doc_d, doc_e]

# Initialize a list for tokenized documents
texts = []

# Tokenize, remove stop words, and stem each document
for doc in doc_set:
    # Convert to lowercase
    raw = doc.lower()
    # Tokenize the document
    tokens = tokenizer.tokenize(raw)
    # Remove stop words
    stopped_tokens = [token for token in tokens if token not in en_stop]
    # Stem tokens
    stemmed_tokens = [p_stemmer.stem(token) for token in stopped_tokens]
    # Append preprocessed tokens to the list
    texts.append(stemmed_tokens)

# Construct a document-term matrix
dictionary = corpora.Dictionary(texts)
print("Token IDs:", dictionary.token2id)

# Convert tokenized documents into a document-term matrix
corpus = [dictionary.doc2bow(text) for text in texts]
print("Document-Term Matrix for the first document:", corpus[0])

# Generate LDA model
ldamodel = gensim.models.ldamodel.LdaModel(corpus, num_topics=2, id2word=dictionary, passes=20)

# Print topics generated by LDA
print("Topics generated by LDA:")
print(ldamodel.print_topics(num_topics=2, num_words=4))
