import pandas as pd
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem.snowball import SnowballStemmer
import numpy as np
import string
import sys

# Download necessary NLTK resources
nltk.download('stopwords')
nltk.download('punkt')

def parse_file(file_path):
    # Read content from the file
    with open(file_path, 'r') as file:
        content = file.readlines()

    # Tokenize words and remove stopwords and punctuation
    stop_words = set(stopwords.words('english'))
    punctuation = set(string.punctuation)
    punct_stopwords = punctuation.union(['i','\x89','_CA','_TX', '_IL','_NY', '_PA', '_GA', '_Ontario', '_MA', '_FL', '_DC','__', '___']).union(stop_words)
    stemmer = SnowballStemmer("english")

    words = []
    for line in content:
        tokens = word_tokenize(line)
        cleaned_line = [stemmer.stem(w) for w in tokens if w.lower() not in punct_stopwords and len(w) > 1]
        if len(cleaned_line) > 1:
            words.append(cleaned_line)

    return [word for word in words if len(word) > 1 and word[0] in cities]

def train_df(train_data):
    # Create a DataFrame to store word frequencies
    word_list = set([w for word in train_data for w in word])
    data = pd.DataFrame(0, index=word_list, columns=cities)

    # Populate the DataFrame with word counts
    for word in train_data:
        current_city = word[0]
        for w in word[1:]:
            data.at[w, current_city] += 1

    return data

def test_dict(file_path):
    # Read content from the file and preprocess
    with open(file_path, 'r') as file:
        content = file.readlines()

    test_data = {}
    stop_words = set(stopwords.words('english'))
    punctuation = set(string.punctuation)
    punct_stopwords = punctuation.union(['i','\x89','_CA','_TX', '_IL','_NY', '_PA', '_GA', '_Ontario', '_MA', '_FL', '_DC','__', '___']).union(stop_words)
    stemmer = SnowballStemmer("english")

    # Tokenize words, remove stopwords and punctuation, and stem
    for line in content:
        tokens = word_tokenize(line)
        cleaned_line = [stemmer.stem(w) for w in tokens if w.lower() not in punct_stopwords]
        if cleaned_line:
            cleaned_line.pop(0)
            test_data[line] = cleaned_line

    return test_data

def get_city_prob(train_data):
    # Calculate the probability of each city based on training data
    for word in train_data:
        city_prob[word[0]] += 1

    total = sum(city_prob.values())
    for city in city_prob:
        city_prob[city] = float(city_prob[city] / total)
