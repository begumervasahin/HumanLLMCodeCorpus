import numpy as np
import matplotlib.pyplot as plt
import pandas as pd
import re
import nltk
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.model_selection import train_test_split
from sklearn.naive_bayes import GaussianNB
from sklearn.metrics import confusion_matrix

# Download NLTK resources needed for text preprocessing
nltk.download('stopwords')

# Read the dataset
reviews_dataset = pd.read_csv('Restaurant_Reviews.tsv', delimiter='\t', quoting=3)

# Data preprocessing: Cleaning and stemming the text data
def clean_and_stem_text(text):
    # Remove non-alphabetic characters
    cleaned_text = re.sub('[^a-zA-Z]', ' ', text)
    # Convert text to lowercase
    cleaned_text = cleaned_text.lower()
    # Split text into words
    words = cleaned_text.split()
    # Initialize Porter stemmer
    stemmer = nltk.PorterStemmer()
    # Stemming and remove stopwords
    stemmed_words = [stemmer.stem(word) for word in words if word not in set(nltk.corpus.stopwords.words('english'))]
    # Join stemmed words back into a single string
    cleaned_review = ' '.join(stemmed_words)
    return cleaned_review

# Apply text preprocessing to each review in the dataset
corpus = [clean_and_stem_text(review) for review in reviews_dataset['Review']]

# Feature extraction: Convert text data into numerical features
vectorizer = CountVectorizer(max_features=1500)  # Initialize CountVectorizer with maximum 1500 features
X_features = vectorizer.fit_transform(corpus).toarray()  # Convert corpus into a matrix of token counts
y_labels = reviews_dataset['Liked'].values  # Extract labels from the dataset

# Split dataset into training and testing sets
X_train, X_test, y_train, y_test = train_test_split(X_features, y_labels, test_size=0.20, random_state=0)

# Train Naive Bayes classifier on the training set
naive_bayes_classifier = GaussianNB()
naive_bayes_classifier.fit(X_train, y_train)

# Predict labels for the test set
predicted_labels = naive_bayes_classifier.predict(X_test)

# Compute confusion matrix
confusion_matrix_result = confusion_matrix(y_test, predicted_labels)

# Print confusion matrix
print("Confusion Matrix:")
print(confusion_matrix_result)
