import re
from tqdm import tqdm

# Open the dataset files for reading and writing
try:
    full_dataset = open("twitter-sentiment-dataset/sentiment-dataset.csv", "r")
    pos_dataset = open("twitter-sentiment-dataset/tw-data.pos", "w")
    neg_dataset = open("twitter-sentiment-dataset/tw-data.neg", "w")
except IOError:
    print("Failed to open file")
    quit()

# Read all lines from the CSV dataset
csv_lines = full_dataset.readlines()

# Counter for progress tracking
i = 0.0

# Iterate over each line in the dataset
for line in tqdm(csv_lines):
    i += 1.0
    # Split the line by commas, considering only the first three occurrences
    line = line.split(",", 3)
    tweet = line[3].strip()  # Extract the tweet text and remove leading/trailing whitespaces
    new_tweet = ''
    
    # Process each word in the tweet
    for word in tweet.split():
        # Replace mentions with <NAME/>
        if re.match('^.*@.*', word):
            word = '<NAME/>'
        # Replace links with <LINK/>
        if re.match('^.*http://.*', word):
            word = '<LINK/>'
        # Handle HTML entities
        word = word.replace('&quot;', '\"')
        word = word.replace('&amp;', '&')
        word = word.replace('&gt;', '>')
        word = word.replace('&lt;', '<')
        new_tweet = ' '.join([new_tweet, word])  # Reconstruct the tweet with processed words
    
    tweet = new_tweet.strip() + '\n'  # Remove leading/trailing whitespaces and add newline
    # Write the tweet to the appropriate file based on sentiment label
    if line[1].strip() == '1':
        pos_dataset.write(tweet)
    else:
        neg_dataset.write(tweet)

# Close all opened files
full_dataset.close()
pos_dataset.close()
neg_dataset.close()
