import tweepy
import progressbar
from config import *
from colorama import init, Fore

# Initialize colorama for colored output
init()

class TwitterSentiment:
    '''
    This class fetches frequent tweets of a particular Twitter user, analyzes the tweets,
    and returns the most frequently used words with their frequencies.
    '''
    auth = tweepy.OAuthHandler(API_KEYS, API_SECRET)
    auth.secure = True
    auth.set_access_token(ACCESS_TOKEN, ACCESS_TOKEN_SECRET)
    api = tweepy.API(auth)

    def __init__(self, username, number=3200):
        '''
        Initializes the username whose data is to be fetched (tweets).
        '''
        self.username = username
        self.no_tweets = number

    def fetch_data(self):
        '''
        Fetches data from Twitter for a specific user and stores it in a dictionary.
        '''
        try:
            tweet_data = dict()
            all_tweets = []
            bar = progressbar.ProgressBar(max_value=self.no_tweets)
            print()

            if self.no_tweets <= 200:
                new_tweets = TwitterSentiment.api.user_timeline(self.username, count=self.no_tweets)
                all_tweets.extend(new_tweets)
                if not all_tweets:
                    tweet_data[0] = "No tweets so far", "00/00/00"
                    return tweet_data
            else:
                new_tweets = TwitterSentiment.api.user_timeline(self.username, count=200)
                all_tweets.extend(new_tweets)
                self.no_tweets -= 200
                oldest = all_tweets[-1].id - 1
                
                while new_tweets:
                    if self.no_tweets <= 200:
                        new_tweets = TwitterSentiment.api.user_timeline(self.username, count=self.no_tweets)
                        all_tweets.extend(new_tweets)
                    else:
                        new_tweets = TwitterSentiment.api.user_timeline(self.username, count=200, max_id=oldest)
                        all_tweets.extend(new_tweets)
                        oldest = all_tweets[-1].id - 1
                        self.no_tweets -= 200

            for tweet in all_tweets:
                tweet_data[tweet.id] = (tweet.text, tweet.created_at)
                
            return tweet_data
        except tweepy.error.TweepError as e:
            print(Fore.RED + "No such Twitter handle or it is a private account, or you are not following them. Please double-check the spelling and try again.")

    def display_tweets(self):
        '''
        Displays fetched tweets.
        '''
        tweet_dict = self.fetch_data()
        header = '{} {} {} {}'.format(
            Fore.BLUE + "\nTWITTER ID".ljust(30),
            Fore.BLUE + "DATE POSTED".ljust(30),
            Fore.BLUE + "THE TWEET\n\n",
            "==" * 70
        )
        print(header)

        try:
            for key in tweet_dict:
                print('{} {} {}'.format(
                    Fore.YELLOW + str(key).ljust(30),
                    Fore.GREEN + str(tweet_dict[key][1]).ljust(30),
                    Fore.RED + tweet_dict[key][0])
                )
        except TypeError as e:
            print(Fore.RED + "No tweets for this particular Twitter handle")

    def analyse_data(self):
        '''
        Analyses the fetched tweets and returns the most frequent words (eliminating stop words).
        '''
        tweet_dict = self.fetch_data()
        freq_dict = dict()

        for key in tweet_dict:
            tweet_words = str(tweet_dict[key][0]).lower().split()
            for word in tweet_words:
                if word in stop_words:
                    continue
                elif word not in freq_dict:
                    freq_dict[word] = 1
                else:
                    freq_dict[word] += 1

        freq_list = [(freq_dict[key], key) for key in freq_dict]
        freq_list.sort()
        freq_list = freq_list[::-1]
        freq_list = freq_list[:10]

        print('{} {} {} {} {} {}'.format(
            Fore.BLUE + "\n\n",
            "COMMON TWEETED WORD".ljust(30),
            Fore.BLUE + "THE FREQUENCY\n".ljust(30),
            '\n\n',
            "==" * 70,
            "\n\n")
        )

        for freq, word in freq_list:
            print(' {} {}'.format(
                Fore.YELLOW + word.ljust(30),
                Fore.GREEN + str(freq).ljust(30))
            )
