import re
import glob
import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
from sklearn.model_selection import RandomizedSearchCV
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import confusion_matrix
from nltk.corpus import stopwords
import scipy.stats as st

def clean_text(text):
    """
    Clean the given text by removing URLs, special characters, and extra spaces.
    Convert text to lowercase.
    
    :param text: The text to be cleaned.
    :return: Cleaned text.
    """
    text = re.sub(r'\s+', ' ', text)
    text = re.sub(r'www\.\S+|https?://\S+', '', text)
    text = re.sub(r'[!?,\'"-]', ' ', text)
    text = re.sub(r'<br\s*/><br\s*/>', ' ', text)
    text = text.lower()
    return text

def read_file_and_clean(file_path):
    """
    Read text data from a file, clean it, and filter out stopwords.
    
    :param file_path: Path to the file to be read.
    :return: A string containing the cleaned and filtered text.
    """
    with open(file_path, 'r', encoding='utf-8') as file:
        content = file.read()
    cleaned_content = clean_text(content)
    words = cleaned_content.split()
    filtered_words = [word for word in words if word not in set(stopwords.words('english'))]
    return ' '.join(filtered_words)

def prepare_data(neg_files, pos_files):
    """
    Create a DataFrame with the text data and sentiment labels.
    
    :param neg_files: List of paths to negative review files.
    :param pos_files: List of paths to positive review files.
    :return: A pandas DataFrame with columns 'text' and 'label'.
    """
    texts = []
    labels = []
    
    for file_path in neg_files:
        texts.append(read_file_and_clean(file_path))
        labels.append(0)
    
    for file_path in pos_files:
        texts.append(read_file_and_clean(file_path))
        labels.append(1)
    
    return pd.DataFrame({'text': texts, 'label': labels})

def display_confusion_matrix(cm, class_names):
    """
    Display a confusion matrix using matplotlib.
    
    :param cm: Confusion matrix to display.
    :param class_names: List of class names for the axis labels.
    """
    plt.imshow(cm, interpolation='nearest', cmap=plt.cm.Blues)
    plt.title('Confusion Matrix')
    plt.colorbar()
    tick_marks = np.arange(len(class_names))
    plt.xticks(tick_marks, class_names, rotation=45)
    plt.yticks(tick_marks, class_names)
    plt.tight_layout()
    plt.ylabel('True Label')
    plt.xlabel('Predicted Label')
    plt.show()
