def load_turkish_dictionary(file_path):
    """
    Loads a Turkish dictionary from the specified file path.
    
    Parameters:
        file_path (str): The path to the file containing the Turkish dictionary.
        
    Returns:
        dict: A dictionary containing Turkish words and their frequencies.
    """
    turkish_dictionary = {}
    with open(file_path, 'r', encoding='utf-8') as dictionary_file:
        for line in dictionary_file:
            if line[0] in ['0', '1']:
                frequency, word = line.strip().split()
                turkish_dictionary[word] = int(frequency)
    return turkish_dictionary

def is_stop_word(word, dictionary):
    """
    Checks if the given word is a stop word in the specified dictionary.
    
    Parameters:
        word (str): The word to check.
        dictionary (dict): The dictionary containing Turkish words and their frequencies.
        
    Returns:
        bool: True if the word is a stop word, False otherwise.
    """
    return word in dictionary

def get_word_frequency(word, dictionary):
    """
    Returns the frequency of the given word in the specified dictionary.
    
    Parameters:
        word (str): The word to get the frequency for.
        dictionary (dict): The dictionary containing Turkish words and their frequencies.
        
    Returns:
        int: The frequency of the word, or 0 if the word is not in the dictionary.
    """
    return dictionary.get(word, 0)

    # Load the Turkish dictionary
    turkish_dictionary = load_turkish_dictionary("dosyalar/derlemtr2016-10000.txt")
    
    # Open file containing stop words
    with open("dosyalar/turkce-stop-words", encoding="utf-8") as stop_words_file:
        # Iterate through each stop word in the file
        for stop_word in stop_words_file:
            # Remove any leading or trailing whitespace
            stop_word = stop_word.strip()
            # Check if the stop word is not in the dictionary
            if not is_stop_word(stop_word, turkish_dictionary):
                # Print a message indicating that the stop word is not in the dictionary
                print('{} is not in the dictionary'.format(stop_word))
