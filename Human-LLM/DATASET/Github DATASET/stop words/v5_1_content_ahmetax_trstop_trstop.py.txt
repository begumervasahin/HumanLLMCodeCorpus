# Function to read word frequencies from a file and store them in a dictionary
def read_word_frequencies(file_path):
    word_frequencies = {}
    with open(file_path, 'r', encoding='utf-8') as file:
        for line in file:
            if line.startswith(('0', '1')):
                frequency, word = line.strip().split()
                word_frequencies[word] = int(frequency)
    return word_frequencies

# Function to check if a word is a stop word
def is_stop_word(word, dictionary):
    """
    Checks if a word is in the dictionary.
    
    Parameters:
        word (str): The word to check.
        dictionary (dict): The dictionary of word frequencies.
    
    Returns:
        bool: True if the word is a stop word, False otherwise.
    """
    return word in dictionary

# Function to get the frequency of a word
def get_word_freq(word, dictionary):
    """
    Returns the frequency of a word.
    
    Parameters:
        word (str): The word to get the frequency of.
        dictionary (dict): The dictionary of word frequencies.
    
    Returns:
        int: The frequency of the word, or 0 if the word is not in the dictionary.
    """
    if is_stop_word(word, dictionary):
        return dictionary[word]
    else:
        return 0

if __name__ == '__main__':
    # Read word frequencies from the file and store them in a dictionary
    word_frequencies = read_word_frequencies("dosyalar/derlemtr2016-10000.txt")
    
    # Open the file containing stop words
    with open("dosyalar/turkce-stop-words", encoding="utf-8") as stop_words_file:
        # Iterate over each word in the file
        for word in stop_words_file:
            word = word.strip()
            # Check if the word is not in the dictionary
            if word not in word_frequencies:
                print(f"'{word}' is not in the dictionary")
