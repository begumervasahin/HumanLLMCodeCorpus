# Importing necessary module for package version retrieval
import pkg_resources

# Defining a function to get the version of a package
def get_package_version(package_name):
    try:
        # Attempting to retrieve the version of the specified package
        version = pkg_resources.get_distribution(package_name).version
        # Returning the retrieved version
        return version
    except pkg_resources.DistributionNotFound:
        # Handling the case where the package is not found
        return "Package '{}' not found.".format(package_name)

# Main entry point of the script
if __name__ == "__main__":
    # Specifying the name of the package to retrieve its version
    package_name = "your_package_name"  # Replace 'your_package_name' with the actual package name
    # Calling the function to get the version of the specified package
    version = get_package_version(package_name)
    # Printing the version of the specified package
    print("Version of '{}': {}".format(package_name, version))
