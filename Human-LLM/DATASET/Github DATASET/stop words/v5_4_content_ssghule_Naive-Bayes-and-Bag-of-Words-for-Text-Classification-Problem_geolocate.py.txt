import pandas as pd
import numpy as np
import nltk
from nltk.tokenize import word_tokenize
from nltk.corpus import stopwords
from nltk.stem.snowball import SnowballStemmer
import string
import sys

# Download NLTK resources
nltk.download('stopwords')
nltk.download('punkt')

# Define global variables
OLD_CITIES = ['Los_Angeles,_CA', 'San_Francisco,_CA', 'San_Diego,_CA', 'Houston,_TX', 'Chicago,_IL', 'Philadelphia,_PA', 'Toronto,_Ontario', 'Atlanta,_GA', 'Boston,_MA', 'Orlando,_FL', 'Washington,_DC', 'Manhattan,_NY']
CITIES = ["los_angel", "san_francisco", "san_diego", "houston", "chicago", "philadelphia", "toronto", "atlanta", "boston", "orlando", "washington", "manhattan"]
CITY_PROB = {city: 0 for city in CITIES}

def parse_file(file_path):
    with open(file_path, 'r') as file:
        content = file.readlines()

    stop_words = set(stopwords.words('english'))
    punctuation = set(string.punctuation)
    punct_stopwords = punctuation.union(['i','\x89','_CA','_TX', '_IL','_NY', '_PA', '_GA', '_Ontario', '_MA', '_FL', '_DC','__', '___']).union(stop_words)
    stemmer = SnowballStemmer("english")

    words = []
    for line in content:
        tokens = word_tokenize(line)
        cleaned_line = [stemmer.stem(w) for w in tokens if w.lower() not in punct_stopwords and len(w) > 1]
        if len(cleaned_line) > 1:
            words.append(cleaned_line)

    return [word for word in words if len(word) > 1 and word[0] in CITIES]

def train_df(train_data):
    word_list = set([w for word in train_data for w in word])
    data = pd.DataFrame(0, index=word_list, columns=CITIES)

    for word in train_data:
        current_city = word[0]
        for w in word[1:]:
            data.at[w, current_city] += 1

    return data

def test_dict(file_path):
    with open(file_path, 'r') as file:
        content = file.readlines()

    test_data = {}
    stop_words = set(stopwords.words('english'))
    punctuation = set(string.punctuation)
    punct_stopwords = punctuation.union(['i','\x89','_CA','_TX', '_IL','_NY', '_PA', '_GA', '_Ontario', '_MA', '_FL', '_DC','__', '___']).union(stop_words)
    stemmer = SnowballStemmer("english")

    for line in content:
        tokens = word_tokenize(line)
        cleaned_line = [stemmer.stem(w) for w in tokens if w.lower() not in punct_stopwords]
        if cleaned_line:
            cleaned_line.pop(0)
            test_data[line] = cleaned_line

    return test_data

def get_city_prob(train_data):
    for word in train_data:
        CITY_PROB[word[0]] += 1

    total = sum(CITY_PROB.values())
    for city in CITY_PROB:
        CITY_PROB[city] = float(CITY_PROB[city] / total)

def bayes(train_df, test_data, output_file):
    with open(output_file, 'w') as file:
        for tweet in test_data:
            label = ''
            words = test_data[tweet]
            if not words:
                continue
            else:
                word_posterior = [[(train_df.at[w, city[i]] / sum(train_df.loc[:, city[i]]) * CITY_PROB[city[i]]) if w in train_df.index else 0.00001 * CITY_PROB[city[i]] for i in range(0, 12)] for w in words]
                probabilities = np.prod(word_posterior, axis=0)
                label = OLD_CITIES[np.argmax(probabilities)]
                file.write(label + " " + tweet)

    print("Output written to ", output_file)

# Main function
if __name__ == "__main__":
    # Parse command line arguments
    f1 = str(sys.argv[1])
    f2 = str(sys.argv[2])
    f3 = str(sys.argv[3])

    # Parse and preprocess training and testing data
    training_data = parse_file(f1)
    training_df = train_df(training_data)
    testing_data = test_dict(f2)

    # Calculate city probabilities
    get_city_prob(training_data)

    # Apply Naive Bayes classifier
    training_df += (0.00001)
    bayes(training_df, testing_data, f3)

    # Print top five words for each city
    print('Top five words for each city: ')
    for city_name in CITIES:
        print(OLD_CITIES[CITIES.index(city_name)], training_df.nlargest(5, city_name).index.tolist())
