import os
import json
import operator

def list_files(directory):
    # Create an empty list to store file paths
    files_list = []
    
    # Recursively walk through the directory and its subdirectories
    for root, _, files in os.walk(directory):
        # For each file found, add its full path to the files_list
        for file in files:
            files_list.append(os.path.join(root, file))
    
    return files_list

def main():
    # Define the directory containing the files to process
    directory = 'extracted'
    
    # Get a list of all files in the specified directory and its subdirectories
    files = list_files(directory)

    # Initialize dictionaries to store word counts and document frequencies
    word_counts = {}
    doc_frequencies = {}

    # Iterate over each file
    for file in files:
        # Open the file for reading
        with open(file, 'r') as f:
            # Read each line in the file
            for line in f:
                # Parse the line as JSON
                json_data = json.loads(line)
                # Extract the 'text' field from the JSON data
                text = json_data.get('text', '')

                # Process each word in the text
                for word in text.split():
                    # Increment the count for the word in the word_counts dictionary
                    word_counts[word] = word_counts.get(word, 0) + 1
                    # Increment the document frequency for the word in the doc_frequencies dictionary
                    doc_frequencies[word] = doc_frequencies.get(word, 0) + 1

    # Calculate the score for each word based on its counts and document frequencies
    scores = {}
    for word in doc_frequencies:
        scores[word] = word_counts[word] * doc_frequencies[word] * doc_frequencies[word]

    # Sort the words based on their scores in descending order
    sorted_scores = sorted(scores.items(), key=operator.itemgetter(1), reverse=True)

    # Print the top 50 words along with their scores
    for word, score in sorted_scores[:50]:
        print(f"{word}: {score}")

# Entry point of the program
if __name__ == "__main__":
    main()
