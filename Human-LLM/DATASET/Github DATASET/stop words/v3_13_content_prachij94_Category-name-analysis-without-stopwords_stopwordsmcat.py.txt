import pandas as pd

# Read input Excel files
working_mcat_subcat_df = pd.read_excel("C:/Users/IMART/Downloads/Working MCAT_subcat.xlsx", sheet_name="Sheet2")
subcat_super_pmcat_df = pd.read_excel("C:/Users/IMART/Downloads/Subcat_super_PMCAT.xlsx", sheet_name="Export Worksheet")
word_frequency_df = pd.read_excel("C:/Users/IMART/Downloads/wordfrequencyfinal.xlsx", sheet_name="wordfrequencyfinal")

# Convert relevant columns to lowercase for consistency
word_frequency_df["HIGH FREQUENCY WORDS"] = word_frequency_df["HIGH FREQUENCY WORDS"].str.lower()
subcat_super_pmcat_df["GLCAT_MCAT_NAME"] = subcat_super_pmcat_df["GLCAT_MCAT_NAME"].str.lower()
working_mcat_subcat_df["GLCAT_MCAT_NAME"] = working_mcat_subcat_df["GLCAT_MCAT_NAME"].str.lower()

# Create a dictionary for easy lookup from word_frequency_df
word_freq_dict = word_frequency_df.set_index('SUBCAT ID')['HIGH FREQUENCY WORDS'].to_dict()

# Merge logic for subcat_super_pmcat_df
merged_mcat_subcat_df = pd.DataFrame(columns=['MCAT_Name', 'Subcat_ID', 'MCAT_Name&HighFrequencyWords'])

for index, row in subcat_super_pmcat_df.iterrows():
    subcat_id = row["SUBCAT_ID"]
    if subcat_id in word_freq_dict:
        mcat_name = row["GLCAT_MCAT_NAME"]
        unique_words = set(mcat_name.split())
        unique_words.difference_update(word_freq_dict[subcat_id].split())
        cleaned_string = " ".join(unique_words)
        merged_mcat_subcat_df = merged_mcat_subcat_df.append({
            'MCAT_Name': row["GLCAT_MCAT_NAME"],
            'Subcat_ID': subcat_id,
            'MCAT_Name&HighFrequencyWords': cleaned_string + " " + word_freq_dict[subcat_id]
        }, ignore_index=True)
    else:
        merged_mcat_subcat_df = merged_mcat_subcat_df.append({
            'MCAT_Name': row["GLCAT_MCAT_NAME"],
            'Subcat_ID': subcat_id,
            'MCAT_Name&HighFrequencyWords': row["GLCAT_MCAT_NAME"]
        }, ignore_index=True)

# Merge logic for working_mcat_subcat_df
merged_pmcat_df = pd.DataFrame(columns=['MCAT_ID', 'PMCAT', 'MCAT_Name', 'Subcat_ID', 'Stopword', 'Attribute'])

for index, row in working_mcat_subcat_df.iterrows():
    subcat_id = row["FK_GLCAT_CAT_ID"]
    attribute_string = ""
    if subcat_id in word_freq_dict:
        stopwords = word_freq_dict[subcat_id]
    else:
        stopwords = ""
    mcat_name = row["GLCAT_MCAT_NAME"]
    for word in mcat_name.split():
        if word not in stopwords.split():
            attribute_string += " " + word
    merged_pmcat_df = merged_pmcat_df.append({
        'MCAT_ID': row["GLCAT_MCAT_ID"],
        'PMCAT': row["PMCAT"],
        'MCAT_Name': row["GLCAT_MCAT_NAME"],
        'Subcat_ID': subcat_id,
        'Stopword': stopwords,
        'Attribute': attribute_string.strip()
    }, ignore_index=True)

# Write the final DataFrame to a CSV file
merged_pmcat_df['Attribute'] = merged_pmcat_df['Attribute'].str.lstrip()
merged_pmcat_df.to_csv("stopwordsremovedmcats.csv", index=False)
