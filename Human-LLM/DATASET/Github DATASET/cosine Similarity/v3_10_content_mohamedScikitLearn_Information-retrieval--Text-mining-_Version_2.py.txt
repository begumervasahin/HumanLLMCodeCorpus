import nltk
import string
from sklearn.feature_extraction.text import CountVectorizer, TfidfTransformer, TfidfVectorizer

# Initialize NLTK resources
nltk.download('punkt')
nltk.download('wordnet')

# Sample data
dataAll = [
    "This is a sample document for testing the TF-IDF vectorization process.",
    "TF-IDF stands for Term Frequency-Inverse Document Frequency.",
    "It is commonly used in natural language processing and information retrieval.",
]

# Token normalization functions
def normalize_text(text):
    tokens = nltk.word_tokenize(text.translate(str.maketrans('', '', string.punctuation)).lower())
    lemmatizer = nltk.stem.WordNetLemmatizer()
    return [lemmatizer.lemmatize(token) for token in tokens]

# TF-IDF vectorization
vectorizer = CountVectorizer(tokenizer=normalize_text, stop_words='english')
tf_matrix = vectorizer.fit_transform(dataAll).toarray()

print("Vocabulary:")
print(vectorizer.vocabulary_)
print("\nTerm Frequency (TF) Matrix:")
print(tf_matrix)

# TF-IDF transformation
tfidf_transformer = TfidfTransformer(norm="l2")
tfidf_matrix = tfidf_transformer.fit_transform(tf_matrix).toarray()

print("\nIDF Values:")
print(tfidf_transformer.idf_)
print("\nTF-IDF Matrix:")
print(tfidf_matrix)

# Cosine similarity
def calculate_cosine_similarity(text_list):
    vectorizer = TfidfVectorizer(tokenizer=normalize_text, stop_words='english')
    tfidf_matrix = vectorizer.fit_transform(text_list)
    return (tfidf_matrix * tfidf_matrix.T).toarray()

print("\nCosine Similarity Matrix:")
print(calculate_cosine_similarity(dataAll))
