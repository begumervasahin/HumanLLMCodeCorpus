import numpy as np
import pandas as pd
import nltk
from sklearn.feature_extraction.text import TfidfTransformer, CountVectorizer
from sklearn.metrics.pairwise import cosine_similarity

# Load the article data from a CSV file
df_articles = pd.read_csv('article_data.csv')

# Extract the target text from the first row
target_text = df_articles.content.iloc[0]

# Tokenize and preprocess the text using NLTK
nltk.download('punkt')
texts = [article for article in df_articles.content]

# Compute TF-IDF matrix
vectorizer = CountVectorizer(stop_words='english')
vectorizer.fit_transform([target_text])
frequency_term_matrix = vectorizer.transform(texts)

tfidf_transformer = TfidfTransformer(norm="l2")
tfidf_transformer.fit(frequency_term_matrix)
tf_idf_matrix = tfidf_transformer.transform(frequency_term_matrix)

# Compute cosine similarity between the target text and all other texts
cosine_similarities = cosine_similarity(tf_idf_matrix[0:1], tf_idf_matrix)
cos_sim_df = pd.DataFrame(cosine_similarities[0], columns=['Cosine_Similarity'])

# Concatenate cosine similarity with the original DataFrame
df_final = pd.concat([df_articles, cos_sim_df], axis=1)

# Sort DataFrame by cosine similarity in descending order
df_final_sorted = df_final.sort_values(by='Cosine_Similarity', ascending=False)

# Print the top 6 most similar articles
print(df_final_sorted.head(6))
