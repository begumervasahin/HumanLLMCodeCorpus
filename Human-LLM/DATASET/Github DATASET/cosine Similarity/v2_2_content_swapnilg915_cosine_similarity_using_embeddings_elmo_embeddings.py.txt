import regex as re
import numpy as np
import tensorflow as tf
import tensorflow_hub as hub
from keras import backend as K
from sklearn.metrics.pairwise import cosine_similarity
from nltk.corpus import stopwords
from nltk.tokenize import word_tokenize
from nltk.stem import WordNetLemmatizer
from abydos.distance import Jaccard

class ElmoSimilarity:
    def __init__(self):
        self.stop_words = set(stopwords.words('english'))
        self.lemmatizer = WordNetLemmatizer()
        self.elmo = hub.Module("https://tfhub.dev/google/elmo/2", trainable=True)

    def preprocess_tokens(self, tokens, lemma=False):
        tokens = str(tokens)
        tokens = re.sub(r"([\w].)([\~\!\@\#\$\%\^\&\*\(\)\-\=\+\[\{\]\}\:\;\'\"\,\<\.\>\/\?])", r"\1 \2", tokens)
        tokens = re.sub(r"\s+", " ", tokens)
        tokens = word_tokenize(tokens.lower())
        if lemma:
            tokens = [self.lemmatizer.lemmatize(token, 'v') for token in tokens if token.isalpha() and token not in self.stop_words]
        else:
            tokens = [token for token in tokens if token.isalpha() and token not in self.stop_words]
        return " ".join(tokens)

    def cosine_similarity(self, a, b):
        return np.inner(a, b) / (np.linalg.norm(a) * np.linalg.norm(b))

    def embed_sentences(self, sentences):
        elmo = hub.Module("https://tfhub.dev/google/elmo/2", trainable=True)
        executable = elmo(sentences, signature="default", as_dict=True)["elmo"]
        with tf.Session() as sess:
            sess.run([tf.global_variables_initializer(), tf.tables_initializer()])
            return sess.run(executable)[0]

    def main(self):
        jd = 'machine learning'
        resume = 'lead learning'
        print("=" * 75)
        print("\n ", jd, "\n ", resume)
        jd_embeddings = self.embed_sentences([self.preprocess_tokens(jd)])
        print("\n jd_embeddings --- ", jd_embeddings, jd_embeddings.shape)
        resume_embeddings = self.embed_sentences([self.preprocess_tokens(resume)])
        sim = self.cosine_similarity(jd_embeddings, resume_embeddings)
        print("\n elmo similarity --- ", sim)

if __name__ == '__main__':
    obj = ElmoSimilarity()
    obj.main()
