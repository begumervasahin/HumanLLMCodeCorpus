import tensorflow as tf

# Constants
HOP = 2048
BIN = 2048

# Configuration for TensorFlow session
config = tf.ConfigProto()
config.gpu_options.allow_growth = True

# Create TensorFlow session
sess = tf.Session(config=config)

def generate_mfcc_graph(pcm, num_mfccs=24, sample_rate=44100.0):
    """
    Generate Mel-Frequency Cepstral Coefficients (MFCCs) graph.
    """
    # Short-time Fourier transform (STFT)
    stfts = tf.contrib.signal.stft(pcm, frame_length=BIN, frame_step=HOP, fft_length=BIN)
    spectrograms = tf.abs(stfts, name="spectrograms")

    # Define mel-frequency filterbank
    num_spectrogram_bins = spectrograms.shape[-1].value
    lower_edge_hertz, upper_edge_hertz, num_mel_bins = 80.0, 7600.0, 64
    linear_to_mel_weight_matrix = tf.contrib.signal.linear_to_mel_weight_matrix(
        num_mel_bins, num_spectrogram_bins, sample_rate, lower_edge_hertz, upper_edge_hertz)

    # Apply mel-frequency filterbank to spectrograms
    mel_spectrograms = tf.tensordot(spectrograms, linear_to_mel_weight_matrix, 1)
    mel_spectrograms.set_shape(spectrograms.shape[:-1].concatenate(linear_to_mel_weight_matrix.shape[-1:]))

    # Apply log operation to mel-spectrograms
    log_offset = 1e-6
    log_mel_spectrograms = tf.log(mel_spectrograms + log_offset)

    # Compute MFCCs from log mel-spectrograms
    mfccs = tf.contrib.signal.mfccs_from_log_mel_spectrograms(log_mel_spectrograms)[..., :num_mfccs]
    return mfccs

# Placeholders for input PCM data
full_source_pcm = tf.placeholder(tf.float32, [None], name="source_pcm")
target_pcm = tf.placeholder(tf.float32, [None], name="target_pcm")

# Generate MFCCs for source and target PCM data
full_source_mfccs = generate_mfcc_graph(full_source_pcm)
target_mfccs = generate_mfcc_graph(target_pcm)

# Compute dot product and normalize dot product matrices
dot_product = tf.matmul(full_source_mfccs, tf.transpose(target_mfccs))
full_source_norms = tf.norm(full_source_mfccs, axis=1)
target_norms = tf.norm(target_mfccs, axis=1)
norm_dot_product = tf.matmul(tf.reshape(full_source_norms, (-1, 1)), tf.reshape(target_norms, (1, -1)))

# Compute similarity matrix and angular distance
similarity_matrix = tf.divide(dot_product, norm_dot_product)
angular_distance = tf.acos(similarity_matrix, name="angular_distance")

# Find best match and corresponding start frames
best_match = tf.argmin(angular_distance, axis=0)
start_frames = tf.multiply(best_match, HOP, name="start_frames")

# Get the graph definition and save it to a file
graph_def = sess.graph_def
directory = './'
tf.train.write_graph(graph_def, directory, 'model.pb', as_text=False)
